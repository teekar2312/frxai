    37→// FNH-008: Retry logic with AbortController timeout
    38→async function fetchWithRetry(url: string, timeoutMs = 8000, retries = 2): Promise<Response> {
    39→  return new Promise((resolve, reject) => {
    40→    const attempt = (tryNum: number) => {
    41→      const controller = new AbortController();
    42→      const timer = setTimeout(() => controller.abort(), timeoutMs);
    43→      fetch(url, { signal: controller.signal })
    44→        .then((res) => {
    45→          clearTimeout(timer);
    46→          if (res.status === 429 && tryNum < retries) {
    47→            setTimeout(() => attempt(tryNum + 1), 1000 * (tryNum + 1));
    48→            return;
    49→          }
    50→          resolve(res);
    51→        })
    52→        .catch((err) => {
    53→          clearTimeout(timer);
    54→          if (tryNum < retries) {
    55→            setTimeout(() => attempt(tryNum + 1), 500 * (tryNum + 1));
    56→            return;
    57→          }
    58→          reject(err);
    59→        });
    60→    };
    61→    attempt(0);
    62→  });
    63→}

    65→// FNH-018: Simulated candles with time-weighted volume for backtest fallback
    66→function generateSimulatedCandles(pair: ForexPair, count: number): CandleData[] {
    67→  const base = SIMULATED_BASES[pair];
    68→  const candles: CandleData[] = [];
    69→  let price = base.price * (1 - base.volatility * 2);
    70→  const interval = 3600; // H1 default
    71→  const now = Math.floor(Date.now() / 1000);
    72→
    73→  for (let i = 0; i < count; i++) {
    74→    const open = price;
    75→    const change = (Math.random() - 0.5) * base.volatility * 2;
    76→    const close = open + change;
    77→    const high = Math.max(open, close) + Math.random() * base.volatility;
    78→    const low = Math.min(open, close) - Math.random() * base.volatility;
    79→    const volume = Math.floor(Math.random() * 3000 + 500);
    80→    candles.push({ time: (now - (count - i) * interval) * 1000, open: parseFloat(open.toFixed(5)), high: parseFloat(high.toFixed(5)), low: parseFloat(low.toFixed(5)), close: parseFloat(close.toFixed(5)), volume });
    81→    price = close;
    82→  }
    83→  return candles;
    84→}

    86→async function fetchHistoricalCandles(
    87→  pair: ForexPair,
    88→  resolution: string,
    89→  startDate: string,
    90→  endDate: string
    91→): Promise<CandleData[]> {
    92→  const apiKey = process.env.FINNHUB_API_KEY;
    93→  // FNH-007: Consistent fallback to simulated data
    94→  if (!apiKey) return generateSimulatedCandles(pair, 500);
    95→
    96→  // FNH-002: Use shared symbol mapping
    97→  const finnhubSymbol = PAIR_TO_FINNHUB_SYMBOL[pair] || `OANDA:${pair.slice(0, 3)}_${pair.slice(3)}`;
    98→  // FNH-014: Convert resolution alias to Finnhub format
    99→  const finnhubResolution = toFinnhubResolution(resolution);
   100→  const from = Math.floor(new Date(startDate).getTime() / 1000);
   101→  const to = Math.floor(new Date(endDate).getTime() / 1000);
   102→  const allCandles: CandleData[] = [];
   103→
   104→  const maxCandles = 500;
   105→  const resSeconds = getResolutionSeconds(resolution);
   106→  const totalSeconds = to - from;
   107→  const totalCandlesNeeded = Math.ceil(totalSeconds / resSeconds);
   108→  const batches = Math.ceil(totalCandlesNeeded / maxCandles);
   109→
   110→  for (let batch = 0; batch < batches; batch++) {
   111→    const batchFrom = from + (batch * maxCandles * resSeconds);
   112→    const batchTo = Math.min(batchFrom + (maxCandles * resSeconds), to);
   113→
   114→    const url = `https://finnhub.io/api/v1/stock/candle?symbol=${finnhubSymbol}&resolution=${finnhubResolution}&from=${batchFrom}&to=${batchTo}&token=${apiKey}`;
   115→
   116→    try {
   117→      const res = await fetchWithRetry(url);
   118→      if (!res.ok) continue;
   119→      const data = await res.json();
   120→      if (data.s !== 'ok' || !data.t) continue;
   121→      for (let i = 0; i < data.t.length; i++) {
   122→        allCandles.push({ time: data.t[i] * 1000, open: data.o[i], high: data.h[i], low: data.l[i], close: data.c[i], volume: data.v[i] || 0 });
   123→      }
   124→    } catch (err) {
   125→      logApiError('Backtest-Fetch', err);
   126→      continue;
   127→    }
   128→
   129→    if (batch < batches - 1) {
   130→      await new Promise((r) => setTimeout(r, 300));
   131→    }
   132→  }
   133→
   134→  allCandles.sort((a, b) => a.time - b.time);
   135→  const unique = allCandles.filter((c, i, arr) => i === 0 || c.time !== arr[i - 1].time);
   136→
   137→  // FNH-007: Fallback to simulated if insufficient real data
   138→  if (unique.length < 50) return generateSimulatedCandles(pair, Math.max(500, 100));
   139→  return unique;
   140→}

   147→function generateSignal(
   148→  strategy: StrategyName,
   149→  candles: OHLCV[],
   150→  index: number,
   151→  indicators: Record<string, unknown>
   152→): SignalResult {
   153→  const close = candles[index].close;
   154→
   155→  switch (strategy) {
   156→    case 'MA_RIBBON': {
   157→      const ema5 = indicators.ema5 as number[];
   158→      const ema9 = indicators.ema9 as number[];
   159→      const ema21 = indicators.ema21 as number[];
   160→      const ema50 = indicators.ema50 as number[];
   161→      if (!ema5 || !ema21 || !ema50) return { direction: null, strength: 0 };
   162→      const e5 = ema5[index], e21 = ema21[index], e50 = ema50[index];
   163→      if (isNaN(e5) || isNaN(e21) || isNaN(e50)) return { direction: null, strength: 0 };
   164→      const bullish = e5 > e21 && e21 > e50;
   165→      const bearish = e5 < e21 && e21 < e50;
   166→      return {
   167→        direction: bullish ? 'BUY' : bearish ? 'SELL' : null,
   168→        strength: Math.abs(e5 - e50) / close,
   169→      };
   170→    }
   171→
   172→    case 'EMA_CROSSOVER': {
   173→      const emaFast = indicators.ema9 as number[];
   174→      const emaSlow = indicators.ema21 as number[];
   175→      if (!emaFast || !emaSlow) return { direction: null, strength: 0 };
   176→      const fast = emaFast[index], slow = emaSlow[index];
   177→      const prevFast = emaFast[index - 1], prevSlow = emaSlow[index - 1];
   178→      if (isNaN(fast) || isNaN(slow) || isNaN(prevFast) || isNaN(prevSlow)) return { direction: null, strength: 0 };
   179→      // Crossover detection
   180→      if (prevFast <= prevSlow && fast > slow) return { direction: 'BUY', strength: (fast - slow) / close };
   181→      if (prevFast >= prevSlow && fast < slow) return { direction: 'SELL', strength: (slow - fast) / close };
   182→      // Trend continuation
   183→      if (fast > slow) return { direction: 'BUY', strength: (fast - slow) / close * 0.3 };
   184→      if (fast < slow) return { direction: 'SELL', strength: (slow - fast) / close * 0.3 };
   185→      return { direction: null, strength: 0 };
   186→    }
   187→
   188→    case 'MOMENTUM_SCALPING': {
   189→      const rsiVal = indicators.rsi as number[];
   190→      const mom = indicators.momentum as number[];
   191→      const macdHist = indicators.macdHistogram as number[];
   192→      if (!rsiVal || !mom || !macdHist) return { direction: null, strength: 0 };
   193→      const r = rsiVal[index], m = mom[index], h = macdHist[index];
   194→      if (isNaN(r) || isNaN(m) || isNaN(h)) return { direction: null, strength: 0 };
   195→      if (r < 30 && m < 0 && h > 0) return { direction: 'BUY', strength: 0.8 };
   196→      if (r > 70 && m > 0 && h < 0) return { direction: 'SELL', strength: 0.8 };
   197→      if (r < 40 && h > 0) return { direction: 'BUY', strength: 0.5 };
   198→      if (r > 60 && h < 0) return { direction: 'SELL', strength: 0.5 };
   199→      return { direction: null, strength: 0 };
   200→    }
   201→
   202→    case 'PIVOT_POINT': {
   203→      const pp = indicators.pivotPoints as { pp: number; r1: number; r2: number; s1: number; s2: number }[];
   204→      if (!pp || pp.length === 0) return { direction: null, strength: 0 };
   205→      const latestPp = pp[pp.length - 1];
   206→      if (close > latestPp.r1) return { direction: 'BUY', strength: (close - latestPp.pp) / close };
   207→      if (close < latestPp.s1) return { direction: 'SELL', strength: (latestPp.pp - close) / close };
   208→      if (close > latestPp.pp) return { direction: 'BUY', strength: 0.3 };
   209→      if (close < latestPp.pp) return { direction: 'SELL', strength: 0.3 };
   210→      return { direction: null, strength: 0 };
   211→    }
   212→
   213→    case 'RMI_TREND_SYNC': {
   214→      const stc = indicators.stc as number[];
   215→      const stDir = indicators.supertrendDirection as number[];
   216→      if (!stc || !stDir) return { direction: null, strength: 0 };
   217→      const s = stc[index], d = stDir[index];
   218→      if (isNaN(s) || isNaN(d)) return { direction: null, strength: 0 };
   219→      const stBullish = d === 1;
   220→      const stcOversold = s < 25;
   221→      const stcOverbought = s > 75;
   222→      if (stBullish && stcOversold) return { direction: 'BUY', strength: 0.9 };
   223→      if (!stBullish && stcOverbought) return { direction: 'SELL', strength: 0.9 };
   224→      if (stBullish && s < 50) return { direction: 'BUY', strength: 0.6 };
   225→      if (!stBullish && s > 50) return { direction: 'SELL', strength: 0.6 };
   226→      return { direction: null, strength: 0 };
   227→    }
   228→
   229→    case 'LINEAR_REGRESSION': {
   230→      const lrcUpper = indicators.lrcUpper as number[];
   231→      const lrcMiddle = indicators.lrcMiddle as number[];
   232→      const lrcLower = indicators.lrcLower as number[];
   233→      if (!lrcUpper || !lrcMiddle || !lrcLower) return { direction: null, strength: 0 };
   234→      const u = lrcUpper[index], m = lrcMiddle[index], l = lrcLower[index];
   235→      if (isNaN(u) || isNaN(m) || isNaN(l)) return { direction: null, strength: 0 };
   236→      if (close <= l) return { direction: 'BUY', strength: (m - l) / close };
   237→      if (close >= u) return { direction: 'SELL', strength: (u - m) / close };
   238→      return { direction: null, strength: 0 };
   239→    }
   240→
   241→    case 'EMA_RSI_FILTER': {
   242→      const emaFast = indicators.ema9 as number[];
   243→      const emaSlow = indicators.ema21 as number[];
   244→      const rsiVal = indicators.rsi as number[];
   245→      if (!emaFast || !emaSlow || !rsiVal) return { direction: null, strength: 0 };
   246→      const ef = emaFast[index], es = emaSlow[index], r = rsiVal[index];
   247→      if (isNaN(ef) || isNaN(es) || isNaN(r)) return { direction: null, strength: 0 };
   248→      if (ef > es && r > 40 && r < 70) return { direction: 'BUY', strength: 0.7 };
   249→      if (ef < es && r > 30 && r < 60) return { direction: 'SELL', strength: 0.7 };
   250→      return { direction: null, strength: 0 };
   251→    }
   252→
   253→    default:
   254→      return { direction: null, strength: 0 };
   255→  }
   256→}

   258→function precomputeIndicators(candles: OHLCV[]): Record<string, unknown> {
   259→  const closes = candles.map((c) => c.close);
   260→  return {
   261→    ema5: ema(closes, 5),
   262→    ema9: ema(closes, 9),
   263→    ema13: ema(closes, 13),
   264→    ema21: ema(closes, 21),
   265→    ema50: ema(closes, 50),
   266→    rsi: rsi(closes, 14),
   267→    momentum: closes.map((v, i) => (i < 10 ? NaN : v - closes[i - 10])),
   268→    macdHistogram: macd(closes, 12, 26, 9).histogram,
   269→    stc: schaffTrendCycle(closes, 23, 50, 10),
   270→    supertrendDirection: supertrend(candles, 10, 3).direction,
   271→    lrcUpper: linearRegressionChannel(closes, 20).upper,
   272→    lrcMiddle: linearRegressionChannel(closes, 20).middle,
   273→    lrcLower: linearRegressionChannel(closes, 20).lower,
   274→    atr: atr(candles, 14),
   275→  };
   276→}

   278→function runBacktestSimulation(
   279→  candles: OHLCV[],
   280→  config: BacktestConfig,
   281→  pipConfig: { standard: number; pipSize: number }
   282→): { trades: BacktestTrade[]; equityCurve: number[] } {
   283→  const indicators = precomputeIndicators(candles);
   284→  const atrValues = indicators.atr as number[];
   285→  const trades: BacktestTrade[] = [];
   286→  const equityCurve: number[] = [];
   287→  let balance = config.initialBalance;
   288→  let openTrade: BacktestTrade | null = null;
   289→  let tradeId = 0;
   290→  const minLookback = 60; // Minimum candles before trading starts
   291→  const maxOpenPositions = 1; // Simplified: one position at a time
   292→
   293→  for (let i = minLookback; i < candles.length; i++) {
   294→    const candle = candles[i];
   295→
   296→    // Check open trade for SL/TP
   297→    if (openTrade) {
   298→      const sl = openTrade.stopLoss;
   299→      const tp = openTrade.takeProfit;
   300→      const direction = openTrade.direction;
   301→
   302→      let closed = false;
   303→      let closePrice = candle.close;
   304→      let reason = 'signal';
   305→
   306→      if (direction === 'BUY') {
   307→        if (candle.low <= sl) { closePrice = sl; reason = 'stop_loss'; closed = true; }
   308→        else if (candle.high >= tp) { closePrice = tp; reason = 'take_profit'; closed = true; }
   309→      } else {
   310→        if (candle.high >= sl) { closePrice = sl; reason = 'stop_loss'; closed = true; }
   311→        else if (candle.low <= tp) { closePrice = tp; reason = 'take_profit'; closed = true; }
   312→      }
   313→
   314→      // Also check for signal-based close
   315→      if (!closed) {
   316→        const signal = generateSignal(config.strategy, candles, i, indicators);
   317→        if (signal.direction && signal.direction !== openTrade.direction && signal.strength > 0.4) {
   318→          closePrice = candle.close;
   319→          reason = 'reverse_signal';
   320→          closed = true;
   321→        }
   322→      }
   323→
   324→      if (closed) {
   325→        const pipDiff = direction === 'BUY'
   326→          ? (closePrice - openTrade.entryPrice) / pipConfig.pipSize
   327→          : (openTrade.entryPrice - closePrice) / pipConfig.pipSize;
   328→        const pnl = pipDiff * pipConfig.standard * openTrade.lotSize - FINEX_CONFIG.commissionPerLot * openTrade.lotSize;
   329→        balance += pnl;
   330→
   331→        openTrade.exitIndex = i;
   332→        openTrade.exitPrice = closePrice;
   333→        openTrade.pnl = pnl;
   334→        openTrade.pnlPips = pipDiff;
   335→        openTrade.reason = reason;
   336→        openTrade.exitTime = candle.time;
   337→        trades.push(openTrade);
   338→        openTrade = null;
   339→      }
   340→    }
   341→
   342→    // Open new trade if no position
   343→    if (!openTrade && i < candles.length - 5) {
   344→      const signal = generateSignal(config.strategy, candles, i, indicators);
   345→      if (signal.direction && signal.strength > 0.2) {
   346→        const atrVal = atrValues[i];
   347→        if (isNaN(atrVal) || atrVal === 0) continue;
   348→
   349→        const slPips = Math.max(config.stopLossPips, Math.round(atrVal / pipConfig.pipSize * 1.5));
   350→        const tpPips = Math.round(slPips * 1.5);
   351→        const riskAmount = balance * (config.riskPerTrade / 100);
   352→        const lotSize = Math.max(
   353→          FINEX_CONFIG.minLot,
   354→          Math.min(
   355→            FINEX_CONFIG.maxLotPerOrder,
   356→            parseFloat((riskAmount / (slPips * pipConfig.standard)).toFixed(2))
   357→          )
   358→        );
   359→
   360→        const entryPrice = candle.close;
   361→        let stopLoss: number, takeProfit: number;
   362→        if (signal.direction === 'BUY') {
   363→          stopLoss = entryPrice - slPips * pipConfig.pipSize;
   364→          takeProfit = entryPrice + tpPips * pipConfig.pipSize;
   365→        } else {
   366→          stopLoss = entryPrice + slPips * pipConfig.pipSize;
   367→          takeProfit = entryPrice - tpPips * pipConfig.pipSize;
   368→        }
   369→
   370→        openTrade = {
   371→          id: ++tradeId,
   372→          entryIndex: i,
   373→          exitIndex: 0,
   374→          direction: signal.direction,
   375→          entryPrice,
   376→          exitPrice: 0,
   377→          lotSize,
   378→          stopLoss,
   379→          takeProfit,
   380→          pnl: 0,
   381→          pnlPips: 0,
   382→          reason: '',
   383→          entryTime: candle.time,
   384→          exitTime: 0,
   385→        };
   386→      }
   387→    }
   388→
   389→    equityCurve.push(balance);
   390→  }
   391→
   392→  // Close any remaining open trade at last price
   393→  if (openTrade) {
   394→    const lastCandle = candles[candles.length - 1];
   395→    const pipDiff = openTrade.direction === 'BUY'
   396→      ? (lastCandle.close - openTrade.entryPrice) / pipConfig.pipSize
   397→      : (openTrade.entryPrice - lastCandle.close) / pipConfig.pipSize;
   398→    const pnl = pipDiff * pipConfig.standard * openTrade.lotSize - FINEX_CONFIG.commissionPerLot * openTrade.lotSize;
   399→    balance += pnl;
   400→
   401→    openTrade.exitIndex = candles.length - 1;
   402→    openTrade.exitPrice = lastCandle.close;
   403→    openTrade.pnl = pnl;
   404→    openTrade.pnlPips = pipDiff;
   405→    openTrade.reason = 'end_of_data';
   406→    openTrade.exitTime = lastCandle.time;
   407→    trades.push(openTrade);
   408→    equityCurve[equityCurve.length - 1] = balance;
   409→  }
   410→
   411→  return { trades, equityCurve };
   412→}

   414→function calculateBacktestStats(
   415→  trades: BacktestTrade[],
   416→  equityCurve: number[],
   417→  config: BacktestConfig
   418→): Omit<BacktestResult, 'id'> {
   419→  const wins = trades.filter((t) => t.pnl > 0);
   420→  const losses = trades.filter((t) => t.pnl <= 0);
   421→  const totalPnl = trades.reduce((sum, t) => sum + t.pnl, 0);
   422→  const winRate = trades.length > 0 ? (wins.length / trades.length) * 100 : 0;
   423→
   424→  // Max drawdown
   425→  let maxDrawdown = 0;
   426→  let peak = config.initialBalance;
   427→  for (const eq of equityCurve) {
   428→    if (eq > peak) peak = eq;
   429→    const dd = ((peak - eq) / peak) * 100;
   430→    if (dd > maxDrawdown) maxDrawdown = dd;
   431→  }
   432→
   433→  // Sharpe ratio (simplified - annualized)
   434→  let sharpeRatio: number | null = null;
   435→  if (trades.length > 2) {
   436→    const returns = trades.map((t) => t.pnl / config.initialBalance);
   437→    const avgReturn = returns.reduce((a, b) => a + b, 0) / returns.length;
   438→    const stdReturn = Math.sqrt(returns.reduce((sum, r) => sum + (r - avgReturn) ** 2, 0) / (returns.length - 1));
   439→    if (stdReturn > 0) {
   440→      sharpeRatio = parseFloat(((avgReturn / stdReturn) * Math.sqrt(252)).toFixed(2));
   441→    }
   442→  }
   443→
   444→  // Profit factor
   445→  let profitFactor: number | null = null;
   446→  const totalWins = wins.reduce((sum, t) => sum + t.pnl, 0);
   447→  const totalLosses = Math.abs(losses.reduce((sum, t) => sum + t.pnl, 0));
   448→  if (totalLosses > 0) {
   449→    profitFactor = parseFloat((totalWins / totalLosses).toFixed(2));
   450→  } else if (totalWins > 0) {
   451→    profitFactor = 999;
   452→  }
   453→
   454→  // Avg win/loss
   455→  const avgWin = wins.length > 0 ? parseFloat((wins.reduce((s, t) => s + t.pnl, 0) / wins.length).toFixed(2)) : null;
   456→  const avgLoss = losses.length > 0 ? parseFloat((losses.reduce((s, t) => s + t.pnl, 0) / losses.length).toFixed(2)) : null;
   457→
   458→  // Max consecutive wins/losses
   459→  let maxConsecWins = 0, maxConsecLosses = 0;
   460→  let consecWins = 0, consecLosses = 0;
   461→  for (const t of trades) {
   462→    if (t.pnl > 0) { consecWins++; consecLosses = 0; }
   463→    else { consecLosses++; consecWins = 0; }
   464→    maxConsecWins = Math.max(maxConsecWins, consecWins);
   465→    maxConsecLosses = Math.max(maxConsecLosses, consecLosses);
   466→  }
   467→
   468→  return {
   469→    name: `${config.strategy} - ${config.pair} - ${config.timeframe}`,
   470→    pair: config.pair,
   471→    strategy: config.strategy,
   472→    timeframe: config.timeframe,
   473→    startDate: config.startDate,
   474→    endDate: config.endDate,
   475→    initialBalance: config.initialBalance,
   476→    finalBalance: parseFloat(equityCurve[equityCurve.length - 1]?.toFixed(2) || config.initialBalance.toFixed(2)),
   477→    totalTrades: trades.length,
   478→    winningTrades: wins.length,
   479→    losingTrades: losses.length,
   480→    winRate: parseFloat(winRate.toFixed(2)),
   481→    totalPnl: parseFloat(totalPnl.toFixed(2)),
   482→    maxDrawdown: parseFloat(maxDrawdown.toFixed(2)),
   483→    sharpeRatio,
   484→    profitFactor,
   485→    avgWin,
   486→    avgLoss,
   487→    maxConsecutiveWins: maxConsecWins,
   488→    maxConsecutiveLosses: maxConsecLosses,
   489→  };
   490→}

   492→export async function POST(request: NextRequest) {
   493→  const auth = requireAuthForMutation(request);
   494→  if (!auth.authorized) return auth.error!;
   495→  if (!request.headers.get('content-type')?.includes('application/json')) {
   496→    return NextResponse.json({ error: 'Content-Type must be application/json' }, { status: 415 });
   497→  }
   498→  const rateCheck = checkRateLimit(clientIp(request), 'general');
   499→  if (!rateCheck.allowed) return rateLimitedResponse(rateCheck.retryAfterMs);
   500→  try {
   501→    const body = await request.json();
   502→    const config = body as BacktestConfig;
   503→
   504→    // M-3: Validate pair, strategy, and timeframe against allowed lists
   505→    const { FOREX_PAIRS } = await import('@/lib/trading-types');
   506→    if (!FOREX_PAIRS.includes(config.pair)) {
   507→      return NextResponse.json({ error: `Invalid pair. Must be one of: ${FOREX_PAIRS.join(', ')}` }, { status: 400 });
   508→    }
   509→    const VALID_STRATEGIES: string[] = ['MA_RIBBON', 'MOMENTUM_SCALPING', 'PIVOT_POINT', 'EMA_CROSSOVER', 'RMI_TREND_SYNC', 'LINEAR_REGRESSION', 'EMA_RSI_FILTER'];
   510→    if (!VALID_STRATEGIES.includes(config.strategy)) {
   511→      return NextResponse.json({ error: `Invalid strategy. Must be one of: ${VALID_STRATEGIES.join(', ')}` }, { status: 400 });
   512→    }
   513→    const VALID_TIMEFRAMES = ['M1', 'M5', 'M15', 'M30', 'H1', 'H4', 'D1', 'W1'];
   514→    if (!VALID_TIMEFRAMES.includes(config.timeframe)) {
   515→      return NextResponse.json({ error: `Invalid timeframe. Must be one of: ${VALID_TIMEFRAMES.join(', ')}` }, { status: 400 });
   516→    }
   517→
   518→    if (!config.pair || !config.strategy || !config.timeframe || !config.startDate || !config.endDate) {
   519→      return NextResponse.json(
   520→        { error: 'pair, strategy, timeframe, startDate, and endDate are required' },
   521→        { status: 400 }
   522→      );
   523→    }
   524→
   525→    // BT-01: Validate numeric fields
   526→    const { stopLossPips, riskPerTrade, initialBalance, maxPositions: configMaxPositions } = body;
   527→    if (!stopLossPips || stopLossPips <= 0 || stopLossPips > 500) {
   528→      return NextResponse.json({ error: 'stopLossPips must be between 1 and 500' }, { status: 400 });
   529→    }
   530→    if (typeof riskPerTrade !== 'number' || riskPerTrade < 0.1 || riskPerTrade > 100) {
   531→      return NextResponse.json({ error: 'riskPerTrade must be between 0.1 and 100' }, { status: 400 });
   532→    }
   533→    if (!initialBalance || initialBalance < 100) {
   534→      return NextResponse.json({ error: 'initialBalance must be at least 100' }, { status: 400 });
   535→    }
   536→
   537→    // BT-05: Limit backtest date range
   538→    const maxDays = 730; // 2 years
   539→    const dayDiff = (new Date(config.endDate).getTime() - new Date(config.startDate).getTime()) / (1000 * 60 * 60 * 24);
   540→    if (dayDiff > maxDays) {
   541→      return NextResponse.json({ error: `Date range too large (max ${maxDays} days)` }, { status: 400 });
   542→    }
   543→    if (dayDiff < 1) {
   544→      return NextResponse.json({ error: 'Date range must be at least 1 day' }, { status: 400 });
   545→    }
   546→
   547→    // Fetch historical candles
   548→    const candles = await fetchHistoricalCandles(
   549→      config.pair,
   550→      config.timeframe,
   551→      config.startDate,
   552→      config.endDate
   553→    );
   554→
   555→    if (candles.length < 100) {
   556→      return NextResponse.json(
   557→        { error: `Insufficient historical data: ${candles.length} candles. Need at least 100.` },
   558→        { status: 400 }
   559→      );
   560→    }
   561→
   562→    const ohlcv: OHLCV[] = candles.map((c) => ({
   563→      time: c.time,
   564→      open: c.open,
   565→      high: c.high,
   566→      low: c.low,
   567→      close: c.close,
   568→      volume: c.volume || 0,
   569→    }));
   570→
   571→    const pipConfig = PAIR_PIP_VALUES[config.pair] || { standard: 10, pipSize: 0.0001 };
   572→
   573→    // Run the simulation
   574→    const { trades, equityCurve } = runBacktestSimulation(ohlcv, config, pipConfig);
   575→
   576→    // Calculate statistics
   577→    const stats = calculateBacktestStats(trades, equityCurve, config);
   578→
   579→    // Store result in DB
   580→    const dbResult = await db.backtestResult.create({
   581→      data: {
   582→        ...stats,
   583→        startDate: new Date(config.startDate),
   584→        endDate: new Date(config.endDate),
   585→        parameters: JSON.stringify({
   586→          stopLossPips: config.stopLossPips,
   587→          takeProfitPips: config.takeProfitPips,
   588→          riskPerTrade: config.riskPerTrade,
   589→          maxPositions: config.maxPositions,
   590→        }),
   591→      },
   592→    });
   593→
   594→    // Log the backtest
   595→    try {
   596→      await db.activityLog.create({
   597→        data: {
   598→          level: 'info',
   599→          category: 'analysis',
   600→          message: `Backtest completed: ${config.strategy} on ${config.pair} - ${trades.length} trades, win rate: ${stats.winRate}%, PnL: $${stats.totalPnl.toFixed(2)}`,
   601→          pair: config.pair,
   602→          metadata: JSON.stringify({
   603→            backtestId: dbResult.id,
   604→            trades: trades.length,
   605→            winRate: stats.winRate,
   606→            totalPnl: stats.totalPnl,
   607→            maxDrawdown: stats.maxDrawdown,
   608→            sharpeRatio: stats.sharpeRatio,
   609→          }),
   610→        },
   611→      });
   612→    } catch {
   613→      // Non-critical
   614→    }
   615→
   616→    return NextResponse.json({
   617→      success: true,
   618→      result: { ...stats, id: dbResult.id },
   619→      trades,
   620→      candlesUsed: candles.length,
   621→      equityCurve,
   622→    });
   623→  } catch (error) {
   624→    logApiError('Backtest', error);
   625→    return NextResponse.json(
   626→      { error: 'Backtest failed' },
   627→      { status: 500 }
   628→    );
   629→  }
   630→}
