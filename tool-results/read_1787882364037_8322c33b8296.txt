    65→// ============================================
    66→// ASYNC MUTEX
    67→// ============================================
    68→
    69→/**
    70→ * Simple async mutex to prevent concurrent MT5 API calls.
    71→ * The MT5 Python module is not thread-safe, so all calls must be serialized.
    72→ */
    73→export class AsyncMutex {
    74→  private _queue: Array<() => void> = []
    75→  private _locked = false
    76→
    77→  /** Acquire the mutex lock. Returns a release function. */
    78→  async acquire(): Promise<() => void> {
    79→    return new Promise<() => void>((resolve) => {
    80→      const tryAcquire = () => {
    81→        if (!this._locked) {
    82→          this._locked = true
    83→          resolve(this._release)
    84→        } else {
    85→          this._queue.push(tryAcquire)
    86→        }
    87→      }
    88→      tryAcquire()
    89→    })
    90→  }
    91→
    92→  /** Run a function exclusively within the mutex lock. */
    93→  async runExclusive<T>(fn: () => Promise<T>): Promise<T> {
    94→    const release = await this.acquire()
    95→    try {
    96→      return await fn()
    97→    } finally {
    98→      release()
    99→    }
   100→  }
   101→
   102→  private _release = (): void => {
   103→    this._locked = false
   104→    const next = this._queue.shift()
   105→    if (next) next()
   106→  }
   107→
   108→  get locked(): boolean {
   109→    return this._locked
   110→  }
   111→
   112→  get queueLength(): number {
   113→    return this._queue.length
   114→  }
   115→}

   572→/** Convert WIB (UTC+7) hours/minutes to UTC hours/minutes */
   573→function wibToUtc(wibHour: number, wibMinute: number = 0): { hour: number; minute: number } {
   574→  let utcHour = wibHour - 7
   575→  let utcMinute = wibMinute
   576→  if (utcHour < 0) utcHour += 24
   577→  return { hour: utcHour, minute: utcMinute }
   578→}
   579→
   580→// Pre-computed UTC boundaries for IDX trading
   581→const PHASE_BOUNDARIES_UTC = {
   582→  preOpenStart: wibToUtc(8, 45),     // 01:45 UTC
   583→  morningOpen: wibToUtc(9, 0),        // 02:00 UTC
   584→  preCloseStart: wibToUtc(11, 30),     // 04:30 UTC
   585→  preCloseEnd: wibToUtc(11, 30, 30),   // 04:30:30 UTC (treated as 04:30 since we only have minutes)
   586→  afternoonOpen: wibToUtc(13, 30),     // 06:30 UTC
   587→  marketClose: wibToUtc(15, 0),        // 08:00 UTC
   588→} as const
   589→
   590→/**
   591→ * Determine the current IDX trading phase based on UTC time.
   592→ * IDX Schedule (WIB / UTC):
   593→ *   PRE_OPEN:     08:45-09:00 WIB  = 01:45-02:00 UTC
   594→ *   OPEN:         09:00-11:30 WIB  = 02:00-04:30 UTC
   595→ *   PRE_CLOSE:    11:30-11:30:30    = ~04:30 UTC (30s window)
   596→ *   CLOSED:       11:30-13:30 WIB  = 04:30-06:30 UTC (lunch break)
   597→ *   OPEN:         13:30-15:00 WIB  = 06:30-08:00 UTC
   598→ *   AFTER_HOURS:  15:00+ WIB       = 08:00+ UTC
   599→ */
   600→export function getTradingPhase(now?: Date): TradingPhase {
   601→  const d = now || new Date()
   602→  // Get UTC hours and minutes as decimal for comparison
   603→  const utcDecimal = d.getUTCHours() + d.getUTCMinutes() / 60 + d.getUTCSeconds() / 3600
   604→
   605→  const { preOpenStart, morningOpen, preCloseStart, preCloseEnd, afternoonOpen, marketClose } = PHASE_BOUNDARIES_UTC
   606→
   607→  // Convert boundaries to decimal hours for comparison
   608→  const preOpenDec = preOpenStart.hour + preOpenStart.minute / 60
   609→  const morningDec = morningOpen.hour + morningOpen.minute / 60
   610→  const preCloseDec = preCloseStart.hour + preCloseStart.minute / 60
   611→  const preCloseEndDec = preCloseStart.hour + preCloseStart.minute / 60 + 30 / 3600 // 30 seconds
   612→  const afternoonDec = afternoonOpen.hour + afternoonOpen.minute / 60
   613→  const closeDec = marketClose.hour + marketClose.minute / 60
   614→
   615→  if (utcDecimal >= preOpenDec && utcDecimal < morningDec) {
   616→    return "PRE_OPEN"
   617→  }
   618→  if (utcDecimal >= morningDec && utcDecimal < preCloseDec) {
   619→    return "OPEN"
   620→  }
   621→  // PRE_CLOSE is a 30-second window; treat it as part of PRE_CLOSE
   622→  if (utcDecimal >= preCloseDec && utcDecimal < preCloseEndDec) {
   623→    return "PRE_CLOSE"
   624→  }
   625→  // Lunch break CLOSED
   626→  if (utcDecimal >= preCloseDec && utcDecimal < afternoonDec) {
   627→    return "CLOSED"
   628→  }
   629→  // Afternoon session OPEN
   630→  if (utcDecimal >= afternoonDec && utcDecimal < closeDec) {
   631→    return "OPEN"
   632→  }
   633→  // After hours
   634→  return "AFTER_HOURS"
   635→}
   636→
   637→/** Check if the IDX market is currently open for trading. */
   638→export function isMarketOpen(now?: Date): boolean {
   639→  const phase = getTradingPhase(now)
   640→  return phase === "OPEN" || phase === "PRE_OPEN"
   641→}

   645→// ============================================
   646→
   647→let _errorCodesSeeded = false
   648→
   649→/**
   650→ * Seed all MT5 error codes into the Mt5ErrorCode table.
   651→ * Uses upsert for idempotency. Safe to call multiple times.
   652→ * Uses lazy initialization - no top-level await.
   653→ */
   654→export async function seedMt5ErrorCodes(): Promise<void> {
   655→  if (_errorCodesSeeded) return
   656→
   657→  try {
   658→    for (const entry of MT5_ERROR_CODES) {
   659→      await db.mt5ErrorCode.upsert({
   660→        where: { code: entry.code },
   661→        update: {
   662→          description: entry.description,
   663→          severity: entry.severity,
   664→          category: entry.category,
   665→          remediation: entry.remediation,
   666→          retryable: entry.retryable,
   667→        },
   668→        create: {
   669→          code: entry.code,
   670→          description: entry.description,
   671→          severity: entry.severity,
   672→          category: entry.category,
   673→          remediation: entry.remediation,
   674→          retryable: entry.retryable,
   675→        },
   676→      })
   677→    }
   678→    _errorCodesSeeded = true
   679→    logger.info("MT5_CONNECTION", `Seeded ${MT5_ERROR_CODES.length} MT5 error codes into database`)
   680→  } catch (err) {
   681→    const msg = err instanceof Error ? err.message : String(err)
   682→    logger.error("MT5_CONNECTION", `Failed to seed MT5 error codes: ${msg}`, {
   683→      details: err instanceof Error ? err.stack : undefined,
   684→    })
   685→    // Don't throw - allow system to continue operating
   686→  }
   687→}

   718→// Status change listeners (for WebSocket broadcast)
   719→type StatusListener = (status: Mt5Status, metrics: ConnectionMetrics) => void
   720→const listeners: StatusListener[] = []
   721→
   722→export function onStatusChange(fn: StatusListener) {
   723→  listeners.push(fn)
   724→  return () => {
   725→    const idx = listeners.indexOf(fn)
   726→    if (idx >= 0) listeners.splice(idx, 1)
   727→  }
   728→}

   731→  for (const fn of listeners) {
   732→    try {
   733→      fn(status, metrics)
   734→    } catch {
   735→      /* ignore listener errors */
   736→    }
   737→  }
   738→}

   816→  /**
   817→   * Validate MT5 API return data for silent failures.
   818→   * Checks for: empty arrays, null values, zero ticks outside trading hours.
   819→   */
   820→  validateReturn<T>(
   821→    data: T,
   822→    context: string,
   823→    options?: {
   824→      expectArray?: boolean
   825→      expectNonEmpty?: boolean
   826→      expectNonNull?: boolean
   827→      symbol?: string
   828→    }
   829→  ): { valid: boolean; issue?: string; data: T } {
   830→    const { expectArray, expectNonEmpty, expectNonNull, symbol } = options || {}
   831→
   832→    // Null/undefined check
   833→    if (data == null) {
   834→      const msg = `MT5 API silent failure: ${context} returned ${data === null ? "null" : "undefined"}`
   835→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context, value: String(data) } })
   836→      return { valid: false, issue: msg, data }
   837→    }
   838→
   839→    // Array checks
   840→    if (expectArray && !Array.isArray(data)) {
   841→      const msg = `MT5 API silent failure: ${context} expected array but got ${typeof data}`
   842→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context, type: typeof data } })
   843→      return { valid: false, issue: msg, data }
   844→    }
   845→
   846→    if (expectArray && expectNonEmpty && Array.isArray(data) && data.length === 0) {
   847→      // Empty array is OK if market is closed
   848→      if (!this.metrics.marketIsOpen) {
   849→        logger.debug("MT5_CONNECTION", `${context} returned empty array but market is closed (expected)`, { symbol })
   850→        return { valid: true, data }
   851→      }
   852→      // During trading hours, empty array = silent failure
   853→      const msg = `MT5 API silent failure: ${context} returned empty array during market hours`
   854→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context, tradingPhase: this.metrics.currentTradingPhase } })
   855→      return { valid: false, issue: msg, data }
   856→    }
   857→
   858→    // Zero-tick check for tick data
   859→    if (expectNonNull && (data as unknown) === 0) {
   860→      if (!this.metrics.marketIsOpen) {
   861→        return { valid: true, data }
   862→      }
   863→      const msg = `MT5 API silent failure: ${context} returned zero value during market hours`
   864→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context } })
   865→      return { valid: false, issue: msg, data }
   866→    }
   867→
   868→    return { valid: true, data }
   869→  }

   871→  async connect(login: number, password: string, server?: string): Promise<{ success: boolean; error?: string }> {
   872→    if (this.status === "CONNECTED" || this.status === "DEGRADED") {
   873→      return { success: true }
   874→    }
   875→

   877→      ...DEFAULT_CONFIG,
   878→      login,
   879→      password,
   880→      server: server || DEFAULT_CONFIG.server,

   884→    await seedMt5ErrorCodes()

   887→      metadata: { login: this.config.login, server: this.config.server },
   888→    })
   889→
   890→    try {
   891→      const result = await this.mutex.runExclusive(() => this.attemptConnection())
   892→      if (result.success) {
   893→        await this.onConnected()

   895→      } else {
   896→        await this.onConnectionFailed(result.error || "Unknown error")
   897→        return { success: false, error: result.error }
   898→      }
   899→    } catch (err) {
   900→      const msg = err instanceof Error ? err.message : String(err)
   901→      await this.onConnectionFailed(msg)
   902→      return { success: false, error: msg }
   903→    }

   956→  private async attemptConnection(): Promise<{ success: boolean; error?: string }> {
   957→    if (!this.config) {
   958→      return { success: false, error: "No configuration provided" }
   959→    }
   960→
   961→    // Simulate connection latency
   962→    const startMs = Date.now()
   963→    const latency = Math.floor(20 + Math.random() * 80)
   964→    await new Promise((r) => setTimeout(r, latency))
   965→    this.metrics.latencyMs = latency
   966→
   967→    // Validate credentials format
   968→    if (!this.config.login || this.config.login <= 0) {
   969→      return { success: false, error: "Invalid login credentials" }
   970→    }
   971→
   972→    if (!this.config.password || this.config.password.length < 1) {
   973→      return { success: false, error: "Password is required" }
   974→    }
   975→
   976→    return { success: true }

   979→  private async onConnected(): Promise<void> {
   980→    this.reconnectAttempt = 0
   981→    this.metrics.connectedAt = new Date()
   982→    this.metrics.lastError = null
   983→    this.metrics.consecutiveHeartbeatFailures = 0
   984→    this.metrics.reconnectCount =
   985→      this.status === "RECONNECTING" ? this.metrics.reconnectCount : 0
   986→    this.status = "CONNECTED"
   987→
   988→    // Start heartbeat
   989→    this.heartbeatTimer = setInterval(
   990→      () => this.heartbeat(),
   991→      this.config?.heartbeatIntervalMs || 5000
   992→    )
   993→
   994→    // Start uptime counter
   995→    this.uptimeTimer = setInterval(() => {
   996→      if (this.metrics.connectedAt) {
   997→        this.metrics.uptimeSeconds = Math.floor(
   998→          (Date.now() - this.metrics.connectedAt.getTime()) / 1000
   999→        )
  1000→      }
  1001→    }, 1000)
  1002→
  1003→    // Start trading phase monitor
  1004→    this.tradingPhaseTimer = setInterval(
  1005→      () => this.checkTradingPhase(),
  1006→      this.config?.tradingPhaseCheckIntervalMs || 30000
  1007→    )
  1008→
  1009→    // Initialize trading phase
  1010→    this.updateTradingPhase()
  1011→
  1012→    await this.persistState()
  1013→    await this.logConnectionEvent("CONNECTED", "Successfully connected to MT5 server")
  1014→
  1015→    logger.info("MT5_CONNECTION", `Connected to ${this.config?.server} (latency: ${this.metrics.latencyMs}ms)`, {
  1016→      metadata: {
  1017→        latencyMs: this.metrics.latencyMs,
  1018→        accountLogin: this.config?.login,
  1019→      },
  1020→    })
  1021→
  1022→    emitStatusChange(this.status, this.metrics)
  1023→  }

  1025→  private async onConnectionFailed(error: string): Promise<void> {
  1026→    // Detect auth failure
  1027→    if (error.includes("auth") || error.includes("credentials") || error.includes("login")) {
  1028→      this.status = "AUTH_FAILED"
  1029→      this.metrics.lastError = error
  1030→      await this.setStatus("AUTH_FAILED")
  1031→      await this.logConnectionEvent("AUTH_FAILED", `Authentication failed: ${error}`)
  1032→      logger.critical("MT5_CONNECTION", `Authentication failed: ${error}`)
  1033→      emitStatusChange(this.status, this.metrics)
  1034→      return
  1035→    }
  1036→
  1037→    this.status = "ERROR"
  1038→    this.metrics.lastError = error
  1039→    await this.setStatus("ERROR")
  1040→    await this.logConnectionEvent("ERROR", `Connection error: ${error}`)
  1041→    logger.error("MT5_CONNECTION", `Connection failed: ${error}`)
  1042→    emitStatusChange(this.status, this.metrics)
  1043→
  1044→    // Auto-reconnect
  1045→    if (!this.isShuttingDown) {
  1046→      this.scheduleReconnect()

  1050→  // ---- DEGRADED State Management ----
  1051→
  1052→  /**
  1053→   * Check if we should enter or exit DEGRADED state.
  1054→   * DEGRADED triggers when:
  1055→   *   - Latency exceeds threshold (default 200ms)
  1056→   *   - Consecutive heartbeat failures >= threshold (default 2)
  1057→   *
  1058→   * PROACTIVE_MC_70: When in DEGRADED state, log a risk event suggesting
  1059→   * margin call awareness if margin usage is approaching 70%.
  1060→   * PROACTIVE_MC_60: Similar at 60% - more urgent.
  1061→   */
  1062→  private async evaluateDegradedState(latencyMs: number, heartbeatOk: boolean): Promise<void> {
  1063→    const threshold = this.config?.degradedLatencyThresholdMs || 200
  1064→    const hbThreshold = this.config?.degradedHeartbeatFailureThreshold || 2
  1065→
  1066→    const wasDegraded = this.status === "DEGRADED"
  1067→    const wasConnected = this.status === "CONNECTED"
  1068→
  1069→    if (heartbeatOk) {
  1070→      this.metrics.consecutiveHeartbeatFailures = 0
  1071→    } else {
  1072→      this.metrics.consecutiveHeartbeatFailures++
  1073→    }
  1074→
  1075→    const shouldDegrade =
  1076→      latencyMs > threshold || this.metrics.consecutiveHeartbeatFailures >= hbThreshold
  1077→
  1078→    const shouldRecover =
  1079→      !shouldDegrade &&
  1080→      latencyMs <= threshold * 0.8 &&
  1081→      this.metrics.consecutiveHeartbeatFailures === 0
  1082→
  1083→    if (shouldDegrade && (wasConnected || wasDegraded) && this.status !== "DEGRADED") {
  1084→      // Transition to DEGRADED
  1085→      const reason = latencyMs > threshold
  1086→        ? `High latency: ${latencyMs}ms > ${threshold}ms threshold`
  1087→        : `${this.metrics.consecutiveHeartbeatFailures} consecutive heartbeat failures >= ${hbThreshold}`
  1088→
  1089→      this.status = "DEGRADED"
  1090→      await this.setStatus("DEGRADED")
  1091→      await this.logConnectionEvent("DEGRADED", reason)
  1092→      logger.warn("MT5_CONNECTION", `Connection DEGRADED: ${reason}`, {
  1093→        metadata: {
  1094→          latencyMs,
  1095→          consecutiveFailures: this.metrics.consecutiveHeartbeatFailures,
  1096→          threshold,
  1097→        },
  1098→      })
  1099→
  1100→      // PROACTIVE_MC_70 concept: log risk awareness when in DEGRADED state
  1101→      await this.logProactiveMarginCall("PROACTIVE_MC_70", 70)
  1102→
  1103→      emitStatusChange(this.status, this.metrics)
  1104→    } else if (shouldRecover && this.status === "DEGRADED") {
  1105→      // Recover from DEGRADED back to CONNECTED
  1106→      this.status = "CONNECTED"
  1107→      await this.setStatus("CONNECTED")
  1108→      await this.logConnectionEvent("RECOVERED", `Recovered from DEGRADED (latency: ${latencyMs}ms)`)
  1109→      logger.info("MT5_CONNECTION", `Connection recovered from DEGRADED (latency: ${latencyMs}ms)`, {
  1110→        metadata: { latencyMs },
  1111→      })
  1112→      emitStatusChange(this.status, this.metrics)
  1113→    } else if (shouldDegrade && this.status === "DEGRADED") {
  1114→      // Still in DEGRADED state - check for PROACTIVE_MC_60 if failures increasing
  1115→      if (this.metrics.consecutiveHeartbeatFailures >= hbThreshold + 1) {
  1116→        await this.logProactiveMarginCall("PROACTIVE_MC_60", 60)
  1117→      }
  1118→    }
  1119→  }
  1120→
  1121→  /**
  1122→   * Log a proactive margin call awareness event.
  1123→   * These events warn that if the system is already DEGRADED,
  1124→   * margin-related operations carry additional risk.
  1125→   */
  1126→  private async logProactiveMarginCall(eventType: string, marginPct: number): Promise<void> {
  1127→    try {
  1128→      await db.riskEvent.create({
  1129→        data: {
  1130→          eventType,
  1131→          severity: marginPct <= 60 ? "HIGH" : "MEDIUM",
  1132→          message: `Connection DEGRADED - Proactive margin call awareness at ${marginPct}% threshold. Reduce position sizes and avoid new trades until connection recovers.`,
  1133→          details: JSON.stringify({
  1134→            latencyMs: this.metrics.latencyMs,
  1135→            consecutiveFailures: this.metrics.consecutiveHeartbeatFailures,
  1136→            marginThreshold: marginPct,
  1137→            recommendation: marginPct <= 60
  1138→              ? "CRITICAL: Avoid all new trades, consider reducing open positions"
  1139→              : "WARNING: Reduce new trade sizes, monitor existing positions closely",
  1140→          }),
  1141→          actionTaken: "NOTIFICATION_SENT",
  1142→        },
  1143→      })
  1144→    } catch {
  1145→      // Best effort
  1146→    }
  1147→  }

  1156→  private async checkTradingPhase(): Promise<void> {
  1157→    const prevPhase = this.metrics.currentTradingPhase
  1158→    const prevOpen = this.metrics.marketIsOpen
  1159→
  1160→    this.updateTradingPhase()
  1161→
  1162→    const newPhase = this.metrics.currentTradingPhase
  1163→    const newOpen = this.metrics.marketIsOpen
  1164→
  1165→    // Log phase transitions
  1166→    if (prevPhase !== newPhase) {
  1167→      logger.info("MT5_CONNECTION", `Trading phase changed: ${prevPhase} -> ${newPhase}`, {
  1168→        metadata: { previousPhase: prevPhase, newPhase, marketOpen: newOpen },
  1169→      })
  1170→    }
  1171→
  1172→    // Log MARKET_OPEN event
  1173→    if (!prevOpen && newOpen) {
  1174→      await this.logConnectionEvent("MARKET_OPEN", `Market opened (phase: ${newPhase})`)
  1175→      logger.info("MT5_CONNECTION", `MARKET_OPEN - IDX market is now open (phase: ${newPhase})`)
  1176→    }
  1177→
  1178→    // Log MARKET_CLOSE event
  1179→    if (prevOpen && !newOpen) {
  1180→      await this.logConnectionEvent("MARKET_CLOSE", `Market closed (phase: ${newPhase})`)
  1181→      logger.info("MT5_CONNECTION", `MARKET_CLOSE - IDX market is now closed (phase: ${newPhase})`)
  1182→    }
  1183→
  1184→    // Always persist the trading phase and market open state
  1185→    try {
  1186→      const existing = await db.mt5ConnectionState.findFirst()
  1187→      if (existing) {
  1188→        await db.mt5ConnectionState.update({
  1189→          where: { id: existing.id },
  1190→          data: {
  1191→            isMarketOpen: newOpen,
  1192→            tradingPhase: newPhase,
  1193→          },
  1194→        })
  1195→      }
  1196→    } catch {
  1197→      // Best effort
  1198→    }
  1199→  }

  1201→  // ---- Reconnection ----
  1202→
  1203→  private scheduleReconnect(): void {
  1204→    if (this.isShuttingDown) return
  1205→    if (!this.config) return
  1206→
  1207→    this.reconnectAttempt++
  1208→    if (this.reconnectAttempt > this.config.maxReconnectAttempts) {
  1209→      logger.fatal(
  1210→        "MT5_CONNECTION",
  1211→        `Max reconnect attempts (${this.config.maxReconnectAttempts}) reached. Giving up.`
  1212→      )
  1213→      this.status = "ERROR"
  1214→      emitStatusChange(this.status, this.metrics)
  1215→      return
  1216→    }
  1217→
  1218→    // Exponential backoff
  1219→    const delay = Math.min(
  1220→      this.config.baseReconnectDelayMs * Math.pow(2, this.reconnectAttempt - 1),
  1221→      this.config.maxReconnectDelayMs
  1222→    )
  1223→
  1224→    const prevStatus = this.status
  1225→    this.status = "RECONNECTING"
  1226→    emitStatusChange(this.status, this.metrics)
  1227→
  1228→    logger.warn(
  1229→      "MT5_CONNECTION",
  1230→      `Reconnecting in ${delay}ms (attempt ${this.reconnectAttempt}/${this.config.maxReconnectAttempts})`,
  1231→      {
  1232→        metadata: { attempt: this.reconnectAttempt, delayMs: delay, previousStatus: prevStatus },
  1233→      }
  1234→    )
  1235→
  1236→    this.reconnectTimer = setTimeout(async () => {
  1237→      await this.logConnectionEvent("RECONNECTING", `Reconnect attempt ${this.reconnectAttempt}`)
  1238→      try {
  1239→        const result = await this.mutex.runExclusive(() => this.attemptConnection())
  1240→        if (result.success) {
  1241→          await this.onConnected()
  1242→        } else {
  1243→          this.metrics.reconnectCount++
  1244→          await this.onConnectionFailed(result.error || "Reconnection failed")
  1245→        }
  1246→      } catch (err) {
  1247→        this.metrics.reconnectCount++
  1248→        const msg = err instanceof Error ? err.message : String(err)
  1249→        await this.onConnectionFailed(msg)
  1250→      }
  1251→    }, delay)
  1252→  }
  1253→
  1254→  // ---- Heartbeat ----
  1255→
  1256→  private async heartbeat(): Promise<void> {
  1257→    if (this.status !== "CONNECTED" && this.status !== "DEGRADED") return
  1258→
  1259→    try {
  1260→      const start = Date.now()
  1261→      await new Promise((r) => setTimeout(r, 5 + Math.random() * 15))
  1262→      const latency = Date.now() - start
  1263→      this.metrics.latencyMs = latency
  1264→      this.metrics.lastHeartbeat = new Date()
  1265→
  1266→      await this.logConnectionEvent("HEARTBEAT_OK", `Heartbeat OK (${latency}ms)`)
  1267→
  1268→      // Evaluate DEGRADED state
  1269→      await this.evaluateDegradedState(latency, true)
  1270→    } catch (err) {
  1271→      const msg = err instanceof Error ? err.message : String(err)
  1272→      await this.logConnectionEvent("HEARTBEAT_FAIL", `Heartbeat failed: ${msg}`)
  1273→      logger.warn("MT5_CONNECTION", `Heartbeat failed: ${msg}`)
  1274→
  1275→      // Evaluate DEGRADED state with failure
  1276→      await this.evaluateDegradedState(this.metrics.latencyMs, false)
  1277→
  1278→      // If fully disconnected (not just degraded), go to DISCONNECTED
  1279→      if (this.metrics.consecutiveHeartbeatFailures > (this.config?.degradedHeartbeatFailureThreshold || 2) + 2) {
  1280→        this.status = "DISCONNECTED"
  1281→        await this.setStatus("DISCONNECTED")
  1282→        emitStatusChange(this.status, this.metrics)
  1283→        this.scheduleReconnect()
  1284→      }
  1285→    }
  1286→  }
