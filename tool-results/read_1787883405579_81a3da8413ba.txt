    65→// ============================================
    66→// ASYNC MUTEX
    67→// ============================================
    68→
    69→/**
    70→ * Simple async mutex to prevent concurrent MT5 API calls.
    71→ * The MT5 Python module is not thread-safe, so all calls must be serialized.
    72→ */
    73→export class AsyncMutex {
    74→  private _queue: Array<() => void> = []
    75→  private _locked = false
    76→
    77→  /** Acquire the mutex lock. Returns a release function. */
    78→  async acquire(): Promise<() => void> {
    79→    return new Promise<() => void>((resolve) => {
    80→      const tryAcquire = () => {
    81→        if (!this._locked) {
    82→          this._locked = true
    83→          resolve(this._release)
    84→        } else {
    85→          this._queue.push(tryAcquire)
    86→        }
    87→      }
    88→      tryAcquire()
    89→    })
    90→  }
    91→
    92→  /** Run a function exclusively within the mutex lock. */
    93→  async runExclusive<T>(fn: () => Promise<T>): Promise<T> {
    94→    const release = await this.acquire()
    95→    try {
    96→      return await fn()
    97→    } finally {
    98→      release()
    99→    }
   100→  }
   101→
   102→  private _release = (): void => {
   103→    this._locked = false
   104→    const next = this._queue.shift()
   105→    if (next) next()
   106→  }
   107→
   108→  get locked(): boolean {
   109→    return this._locked
   110→  }
   111→
   112→  get queueLength(): number {
   113→    return this._queue.length
   114→  }
   115→}

   572→/** Convert WIB (UTC+7) hours/minutes to UTC hours/minutes */
   573→function wibToUtc(wibHour: number, wibMinute: number = 0): { hour: number; minute: number } {
   574→  let utcHour = wibHour - 7
   575→  let utcMinute = wibMinute
   576→  if (utcHour < 0) utcHour += 24
   577→  return { hour: utcHour, minute: utcMinute }
   578→}
   579→
   580→// Pre-computed UTC boundaries for IDX trading
   581→const PHASE_BOUNDARIES_UTC = {
   582→  preOpenStart: wibToUtc(8, 45),     // 01:45 UTC
   583→  morningOpen: wibToUtc(9, 0),        // 02:00 UTC
   584→  preCloseStart: wibToUtc(11, 30),     // 04:30 UTC
   585→  preCloseEnd: wibToUtc(11, 30, 30),   // 04:30:30 UTC (treated as 04:30 since we only have minutes)
   586→  afternoonOpen: wibToUtc(13, 30),     // 06:30 UTC
   587→  marketClose: wibToUtc(15, 0),        // 08:00 UTC
   588→} as const
   589→
   590→/**
   591→ * Determine the current IDX trading phase based on UTC time.
   592→ * IDX Schedule (WIB / UTC):
   593→ *   PRE_OPEN:     08:45-09:00 WIB  = 01:45-02:00 UTC
   594→ *   OPEN:         09:00-11:30 WIB  = 02:00-04:30 UTC
   595→ *   PRE_CLOSE:    11:30-11:30:30    = ~04:30 UTC (30s window)
   596→ *   CLOSED:       11:30-13:30 WIB  = 04:30-06:30 UTC (lunch break)
   597→ *   OPEN:         13:30-15:00 WIB  = 06:30-08:00 UTC
   598→ *   AFTER_HOURS:  15:00+ WIB       = 08:00+ UTC
   599→ */
   600→export function getTradingPhase(now?: Date): TradingPhase {
   601→  const d = now || new Date()
   602→  // Get UTC hours and minutes as decimal for comparison
   603→  const utcDecimal = d.getUTCHours() + d.getUTCMinutes() / 60 + d.getUTCSeconds() / 3600
   604→
   605→  const { preOpenStart, morningOpen, preCloseStart, preCloseEnd, afternoonOpen, marketClose } = PHASE_BOUNDARIES_UTC
   606→
   607→  // Convert boundaries to decimal hours for comparison
   608→  const preOpenDec = preOpenStart.hour + preOpenStart.minute / 60
   609→  const morningDec = morningOpen.hour + morningOpen.minute / 60
   610→  const preCloseDec = preCloseStart.hour + preCloseStart.minute / 60
   611→  const preCloseEndDec = preCloseStart.hour + preCloseStart.minute / 60 + 30 / 3600 // 30 seconds
   612→  const afternoonDec = afternoonOpen.hour + afternoonOpen.minute / 60
   613→  const closeDec = marketClose.hour + marketClose.minute / 60
   614→
   615→  if (utcDecimal >= preOpenDec && utcDecimal < morningDec) {
   616→    return "PRE_OPEN"
   617→  }
   618→  if (utcDecimal >= morningDec && utcDecimal < preCloseDec) {
   619→    return "OPEN"
   620→  }
   621→  // PRE_CLOSE is a 30-second window; treat it as part of PRE_CLOSE
   622→  if (utcDecimal >= preCloseDec && utcDecimal < preCloseEndDec) {
   623→    return "PRE_CLOSE"
   624→  }
   625→  // Lunch break CLOSED
   626→  if (utcDecimal >= preCloseDec && utcDecimal < afternoonDec) {
   627→    return "CLOSED"
   628→  }
   629→  // Afternoon session OPEN
   630→  if (utcDecimal >= afternoonDec && utcDecimal < closeDec) {
   631→    return "OPEN"
   632→  }
   633→  // After hours
   634→  return "AFTER_HOURS"
   635→}

   637→/** Check if the IDX market is currently open for trading. */
   638→export function isMarketOpen(now?: Date): boolean {
   639→  const phase = getTradingPhase(now)
   640→  return phase === "OPEN" || phase === "PRE_OPEN"
   641→}

   819→  /**
   820→   * Validate MT5 API return data for silent failures.
   821→   * Checks for: empty arrays, null values, zero ticks outside trading hours.
   822→   */
   823→  validateReturn<T>(
   824→    data: T,
   825→    context: string,
   826→    options?: {
   827→      expectArray?: boolean
   828→      expectNonEmpty?: boolean
   829→      expectNonNull?: boolean
   830→      symbol?: string
   831→    }
   832→  ): { valid: boolean; issue?: string; data: T } {
   833→    const { expectArray, expectNonEmpty, expectNonNull, symbol } = options || {}
   834→
   835→    // Null/undefined check
   836→    if (data == null) {
   837→      const msg = `MT5 API silent failure: ${context} returned ${data === null ? "null" : "undefined"}`
   838→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context, value: String(data) } })
   839→      return { valid: false, issue: msg, data }
   840→    }
   841→
   842→    // Array checks
   843→    if (expectArray && !Array.isArray(data)) {
   844→      const msg = `MT5 API silent failure: ${context} expected array but got ${typeof data}`
   845→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context, type: typeof data } })
   846→      return { valid: false, issue: msg, data }
   847→    }
   848→
   849→    if (expectArray && expectNonEmpty && Array.isArray(data) && data.length === 0) {
   850→      // Empty array is OK if market is closed
   851→      if (!this.metrics.marketIsOpen) {
   852→        logger.debug("MT5_CONNECTION", `${context} returned empty array but market is closed (expected)`, { symbol })
   853→        return { valid: true, data }
   854→      }
   855→      // During trading hours, empty array = silent failure
   856→      const msg = `MT5 API silent failure: ${context} returned empty array during market hours`
   857→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context, tradingPhase: this.metrics.currentTradingPhase } })
   858→      return { valid: false, issue: msg, data }
   859→    }
   860→
   861→    // Zero-tick check for tick data
   862→    if (expectNonNull && (data as unknown) === 0) {
   863→      if (!this.metrics.marketIsOpen) {
   864→        return { valid: true, data }
   865→      }
   866→      const msg = `MT5 API silent failure: ${context} returned zero value during market hours`
   867→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context } })
   868→      return { valid: false, issue: msg, data }
   869→    }
   870→
   871→    return { valid: true, data }
   872→  }

  1053→  // ---- DEGRADED State Management ----
  1054→
  1055→  /**
  1056→   * Check if we should enter or exit DEGRADED state.
  1057→   * DEGRADED triggers when:
  1058→   *   - Latency exceeds threshold (default 200ms)
  1059→   *   - Consecutive heartbeat failures >= threshold (default 2)
  1060→   *
  1061→   * PROACTIVE_MC_70: When in DEGRADED state, log a risk event suggesting
  1062→   * margin call awareness if margin usage is approaching 70%.
  1063→   * PROACTIVE_MC_60: Similar at 60% - more urgent.
  1064→   */
  1065→  private async evaluateDegradedState(latencyMs: number, heartbeatOk: boolean): Promise<void> {
  1066→    const threshold = this.config?.degradedLatencyThresholdMs || 200
  1067→    const hbThreshold = this.config?.degradedHeartbeatFailureThreshold || 2
  1068→
  1069→    const wasDegraded = this.status === "DEGRADED"
  1070→    const wasConnected = this.status === "CONNECTED"
  1071→
  1072→    if (heartbeatOk) {
  1073→      this.metrics.consecutiveHeartbeatFailures = 0
  1074→    } else {
  1075→      this.metrics.consecutiveHeartbeatFailures++
  1076→    }
  1077→
  1078→    const shouldDegrade =
  1079→      latencyMs > threshold || this.metrics.consecutiveHeartbeatFailures >= hbThreshold
  1080→
  1081→    const shouldRecover =
  1082→      !shouldDegrade &&
  1083→      latencyMs <= threshold * 0.8 &&
  1084→      this.metrics.consecutiveHeartbeatFailures === 0
  1085→
  1086→    if (shouldDegrade && (wasConnected || wasDegraded) && this.status !== "DEGRADED") {
  1087→      // Transition to DEGRADED
  1088→      const reason = latencyMs > threshold
  1089→        ? `High latency: ${latencyMs}ms > ${threshold}ms threshold`
  1090→        : `${this.metrics.consecutiveHeartbeatFailures} consecutive heartbeat failures >= ${hbThreshold}`
  1091→
  1092→      this.status = "DEGRADED"
  1093→      await this.setStatus("DEGRADED")
  1094→      await this.logConnectionEvent("DEGRADED", reason)
  1095→      logger.warn("MT5_CONNECTION", `Connection DEGRADED: ${reason}`, {
  1096→        metadata: {
  1097→          latencyMs,
  1098→          consecutiveFailures: this.metrics.consecutiveHeartbeatFailures,
  1099→          threshold,
  1100→        },
  1101→      })
  1102→
  1103→      // PROACTIVE_MC_70 concept: log risk awareness when in DEGRADED state
  1104→      await this.logProactiveMarginCall("PROACTIVE_MC_70", 70)
  1105→
  1106→      emitStatusChange(this.status, this.metrics)
  1107→    } else if (shouldRecover && this.status === "DEGRADED") {
  1108→      // Recover from DEGRADED back to CONNECTED
  1109→      this.status = "CONNECTED"
  1110→      await this.setStatus("CONNECTED")
  1111→      await this.logConnectionEvent("RECOVERED", `Recovered from DEGRADED (latency: ${latencyMs}ms)`)
  1112→      logger.info("MT5_CONNECTION", `Connection recovered from DEGRADED (latency: ${latencyMs}ms)`, {
  1113→        metadata: { latencyMs },
  1114→      })
  1115→      emitStatusChange(this.status, this.metrics)
  1116→    } else if (shouldDegrade && this.status === "DEGRADED") {
  1117→      // Still in DEGRADED state - check for PROACTIVE_MC_60 if failures increasing
  1118→      if (this.metrics.consecutiveHeartbeatFailures >= hbThreshold + 1) {
  1119→        await this.logProactiveMarginCall("PROACTIVE_MC_60", 60)
  1120→      }
  1121→    }
  1122→  }

  1209→
  1210→    this.reconnectAttempt++
  1211→    if (this.reconnectAttempt > this.config.maxReconnectAttempts) {
  1212→      logger.fatal(
  1213→        "MT5_CONNECTION",
  1214→        `Max reconnect attempts (${this.config.maxReconnectAttempts}) reached. Giving up.`
  1215→      )
  1216→      this.status = "ERROR"
  1217→      emitStatusChange(this.status, this.metrics)
  1218→      return
  1219→    }
  1220→
  1221→    // Exponential backoff
  1222→    const delay = Math.min(
  1223→      this.config.baseReconnectDelayMs * Math.pow(2, this.reconnectAttempt - 1),
  1224→      this.config.maxReconnectDelayMs
  1225→    )
  1226→
  1227→    const prevStatus = this.status
  1228→    this.status = "RECONNECTING"
  1229→    emitStatusChange(this.status, this.metrics)
  1230→
  1231→    logger.warn(
  1232→      "MT5_CONNECTION",
  1233→      `Reconnecting in ${delay}ms (attempt ${this.reconnectAttempt}/${this.config.maxReconnectAttempts})`,
  1234→      {
  1235→        metadata: { attempt: this.reconnectAttempt, delayMs: delay, previousStatus: prevStatus },
  1236→      }
  1237→    )
  1238→
  1239→    this.reconnectTimer = setTimeout(async () => {
  1240→      await this.logConnectionEvent("RECONNECTING", `Reconnect attempt ${this.reconnectAttempt}`)
  1241→      try {
  1242→        const result = await this.mutex.runExclusive(() => this.attemptConnection())
  1243→        if (result.success) {
  1244→          await this.onConnected()
  1245→        } else {
  1246→          this.metrics.reconnectCount++
  1247→          await this.onConnectionFailed(result.error || "Reconnection failed")
  1248→        }
  1249→      } catch (err) {
  1250→        this.metrics.reconnectCount++
  1251→        const msg = err instanceof Error ? err.message : String(err)
  1252→        await this.onConnectionFailed(msg)
  1253→      }
  1254→    }, delay)
  1255→  }

  1257→  // ---- Heartbeat ----
  1258→
  1259→  private async heartbeat(): Promise<void> {
  1260→    if (this.status !== "CONNECTED" && this.status !== "DEGRADED") return
  1261→
  1262→    try {
  1263→      const start = Date.now()
  1264→      await new Promise((r) => setTimeout(r, 5 + Math.random() * 15))
  1265→      const latency = Date.now() - start
  1266→      this.metrics.latencyMs = latency
  1267→      this.metrics.lastHeartbeat = new Date()
  1268→
  1269→      await this.logConnectionEvent("HEARTBEAT_OK", `Heartbeat OK (${latency}ms)`)
  1270→
  1271→      // Deep Audit Fix #3: Throttle DB persistence on heartbeat
  1272→      // Only persist state every 30s instead of every 5s heartbeat
  1273→      this._heartbeatCountSinceLastPersist++
  1274→      if (this._heartbeatCountSinceLastPersist >= 6) {
  1275→        this._heartbeatCountSinceLastPersist = 0
  1276→        await this.persistState()
  1277→      }
  1278→
  1279→      // Evaluate DEGRADED state
  1280→      await this.evaluateDegradedState(latency, true)
  1281→    } catch (err) {
  1282→      const msg = err instanceof Error ? err.message : String(err)
  1283→      await this.logConnectionEvent("HEARTBEAT_FAIL", `Heartbeat failed: ${msg}`)
  1284→      logger.warn("MT5_CONNECTION", `Heartbeat failed: ${msg}`)
  1285→
  1286→      // Evaluate DEGRADED state with failure
  1287→      await this.evaluateDegradedState(this.metrics.latencyMs, false)
  1288→
  1289→      // If fully disconnected (not just degraded), go to DISCONNECTED
  1290→      if (this.metrics.consecutiveHeartbeatFailures > (this.config?.degradedHeartbeatFailureThreshold || 2) + 2) {
  1291→        this.status = "DISCONNECTED"
  1292→        await this.setStatus("DISCONNECTED")
  1293→        emitStatusChange(this.status, this.metrics)
  1294→        this.scheduleReconnect()
  1295→      }
  1296→    }
  1297→  }
