   242→   242→/**
   243→   243→ * Determine the proactive margin zone based on margin level percentage.
   244→   244→ * Zone 1: PROACTIVE at 70% - warning, log event, continue trading
   245→   245→ * Zone 2: PROACTIVE at 60% - strong warning, reduce new position sizes by 50%, log event
   246→   246→ * Zone 3: MARGIN CALL at 50% - critical, block new trades
   247→   247→ * Zone 4: STOP OUT at 20% - fatal, emergency close all positions
   248→   248→ */
   249→   249→export function determineProactiveMarginZone(
   250→   250→  marginLevelPercent: number,
   251→   251→  config: RiskConfigData,
   252→   252→): ProactiveMarginZone {
   253→   253→  // When no margin is used, we're in SAFE zone
   254→   254→  if (marginLevelPercent === 0) {
   255→   255→    return "SAFE"
   256→   256→  }
   257→   257→  if (marginLevelPercent <= config.stopOutLevel) {
   258→   258→    return "STOP_OUT"
   259→   259→  }
   260→   260→  if (marginLevelPercent <= config.marginCallLevel) {
   261→   261→    return "MARGIN_CALL"
   262→   262→  }
   263→   263→  if (config.proactiveMcLevel60 && marginLevelPercent <= 60) {
   264→   264→    return "PROACTIVE_60"
   265→   265→  }
   266→   266→  if (config.proactiveMcLevel70 && marginLevelPercent <= 70) {
   267→   267→    return "PROACTIVE_70"
   268→   268→  }
   269→   269→  return "SAFE"
   270→   270→}

   280→   280→  const zone = determineProactiveMarginZone(marginLevelPercent, config)

   282→   282→

   289→   289→        eventType: "PROACTIVE_MC_70",
   290→   290→        createdAt: { gte: fiveMinutesAgo },
   291→   291→      },
   292→   292→    })
   293→   293→    if (!recent) {
   294→   294→      await logRiskEvent({
   295→   295→        eventType: "PROACTIVE_MC_70",
   296→   296→        severity: "MEDIUM",

   299→   299→        actionTaken: "NOTIFICATION_SENT",

   302→   302→    }
   303→   303→    actionTaken = "NOTIFICATION_SENT"
   304→   304→  } else if (zone === "PROACTIVE_60") {

   306→   306→      where: {
   307→   307→        eventType: "PROACTIVE_MC_60",

   312→   312→      await logRiskEvent({
   313→   313→        eventType: "PROACTIVE_MC_60",

   321→   321→    actionTaken = "REDUCED_SIZE"
   322→   322→  } else if (zone === "MARGIN_CALL") {

   328→   328→    })

   363→   363→/**
   364→   364→ * Calculate dynamic risk scaling factor based on recent performance.
   365→   365→ * - If daily loss > 1% of limit, scale down by 50%
   366→   366→ * - If drawdown > 80% of max, scale down by 75%
   367→   367→ * - If winning streak > 3, can scale up by 25% (max 1.5x)
   368→   368→ */
   369→   369→export function calculateScalingFactor(params: {
   370→   370→  dailyPnl: number
   371→   371→  dailyLossLimit: number
   372→   372→  currentDrawdown: number
   373→   373→  maxDrawdown: number
   374→   374→  equity: number
   375→   375→}): number {
   376→   376→  let factor = 1.0
   377→   377→
   378→   378→  // Scale down if daily loss is more than 1% of the limit
   379→   379→  if (params.dailyPnl < 0 && params.dailyLossLimit > 0) {
   380→   380→    const dailyLossPct = Math.abs(params.dailyPnl) / params.dailyLossLimit
   381→   381→    if (dailyLossPct > 0.5) {
   382→   382→      factor = Math.min(factor, 0.5) // 50% reduction
   383→   383→    }
   384→   384→  }
   385→   385→
   386→   386→  // Scale down if drawdown > 80% of max
   387→   387→  if (params.maxDrawdown > 0 && params.currentDrawdown > params.maxDrawdown * 0.8) {
   388→   388→    factor = Math.min(factor, 0.25) // 75% reduction
   389→   389→  }
   390→   390→
   391→   391→  // Scale up if on a winning streak > 3 consecutive wins
   392→   392→  // This is a bonus that can only increase factor up to 1.5x
   393→   393→  // The winning streak check requires trade history, so we apply a
   394→   394→  // conservative boost. The actual streak check is in the caller.
   395→   395→  // Here we use a positive daily P&L as a proxy for the upward boost.
   396→   396→  if (params.dailyPnl > 0 && factor >= 1.0) {
   397→   397→    // Only allow scaling up if no drawdown pressure
   398→   398→    if (params.currentDrawdown < params.maxDrawdown * 0.3) {
   399→   399→      factor = Math.min(factor * 1.25, 1.5)
   400→   400→    }
   401→   401→  }
   402→   402→
   403→   403→  return Math.round(factor * 1000) / 1000
   404→   404→}

   406→   406→/**
   407→   407→ * Calculate sector exposure breakdown for all open positions.
   408→   408→ */
   409→   409→export function calculateSectorExposure(
   410→   410→  openTrades: PositionRiskBreakdown[],
   411→   411→  equity: number,
   412→   412→): SectorExposureEntry[] {
   413→   413→  const sectorMap = new Map<string, { margin: number; count: number }>()
   414→   414→
   415→   415→  for (const pos of openTrades) {
   416→   416→    const sector = SYMBOL_SECTORS[pos.symbol] || "Unknown"
   417→   417→    const existing = sectorMap.get(sector)
   418→   418→    if (existing) {
   419→   419→      existing.margin += pos.margin
   420→   420→      existing.count += 1
   421→   421→    } else {
   422→   422→      sectorMap.set(sector, { margin: pos.margin, count: 1 })
   423→   423→    }
   424→   424→  }
   425→   425→
   426→   426→  const entries: SectorExposureEntry[] = []
   427→   427→  for (const [sector, data] of sectorMap.entries()) {
   428→   428→    entries.push({
   429→   429→      sector,
   430→   430→      exposurePct: equity > 0 ? Math.round((data.margin / equity) * 10000) / 100 : 0,
   431→   431→      positionCount: data.count,
   432→   432→      marginUsed: Math.round(data.margin * 100) / 100,
   433→   433→    })
   434→   434→  }
   435→   435→
   436→   436→  // Sort by exposure descending
   437→   437→  entries.sort((a, b) => b.exposurePct - a.exposurePct)
   438→   438→  return entries
   439→   439→}

   441→   441→/**
   442→   442→ * Calculate a full risk snapshot of the current account state.
   443→   443→ */
   444→   444→export async function getRiskSnapshot(): Promise<RiskSnapshot> {
   445→   445→  const config = await getRiskConfig()
   446→   446→  const openTrades = await db.trade.findMany({ where: { status: "OPEN" } })
   447→   447→  const allClosed = await db.trade.findMany({ where: { status: "CLOSED" } })
   448→   448→
   449→   449→  const BASE_BALANCE = 10000
   450→   450→  const totalClosedPnl = allClosed.reduce((s, t) => s + t.pnl, 0)
   451→   451→  const currentBalance = Math.round((BASE_BALANCE + totalClosedPnl) * 100) / 100
   452→   452→  const totalOpenPnl = openTrades.reduce((s, t) => s + t.pnl, 0)
   453→   453→  const equity = Math.round((currentBalance + totalOpenPnl) * 100) / 100
   454→   454→  const totalMarginUsed = openTrades.reduce((s, t) => s + t.margin, 0)
   455→   455→  const freeMargin = Math.max(0, Math.round((equity - totalMarginUsed) * 100) / 100)
   456→   456→  const marginLevelPercent = totalMarginUsed > 0 ? Math.round((equity / totalMarginUsed) * 10000) / 100 : 0
   457→   457→
   458→   458→  // ---- Time-based P&L ----
   459→   459→  const now = new Date()
   460→   460→  const todayStr = now.toISOString().split("T")[0]
   461→   461→  const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000)
   462→   462→  const monthAgo = new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000)
   463→   463→
   464→   464→  const closedToday = allClosed.filter((t) => t.closeTime && t.closeTime >= new Date(todayStr))
   465→   465→  const closedWeek = allClosed.filter((t) => t.closeTime && t.closeTime >= weekAgo)
   466→   466→  const closedMonth = allClosed.filter((t) => t.closeTime && t.closeTime >= monthAgo)
   467→   467→
   468→   468→  const dailyPnl = closedToday.reduce((s, t) => s + t.pnl, 0) + totalOpenPnl
   469→   469→  const weeklyPnl = closedWeek.reduce((s, t) => s + t.pnl, 0) + totalOpenPnl
   470→   470→  const monthlyPnl = closedMonth.reduce((s, t) => s + t.pnl, 0) + totalOpenPnl
   471→   471→
   472→   472→  // ---- Drawdown calculation ----
   473→   473→  let peak = currentBalance
   474→   474→  let maxDrawdown = 0
   475→   475→  let runningBalance = currentBalance - totalClosedPnl
   476→   476→  for (const trade of allClosed) {
   477→   477→    runningBalance += trade.pnl
   478→   478→    if (runningBalance > peak) peak = runningBalance
   479→   479→    const dd = peak > 0 ? ((peak - runningBalance) / peak) * 100 : 0
   480→   480→    if (dd > maxDrawdown) maxDrawdown = dd
   481→   481→  }
   482→   482→  // Factor in open P&L for current drawdown
   483→   483→  if (equity < peak) {
   484→   484→    const currentDD = ((peak - equity) / peak) * 100
   485→   485→    if (currentDD > maxDrawdown) maxDrawdown = currentDD
   486→   486→  }
   487→   487→  const currentDrawdown = peak > 0 ? Math.round(((peak - equity) / peak) * 10000) / 100 : 0
   488→   488→
   489→   489→  // ---- Proactive margin monitoring (4 zones) ----
   490→   490→  const proactiveMarginZone = determineProactiveMarginZone(marginLevelPercent, config)
   491→   491→
   492→   492→  // Process zone transitions and log events
   493→   493→  if (proactiveMarginZone !== "SAFE") {
   494→   494→    await processProactiveMarginMonitoring(marginLevelPercent, config)
   495→   495→  }
   496→   496→
   497→   497→  const isMarginCallWarning =
   498→   498→    proactiveMarginZone === "MARGIN_CALL" ||
   499→   499→    proactiveMarginZone === "STOP_OUT"
   500→   500→  const isStopOutWarning = proactiveMarginZone === "STOP_OUT"
   501→   501→
   502→   502→  // ---- Risk Score (deterministic, not random) ----
   503→   503→  const riskScore = calculateRiskScore({
   504→   504→    dailyPnlPercent: (dailyPnl / equity) * 100,
   505→   505→    marginUsagePercent: equity > 0 ? (totalMarginUsed / equity) * 100 : 0,
   506→   506→    currentDrawdown,
   507→   507→    maxDrawdown,
   508→   508→    config,
   509→   509→    openPositionCount: openTrades.length,
   510→   510→    marginLevelPercent,
   511→   511→  })
   512→   512→
   513→   513→  const riskLevel = getRiskLevel(riskScore)
   514→   514→
   515→   515→  // ---- Daily loss limit ----
   516→   516→  const dailyLossLimit = (config.maxDailyLoss / 100) * equity
   517→   517→  const isDailyLimitReached = dailyPnl < 0 && Math.abs(dailyPnl) >= dailyLossLimit
   518→   518→
   519→   519→  // ---- Trading allowed? ----
   520→   520→  let isTradingAllowed = true
   521→   521→  let tradingBlockReason: string | undefined
   522→   522→
   523→   523→  if (isStopOutWarning) {
   524→   524→    isTradingAllowed = false
   525→   525→    tradingBlockReason = `Stop out level reached (${config.stopOutLevel}%)`
   526→   526→  }
   527→   527→  if (proactiveMarginZone === "MARGIN_CALL") {
   528→   528→    isTradingAllowed = false
   529→   529→    tradingBlockReason = `Margin call level reached (${config.marginCallLevel}%)`
   530→   530→  }
   531→   531→  if (isDailyLimitReached) {
   532→   532→    isTradingAllowed = false
   533→   533→    tradingBlockReason = `Daily loss limit reached (${config.maxDailyLoss}% / $${dailyLossLimit.toFixed(2)})`
   534→   534→  }
   535→   535→  if (currentDrawdown >= config.maxDrawdown) {
   536→   536→    isTradingAllowed = false
   537→   537→    tradingBlockReason = `Max drawdown limit reached (${config.maxDrawdown}%)`
   538→   538→  }
   539→   539→  if (openTrades.length >= config.maxOpenPositions) {
   540→   540→    isTradingAllowed = false
   541→   541→    tradingBlockReason = `Max open positions reached (${config.maxOpenPositions})`
   542→   542→  }
   543→   543→
   544→   544→  // ---- Recent risk events ----
   545→   545→  const recentEvents = await db.riskEvent.findMany({
   546→   546→    orderBy: { createdAt: "desc" },
   547→   547→    take: 10,
   548→   548→  })
   549→   549→
   550→   550→  // ---- Recommendations ----
   551→   551→  const recommendations = generateRecommendations({
   552→   552→    riskScore,
   553→   553→    riskLevel,
   554→   554→    dailyPnl,
   555→   555→    dailyPnlPercent: (dailyPnl / equity) * 100,
   556→   556→    currentDrawdown,
   557→   557→    marginUsagePercent: equity > 0 ? (totalMarginUsed / equity) * 100 : 0,
   558→   558→    config,
   559→   559→    isDailyLimitReached,
   560→   560→    isMarginCallWarning,
   561→   561→    isStopOutWarning,
   562→   562→    isTradingAllowed,
   563→   563→    tradingBlockReason,
   564→   564→    proactiveMarginZone,
   565→   565→  })
   566→   566→
   567→   567→  // ---- Position risk breakdown ----
   568→   568→  const positions: PositionRiskBreakdown[] = openTrades.map((t) => {
   569→   569→    let riskAmount = 0
   570→   570→    if (t.sl) {
   571→   571→      riskAmount = t.direction === "BUY"
   572→   572→        ? (t.entryPrice - t.sl) * t.lotSize * 100000
   573→   573→        : (t.sl - t.entryPrice) * t.lotSize * 100000
   574→   574→    }
   575→   575→    return {
   576→   576→      tradeId: t.id,
   577→   577→      symbol: t.symbol,
   578→   578→      direction: t.direction,
   579→   579→      lotSize: t.lotSize,
   580→   580→      entryPrice: t.entryPrice,
   581→   581→      currentPrice: t.currentPrice,
   582→   582→      sl: t.sl,
   583→   583→      tp: t.tp,
   584→   584→      margin: t.margin,
   585→   585→      pnl: t.pnl,
   586→   586→      pnlPercent: t.pnlPercent,
   587→   587→      riskAmount: Math.round(Math.abs(riskAmount) * 100) / 100,
   588→   588→      riskPercent: equity > 0 ? Math.round((Math.abs(riskAmount) / equity) * 10000) / 100 : 0,
   589→   589→      strategy: t.strategy,
   590→   590→      trailingStop: t.trailingStop,
   591→   591→    }
   592→   592→  })
   593→   593→
   594→   594→  // ---- Sector exposure ----
   595→   595→  const sectorExposure = calculateSectorExposure(positions, equity)
   596→   596→
   597→   597→  // ---- Portfolio total risk % ----
   598→   598→  const portfolioTotalRisk = positions.reduce((s, p) => s + p.riskAmount, 0)
   599→   599→  const portfolioTotalRiskPct = equity > 0
   600→   600→    ? Math.round((portfolioTotalRisk / equity) * 10000) / 100
   601→   601→    : 0
   602→   602→
   603→   603→  // ---- Current effective leverage ----
   604→   604→  const totalNotionalValue = openTrades.reduce(
   605→   605→    (s, t) => s + t.entryPrice * t.lotSize * 100000, 0,
   606→   606→  )
   607→   607→  const leverageUsed = equity > 0
   608→   608→    ? Math.round((totalNotionalValue / equity) * 100) / 100
   609→   609→    : 0
   610→   610→
   611→   611→  // ---- Current reserve capital % ----
   612→   612→  const reserveCapitalPct = equity > 0
   613→   613→    ? Math.round((freeMargin / equity) * 10000) / 100
   614→   614→    : 0
   615→   615→
   616→   616→  // ---- Dynamic risk scaling factor ----
   617→   617→  // Check winning streak for upward scaling
   618→   618→  const recentClosed = allClosed
   619→   619→    .filter((t) => t.closeTime && t.closeTime >= new Date(todayStr))
   620→   620→    .sort((a, b) => (b.closeTime?.getTime() || 0) - (a.closeTime?.getTime() || 0))
   621→   621→  let winningStreak = 0
   622→   622→  for (const t of recentClosed) {
   623→   623→    if (t.pnl > 0) {
   624→   624→      winningStreak++
   625→   625→    } else {
   626→   626→      break
   627→   627→    }
   628→   628→  }
   629→   629→  let scalingFactor = calculateScalingFactor({
   630→   630→    dailyPnl,
   631→   631→    dailyLossLimit,
   632→   632→    currentDrawdown,
   633→   633→    maxDrawdown: config.maxDrawdown,
   634→   634→    equity,
   635→   635→  })
   636→   636→  // Apply winning streak boost
   637→   637→  if (winningStreak > 3 && scalingFactor >= 1.0 && currentDrawdown < config.maxDrawdown * 0.3) {
   638→   638→    scalingFactor = Math.min(scalingFactor * 1.25, 1.5)
   639→   639→    scalingFactor = Math.round(scalingFactor * 1000) / 1000
   640→   640→  }
   641→   641→
   642→   642→  return {
   643→   643→    equity: Math.round(equity * 100) / 100,
   644→   644→    balance: Math.round(currentBalance * 100) / 100,
   645→   645→    freeMargin,
   646→   646→    marginUsed: Math.round(totalMarginUsed * 100) / 100,
   647→   647→    marginLevelPercent,
   648→   648→    dailyPnl: Math.round(dailyPnl * 100) / 100,
   649→   649→    dailyPnlPercent: Math.round((dailyPnl / equity) * 10000) / 100,
   650→   650→    weeklyPnl: Math.round(weeklyPnl * 100) / 100,
   651→   651→    weeklyPnlPercent: Math.round((weeklyPnl / equity) * 10000) / 100,
   652→   652→    monthlyPnl: Math.round(monthlyPnl * 100) / 100,
   653→   653→    monthlyPnlPercent: Math.round((monthlyPnl / equity) * 10000) / 100,
   654→   654→    currentDrawdown: Math.round(currentDrawdown * 100) / 100,
   655→   655→    maxDrawdown: Math.round(maxDrawdown * 100) / 100,
   656→   656→    maxAllowedDrawdown: config.maxDrawdown,
   657→   657→    riskScore,
   658→   658→    riskLevel,
   659→   659→    openPositions: openTrades.length,
   660→   660→    maxPositionsAllowed: config.maxOpenPositions,
   661→   661→    marginUsagePercent: Math.round((equity > 0 ? (totalMarginUsed / equity) * 100 : 0) * 100) / 100,
   662→   662→    maxMarginAllowed: config.maxMarginUsage,
   663→   663→    dailyLossRemaining: Math.round(Math.max(0, dailyLossLimit - Math.abs(Math.min(0, dailyPnl))) * 100) / 100,
   664→   664→    dailyLossLimit: Math.round(dailyLossLimit * 100) / 100,
   665→   665→    isDailyLimitReached,
   666→   666→    isMarginCallWarning,
   667→   667→    isStopOutWarning,
   668→   668→    isTradingAllowed,
   669→   669→    tradingBlockReason,
   670→   670→    recentRiskEvents: recentEvents.map((e) => ({
   671→   671→      eventType: e.eventType,
   672→   672→      severity: e.severity,
   673→   673→      message: e.message,
   674→   674→      createdAt: e.createdAt.toISOString(),
   675→   675→      resolved: e.resolved,
   676→   676→    })),
   677→   677→    recommendations,
   678→   678→    positions,
   679→   679→    // Deep audit fields
   680→   680→    proactiveMarginZone,
   681→   681→    sectorExposure,
   682→   682→    portfolioTotalRiskPct,
   683→   683→    leverageUsed,
   684→   684→    reserveCapitalPct,
   685→   685→    scalingFactor,
   686→   686→  }
   687→   687→}
   688→   688→
   689→   689→/**
   690→   690→ * Pre-trade risk validation. Call this BEFORE creating any trade.
   691→   691→ *
   692→   692→ * Checks (in order):
   693→   693→ *  1. Trading allowed? (global block)
   694→   694→ *  2. Position limit
   695→   695→ *  3. Lot size limits (per trade, per symbol)
   696→   696→ *  4. Minimum lot check
   697→   697→ *  5. Calculate risk amount (including slippage)
   698→   698→ *  6. Risk per trade limit
   699→   699→ *  7. No stop loss warning
   700→   700→ *  8. Correlation risk (sector)
   701→   701→ *  9. Margin usage after trade
   702→   702→ * 10. Cooldown after loss
   703→   703→ * 11. Portfolio-level risk cap
   704→   704→ * 12. Leverage utilization cap
   705→   705→ * 13. Single stock concentration limit
   706→   706→ * 14. Sector concentration limit
   707→   707→ * 15. Reserve capital check
   708→   708→ */
   709→   709→export async function preTradeCheck(params: {
   710→   710→  symbol: string
   711→   711→  direction: string
   712→   712→  lotSize: number
   713→   713→  entryPrice: number
   714→   714→  sl?: number | null
   715→   715→  tp?: number | null
   716→   716→  strategy?: string | null
   717→   717→  aiConfidence?: number | null
   718→   718→  expectedSlippage?: number | null
   719→   719→}): Promise<PreTradeCheck> {
   720→   720→  const config = await getRiskConfig()
   721→   721→  const snapshot = await getRiskSnapshot()
   722→   722→  const warnings: string[] = []
   723→   723→  let positionSizeReduction = 0
   724→   724→
   725→   725→  // ---- 1. Trading allowed? ----
   726→   726→  if (!snapshot.isTradingAllowed) {
   727→   727→    return {
   728→   728→      approved: false,
   729→   729→      reason: snapshot.tradingBlockReason || "Trading is currently blocked by risk controls",
   730→   730→      riskAmount: 0,
   731→   731→      riskPercent: 0,
   732→   732→      suggestedLotSize: 0,
   733→   733→      warnings,
   734→   734→    }
   735→   735→  }
   736→   736→
   737→   737→  // ---- 1b. Proactive 60% zone: reduce new position sizes by 50% ----
   738→   738→  if (snapshot.proactiveMarginZone === "PROACTIVE_60") {
   739→   739→    positionSizeReduction = 0.5
   740→   740→    warnings.push("Margin level in PROACTIVE_60 zone: new position sizes reduced by 50%")
   741→   741→    logger.warn("RISK_MANAGEMENT", `PROACTIVE_60 zone: reducing ${params.symbol} position size by 50%`, {
   742→   742→      symbol: params.symbol,
   743→   743→      metadata: { marginLevelPercent: snapshot.marginLevelPercent },
   744→   744→    })
   745→   745→  }
   746→   746→
   747→   747→  // ---- 2. Position limit ----
   748→   748→  if (snapshot.openPositions >= config.maxOpenPositions) {
   749→   749→    return {
   750→   750→      approved: false,
   751→   751→      reason: `Max open positions reached (${config.maxOpenPositions})`,
   752→   752→      riskAmount: 0,
   753→   753→      riskPercent: 0,
   754→   754→      suggestedLotSize: 0,
   755→   755→      warnings,
   756→   756→    }
   757→   757→  }
   758→   758→
   759→   759→  // ---- 3. Lot size limits ----
   760→   760→  if (params.lotSize > config.maxLotPerTrade) {
   761→   761→    return {
   762→   762→      approved: false,
   763→   763→      reason: `Lot size ${params.lotSize} exceeds max per trade (${config.maxLotPerTrade})`,
   764→   764→      riskAmount: 0,
   765→   765→      riskPercent: 0,
   766→   766→      suggestedLotSize: 0,
   767→   767→      warnings,
   768→   768→    }
   769→   769→  }
   770→   770→
   771→   771→  // 3b. Check total lot per symbol
   772→   772→  const sameSymbolTrades = snapshot.positions.filter((p) => p.symbol === params.symbol)
   773→   773→  const totalSymbolLots = sameSymbolTrades.reduce((s, p) => s + p.lotSize, 0) + params.lotSize
   774→   774→  if (totalSymbolLots > config.maxLotPerSymbol) {
   775→   775→    return {
   776→   776→      approved: false,
   777→   777→      reason: `Total lot for ${params.symbol} (${totalSymbolLots}) would exceed max (${config.maxLotPerSymbol})`,
   778→   778→      riskAmount: 0,
   779→   779→      riskPercent: 0,
   780→   780→      suggestedLotSize: 0,
   781→   781→      warnings,
   782→   782→    }
   783→   783→  }
   784→   784→
   785→   785→  // ---- 4. Minimum lot check (FINEX: 0.01) ----
   786→   786→  if (params.lotSize < 0.01) {
   787→   787→    return {
   788→   788→      approved: false,
   789→   789→      reason: "Minimum lot size is 0.01",
   790→   790→      riskAmount: 0,
   791→   791→      riskPercent: 0,
   792→   792→      suggestedLotSize: 0.01,
   793→   793→      warnings,
   794→   794→    }
   795→   795→  }
   796→   796→
   797→   797→  // ---- 5. Calculate risk amount (including slippage) ----
   798→   798→  let riskAmount = 0
   799→   799→  if (params.sl) {
   800→   800→    riskAmount = params.direction === "BUY"
   801→   801→      ? (params.entryPrice - params.sl) * params.lotSize * 100000
   802→   802→      : (params.sl - params.entryPrice) * params.lotSize * 100000
   803→   803→    riskAmount = Math.abs(riskAmount)
   804→   804→  }
   805→   805→
   806→   806→  // 5b. Slippage modeling: if expected slippage exceeds tolerance, add to risk
   807→   807→  let slippageCost = 0
   808→   808→  if (params.expectedSlippage != null && params.expectedSlippage > 0) {
   809→   809→    if (params.expectedSlippage > config.slippageTolerancePips) {
   810→   810→      slippageCost = (params.expectedSlippage * params.lotSize * 100000) / params.entryPrice
   811→   811→      riskAmount += slippageCost
   812→   812→      warnings.push(
   813→   813→        `Expected slippage (${params.expectedSlippage.toFixed(1)} pips) exceeds tolerance (${config.slippageTolerancePips} pips). Added $${slippageCost.toFixed(2)} to risk amount.`,
   814→   814→      )
   815→   815→      await logRiskEvent({
   816→   816→        eventType: "SLIPPAGE_WARNING",
   817→   817→        severity: "MEDIUM",
   818→   818→        message: `High slippage expected for ${params.symbol}: ${params.expectedSlippage.toFixed(1)} pips (tolerance: ${config.slippageTolerancePips})`,
   819→   819→        details: `Slippage cost $${slippageCost.toFixed(2)} added to risk. Total risk for this trade: $${riskAmount.toFixed(2)}.`,
   820→   820→        actionTaken: "NONE",
   821→   821→      })
   822→   822→    } else {
   823→   823→      // Even within tolerance, include small slippage cost in risk
   824→   824→      slippageCost = (params.expectedSlippage * params.lotSize * 100000) / params.entryPrice
   825→   825→      riskAmount += slippageCost
   826→   826→    }
   827→   827→  }
   828→   828→
   829→   829→  const riskPercent = snapshot.equity > 0 ? (riskAmount / snapshot.equity) * 100 : 0
   830→   830→
   831→   831→  // ---- 6. Risk per trade limit ----
   832→   832→  const maxRiskDollar = (config.maxRiskPerTrade / 100) * snapshot.equity
   833→   833→  let suggestedLotSize = params.lotSize
   834→   834→
   835→   835→  if (riskAmount > maxRiskDollar && params.sl) {
   836→   836→    // Calculate safe lot size
   837→   837→    const pipRisk = params.direction === "BUY"
   838→   838→      ? params.entryPrice - (params.sl || 0)
   839→   839→      : (params.sl || 0) - params.entryPrice
   840→   840→    if (pipRisk > 0) {
   841→   841→      suggestedLotSize = Math.max(0.01, Math.floor((maxRiskDollar / (pipRisk * 100000)) * 100) / 100)
   842→   842→    }
   843→   843→    warnings.push(`Risk $${riskAmount.toFixed(2)} exceeds max $${maxRiskDollar.toFixed(2)} (${config.maxRiskPerTrade}%)`)
   844→   844→  }
   845→   845→
   846→   846→  // ---- 7. No stop loss? ----
   847→   847→  if (!params.sl) {
   848→   848→    warnings.push("No stop loss set - this increases uncontrolled risk")
   849→   849→  }
   850→   850→
   851→   851→  // ---- 8. Correlation risk (existing sector check) ----
   852→   852→  const sector = SYMBOL_SECTORS[params.symbol]
   853→   853→  if (sector) {
   854→   854→    const sectorExposure = snapshot.positions
   855→   855→      .filter((p) => SYMBOL_SECTORS[p.symbol] === sector)
   856→   856→      .reduce((s, p) => s + p.margin, 0)
   857→   857→    const newMargin = (params.entryPrice * suggestedLotSize * 100000) / 25
   858→   858→    const sectorExposurePct = ((sectorExposure + newMargin) / snapshot.equity) * 100
   859→   859→    if (sectorExposurePct > config.maxCorrelatedExposure) {
   860→   860→      warnings.push(`Sector (${sector}) exposure would be ${sectorExposurePct.toFixed(1)}% (max ${config.maxCorrelatedExposure}%)`)
   861→   861→    }
   862→   862→  }
   863→   863→
   864→   864→  // ---- 9. Margin usage after trade ----
   865→   865→  const newMargin = (params.entryPrice * suggestedLotSize * 100000) / 25
   866→   866→  const projectedMarginUsage = ((snapshot.marginUsed + newMargin) / snapshot.equity) * 100
   867→   867→  if (projectedMarginUsage > config.maxMarginUsage) {
   868→   868→    return {
   869→   869→      approved: false,
   870→   870→      reason: `Margin usage would be ${projectedMarginUsage.toFixed(1)}% (max ${config.maxMarginUsage}%)`,
   871→   871→      riskAmount,
   872→   872→      riskPercent,
   873→   873→      suggestedLotSize: 0,
   874→   874→      warnings,
   875→   875→    }
   876→   876→  }
   877→   877→
   878→   878→  // ---- 10. Cooldown after loss ----
   879→   879→  if (config.cooldownAfterLossMinutes > 0) {
   880→   880→    const recentLosses = await db.trade.findMany({
   881→   881→      where: {
   882→   882→        status: "CLOSED",
   883→   883→        pnl: { lt: 0 },
   884→   884→        closeTime: { gte: new Date(Date.now() - config.cooldownAfterLossMinutes * 60 * 1000) },
   885→   885→      },
   886→   886→    })
   887→   887→    if (recentLosses.length >= 3) {
   888→   888→      return {
   889→   889→        approved: false,
   890→   890→        reason: `Cooldown active: ${recentLosses.length} losses in last ${config.cooldownAfterLossMinutes} min`,
   891→   891→        riskAmount,
   892→   892→        riskPercent,
   893→   893→        suggestedLotSize: 0,
   894→   894→        warnings: ["Multiple recent losses detected. Waiting for cooldown period."],
   895→   895→      }
   896→   896→    }
   897→   897→  }
   898→   898→
   899→   899→  // ---- 11. Portfolio-level risk cap ----
   900→   900→  const currentPortfolioRisk = snapshot.positions.reduce((s, p) => s + p.riskAmount, 0)
   901→   901→  const projectedTotalRisk = currentPortfolioRisk + riskAmount
   902→   902→  const projectedPortfolioRiskPct = (projectedTotalRisk / snapshot.equity) * 100
   903→   903→  if (projectedPortfolioRiskPct > config.maxPortfolioRiskPct) {
   904→   904→    await logRiskEvent({
   905→   905→      eventType: "PORTFOLIO_RISK_CAP",
   906→   906→      severity: "HIGH",
   907→   907→      message: `Portfolio risk cap would be exceeded: ${projectedPortfolioRiskPct.toFixed(2)}% (max ${config.maxPortfolioRiskPct}%)`,
   908→   908→      details: `Current portfolio risk: $${currentPortfolioRisk.toFixed(2)} (${((currentPortfolioRisk / snapshot.equity) * 100).toFixed(2)}%). New trade risk: $${riskAmount.toFixed(2)} (${riskPercent.toFixed(2)}%). Combined: $${projectedTotalRisk.toFixed(2)} (${projectedPortfolioRiskPct.toFixed(2)}%).`,
   909→   909→      actionTaken: "TRADE_BLOCKED",
   910→   910→    })
   911→   911→    return {
   912→   912→      approved: false,
   913→   913→      reason: `Portfolio risk would be ${projectedPortfolioRiskPct.toFixed(2)}% (max ${config.maxPortfolioRiskPct}% of equity)`,
   914→   914→      riskAmount,
   915→   915→      riskPercent,
   916→   916→      suggestedLotSize: 0,
   917→   917→      warnings,
   918→   918→    }
   919→   919→  }
   920→   920→
   921→   921→  // ---- 12. Leverage utilization cap ----
   922→   922→  const effectiveLeverage = (params.entryPrice * suggestedLotSize * 100000) / snapshot.equity
   923→   923→  if (effectiveLeverage > config.maxLeveragePerTrade) {
   924→   924→    // Suggest smaller lot size
   925→   925→    const safeLotForLeverage = (config.maxLeveragePerTrade * snapshot.equity) / (params.entryPrice * 100000)
   926→   926→    const suggestedLotFromLeverage = Math.max(0.01, Math.floor(safeLotForLeverage * 100) / 100)
   927→   927→    await logRiskEvent({
   928→   928→      eventType: "LEVERAGE_CAP",
   929→   929→      severity: "HIGH",
   930→   930→      message: `Leverage cap exceeded for ${params.symbol}: ${effectiveLeverage.toFixed(1)}x (max ${config.maxLeveragePerTrade}x)`,
   931→   931→      details: `Entry ${params.entryPrice} x ${suggestedLotSize} lots = ${effectiveLeverage.toFixed(1)}x leverage. Suggested max lot: ${suggestedLotFromLeverage}.`,
   932→   932→      actionTaken: "TRADE_BLOCKED",
   933→   933→    })
   934→   934→    return {
   935→   935→      approved: false,
   936→   936→      reason: `Effective leverage ${effectiveLeverage.toFixed(1)}x exceeds max ${config.maxLeveragePerTrade}x. Suggested lot size: ${suggestedLotFromLeverage}`,
   937→   937→      riskAmount,
   938→   938→      riskPercent,
   939→   939→      suggestedLotSize: suggestedLotFromLeverage,
   940→   940→      warnings: [`Leverage ${effectiveLeverage.toFixed(1)}x exceeds cap ${config.maxLeveragePerTrade}x. Reduce lot to ${suggestedLotFromLeverage}.`],
   941→   941→    }
   942→   942→  }
   943→   943→
   944→   944→  // ---- 13. Single stock concentration limit ----
   945→   945→  const sameSymbolMargin = sameSymbolTrades.reduce((s, p) => s + p.margin, 0)
   946→   946→  const tradeMargin = (params.entryPrice * suggestedLotSize * 100000) / 25
   947→   947→  const totalSymbolMargin = sameSymbolMargin + tradeMargin
   948→   948→  const singleStockPct = (totalSymbolMargin / snapshot.equity) * 100
   949→   949→  if (singleStockPct > config.maxSingleStockPct) {
   950→   950→    await logRiskEvent({
   951→   951→      eventType: "CONCENTRATION_LIMIT",
   952→   952→      severity: "HIGH",
   953→   953→      message: `Single stock concentration exceeded for ${params.symbol}: ${singleStockPct.toFixed(2)}% (max ${config.maxSingleStockPct}%)`,
   954→   954→      details: `Current margin in ${params.symbol}: $${sameSymbolMargin.toFixed(2)}. New trade margin: $${tradeMargin.toFixed(2)}. Total: $${totalSymbolMargin.toFixed(2)} = ${singleStockPct.toFixed(2)}% of equity.`,
   955→   955→      actionTaken: "TRADE_BLOCKED",
   956→   956→    })
   957→   957→    return {
   958→   958→      approved: false,
   959→   959→      reason: `Single stock ${params.symbol} concentration would be ${singleStockPct.toFixed(2)}% (max ${config.maxSingleStockPct}% of equity)`,
   960→   960→      riskAmount,
   961→   961→      riskPercent,
   962→   962→      suggestedLotSize: 0,
   963→   963→      warnings,
   964→   964→    }
   965→   965→  }
   966→   966→
   967→   967→  // ---- 14. Sector concentration limit ----
   968→   968→  if (sector) {
   969→   969→    const sectorMarginUsed = snapshot.sectorExposure.find((se) => se.sector === sector)?.marginUsed || 0
   970→   970→    const totalSectorMargin = sectorMarginUsed + tradeMargin
   971→   971→    const sectorPct = (totalSectorMargin / snapshot.equity) * 100
   972→   972→    if (sectorPct > config.maxSectorPct) {
   973→   973→      await logRiskEvent({
   974→   974→        eventType: "CONCENTRATION_LIMIT",
   975→   975→        severity: "HIGH",
   976→   976→        message: `Sector concentration exceeded for ${sector}: ${sectorPct.toFixed(2)}% (max ${config.maxSectorPct}%)`,
   977→   977→        details: `Current sector (${sector}) margin: $${sectorMarginUsed.toFixed(2)}. New trade margin: $${tradeMargin.toFixed(2)}. Total: $${totalSectorMargin.toFixed(2)} = ${sectorPct.toFixed(2)}% of equity.`,
   978→   978→        actionTaken: "TRADE_BLOCKED",
   979→   979→      })
   980→   980→      return {
   981→   981→        approved: false,
   982→   982→        reason: `Sector ${sector} concentration would be ${sectorPct.toFixed(2)}% (max ${config.maxSectorPct}% of equity)`,
   983→   983→        riskAmount,
   984→   984→        riskPercent,
   985→   985→        suggestedLotSize: 0,
   986→   986→        warnings,
   987→   987→      }
   988→   988→    }
   989→   989→  }
   990→   990→
   991→   991→  // ---- 15. Reserve capital check ----
   992→   992→  const requiredReserve = (config.reserveCapitalPct / 100) * snapshot.equity
   993→   993→  const freeMarginAfterTrade = snapshot.freeMargin - tradeMargin
   994→   994→  if (freeMarginAfterTrade < requiredReserve) {
   995→   995→    const currentReservePct = (snapshot.freeMargin / snapshot.equity) * 100
   996→   996→    return {
   997→   997→      approved: false,
   998→   998→      reason: `Insufficient reserve capital: $${freeMarginAfterTrade.toFixed(2)} remaining (need ${config.reserveCapitalPct}% = $${requiredReserve.toFixed(2)}). Current reserve: ${currentReservePct.toFixed(1)}%.`,
   999→   999→      riskAmount,
  1000→  1000→      riskPercent,
  1001→  1001→      suggestedLotSize: 0,
  1002→  1002→      warnings: [`Reserve capital requirement: ${config.reserveCapitalPct}% of equity ($${requiredReserve.toFixed(2)}).`],
  1003→  1003→    }
  1004→  1004→  }
  1005→  1005→
  1006→  1006→  // ---- Apply position size reduction from PROACTIVE_60 zone ----
  1007→  1007→  if (positionSizeReduction > 0) {
  1008→  1008→    suggestedLotSize = Math.max(0.01, Math.round(suggestedLotSize * (1 - positionSizeReduction) * 100) / 100)
  1009→  1009→    // Recalculate risk amount with reduced lot
  1010→  1010→    if (params.sl) {
  1011→  1011→      riskAmount = params.direction === "BUY"
  1012→  1012→        ? (params.entryPrice - params.sl) * suggestedLotSize * 100000
  1013→  1013→        : (params.sl - params.entryPrice) * suggestedLotSize * 100000
  1014→  1014→      riskAmount = Math.abs(riskAmount) + slippageCost
  1015→  1015→    }
  1016→  1016→  }
  1017→  1017→
  1018→  1018→  // ---- Apply dynamic scaling factor ----
  1019→  1019→  if (snapshot.scalingFactor < 1.0) {
  1020→  1020→    const scaledLot = Math.max(0.01, Math.round(suggestedLotSize * snapshot.scalingFactor * 100) / 100)
  1021→  1021→    if (scaledLot < suggestedLotSize) {
  1022→  1022→      warnings.push(
  1023→  1023→        `Dynamic risk scaling applied: lot reduced from ${suggestedLotSize} to ${scaledLot} (factor: ${snapshot.scalingFactor})`,
  1024→  1024→      )
  1025→  1025→      suggestedLotSize = scaledLot
  1026→  1026→    }
  1027→  1027→  }
  1028→  1028→
  1029→  1029→  // ---- Approved ----
  1030→  1030→  return {
  1031→  1031→    approved: true,
  1032→  1032→    riskAmount: Math.round(riskAmount * 100) / 100,
  1033→  1033→    riskPercent: Math.round(riskPercent * 100) / 100,
  1034→  1034→    suggestedLotSize: Math.round(suggestedLotSize * 100) / 100,
  1035→  1035→    warnings,
  1036→  1036→    positionSizeReduction: positionSizeReduction > 0 ? positionSizeReduction : undefined,
  1037→  1037→  }
  1038→  1038→}

  1077→  1077→function calculateRiskScore(factors: {
  1078→  1078→  dailyPnlPercent: number
  1079→  1079→  marginUsagePercent: number
  1080→  1080→  currentDrawdown: number
  1081→  1081→  maxDrawdown: number
  1082→  1082→  config: RiskConfigData
  1083→  1083→  openPositionCount: number
  1084→  1084→  marginLevelPercent: number
  1085→  1085→}): number {
  1086→  1086→  let score = 0
  1087→  1087→
  1088→  1088→  // Daily loss contribution (0-3 points)
  1089→  1089→  const dailyLossPct = Math.abs(Math.min(0, factors.dailyPnlPercent))
  1090→  1090→  score += Math.min(3, (dailyLossPct / factors.config.maxDailyLoss) * 3)
  1091→  1091→
  1092→  1092→  // Margin usage (0-2 points)
  1093→  1093→  score += Math.min(2, (factors.marginUsagePercent / factors.config.maxMarginUsage) * 2)
  1094→  1094→
  1095→  1095→  // Drawdown (0-3 points)
  1096→  1096→  score += Math.min(3, (factors.currentDrawdown / factors.config.maxDrawdown) * 3)
  1097→  1097→
  1098→  1098→  // Margin level proximity to stop out (0-2 points)
  1099→  1099→  // Enhanced to account for 4-zone system
  1100→  1100→  if (factors.marginLevelPercent > 0) {
  1101→  1101→    if (factors.marginLevelPercent <= factors.config.stopOutLevel) {
  1102→  1102→      score += 2 // STOP_OUT
  1103→  1103→    } else if (factors.marginLevelPercent <= factors.config.marginCallLevel) {
  1104→  1104→      score += 2 // MARGIN_CALL
  1105→  1105→    } else if (factors.config.proactiveMcLevel60 && factors.marginLevelPercent <= 60) {
  1106→  1106→      score += 1.5 // PROACTIVE_60
  1107→  1107→    } else if (factors.config.proactiveMcLevel70 && factors.marginLevelPercent <= 70) {
  1108→  1108→      score += 0.75 // PROACTIVE_70
  1109→  1109→    }
  1110→  1110→  }
  1111→  1111→
  1112→  1112→  return Math.min(10, Math.round(score * 10) / 10)
  1113→  1113→}

  1115→  1115→function getRiskLevel(score: number): RiskSnapshot["riskLevel"] {
  1116→  1116→  if (score <= 2) return "LOW"
  1117→  1117→  if (score <= 4) return "MODERATE"
  1118→  1118→  if (score <= 6) return "ELEVATED"
  1119→  1119→  if (score <= 8) return "HIGH"
  1120→  1120→  return "CRITICAL"

  1123→  1123→function generateRecommendations(ctx: {

  1125→  1125→  riskLevel: string

  1128→  1128→  currentDrawdown: number
  1129→  1129→  marginUsagePercent: number
  1130→  1130→  config: RiskConfigData

  1166→  1166→
  1167→  1167→  if (ctx.riskLevel === "LOW" && ctx.isTradingAllowed) {

  1171→  1171→
  1172→  1172→  if (!ctx.isTradingAllowed && ctx.tradingBlockReason) {

  1175→  1175→
  1176→  1176→  return recs
  1177→  1177→}
