   290→/**
   291→ * Determine the proactive margin zone based on margin level percentage.
   292→ * Zone 1: PROACTIVE at 70% - warning, log event, continue trading
   293→ * Zone 2: PROACTIVE at 60% - strong warning, reduce new position sizes by 50%, log event
   294→ * Zone 3: MARGIN CALL at 50% - critical, block new trades
   295→ * Zone 4: STOP OUT at 20% - fatal, emergency close all positions
   296→ */
   297→export function determineProactiveMarginZone(
   298→  marginLevelPercent: number,
   299→  config: RiskConfigData,
   300→): ProactiveMarginZone {
   301→  // When no margin is used, we're in SAFE zone
   302→  if (marginLevelPercent === 0) {
   303→    return "SAFE"
   304→  }
   305→  if (marginLevelPercent <= config.stopOutLevel) {
   306→    return "STOP_OUT"
   307→  }
   308→  if (marginLevelPercent <= config.marginCallLevel) {
   309→    return "MARGIN_CALL"
   310→  }
   311→  if (config.proactiveMcLevel60 && marginLevelPercent <= 60) {
   312→    return "PROACTIVE_60"
   313→  }
   314→  if (config.proactiveMcLevel70 && marginLevelPercent <= 70) {
   315→    return "PROACTIVE_70"
   316→  }
   317→  return "SAFE"
   318→}

   320→/**
   321→ * Process proactive margin monitoring: log events for zone transitions.
   322→ * Returns the zone and any actions taken.
   323→ */
   324→export async function processProactiveMarginMonitoring(
   325→  marginLevelPercent: number,
   326→  config: RiskConfigData,
   327→): Promise<{ zone: ProactiveMarginZone; actionTaken: string }> {
   328→  const zone = determineProactiveMarginZone(marginLevelPercent, config)
   329→  let actionTaken = "NONE"
   330→
   331→  // Check if we recently logged the same zone to avoid spam
   332→  const fiveMinutesAgo = new Date(Date.now() - 5 * 60 * 1000)
   333→
   334→  if (zone === "PROACTIVE_70") {
   335→    const recent = await db.riskEvent.findFirst({
   336→      where: {
   337→        eventType: "PROACTIVE_MC_70",
   338→        createdAt: { gte: fiveMinutesAgo },
   339→      },
   340→    })
   341→    if (!recent) {
   342→      await logRiskEvent({
   343→        eventType: "PROACTIVE_MC_70",
   344→        severity: "MEDIUM",
   345→        message: `Proactive margin warning: margin level at ${marginLevelPercent.toFixed(1)}% (zone 70%)`,
   346→        details: `Margin level has dropped below 70% threshold. Current level: ${marginLevelPercent.toFixed(1)}%. Trading continues but monitor closely.`,
   347→        actionTaken: "NOTIFICATION_SENT",
   348→      })
   349→      logger.warn("RISK_MANAGEMENT", `Proactive MC 70% triggered: margin level at ${marginLevelPercent.toFixed(1)}%`)
   350→    }
   351→    actionTaken = "NOTIFICATION_SENT"
   352→  } else if (zone === "PROACTIVE_60") {
   353→    const recent = await db.riskEvent.findFirst({
   354→      where: {
   355→        eventType: "PROACTIVE_MC_60",
   356→        createdAt: { gte: fiveMinutesAgo },
   357→      },
   358→    })
   359→    if (!recent) {
   360→      await logRiskEvent({
   361→        eventType: "PROACTIVE_MC_60",
   362→        severity: "HIGH",
   363→        message: `Proactive margin strong warning: margin level at ${marginLevelPercent.toFixed(1)}% (zone 60%)`,
   364→        details: `Margin level has dropped below 60% threshold. Current level: ${marginLevelPercent.toFixed(1)}%. New position sizes will be reduced by 50%.`,
   365→        actionTaken: "REDUCED_SIZE",
   366→      })
   367→      logger.warn("RISK_MANAGEMENT", `Proactive MC 60% triggered: margin level at ${marginLevelPercent.toFixed(1)}%, sizes reduced 50%`)
   368→    }
   369→    actionTaken = "REDUCED_SIZE"
   370→  } else if (zone === "MARGIN_CALL") {
   371→    const recent = await db.riskEvent.findFirst({
   372→      where: {
   373→        eventType: "MARGIN_CALL_WARNING",
   374→        createdAt: { gte: fiveMinutesAgo },
   375→      },
   376→    })
   377→    if (!recent) {
   378→      await logRiskEvent({
   379→        eventType: "MARGIN_CALL_WARNING",
   380→        severity: "CRITICAL",
   381→        message: `Margin call level reached: margin level at ${marginLevelPercent.toFixed(1)}% (threshold: ${config.marginCallLevel}%)`,
   382→        details: `Margin level has reached margin call threshold. All new trades blocked. Monitor for stop out.`,
   383→        actionTaken: "TRADE_BLOCKED",
   384→      })
   385→      logger.critical("RISK_MANAGEMENT", `Margin call at ${marginLevelPercent.toFixed(1)}%: new trades blocked`)
   386→    }
   387→    actionTaken = "TRADE_BLOCKED"
   388→  } else if (zone === "STOP_OUT") {
   389→    const recent = await db.riskEvent.findFirst({
   390→      where: {
   391→        eventType: "STOP_OUT_WARNING",
   392→        createdAt: { gte: fiveMinutesAgo },
   393→      },
   394→    })
   395→    if (!recent) {
   396→      await logRiskEvent({
   397→        eventType: "STOP_OUT_WARNING",
   398→        severity: "CRITICAL",
   399→        message: `STOP OUT level reached: margin level at ${marginLevelPercent.toFixed(1)}% (threshold: ${config.stopOutLevel}%)`,
   400→        details: `Margin level has reached stop out threshold. Emergency: all positions should be closed immediately.`,
   401→        actionTaken: "ALL_POSITIONS_CLOSED",
   402→      })
   403→      logger.critical("RISK_MANAGEMENT", `STOP OUT at ${marginLevelPercent.toFixed(1)}%: emergency close all positions`)
   404→    }
   405→    actionTaken = "ALL_POSITIONS_CLOSED"
   406→  }
   407→
   408→  return { zone, actionTaken }
   409→}

   411→/**
   412→ * Calculate dynamic risk scaling factor based on recent performance.
   413→ * - If daily loss > 1% of limit, scale down by 50%
   414→ * - If drawdown > 80% of max, scale down by 75%
   415→ * - If winning streak > 3, can scale up by 25% (max 1.5x)
   416→ */
   417→export function calculateScalingFactor(params: {
   418→  dailyPnl: number
   419→  dailyLossLimit: number
   420→  currentDrawdown: number
   421→  maxDrawdown: number
   422→  equity: number
   423→}): number {
   424→  let factor = 1.0
   425→
   426→  // Scale down if daily loss is more than 1% of the limit
   427→  if (params.dailyPnl < 0 && params.dailyLossLimit > 0) {
   428→    const dailyLossPct = Math.abs(params.dailyPnl) / params.dailyLossLimit
   429→    if (dailyLossPct > 0.5) {
   430→      factor = Math.min(factor, 0.5) // 50% reduction
   431→    }
   432→  }
   433→
   434→  // Scale down if drawdown > 80% of max
   435→  if (params.maxDrawdown > 0 && params.currentDrawdown > params.maxDrawdown * 0.8) {
   436→    factor = Math.min(factor, 0.25) // 75% reduction
   437→  }
   438→
   439→  // Scale up if on a winning streak > 3 consecutive wins
   440→  // This is a bonus that can only increase factor up to 1.5x
   441→  // The winning streak check requires trade history, so we apply a
   442→  // conservative boost. The actual streak check is in the caller.
   443→  // Here we use a positive daily P&L as a proxy for the upward boost.
   444→  if (params.dailyPnl > 0 && factor >= 1.0) {
   445→    // Only allow scaling up if no drawdown pressure
   446→    if (params.currentDrawdown < params.maxDrawdown * 0.3) {
   447→      factor = Math.min(factor * 1.25, 1.5)
   448→    }
   449→  }
   450→
   451→  return Math.round(factor * 1000) / 1000
   452→}

   454→/**
   455→ * Calculate sector exposure breakdown for all open positions.
   456→ */
   457→export function calculateSectorExposure(
   458→  openTrades: PositionRiskBreakdown[],
   459→  equity: number,
   460→): SectorExposureEntry[] {
   461→  const sectorMap = new Map<string, { margin: number; count: number }>()
   462→
   463→  for (const pos of openTrades) {
   464→    const sector = SYMBOL_SECTORS[pos.symbol] || "Unknown"
   465→    const existing = sectorMap.get(sector)
   466→    if (existing) {
   467→      existing.margin += pos.margin
   468→      existing.count += 1
   469→    } else {
   470→      sectorMap.set(sector, { margin: pos.margin, count: 1 })
   471→    }
   472→  }
   473→
   474→  const entries: SectorExposureEntry[] = []
   475→  for (const [sector, data] of sectorMap.entries()) {
   476→    entries.push({
   477→      sector,
   478→      exposurePct: equity > 0 ? Math.round((data.margin / equity) * 10000) / 100 : 0,
   479→      positionCount: data.count,
   480→      marginUsed: Math.round(data.margin * 100) / 100,
   481→    })
   482→  }
   483→
   484→  // Sort by exposure descending
   485→  entries.sort((a, b) => b.exposurePct - a.exposurePct)
   486→  return entries
   487→}

   489→/**
   490→ * Calculate a full risk snapshot of the current account state.
   491→ */
   492→export async function getRiskSnapshot(): Promise<RiskSnapshot> {
   493→  const config = await getRiskConfig()
   494→  const openTrades = await db.trade.findMany({ where: { status: "OPEN" } })
   495→  const allClosed = await db.trade.findMany({ where: { status: "CLOSED" } })
   496→
   497→  const BASE_BALANCE = 10000
   498→  const totalClosedPnl = allClosed.reduce((s, t) => s + t.pnl, 0)
   499→  const currentBalance = Math.round((BASE_BALANCE + totalClosedPnl) * 100) / 100
   500→  const totalOpenPnl = openTrades.reduce((s, t) => s + t.pnl, 0)
   501→  const equity = Math.round((currentBalance + totalOpenPnl) * 100) / 100
   502→  const totalMarginUsed = openTrades.reduce((s, t) => s + t.margin, 0)
   503→  const freeMargin = Math.max(0, Math.round((equity - totalMarginUsed) * 100) / 100)
   504→  const marginLevelPercent = totalMarginUsed > 0 ? Math.round((equity / totalMarginUsed) * 10000) / 100 : 0
   505→
   506→  // ---- Time-based P&L ----
   507→  const now = new Date()
   508→  const todayStr = now.toISOString().split("T")[0]
   509→  const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000)
   510→  const monthAgo = new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000)
   511→
   512→  const closedToday = allClosed.filter((t) => t.closeTime && t.closeTime >= new Date(todayStr))
   513→  const closedWeek = allClosed.filter((t) => t.closeTime && t.closeTime >= weekAgo)
   514→  const closedMonth = allClosed.filter((t) => t.closeTime && t.closeTime >= monthAgo)
   515→
   516→  const dailyPnl = closedToday.reduce((s, t) => s + t.pnl, 0) + totalOpenPnl
   517→  const weeklyPnl = closedWeek.reduce((s, t) => s + t.pnl, 0) + totalOpenPnl
   518→  const monthlyPnl = closedMonth.reduce((s, t) => s + t.pnl, 0) + totalOpenPnl
   519→
   520→  // ---- Drawdown calculation ----
   521→  let peak = currentBalance
   522→  let maxDrawdown = 0
   523→  let runningBalance = currentBalance - totalClosedPnl
   524→  for (const trade of allClosed) {
   525→    runningBalance += trade.pnl
   526→    if (runningBalance > peak) peak = runningBalance
   527→    const dd = peak > 0 ? ((peak - runningBalance) / peak) * 100 : 0
   528→    if (dd > maxDrawdown) maxDrawdown = dd
   529→  }
   530→  // Factor in open P&L for current drawdown
   531→  if (equity < peak) {
   532→    const currentDD = ((peak - equity) / peak) * 100
   533→    if (currentDD > maxDrawdown) maxDrawdown = currentDD
   534→  }
   535→  const currentDrawdown = peak > 0 ? Math.round(((peak - equity) / peak) * 10000) / 100 : 0
   536→
   537→  // ---- Proactive margin monitoring (4 zones) ----
   538→  const proactiveMarginZone = determineProactiveMarginZone(marginLevelPercent, config)
   539→
   540→  // Process zone transitions and log events
   541→  if (proactiveMarginZone !== "SAFE") {
   542→    await processProactiveMarginMonitoring(marginLevelPercent, config)
   543→  }
   544→
   545→  const isMarginCallWarning =
   546→    proactiveMarginZone === "MARGIN_CALL" ||
   547→    proactiveMarginZone === "STOP_OUT"
   548→  const isStopOutWarning = proactiveMarginZone === "STOP_OUT"
   549→
   550→  // ---- Risk Score (deterministic, not random) ----
   551→  const riskScore = calculateRiskScore({
   552→    dailyPnlPercent: (dailyPnl / equity) * 100,
   553→    marginUsagePercent: equity > 0 ? (totalMarginUsed / equity) * 100 : 0,
   554→    currentDrawdown,
   555→    maxDrawdown,
   556→    config,
   557→    openPositionCount: openTrades.length,
   558→    marginLevelPercent,
   559→  })
   560→
   561→  const riskLevel = getRiskLevel(riskScore)
   562→
   563→  // ---- Daily loss limit ----
   564→  const dailyLossLimit = (config.maxDailyLoss / 100) * equity
   565→  const isDailyLimitReached = dailyPnl < 0 && Math.abs(dailyPnl) >= dailyLossLimit
   566→
   567→  // ---- Trading allowed? ----
   568→  let isTradingAllowed = true
   569→  let tradingBlockReason: string | undefined
   570→
   571→  if (isStopOutWarning) {
   572→    isTradingAllowed = false
   573→    tradingBlockReason = `Stop out level reached (${config.stopOutLevel}%)`
   574→  }
   575→  if (proactiveMarginZone === "MARGIN_CALL") {
   576→    isTradingAllowed = false
   577→    tradingBlockReason = `Margin call level reached (${config.marginCallLevel}%)`
   578→  }
   579→  if (isDailyLimitReached) {
   580→    isTradingAllowed = false
   581→    tradingBlockReason = `Daily loss limit reached (${config.maxDailyLoss}% / $${dailyLossLimit.toFixed(2)})`
   582→  }
   583→  if (currentDrawdown >= config.maxDrawdown) {
   584→    isTradingAllowed = false
   585→    tradingBlockReason = `Max drawdown limit reached (${config.maxDrawdown}%)`
   586→  }
   587→  if (openTrades.length >= config.maxOpenPositions) {
   588→    isTradingAllowed = false
   589→    tradingBlockReason = `Max open positions reached (${config.maxOpenPositions})`
   590→  }
   591→
   592→  // ---- Recent risk events ----
   593→  const recentEvents = await db.riskEvent.findMany({
   594→    orderBy: { createdAt: "desc" },
   595→    take: 10,
   596→  })
   597→
   598→  // ---- Recommendations ----
   599→  const recommendations = generateRecommendations({
   600→    riskScore,
   601→    riskLevel,
   602→    dailyPnl,
   603→    dailyPnlPercent: (dailyPnl / equity) * 100,
   604→    currentDrawdown,
   605→    marginUsagePercent: equity > 0 ? (totalMarginUsed / equity) * 100 : 0,
   606→    config,
   607→    isDailyLimitReached,
   608→    isMarginCallWarning,
   609→    isStopOutWarning,
   610→    isTradingAllowed,
   611→    tradingBlockReason,
   612→    proactiveMarginZone,
   613→  })
   614→
   615→  // ---- Position risk breakdown ----
   616→  const positions: PositionRiskBreakdown[] = openTrades.map((t) => {
   617→    let riskAmount = 0
   618→    if (t.sl) {
   619→      riskAmount = t.direction === "BUY"
   620→        ? (t.entryPrice - t.sl) * t.lotSize * 100000
   621→        : (t.sl - t.entryPrice) * t.lotSize * 100000
   622→    }
   623→    return {
   624→      tradeId: t.id,
   625→      symbol: t.symbol,
   626→      direction: t.direction,
   627→      lotSize: t.lotSize,
   628→      entryPrice: t.entryPrice,
   629→      currentPrice: t.currentPrice,
   630→      sl: t.sl,
   631→      tp: t.tp,
   632→      margin: t.margin,
   633→      pnl: t.pnl,
   634→      pnlPercent: t.pnlPercent,
   635→      riskAmount: Math.round(Math.abs(riskAmount) * 100) / 100,
   636→      riskPercent: equity > 0 ? Math.round((Math.abs(riskAmount) / equity) * 10000) / 100 : 0,
   637→      strategy: t.strategy,
   638→      trailingStop: t.trailingStop,
   639→    }
   640→  })
   641→
   642→  // ---- Sector exposure ----
   643→  const sectorExposure = calculateSectorExposure(positions, equity)
   644→
   645→  // ---- Portfolio total risk % ----
   646→  const portfolioTotalRisk = positions.reduce((s, p) => s + p.riskAmount, 0)
   647→  const portfolioTotalRiskPct = equity > 0
   648→    ? Math.round((portfolioTotalRisk / equity) * 10000) / 100
   649→    : 0
   650→
   651→  // ---- Current effective leverage ----
   652→  const totalNotionalValue = openTrades.reduce(
   653→    (s, t) => s + t.entryPrice * t.lotSize * 100000, 0,
   654→  )
   655→  const leverageUsed = equity > 0
   656→    ? Math.round((totalNotionalValue / equity) * 100) / 100
   657→    : 0
   658→
   659→  // ---- Current reserve capital % ----
   660→  const reserveCapitalPct = equity > 0
   661→    ? Math.round((freeMargin / equity) * 10000) / 100
   662→    : 0
   663→
   664→  // ---- Dynamic risk scaling factor ----
   665→  // Check winning streak for upward scaling
   666→  const recentClosed = allClosed
   667→    .filter((t) => t.closeTime && t.closeTime >= new Date(todayStr))
   668→    .sort((a, b) => (b.closeTime?.getTime() || 0) - (a.closeTime?.getTime() || 0))
   669→  let winningStreak = 0
   670→  for (const t of recentClosed) {
   671→    if (t.pnl > 0) {
   672→      winningStreak++
   673→    } else {
   674→      break
   675→    }
   676→  }
   677→  let scalingFactor = calculateScalingFactor({
   678→    dailyPnl,
   679→    dailyLossLimit,
   680→    currentDrawdown,
   681→    maxDrawdown: config.maxDrawdown,
   682→    equity,
   683→  })
   684→  // Apply winning streak boost
   685→  if (winningStreak > 3 && scalingFactor >= 1.0 && currentDrawdown < config.maxDrawdown * 0.3) {
   686→    scalingFactor = Math.min(scalingFactor * 1.25, 1.5)
   687→    scalingFactor = Math.round(scalingFactor * 1000) / 1000
   688→  }
   689→
   690→  return {
   691→    equity: Math.round(equity * 100) / 100,
   692→    balance: Math.round(currentBalance * 100) / 100,
   693→    freeMargin,
   694→    marginUsed: Math.round(totalMarginUsed * 100) / 100,
   695→    marginLevelPercent,
   696→    dailyPnl: Math.round(dailyPnl * 100) / 100,
   697→    dailyPnlPercent: Math.round((dailyPnl / equity) * 10000) / 100,
   698→    weeklyPnl: Math.round(weeklyPnl * 100) / 100,
   699→    weeklyPnlPercent: Math.round((weeklyPnl / equity) * 10000) / 100,
   700→    monthlyPnl: Math.round(monthlyPnl * 100) / 100,
   701→    monthlyPnlPercent: Math.round((monthlyPnl / equity) * 10000) / 100,
   702→    currentDrawdown: Math.round(currentDrawdown * 100) / 100,
   703→    maxDrawdown: Math.round(maxDrawdown * 100) / 100,
   704→    maxAllowedDrawdown: config.maxDrawdown,
   705→    riskScore,
   706→    riskLevel,
   707→    openPositions: openTrades.length,
   708→    maxPositionsAllowed: config.maxOpenPositions,
   709→    marginUsagePercent: Math.round((equity > 0 ? (totalMarginUsed / equity) * 100 : 0) * 100) / 100,
   710→    maxMarginAllowed: config.maxMarginUsage,
   711→    dailyLossRemaining: Math.round(Math.max(0, dailyLossLimit - Math.abs(Math.min(0, dailyPnl))) * 100) / 100,
   712→    dailyLossLimit: Math.round(dailyLossLimit * 100) / 100,
   713→    isDailyLimitReached,
   714→    isMarginCallWarning,
   715→    isStopOutWarning,
   716→    isTradingAllowed,
   717→    tradingBlockReason,
   718→    recentRiskEvents: recentEvents.map((e) => ({
   719→      eventType: e.eventType,
   720→      severity: e.severity,
   721→      message: e.message,
   722→      createdAt: e.createdAt.toISOString(),
   723→      resolved: e.resolved,
   724→    })),
   725→    recommendations,
   726→    positions,
   727→    // Deep audit fields
   728→    proactiveMarginZone,
   729→    sectorExposure,
   730→    portfolioTotalRiskPct,
   731→    leverageUsed,
   732→    reserveCapitalPct,
   733→    scalingFactor,
   734→    // Phase 3 fields
   735→    ...(await buildPhase3SnapshotFields()),
   736→  }
   737→}
   738→
   739→/**
   740→ * Build Phase 3 snapshot fields by reading from DB and computing volatility regime.
   741→ */
   742→async function buildPhase3SnapshotFields(): Promise<{
   743→  volatilityRegime: string
   744→  volatilityRiskMultiplier: number
   745→  circuitBreakerState: string
   746→  connectionQuality: number
   747→  hasGapRisk: boolean
   748→  unresolvedRiskEvents: number
   749→}> {
   750→  // Default volatility regime when no real data available
   751→  const volRegime = detectVolatilityRegime({ recentVolatility: 0, avgVolatility: 0 })
   752→
   753→  // Read MT5 connection state from DB for circuit breaker & quality
   754→  let circuitBreakerState = "CLOSED"
   755→  let connectionQuality = 100
   756→  try {
   757→    const connState = await db.mt5ConnectionState.findFirst({ orderBy: { createdAt: "desc" } })
   758→    if (connState) {
   759→      circuitBreakerState = connState.circuitState || "CLOSED"
   760→      connectionQuality = Math.round(connState.connectionQuality * 100) / 100
   761→    }
   762→  } catch {
   763→    // Table may not exist or be accessible; use defaults
   764→  }
   765→
   766→  const unresolvedRiskEvents = await db.riskEvent.count({ where: { resolved: false } })
   767→
   768→  return {
   769→    volatilityRegime: volRegime.regime,
   770→    volatilityRiskMultiplier: volRegime.riskMultiplier,
   771→    circuitBreakerState,
   772→    connectionQuality,
   773→    hasGapRisk: false,
   774→    unresolvedRiskEvents,
   775→  }
   776→}

   778→/**
   779→ * Pre-trade risk validation. Call this BEFORE creating any trade.
   780→ *
   781→ * Checks (in order):
   782→ *  1. Trading allowed? (global block)
   783→ *  2. Position limit
   784→ *  3. Lot size limits (per trade, per symbol)
   785→ *  4. Minimum lot check
   786→ *  5. Calculate risk amount (including slippage)
   787→ *  6. Risk per trade limit
   788→ *  7. No stop loss warning
   789→ *  8. Correlation risk (sector)
   790→ *  9. Margin usage after trade
   791→ * 10. Cooldown after loss
   792→ * 11. Portfolio-level risk cap
   793→ * 12. Leverage utilization cap
   794→ * 13. Single stock concentration limit
   795→ * 14. Sector concentration limit
   796→ * 15. Reserve capital check
   797→ */
   798→export async function preTradeCheck(params: {
   799→  symbol: string
   800→  direction: string
   801→  lotSize: number
   802→  entryPrice: number
   803→  sl?: number | null
   804→  tp?: number | null
   805→  strategy?: string | null
   806→  aiConfidence?: number | null
   807→  expectedSlippage?: number | null
   808→}): Promise<PreTradeCheck> {
   809→  const config = await getRiskConfig()
   810→  const snapshot = await getRiskSnapshot()
   811→  const warnings: string[] = []
   812→  let positionSizeReduction = 0
   813→
   814→  // ---- 1. Trading allowed? ----
   815→  if (!snapshot.isTradingAllowed) {
   816→    return {
   817→      approved: false,
   818→      reason: snapshot.tradingBlockReason || "Trading is currently blocked by risk controls",
   819→      riskAmount: 0,
   820→      riskPercent: 0,
   821→      suggestedLotSize: 0,
   822→      warnings,
   823→    }
   824→  }
   825→
   826→  // ---- 1b. Proactive 60% zone: reduce new position sizes by 50% ----
   827→  if (snapshot.proactiveMarginZone === "PROACTIVE_60") {
   828→    positionSizeReduction = 0.5
   829→    warnings.push("Margin level in PROACTIVE_60 zone: new position sizes reduced by 50%")
   830→    logger.warn("RISK_MANAGEMENT", `PROACTIVE_60 zone: reducing ${params.symbol} position size by 50%`, {
   831→      symbol: params.symbol,
   832→      metadata: { marginLevelPercent: snapshot.marginLevelPercent },
   833→    })
   834→  }
   835→
   836→  // ---- 2. Position limit ----
   837→  if (snapshot.openPositions >= config.maxOpenPositions) {
   838→    return {
   839→      approved: false,
   840→      reason: `Max open positions reached (${config.maxOpenPositions})`,
   841→      riskAmount: 0,
   842→      riskPercent: 0,
   843→      suggestedLotSize: 0,
   844→      warnings,
   845→    }
   846→  }
   847→
   848→  // ---- 3. Lot size limits ----
   849→  if (params.lotSize > config.maxLotPerTrade) {
   850→    return {
   851→      approved: false,
   852→      reason: `Lot size ${params.lotSize} exceeds max per trade (${config.maxLotPerTrade})`,
   853→      riskAmount: 0,
   854→      riskPercent: 0,
   855→      suggestedLotSize: 0,
   856→      warnings,
   857→    }
   858→  }
   859→
   860→  // 3b. Check total lot per symbol
   861→  const sameSymbolTrades = snapshot.positions.filter((p) => p.symbol === params.symbol)
   862→  const totalSymbolLots = sameSymbolTrades.reduce((s, p) => s + p.lotSize, 0) + params.lotSize
   863→  if (totalSymbolLots > config.maxLotPerSymbol) {
   864→    return {
   865→      approved: false,
   866→      reason: `Total lot for ${params.symbol} (${totalSymbolLots}) would exceed max (${config.maxLotPerSymbol})`,
   867→      riskAmount: 0,
   868→      riskPercent: 0,
   869→      suggestedLotSize: 0,
   870→      warnings,
   871→    }
   872→  }
   873→
   874→  // ---- 4. Minimum lot check (FINEX: 0.01) ----
   875→  if (params.lotSize < 0.01) {
   876→    return {
   877→      approved: false,
   878→      reason: "Minimum lot size is 0.01",
   879→      riskAmount: 0,
   880→      riskPercent: 0,
   881→      suggestedLotSize: 0.01,
   882→      warnings,
   883→    }
   884→  }
   885→
   886→  // ---- 5. Calculate risk amount (including slippage) ----
   887→  let riskAmount = 0
   888→  if (params.sl) {
   889→    riskAmount = params.direction === "BUY"
   890→      ? (params.entryPrice - params.sl) * params.lotSize * 100000
   891→      : (params.sl - params.entryPrice) * params.lotSize * 100000
   892→    riskAmount = Math.abs(riskAmount)
   893→  }
   894→
   895→  // 5b. Slippage modeling: if expected slippage exceeds tolerance, add to risk
   896→  let slippageCost = 0
   897→  if (params.expectedSlippage != null && params.expectedSlippage > 0) {
   898→    if (params.expectedSlippage > config.slippageTolerancePips) {
   899→      slippageCost = (params.expectedSlippage * params.lotSize * 100000) / params.entryPrice
   900→      riskAmount += slippageCost
   901→      warnings.push(
   902→        `Expected slippage (${params.expectedSlippage.toFixed(1)} pips) exceeds tolerance (${config.slippageTolerancePips} pips). Added $${slippageCost.toFixed(2)} to risk amount.`,
   903→      )
   904→      await logRiskEvent({
   905→        eventType: "SLIPPAGE_WARNING",
   906→        severity: "MEDIUM",
   907→        message: `High slippage expected for ${params.symbol}: ${params.expectedSlippage.toFixed(1)} pips (tolerance: ${config.slippageTolerancePips})`,
   908→        details: `Slippage cost $${slippageCost.toFixed(2)} added to risk. Total risk for this trade: $${riskAmount.toFixed(2)}.`,
   909→        actionTaken: "NONE",
   910→      })
   911→    } else {
   912→      // Even within tolerance, include small slippage cost in risk
   913→      slippageCost = (params.expectedSlippage * params.lotSize * 100000) / params.entryPrice
   914→      riskAmount += slippageCost
   915→    }
   916→  }
   917→
   918→  const riskPercent = snapshot.equity > 0 ? (riskAmount / snapshot.equity) * 100 : 0
   919→
   920→  // ---- 6. Risk per trade limit ----
   921→  const maxRiskDollar = (config.maxRiskPerTrade / 100) * snapshot.equity
   922→  let suggestedLotSize = params.lotSize
   923→
   924→  if (riskAmount > maxRiskDollar && params.sl) {
   925→    // Calculate safe lot size
   926→    const pipRisk = params.direction === "BUY"
   927→      ? params.entryPrice - (params.sl || 0)
   928→      : (params.sl || 0) - params.entryPrice
   929→    if (pipRisk > 0) {
   930→      suggestedLotSize = Math.max(0.01, Math.floor((maxRiskDollar / (pipRisk * 100000)) * 100) / 100)
   931→    }
   932→    warnings.push(`Risk $${riskAmount.toFixed(2)} exceeds max $${maxRiskDollar.toFixed(2)} (${config.maxRiskPerTrade}%)`)
   933→  }
   934→
   935→  // ---- 7. No stop loss? ----
   936→  if (!params.sl) {
   937→    warnings.push("No stop loss set - this increases uncontrolled risk")
   938→  }
   939→
   940→  // ---- 8. Correlation risk (existing sector check) ----
   941→  const sector = SYMBOL_SECTORS[params.symbol]
   942→  if (sector) {
   943→    const sectorExposure = snapshot.positions
   944→      .filter((p) => SYMBOL_SECTORS[p.symbol] === sector)
   945→      .reduce((s, p) => s + p.margin, 0)
   946→    const newMargin = (params.entryPrice * suggestedLotSize * 100000) / 25
   947→    const sectorExposurePct = ((sectorExposure + newMargin) / snapshot.equity) * 100
   948→    if (sectorExposurePct > config.maxCorrelatedExposure) {
   949→      warnings.push(`Sector (${sector}) exposure would be ${sectorExposurePct.toFixed(1)}% (max ${config.maxCorrelatedExposure}%)`)
   950→    }
   951→  }
   952→
   953→  // ---- 9. Margin usage after trade ----
   954→  const newMargin = (params.entryPrice * suggestedLotSize * 100000) / 25
   955→  const projectedMarginUsage = ((snapshot.marginUsed + newMargin) / snapshot.equity) * 100
   956→  if (projectedMarginUsage > config.maxMarginUsage) {
   957→    return {
   958→      approved: false,
   959→      reason: `Margin usage would be ${projectedMarginUsage.toFixed(1)}% (max ${config.maxMarginUsage}%)`,
   960→      riskAmount,
   961→      riskPercent,
   962→      suggestedLotSize: 0,
   963→      warnings,
   964→    }
   965→  }
   966→
   967→  // ---- 10. Cooldown after loss ----
   968→  if (config.cooldownAfterLossMinutes > 0) {
   969→    const recentLosses = await db.trade.findMany({
   970→      where: {
   971→        status: "CLOSED",
   972→        pnl: { lt: 0 },
   973→        closeTime: { gte: new Date(Date.now() - config.cooldownAfterLossMinutes * 60 * 1000) },
   974→      },
   975→    })
   976→    if (recentLosses.length >= 3) {
   977→      return {
   978→        approved: false,
   979→        reason: `Cooldown active: ${recentLosses.length} losses in last ${config.cooldownAfterLossMinutes} min`,
   980→        riskAmount,
   981→        riskPercent,
   982→        suggestedLotSize: 0,
   983→        warnings: ["Multiple recent losses detected. Waiting for cooldown period."],
   984→      }
   985→    }
   986→  }
   987→
   988→  // ---- 11. Portfolio-level risk cap ----
   989→  const currentPortfolioRisk = snapshot.positions.reduce((s, p) => s + p.riskAmount, 0)
   990→  const projectedTotalRisk = currentPortfolioRisk + riskAmount
   991→  const projectedPortfolioRiskPct = (projectedTotalRisk / snapshot.equity) * 100
   992→  if (projectedPortfolioRiskPct > config.maxPortfolioRiskPct) {
   993→    await logRiskEvent({
   994→      eventType: "PORTFOLIO_RISK_CAP",
   995→      severity: "HIGH",
   996→      message: `Portfolio risk cap would be exceeded: ${projectedPortfolioRiskPct.toFixed(2)}% (max ${config.maxPortfolioRiskPct}%)`,
   997→      details: `Current portfolio risk: $${currentPortfolioRisk.toFixed(2)} (${((currentPortfolioRisk / snapshot.equity) * 100).toFixed(2)}%). New trade risk: $${riskAmount.toFixed(2)} (${riskPercent.toFixed(2)}%). Combined: $${projectedTotalRisk.toFixed(2)} (${projectedPortfolioRiskPct.toFixed(2)}%).`,
   998→      actionTaken: "TRADE_BLOCKED",
   999→    })
  1000→    return {
  1001→      approved: false,
  1002→      reason: `Portfolio risk would be ${projectedPortfolioRiskPct.toFixed(2)}% (max ${config.maxPortfolioRiskPct}% of equity)`,
  1003→      riskAmount,
  1004→      riskPercent,
  1005→      suggestedLotSize: 0,
  1006→      warnings,
  1007→    }
  1008→  }
  1009→
  1010→  // ---- 12. Leverage utilization cap ----
  1011→  const effectiveLeverage = (params.entryPrice * suggestedLotSize * 100000) / snapshot.equity
  1012→  if (effectiveLeverage > config.maxLeveragePerTrade) {
  1013→    // Suggest smaller lot size
  1014→    const safeLotForLeverage = (config.maxLeveragePerTrade * snapshot.equity) / (params.entryPrice * 100000)
  1015→    const suggestedLotFromLeverage = Math.max(0.01, Math.floor(safeLotForLeverage * 100) / 100)
  1016→    await logRiskEvent({
  1017→      eventType: "LEVERAGE_CAP",
  1018→      severity: "HIGH",
  1019→      message: `Leverage cap exceeded for ${params.symbol}: ${effectiveLeverage.toFixed(1)}x (max ${config.maxLeveragePerTrade}x)`,
  1020→      details: `Entry ${params.entryPrice} x ${suggestedLotSize} lots = ${effectiveLeverage.toFixed(1)}x leverage. Suggested max lot: ${suggestedLotFromLeverage}.`,
  1021→      actionTaken: "TRADE_BLOCKED",
  1022→    })
  1023→    return {
  1024→      approved: false,
  1025→      reason: `Effective leverage ${effectiveLeverage.toFixed(1)}x exceeds max ${config.maxLeveragePerTrade}x. Suggested lot size: ${suggestedLotFromLeverage}`,
  1026→      riskAmount,
  1027→      riskPercent,
  1028→      suggestedLotSize: suggestedLotFromLeverage,
  1029→      warnings: [`Leverage ${effectiveLeverage.toFixed(1)}x exceeds cap ${config.maxLeveragePerTrade}x. Reduce lot to ${suggestedLotFromLeverage}.`],
  1030→    }
  1031→  }
  1032→
  1033→  // ---- 13. Single stock concentration limit ----
  1034→  const sameSymbolMargin = sameSymbolTrades.reduce((s, p) => s + p.margin, 0)
  1035→  const tradeMargin = (params.entryPrice * suggestedLotSize * 100000) / 25
  1036→  const totalSymbolMargin = sameSymbolMargin + tradeMargin
  1037→  const singleStockPct = (totalSymbolMargin / snapshot.equity) * 100
  1038→  if (singleStockPct > config.maxSingleStockPct) {
  1039→    await logRiskEvent({
  1040→      eventType: "CONCENTRATION_LIMIT",
  1041→      severity: "HIGH",
  1042→      message: `Single stock concentration exceeded for ${params.symbol}: ${singleStockPct.toFixed(2)}% (max ${config.maxSingleStockPct}%)`,
  1043→      details: `Current margin in ${params.symbol}: $${sameSymbolMargin.toFixed(2)}. New trade margin: $${tradeMargin.toFixed(2)}. Total: $${totalSymbolMargin.toFixed(2)} = ${singleStockPct.toFixed(2)}% of equity.`,
  1044→      actionTaken: "TRADE_BLOCKED",
  1045→    })
  1046→    return {
  1047→      approved: false,
  1048→      reason: `Single stock ${params.symbol} concentration would be ${singleStockPct.toFixed(2)}% (max ${config.maxSingleStockPct}% of equity)`,
  1049→      riskAmount,
  1050→      riskPercent,
  1051→      suggestedLotSize: 0,
  1052→      warnings,
  1053→    }
  1054→  }
  1055→
  1056→  // ---- 14. Sector concentration limit ----
  1057→  if (sector) {
  1058→    const sectorMarginUsed = snapshot.sectorExposure.find((se) => se.sector === sector)?.marginUsed || 0
  1059→    const totalSectorMargin = sectorMarginUsed + tradeMargin
  1060→    const sectorPct = (totalSectorMargin / snapshot.equity) * 100
  1061→    if (sectorPct > config.maxSectorPct) {
  1062→      await logRiskEvent({
  1063→        eventType: "CONCENTRATION_LIMIT",
  1064→        severity: "HIGH",
  1065→        message: `Sector concentration exceeded for ${sector}: ${sectorPct.toFixed(2)}% (max ${config.maxSectorPct}%)`,
  1066→        details: `Current sector (${sector}) margin: $${sectorMarginUsed.toFixed(2)}. New trade margin: $${tradeMargin.toFixed(2)}. Total: $${totalSectorMargin.toFixed(2)} = ${sectorPct.toFixed(2)}% of equity.`,
  1067→        actionTaken: "TRADE_BLOCKED",
  1068→      })
  1069→      return {
  1070→        approved: false,
  1071→        reason: `Sector ${sector} concentration would be ${sectorPct.toFixed(2)}% (max ${config.maxSectorPct}% of equity)`,
  1072→        riskAmount,
  1073→        riskPercent,
  1074→        suggestedLotSize: 0,
  1075→        warnings,
  1076→      }
  1077→    }
  1078→  }
  1079→
  1080→  // ---- 15. Reserve capital check ----
  1081→  const requiredReserve = (config.reserveCapitalPct / 100) * snapshot.equity
  1082→  const freeMarginAfterTrade = snapshot.freeMargin - tradeMargin
  1083→  if (freeMarginAfterTrade < requiredReserve) {
  1084→    const currentReservePct = (snapshot.freeMargin / snapshot.equity) * 100
  1085→    return {
  1086→      approved: false,
  1087→      reason: `Insufficient reserve capital: $${freeMarginAfterTrade.toFixed(2)} remaining (need ${config.reserveCapitalPct}% = $${requiredReserve.toFixed(2)}). Current reserve: ${currentReservePct.toFixed(1)}%.`,
  1088→      riskAmount,
  1089→      riskPercent,
  1090→      suggestedLotSize: 0,
  1091→      warnings: [`Reserve capital requirement: ${config.reserveCapitalPct}% of equity ($${requiredReserve.toFixed(2)}).`],
  1092→    }
  1093→  }
  1094→
  1095→  // ---- Apply position size reduction from PROACTIVE_60 zone ----
  1096→  if (positionSizeReduction > 0) {
  1097→    suggestedLotSize = Math.max(0.01, Math.round(suggestedLotSize * (1 - positionSizeReduction) * 100) / 100)
  1098→    // Recalculate risk amount with reduced lot
  1099→    if (params.sl) {
  1100→      riskAmount = params.direction === "BUY"
  1101→        ? (params.entryPrice - params.sl) * suggestedLotSize * 100000
  1102→        : (params.sl - params.entryPrice) * suggestedLotSize * 100000
  1103→      riskAmount = Math.abs(riskAmount) + slippageCost
  1104→    }
  1105→  }
  1106→
  1107→  // ---- Apply dynamic scaling factor ----
  1108→  if (snapshot.scalingFactor < 1.0) {
  1109→    const scaledLot = Math.max(0.01, Math.round(suggestedLotSize * snapshot.scalingFactor * 100) / 100)
  1110→    if (scaledLot < suggestedLotSize) {
  1111→      warnings.push(
  1112→        `Dynamic risk scaling applied: lot reduced from ${suggestedLotSize} to ${scaledLot} (factor: ${snapshot.scalingFactor})`,
  1113→      )
  1114→      suggestedLotSize = scaledLot
  1115→    }
  1116→  }
  1117→
  1118→  // ---- Approved ----
  1119→  return {
  1120→    approved: true,
  1121→    riskAmount: Math.round(riskAmount * 100) / 100,
  1122→    riskPercent: Math.round(riskPercent * 100) / 100,
  1123→    suggestedLotSize: Math.round(suggestedLotSize * 100) / 100,
  1124→    warnings,
  1125→    positionSizeReduction: positionSizeReduction > 0 ? positionSizeReduction : undefined,
  1126→  }
  1127→}

  1166→function calculateRiskScore(factors: {
  1167→  dailyPnlPercent: number
  1168→  marginUsagePercent: number
  1169→  currentDrawdown: number
  1170→  maxDrawdown: number
  1171→  config: RiskConfigData
  1172→  openPositionCount: number
  1173→  marginLevelPercent: number
  1174→}): number {
  1175→  let score = 0
  1176→
  1177→  // Daily loss contribution (0-3 points)
  1178→  const dailyLossPct = Math.abs(Math.min(0, factors.dailyPnlPercent))
  1179→  score += Math.min(3, (dailyLossPct / factors.config.maxDailyLoss) * 3)
  1180→
  1181→  // Margin usage (0-2 points)
  1182→  score += Math.min(2, (factors.marginUsagePercent / factors.config.maxMarginUsage) * 2)
  1183→
  1184→  // Drawdown (0-3 points)
  1185→  score += Math.min(3, (factors.currentDrawdown / factors.config.maxDrawdown) * 3)
  1186→
  1187→  // Margin level proximity to stop out (0-2 points)
  1188→  // Enhanced to account for 4-zone system
  1189→  if (factors.marginLevelPercent > 0) {
  1190→    if (factors.marginLevelPercent <= factors.config.stopOutLevel) {
  1191→      score += 2 // STOP_OUT
  1192→    } else if (factors.marginLevelPercent <= factors.config.marginCallLevel) {
  1193→      score += 2 // MARGIN_CALL
  1194→    } else if (factors.config.proactiveMcLevel60 && factors.marginLevelPercent <= 60) {
  1195→      score += 1.5 // PROACTIVE_60
  1196→    } else if (factors.config.proactiveMcLevel70 && factors.marginLevelPercent <= 70) {
  1197→      score += 0.75 // PROACTIVE_70
  1198→    }
  1199→  }
  1200→
  1201→  return Math.min(10, Math.round(score * 10) / 10)
  1202→}

  1204→function getRiskLevel(score: number): RiskSnapshot["riskLevel"] {
  1205→  if (score <= 2) return "LOW"
  1206→  if (score <= 4) return "MODERATE"
  1207→  if (score <= 6) return "ELEVATED"
  1208→  if (score <= 8) return "HIGH"
  1209→  return "CRITICAL"

  1212→function generateRecommendations(ctx: {

  1272→/**
  1273→ * Assess gap risk for a potential trade entry.
  1274→ *
  1275→ * Uses ATR-based gap estimation (volatility * 2.5 as max expected gap).
  1276→ * If entering near market close (within 30 min of 15:00 WIB), gap risk is increased by 50%.
  1277→ */
  1278→export async function assessGapRisk(params: {
  1279→  symbol: string
  1280→  direction: string
  1281→  entryPrice: number
  1282→  equity: number
  1283→  volatility?: number
  1284→}): Promise<GapRiskResult> {
  1285→  const config = await getRiskConfig()
  1286→  const vol = params.volatility ?? 0.01 // Default 1% daily volatility if not provided
  1287→
  1288→  // ATR-based max expected gap = volatility * 2.5
  1289→  let estimatedMaxGapPct = vol * 2.5 * 100
  1290→
  1291→  // If near market close (within 30 min of 15:00 WIB), increase gap risk by 50%
  1292→  const now = new Date()
  1293→  // WIB is UTC+7
  1294→  const wibOffset = 7 * 60 * 60 * 1000
  1295→  const wibHour = new Date(now.getTime() + wibOffset).getUTCHours()
  1296→  const wibMinute = new Date(now.getTime() + wibOffset).getUTCMinutes()
  1297→  const minutesSinceOpen = wibHour * 60 + wibMinute
  1298→  // Market closes at 15:00 WIB (900 minutes). Within 30 min = >= 870 minutes.
  1299→  const isNearClose = minutesSinceOpen >= 870 && minutesSinceOpen <= 900
  1300→
  1301→  if (isNearClose) {
  1302→    estimatedMaxGapPct *= 1.5
  1303→    logger.info("RISK_MANAGEMENT", `Gap risk boosted near close for ${params.symbol}`, {
  1304→      metadata: { isNearClose: true, estimatedMaxGapPct },
  1305→    })
  1306→  }
  1307→
  1308→  estimatedMaxGapPct = Math.round(estimatedMaxGapPct * 100) / 100
  1309→
  1310→  // Risk amount = estimated gap % * entry price * typical 1 lot position
  1311→  const riskAmount = (estimatedMaxGapPct / 100) * params.entryPrice * 100000
  1312→
  1313→  // Determine severity
  1314→  let severity = "LOW"
  1315→  let hasGapRisk = false
  1316→  let recommendation = "Acceptable gap risk"
  1317→
  1318→  if (estimatedMaxGapPct > config.gapRiskMaxPct) {
  1319→    severity = "HIGH"
  1320→    hasGapRisk = true
  1321→    recommendation = "Avoid new positions near close"
  1322→  } else if (estimatedMaxGapPct > config.gapRiskAlertPct) {
  1323→    severity = "MEDIUM"
  1324→    hasGapRisk = true
  1325→    recommendation = "Reduce position size"
  1326→  }
  1327→
  1328→  if (hasGapRisk && !isMarketOpen(now)) {
  1329→    // Outside market hours, any gap risk is elevated
  1330→    severity = "HIGH"
  1331→    recommendation = "Avoid new positions near close"
  1332→  }
  1333→
  1334→  return {
  1335→    hasGapRisk,
  1336→    estimatedMaxGapPct,
  1337→    riskAmount: Math.round(riskAmount * 100) / 100,
  1338→    recommendation,
  1339→    severity,
  1340→  }
  1341→}

  1344→// PHASE 3: VOLATILITY REGIME DETECTION
  1345→// ============================================
  1346→
  1347→/**
  1348→ * Detect the current volatility regime and return an appropriate risk multiplier.
  1349→ *
  1350→ * - HIGH_VOLATILITY: recentVol > avgVol * 1.5 → riskMultiplier = highVolRiskReduction (default 0.5)
  1351→ * - LOW_VOLATILITY:  recentVol < avgVol * 0.5 → riskMultiplier = lowVolRiskReduction (default 0.8)
  1352→ * - NORMAL:           otherwise                 → riskMultiplier = 1.0
  1353→ */
  1354→export function detectVolatilityRegime(params: {
  1355→  recentVolatility: number
  1356→  avgVolatility: number
  1357→}): VolatilityRegimeResult {
  1358→  const { recentVolatility, avgVolatility } = params
  1359→
  1360→  // When no meaningful data, default to NORMAL
  1361→  if (avgVolatility <= 0 || recentVolatility <= 0) {
  1362→    return {
  1363→      regime: "NORMAL",
  1364→      riskMultiplier: 1.0,
  1365→      reason: "Insufficient volatility data; defaulting to NORMAL regime",
  1366→    }
  1367→  }
  1368→
  1369→  const ratio = recentVolatility / avgVolatility
  1370→
  1371→  if (ratio > 1.5) {
  1372→    return {
  1373→      regime: "HIGH_VOLATILITY",
  1374→      riskMultiplier: 0.5,
  1375→      reason: `Recent volatility (${(recentVolatility * 100).toFixed(2)}%) is ${(ratio).toFixed(1)}x above average (${(avgVolatility * 100).toFixed(2)}%). Reducing risk to 50%.`,
  1376→    }
  1377→  }
  1378→
  1379→  if (ratio < 0.5) {
  1380→    return {
  1381→      regime: "LOW_VOLATILITY",
  1382→      riskMultiplier: 0.8,
  1383→      reason: `Recent volatility (${(recentVolatility * 100).toFixed(2)}%) is ${(1/ratio).toFixed(1)}x below average (${(avgVolatility * 100).toFixed(2)}%). Mild risk reduction to 80%.`,
  1384→    }
  1385→  }
  1386→
  1387→  return {
  1388→    regime: "NORMAL",
  1389→    riskMultiplier: 1.0,
  1390→    reason: `Volatility ratio ${(ratio).toFixed(2)}x within normal bounds (0.5-1.5x). No adjustment needed.`,
  1391→  }
  1392→}

  1398→/**
  1399→ * Auto-resolve risk events that have been unresolved for too long.
  1400→ *
  1401→ * Finds all RiskEvent where resolved=false AND createdAt < (now - maxAgeMinutes).
  1402→ * Sets resolved=true, resolvedAt=now, actionTaken includes "AUTO_RESOLVED".

  1405→export async function autoResolveStaleRiskEvents(maxAgeMinutes: number = 60): Promise<number> {
  1406→  const cutoff = new Date(Date.now() - maxAgeMinutes * 60 * 1000)
  1407→
  1408→  const staleEvents = await db.riskEvent.findMany({
  1409→    where: {
  1410→      resolved: false,
  1411→      createdAt: { lt: cutoff },
  1412→    },
  1413→  })
  1414→
  1415→  if (staleEvents.length === 0) {
  1416→    return 0
  1417→  }
  1418→
  1419→  const now = new Date()
  1420→  let resolvedCount = 0
  1421→
  1422→  for (const event of staleEvents) {
  1423→    try {
  1424→      await db.riskEvent.update({
  1425→        where: { id: event.id },
  1426→        data: {
  1427→          resolved: true,
  1428→          resolvedAt: now,
  1429→          actionTaken: event.actionTaken
  1430→            ? `${event.actionTaken}; AUTO_RESOLVED`
  1431→            : "AUTO_RESOLVED",
  1432→        },
  1433→      })
  1434→      logger.info("RISK_MANAGEMENT", `Auto-resolved stale risk event: ${event.eventType}`, {
  1435→        metadata: {
  1436→          eventId: event.id,
  1437→          eventType: event.eventType,
  1438→          severity: event.severity,
  1439→          ageMinutes: Math.round((now.getTime() - event.createdAt.getTime()) / 60000),
  1440→        },
  1441→      })
  1442→      resolvedCount++
  1443→    } catch (err) {
  1444→      logger.error("RISK_MANAGEMENT", `Failed to auto-resolve risk event ${event.id}`, {
  1445→        details: err instanceof Error ? err.stack : undefined,
  1446→      })
  1447→    }
  1448→  }
  1449→
  1450→  if (resolvedCount > 0) {
  1451→    logger.info("RISK_MANAGEMENT", `Auto-resolved ${resolvedCount} stale risk events (max age: ${maxAgeMinutes}min)`, {
  1452→      metadata: { resolvedCount, maxAgeMinutes },
  1453→    })
  1454→  }
  1455→
  1456→  return resolvedCount
  1457→}

  1463→/**
  1464→ * Calculate a sector-level correlation matrix from open positions.
  1465→ *
  1466→ * Groups positions by sector, calculates sector exposure as % of total margin,
  1467→ * and assigns correlation groups based on position count per sector.
  1468→ */
  1469→export function calculateCorrelationMatrix(
  1470→  openPositions: Array<{ symbol: string; sector: string; margin: number; pnl: number }>,
  1471→): CorrelationMatrixResult {
  1472→  const config = DEFAULT_CONFIG // Use defaults for threshold comparison
  1473→
  1474→  // Group by sector
  1475→  const sectorMap = new Map<string, { margin: number; count: number; pnl: number }>()
  1476→  let totalMargin = 0
  1477→
  1478→  for (const pos of openPositions) {
  1479→    const sector = pos.sector || "Unknown"
  1480→    const existing = sectorMap.get(sector)
  1481→    if (existing) {
  1482→      existing.margin += pos.margin
  1483→      existing.count += 1
  1484→      existing.pnl += pos.pnl
  1485→    } else {
  1486→      sectorMap.set(sector, { margin: pos.margin, count: 1, pnl: pos.pnl })
  1487→    }
  1488→    totalMargin += pos.margin
  1489→  }
  1490→
  1491→  const sectors: CorrelationMatrixResult["sectors"] = []
  1492→
  1493→  for (const [sector, data] of sectorMap.entries()) {
  1494→    const exposure = totalMargin > 0
  1495→      ? Math.round((data.margin / totalMargin) * 10000) / 100
  1496→      : 0
  1497→
  1498→    // Assign correlation group based on position count
  1499→    let correlationGroup: string
  1500→    if (data.count > 3) {
  1501→      correlationGroup = "HIGH_CORRELATION"
  1502→    } else if (data.count >= 2) {
  1503→      correlationGroup = "MEDIUM"
  1504→    } else {
  1505→      correlationGroup = "LOW"
  1506→    }
  1507→
  1508→    sectors.push({
  1509→      sector,
  1510→      exposure,
  1511→      positionCount: data.count,
  1512→      correlationGroup,
  1513→    })
  1514→  }
  1515→
  1516→  // Sort by exposure descending
  1517→  sectors.sort((a, b) => b.exposure - a.exposure)
  1518→
  1519→  return { sectors }
  1520→}
