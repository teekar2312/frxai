    80→const LOT_STEP = 0.01

    84→// ---- Position Sizing ----
    85→
    86→/**
    87→ * Calculate the optimal lot size for a trade.
    88→ * Enhanced with commission-aware sizing, reserve capital enforcement,
    89→ * deployed margin cap, and performance-based dynamic scaling.
    90→ */
    91→export async function calculatePositionSize(params: {
    92→  symbol: string
    93→  direction: "BUY" | "SELL"
    94→  entryPrice: number
    95→  sl?: number | null
    96→  equity: number
    97→  method?: SizingMethod
    98→  fixedDollarRisk?: number
    99→  kellyFraction?: number // default 0.5 (half-Kelly)
   100→  commissionPerLot?: number // default $1 (FINEX)
   101→  reserveCapitalPct?: number // default from RiskConfig
   102→}): Promise<PositionSizeResult> {
   103→  const config = await getRiskConfig()
   104→  const method = params.method || "FIXED_FRACTIONAL"
   105→  const sl = params.sl
   106→  const commissionPerLot = params.commissionPerLot ?? DEFAULT_COMMISSION_PER_LOT
   107→  const reserveCapitalPct = params.reserveCapitalPct ?? config.reserveCapitalPct
   108→
   109→  // --- Performance-Based Dynamic Scaling ---
   110→  const scalingFactor = await calculateScalingFactor()
   111→
   112→  // Calculate pip risk (price distance to stop loss)
   113→  let pipRisk = 0
   114→  if (sl) {
   115→    pipRisk = params.direction === "BUY"
   116→      ? params.entryPrice - sl
   117→      : sl - params.entryPrice
   118→    pipRisk = Math.abs(pipRisk)
   119→  }
   120→
   121→  // If no SL, use a default 2% price risk as fallback
   122→  const effectivePipRisk = pipRisk > 0 ? pipRisk : params.entryPrice * 0.02
   123→
   124→  // --- Base Risk Amount by Method ---
   125→  let baseRiskAmount: number
   126→  let methodReasoning: string
   127→
   128→  switch (method) {
   129→    case "FIXED_FRACTIONAL": {
   130→      baseRiskAmount = (config.maxRiskPerTrade / 100) * params.equity
   131→      methodReasoning = `Fixed fractional: ${config.maxRiskPerTrade}% of equity ($${params.equity.toFixed(2)}) = $${baseRiskAmount.toFixed(2)}`
   132→      break
   133→    }
   134→    case "KELLY": {
   135→      const kellyFraction = params.kellyFraction || 0.5
   136→      const kellyResult = await calculateKelly(params.equity, kellyFraction)
   137→      baseRiskAmount = kellyResult.suggestedRisk
   138→      methodReasoning = `Half-Kelly criterion: Kelly suggests ${kellyResult.kellyPercent.toFixed(1)}%, using ${kellyFraction * 100}% = $${baseRiskAmount.toFixed(2)}`
   139→      break
   140→    }
   141→    case "FIXED_DOLLAR": {
   142→      baseRiskAmount = params.fixedDollarRisk || 100
   143→      methodReasoning = `Fixed dollar risk: $${baseRiskAmount.toFixed(2)} per trade`
   144→      break
   145→    }
   146→    case "ANTI_MARTINGALE": {
   147→      const recentTrades = await getRecentTradeResults(5)
   148→      let multiplier = 1
   149→      for (const t of recentTrades) {
   150→        if (t > 0) multiplier = Math.min(multiplier + 0.5, 3)
   151→        else break
   152→      }
   153→      baseRiskAmount = (config.maxRiskPerTrade / 100) * params.equity * multiplier
   154→      methodReasoning = `Anti-Martingale: base ${config.maxRiskPerTrade}% * ${multiplier.toFixed(1)}x (consecutive wins) = $${baseRiskAmount.toFixed(2)}`
   155→      break
   156→    }
   157→    default:
   158→      baseRiskAmount = (config.maxRiskPerTrade / 100) * params.equity
   159→      methodReasoning = `Default fixed fractional: $${baseRiskAmount.toFixed(2)}`
   160→  }
   161→
   162→  // --- Apply Scaling Factor ---
   163→  const scaledRiskAmount = baseRiskAmount * scalingFactor
   164→  let reasoning = methodReasoning
   165→  if (scalingFactor !== 1.0) {
   166→    reasoning += ` | Scaled ${scalingFactor}x -> $${scaledRiskAmount.toFixed(2)}`
   167→  }
   168→
   169→  // --- Commission-Aware Lot Sizing ---
   170→  // Risk budget must cover both SL risk AND commission cost.
   171→  // totalRisk = SL_risk + commission
   172→  // We need to find a lot size L where:
   173→  //   L * effectivePipRisk * PIP_VALUE_PER_LOT + L * commissionPerLot <= scaledRiskAmount
   174→  //   L * (effectivePipRisk * PIP_VALUE_PER_LOT + commissionPerLot) <= scaledRiskAmount
   175→  //   L <= scaledRiskAmount / (effectivePipRisk * PIP_VALUE_PER_LOT + commissionPerLot)
   176→  const costPerLot = effectivePipRisk * PIP_VALUE_PER_LOT + commissionPerLot
   177→  const rawLotSize = costPerLot > 0 ? scaledRiskAmount / costPerLot : 0
   178→  let suggestedLotSize = Math.floor(rawLotSize / LOT_STEP) * LOT_STEP
   179→  suggestedLotSize = Math.max(MIN_LOT, suggestedLotSize)
   180→
   181→  // Calculate commission cost for the suggested lot
   182→  const commissionCost = Math.round(suggestedLotSize * commissionPerLot * 100) / 100
   183→  const slRiskAmount = Math.round(suggestedLotSize * effectivePipRisk * PIP_VALUE_PER_LOT * 100) / 100
   184→  const netRiskAfterCommission = Math.round((slRiskAmount + commissionCost) * 100) / 100
   185→
   186→  reasoning += ` | Commission: $${commissionCost.toFixed(2)} (${suggestedLotSize} lot x $${commissionPerLot}/lot)`
   187→  reasoning += ` | Net risk (SL + commission): $${netRiskAfterCommission.toFixed(2)}`
   188→
   189→  // --- Reserve Capital Enforcement ---
   190→  const maxDeployable = params.equity * (1 - reserveCapitalPct / 100)
   191→  const contractValue = suggestedLotSize * params.entryPrice * 100000
   192→  const marginRequired = contractValue / LEVERAGE
   193→  let reserveCheckApplied = false
   194→
   195→  if (marginRequired > maxDeployable) {
   196→    // Reduce lot size so margin fits within deployable capital
   197→    const maxLotForReserve = (maxDeployable * LEVERAGE) / (params.entryPrice * 100000)
   198→    const reducedLot = Math.floor(maxLotForReserve / LOT_STEP) * LOT_STEP
   199→    if (reducedLot < suggestedLotSize) {
   200→      suggestedLotSize = Math.max(MIN_LOT, reducedLot)
   201→      reserveCheckApplied = true
   202→      // Recalculate values after reserve reduction
   203→    }
   204→    reasoning += ` | Reserve ${reserveCapitalPct}% enforced: max deployable $${maxDeployable.toFixed(2)}, reduced lot to ${suggestedLotSize}`
   205→  }
   206→
   207→  // --- Max Capital Deployment (Open Positions Check) ---
   208→  const openPositions = await db.trade.findMany({ where: { status: "OPEN" } })
   209→  const existingDeployedMargin = openPositions.reduce((sum, t) => sum + (t.margin || 0), 0)
   210→  const newMarginRequired = (suggestedLotSize * params.entryPrice * 100000) / LEVERAGE
   211→  let deployedMarginCheckApplied = false
   212→
   213→  if (existingDeployedMargin + newMarginRequired > maxDeployable) {
   214→    const availableForNew = maxDeployable - existingDeployedMargin
   215→    if (availableForNew > 0) {
   216→      const maxLotForDeployed = (availableForNew * LEVERAGE) / (params.entryPrice * 100000)
   217→      const reducedLot = Math.floor(maxLotForDeployed / LOT_STEP) * LOT_STEP
   218→      if (reducedLot < suggestedLotSize && reducedLot >= MIN_LOT) {
   219→        suggestedLotSize = reducedLot
   220→        deployedMarginCheckApplied = true
   221→        reasoning += ` | Deployed margin check: $${existingDeployedMargin.toFixed(2)} already deployed, reduced lot to ${suggestedLotSize}`
   222→      } else if (reducedLot < MIN_LOT) {
   223→        // Not enough deployable capital for even minimum lot
   224→        suggestedLotSize = 0
   225→        reasoning += ` | REJECTED: Not enough deployable capital ($${availableForNew.toFixed(2)} available, $${newMarginRequired.toFixed(2)} needed)`
   226→      }
   227→    } else {
   228→      suggestedLotSize = 0
   229→      reasoning += ` | REJECTED: All deployable capital ($${maxDeployable.toFixed(2)}) already deployed ($${existingDeployedMargin.toFixed(2)})`
   230→    }
   231→  }
   232→
   233→  // Cap at max lot per trade
   234→  if (suggestedLotSize > 0) {
   235→    suggestedLotSize = Math.min(suggestedLotSize, config.maxLotPerTrade)
   236→  }
   237→
   238→  // --- Final Derived Values ---
   239→  const finalContractValue = suggestedLotSize * params.entryPrice * 100000
   240→  const finalMarginRequired = finalContractValue / LEVERAGE
   241→  const finalCommissionCost = Math.round(suggestedLotSize * commissionPerLot * 100) / 100
   242→  const finalSlRiskAmount = Math.round(suggestedLotSize * effectivePipRisk * PIP_VALUE_PER_LOT * 100) / 100
   243→  const finalNetRisk = Math.round((finalSlRiskAmount + finalCommissionCost) * 100) / 100
   244→  const riskPercent = params.equity > 0 ? (finalNetRisk / params.equity) * 100 : 0
   245→
   246→  return {
   247→    suggestedLotSize: Math.round(suggestedLotSize * 100) / 100,
   248→    riskAmount: Math.round((baseRiskAmount * scalingFactor) * 100) / 100,
   249→    riskPercent: Math.round(riskPercent * 100) / 100,
   250→    method,
   251→    pipRisk: Math.round(effectivePipRisk * 100) / 100,
   252→    contractValue: Math.round(finalContractValue * 100) / 100,
   253→    marginRequired: Math.round(finalMarginRequired * 100) / 100,
   254→    reasoning,
   255→    commissionCost: finalCommissionCost,
   256→    netRiskAfterCommission: finalNetRisk,
   257→    scalingFactor,
   258→    reserveCheckApplied,
   259→    deployedMarginCheckApplied,
   260→  }
   261→}

   263→/**
   264→ * Kelly Criterion position sizing.
   265→ */
   266→async function calculateKelly(equity: number, fraction: number): Promise<{
   267→  kellyPercent: number
   268→  suggestedRisk: number
   269→}> {
   270→  const closedTrades = await db.trade.findMany({
   271→    where: { status: "CLOSED" },
   272→    orderBy: { closeTime: "desc" },
   273→    take: 100,
   274→  })
   275→
   276→  if (closedTrades.length < 10) {
   277→    return { kellyPercent: 1.0, suggestedRisk: (fraction / 100) * equity }
   278→  }
   279→
   280→  const wins = closedTrades.filter((t) => t.pnl > 0)
   281→  const losses = closedTrades.filter((t) => t.pnl < 0)
   282→  const winRate = wins.length / closedTrades.length
   283→  const avgWin = wins.length > 0 ? wins.reduce((s, t) => s + t.pnl, 0) / wins.length : 0
   284→  const avgLoss = losses.length > 0 ? Math.abs(losses.reduce((s, t) => s + t.pnl, 0) / losses.length) : 1
   285→
   286→  const R = avgLoss > 0 ? avgWin / avgLoss : 0
   287→  const kellyPercent = R > 0 ? (winRate - (1 - winRate) / R) * 100 : 0
   288→  const clampedKelly = Math.max(-5, Math.min(25, kellyPercent))
   289→
   290→  return {
   291→    kellyPercent: clampedKelly,
   292→    suggestedRisk: (Math.max(0, clampedKelly) * fraction / 100) * equity,
   293→  }
   294→}

   308→// ---- Risk of Ruin ----
   309→
   310→/**
   311→ * Estimate the probability of ruin (losing X% of account).
   312→ * Uses the formula: P(ruin) = ((1 - edge) / (1 + edge))^units
   313→ * where edge = W - (1-W)*avgLoss/avgWin and units = account / (risk per trade)
   314→ */
   315→export function calculateRiskOfRuin(input: RiskOfRuinInput): {
   316→  probability: number
   317→  interpretation: string
   318→  recommendation: string
   319→} {
   320→  if (input.winRate <= 0 || input.avgLoss <= 0) {
   321→    return {
   322→      probability: 100,
   323→      interpretation: "Cannot calculate - insufficient winning trades or zero avg loss",
   324→      recommendation: "Improve strategy win rate or reduce loss size",
   325→    }
   326→  }
   327→
   328→  const W = input.winRate / 100
   329→  const L = 1 - W
   330→  const R = input.avgWin / input.avgLoss
   331→
   332→  if (R <= 0) {
   333→    return {
   334→      probability: 100,
   335→      interpretation: "Average win is zero or negative",
   336→      recommendation: "Strategy needs improvement - avg win must exceed avg loss",
   337→    }
   338→  }
   339→
   340→  const edge = W - L / R
   341→  const riskFraction = input.riskPerTrade / 100
   342→
   343→  if (edge <= 0) {
   344→    return {
   345→      probability: 100,
   346→      interpretation: `Negative edge (${(edge * 100).toFixed(2)}%) - strategy is unprofitable statistically`,
   347→      recommendation: "Do not trade this strategy. Win rate and/or reward-risk ratio must improve.",
   348→    }
   349→  }
   350→
   351→  const q = (1 - edge) / (1 + edge)
   352→  const units = 1 / riskFraction
   353→  const ror = Math.pow(Math.max(0, q), units) * 100
   354→
   355→  let interpretation: string
   356→  let recommendation: string
   357→
   358→  if (ror < 1) {
   359→    interpretation = "Very Low Risk"
   360→    recommendation = "Strategy has a strong statistical edge. Safe to trade with current parameters."
   361→  } else if (ror < 5) {
   362→    interpretation = "Low Risk"
   363→    recommendation = "Acceptable risk level. Continue monitoring performance metrics."
   364→  } else if (ror < 15) {
   365→    interpretation = "Moderate Risk"
   366→    recommendation = "Consider reducing position sizes or improving win rate to lower risk of ruin."
   367→  } else if (ror < 30) {
   368→    interpretation = "High Risk"
   369→    recommendation = "Significant risk of account drawdown. Reduce risk per trade significantly."
   370→  } else {
   371→    interpretation = "Extreme Risk"
   372→    recommendation = "Do NOT trade this strategy with current parameters. High probability of account destruction."
   373→  }
   374→
   375→  return {
   376→    probability: Math.min(100, Math.round(ror * 100) / 100),
   377→    interpretation,
   378→    recommendation,
   379→  }
   380→}

   382→// ---- Drawdown Recovery Model ----
   383→
   384→/**
   385→ * Calculate the recovery needed for any drawdown level.
   386→ * Formula: recoveryNeeded = (drawdown / (100 - drawdown)) * 100
   387→ *
   388→ * Examples:
   389→ *   10% DD -> 11.1% recovery needed
   390→ *   20% DD -> 25% recovery needed
   391→ *   50% DD -> 100% recovery needed
   392→ */
   393→export function calculateDrawdownRecovery(drawdownPct: number): DrawdownRecoveryResult {
   394→  const dd = Math.max(0, Math.min(99, drawdownPct)) // Clamp 0-99%
   395→
   396→  const recoveryNeeded = dd >= 100 ? Infinity : (dd / (100 - dd)) * 100
   397→
   398→  // Recovery strategy and risk reduction based on drawdown severity
   399→  let strategy: string
   400→  let riskReductionPct: number
   401→
   402→  if (dd < 5) {
   403→    strategy = "NORMAL: No action needed. Drawdown within acceptable range. Continue normal trading operations."
   404→    riskReductionPct = 0
   405→  } else if (dd < 10) {
   406→    strategy = "CAUTION: Minor drawdown. Review recent trades for patterns. Tighten stop losses by 10%. Avoid high-risk setups."
   407→    riskReductionPct = 10
   408→  } else if (dd < 15) {
   409→    strategy = "ELEVATED: Moderate drawdown. Reduce position sizes by 25%. Focus on highest-confidence setups only. Consider pausing new positions temporarily."
   410→    riskReductionPct = 25
   411→  } else if (dd < 20) {
   412→    strategy = "HIGH: Significant drawdown. Reduce risk to 50% of normal. Trade only A+ setups. Implement stricter entry criteria. Review and potentially pause underperforming strategies."
   413→    riskReductionPct = 50
   414→  } else if (dd < 30) {
   415→    strategy = "CRITICAL: Severe drawdown. Reduce risk to 75% of normal. Consider full trading pause for 24-48 hours. Perform strategy review. Check if market regime has changed."
   416→    riskReductionPct = 75
   417→  } else if (dd < 50) {
   418→    strategy = "EMERGENCY: Extreme drawdown. Stop all new trading immediately. Close weakest positions. Full strategy audit required. Resume only with micro-lots after analysis."
   419→    riskReductionPct = 90
   420→  } else {
   421→    strategy = "CATASTROPHIC: Account severely impaired. ${recoveryNeeded.toFixed(1)}% gain required to recover. Full trading halt. Account preservation is the only priority. Consider account reset or external capital injection after thorough review."
   422→    riskReductionPct = 100
   423→  }
   424→
   425→  // Fix the template literal for 50%+ case
   426→  if (dd >= 50) {
   427→    strategy = `CATASTROPHIC: Account severely impaired. ${recoveryNeeded.toFixed(1)}% gain required to recover. Full trading halt. Account preservation is the only priority. Consider account reset or external capital injection after thorough review.`
   428→  }
   429→
   430→  return {
   431→    drawdownPct: Math.round(dd * 100) / 100,
   432→    recoveryNeeded: Math.round(recoveryNeeded * 100) / 100,
   433→    strategy,
   434→    riskReductionPct,
   435→  }
   436→}

   438→// ---- Performance-Based Dynamic Scaling ----
   439→
   440→/**
   441→ * Calculate a dynamic scaling factor based on recent trading performance.
   442→ * Analyzes the last 30 closed trades (or all if fewer) for:
   443→ *   - Rolling win rate
   444→ *   - Profit factor (gross profit / gross loss)
   445→ *   - Sharpe-like ratio (mean PnL / std dev of PnL)
   446→ *
   447→ * Scaling rules:
   448→ *   WR > 60% AND PF > 1.5  -> scale up to 1.25x (max)
   449→ *   WR > 55% AND PF > 1.2  -> scale up to 1.1x
   450→ *   WR < 40% OR PF < 0.8  -> scale down to 0.5x
   451→ *   WR < 45% OR PF < 1.0  -> scale down to 0.75x
   452→ *   Otherwise              -> 1.0x (neutral)
   453→ */
   454→export async function calculateScalingFactor(): Promise<number> {
   455→  try {
   456→    const closedTrades = await db.trade.findMany({
   457→      where: { status: "CLOSED" },
   458→      orderBy: { closeTime: "desc" },
   459→      take: 30,
   460→    })
   461→
   462→    // Not enough trades to evaluate - use neutral scaling
   463→    if (closedTrades.length < 5) {
   464→      return 1.0
   465→    }
   466→
   467→    // Calculate rolling win rate
   468→    const wins = closedTrades.filter((t) => t.pnl > 0)
   469→    const winRate = (wins.length / closedTrades.length) * 100
   470→
   471→    // Calculate profit factor
   472→    const grossProfit = wins.reduce((sum, t) => sum + t.pnl, 0)
   473→    const losses = closedTrades.filter((t) => t.pnl < 0)
   474→    const grossLoss = Math.abs(losses.reduce((sum, t) => sum + t.pnl, 0))
   475→    const profitFactor = grossLoss > 0 ? grossProfit / grossLoss : 0
   476→
   477→    // Calculate Sharpe-like ratio (mean / stddev of PnL)
   478→    const pnls = closedTrades.map((t) => t.pnl)
   479→    const mean = pnls.reduce((s, v) => s + v, 0) / pnls.length
   480→    const variance = pnls.reduce((s, v) => s + (v - mean) ** 2, 0) / pnls.length
   481→    const stddev = Math.sqrt(variance)
   482→    // We don't use this for scaling decisions directly but log it
   483→    const sharpeLike = stddev > 0 ? mean / stddev : 0
   484→
   485→    // Determine scaling factor based on rules
   486→    let factor: number
   487→    if (winRate > 60 && profitFactor > 1.5) {
   488→      factor = 1.25
   489→    } else if (winRate > 55 && profitFactor > 1.2) {
   490→      factor = 1.1
   491→    } else if (winRate < 40 || profitFactor < 0.8) {
   492→      factor = 0.5
   493→    } else if (winRate < 45 || profitFactor < 1.0) {
   494→      factor = 0.75
   495→    } else {
   496→      factor = 1.0
   497→    }
   498→
   499→    logger.debug("MONEY_MANAGEMENT", `Scaling factor calculated: ${factor}x (WR: ${winRate.toFixed(1)}%, PF: ${profitFactor.toFixed(2)}, Sharpe-like: ${sharpeLike.toFixed(2)})`, {
   500→      metadata: {
   501→        scalingFactor: factor,
   502→        winRate: Math.round(winRate * 100) / 100,
   503→        profitFactor: Math.round(profitFactor * 100) / 100,
   504→        sharpeLike: Math.round(sharpeLike * 100) / 100,
   505→        sampleSize: closedTrades.length,
   506→      },
   507→    })
   508→
   509→    return factor
   510→  } catch (err) {
   511→    logger.error("MONEY_MANAGEMENT", "Error calculating scaling factor, defaulting to 1.0", {
   512→      details: err instanceof Error ? err.stack : undefined,
   513→    })
   514→    return 1.0
   515→  }
   516→}

   518→// ---- Currency Risk Awareness ----
   519→
   520→/**
   521→ * Returns informational awareness about IDR/USD currency risk.

   547→/**
   548→ * Get or create today's daily performance record.
   549→ */
   550→export async function getDailyPerformance(): Promise<DailyPerformanceData> {
   551→  const todayStr = new Date().toISOString().split("T")[0]
   552→
   553→  const existing = await db.dailyPerformance.findUnique({ where: { date: todayStr } })
   554→  if (existing) {
   555→    return mapDailyPerformanceToData(existing)
   556→  }
   557→
   558→  // Create today's record
   559→  const BASE_BALANCE = 10000
   560→  const allClosed = await db.trade.findMany({ where: { status: "CLOSED" } })
   561→  const totalClosedPnl = allClosed.reduce((s, t) => s + t.pnl, 0)
   562→  const startBalance = Math.round((BASE_BALANCE + totalClosedPnl) * 100) / 100
   563→
   564→  const perf = await db.dailyPerformance.create({
   565→    data: {
   566→      date: todayStr,
   567→      startBalance,
   568→      endBalance: startBalance,
   569→      peakEquity: startBalance,
   570→      troughEquity: startBalance,
   571→    },
   572→  })
   573→
   574→  return mapDailyPerformanceToData(perf)
   575→}

   577→/**
   578→ * Update daily performance after a trade event.
   579→ * Enhanced to track commission, slippage, deployed capital, reserve capital,
   580→ * scaling factor, and sizing method used.
   581→ */
   582→export async function updateDailyPerformance(params: {
   583→  type: "OPEN" | "CLOSE"
   584→  pnl?: number
   585→  isWin?: boolean
   586→  commission?: number
   587→  slippage?: number
   588→  sizingMethod?: string
   589→}): Promise<void> {
   590→  const todayStr = new Date().toISOString().split("T")[0]
   591→
   592→  const existing = await db.dailyPerformance.findUnique({ where: { date: todayStr } })
   593→  if (!existing) return
   594→
   595→  const openTrades = await db.trade.findMany({ where: { status: "OPEN" } })
   596→  const unrealizedPnl = openTrades.reduce((s, t) => s + t.pnl, 0)
   597→
   598→  // Calculate deployed capital = sum of margin for all open positions
   599→  const deployedCapital = openTrades.reduce((s, t) => s + (t.margin || 0), 0)
   600→
   601→  // Reserve capital = equity - deployed capital - unrealized PnL
   602→  const currentEquity = existing.startBalance + existing.realizedPnl + unrealizedPnl
   603→  const reserveCapital = Math.max(0, currentEquity - deployedCapital - unrealizedPnl)
   604→
   605→  // Track commission and slippage from closed trades today
   606→  const todayStart = new Date(todayStr)
   607→  const todayClosedTrades = await db.trade.findMany({
   608→    where: {
   609→      status: "CLOSED",
   610→      closeTime: { gte: todayStart },
   611→    },
   612→  })
   613→  const totalCommissionPaid = todayClosedTrades.reduce((s, t) => s + (t.commission || 0), 0)
   614→  const totalSlippageCost = todayClosedTrades.reduce((s, t) => s + (t.slippage || 0), 0)
   615→
   616→  // Get current scaling factor
   617→  const scalingFactor = await calculateScalingFactor()
   618→
   619→  const updateData: Record<string, unknown> = {
   620→    unrealizedPnl: Math.round(unrealizedPnl * 100) / 100,
   621→    deployedCapital: Math.round(deployedCapital * 100) / 100,
   622→    reserveCapital: Math.round(reserveCapital * 100) / 100,
   623→    commissionPaid: Math.round(totalCommissionPaid * 100) / 100,
   624→    slippageCost: Math.round(totalSlippageCost * 100) / 100,
   625→    scalingFactor,
   626→  }
   627→
   628→  // Track the most common sizing method used today
   629→  if (params.sizingMethod) {
   630→    // Store the latest sizing method used; this is a simplification.
   631→    // A more complex approach would count method occurrences.
   632→    updateData.sizingMethodUsed = params.sizingMethod
   633→  }
   634→
   635→  if (params.type === "CLOSE") {
   636→    const newRealized = Math.round((existing.realizedPnl + (params.pnl || 0)) * 100) / 100
   637→    updateData.realizedPnl = newRealized
   638→    updateData.totalPnl = Math.round((newRealized + unrealizedPnl) * 100) / 100
   639→    updateData.pnlPercent = existing.startBalance > 0 ? Math.round((newRealized / existing.startBalance) * 10000) / 100 : 0
   640→    updateData.tradesClosed = existing.tradesClosed + 1
   641→    if (params.isWin) {
   642→      updateData.winTrades = existing.winTrades + 1
   643→    } else {
   644→      updateData.lossTrades = existing.lossTrades + 1
   645→    }
   646→  } else {
   647→    updateData.tradesOpened = existing.tradesOpened + 1
   648→    updateData.totalPnl = Math.round((existing.realizedPnl + unrealizedPnl) * 100) / 100
   649→    updateData.pnlPercent = existing.startBalance > 0 ? Math.round((existing.realizedPnl / existing.startBalance) * 10000) / 100 : 0
   650→  }
   651→
   652→  // Update peak/trough for drawdown
   653→  const finalEquity = existing.startBalance + (updateData.totalPnl as number)
   654→  updateData.endBalance = Math.round(finalEquity * 100) / 100
   655→
   656→  const newPeak = Math.max(existing.peakEquity, finalEquity)
   657→  const newTrough = Math.min(existing.troughEquity, finalEquity)
   658→  updateData.peakEquity = Math.round(newPeak * 100) / 100
   659→  updateData.troughEquity = Math.round(newTrough * 100) / 100
   660→
   661→  if (newPeak > 0) {
   662→    const dd = ((newPeak - newTrough) / newPeak) * 100
   663→    updateData.maxDrawdown = Math.max(existing.maxDrawdown, Math.round(dd * 100) / 100)
   664→  }
   665→
   666→  await db.dailyPerformance.update({
   667→    where: { date: todayStr },
   668→    data: updateData,
   669→  })
   670→
   671→  logger.debug("MONEY_MANAGEMENT", `Daily performance updated: ${params.type} | PnL: ${params.pnl ?? 0} | Commission: $${totalCommissionPaid.toFixed(2)} | Deployed: $${deployedCapital.toFixed(2)} | Reserve: $${reserveCapital.toFixed(2)}`, {
   672→    metadata: { type: params.type, pnl: params.pnl, date: todayStr, scalingFactor },
   673→  })
   674→}
