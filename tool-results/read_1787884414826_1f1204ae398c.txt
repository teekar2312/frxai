    76→    76→// ---- Constants ----

    78→    78→const LEVERAGE = 25

    86→    86→/**
    87→    87→ * Calculate the optimal lot size for a trade.
    88→    88→ * Enhanced with commission-aware sizing, reserve capital enforcement,
    89→    89→ * deployed margin cap, and performance-based dynamic scaling.
    90→    90→ */
    91→    91→export async function calculatePositionSize(params: {
    92→    92→  symbol: string
    93→    93→  direction: "BUY" | "SELL"
    94→    94→  entryPrice: number
    95→    95→  sl?: number | null
    96→    96→  equity: number
    97→    97→  method?: SizingMethod
    98→    98→  fixedDollarRisk?: number
    99→    99→  kellyFraction?: number // default 0.5 (half-Kelly)
   100→   100→  commissionPerLot?: number // default $1 (FINEX)
   101→   101→  reserveCapitalPct?: number // default from RiskConfig
   102→   102→}): Promise<PositionSizeResult> {
   103→   103→  const config = await getRiskConfig()
   104→   104→  const method = params.method || "FIXED_FRACTIONAL"
   105→   105→  const sl = params.sl
   106→   106→  const commissionPerLot = params.commissionPerLot ?? DEFAULT_COMMISSION_PER_LOT
   107→   107→  const reserveCapitalPct = params.reserveCapitalPct ?? config.reserveCapitalPct
   108→   108→
   109→   109→  // --- Performance-Based Dynamic Scaling ---
   110→   110→  const scalingFactor = await calculateScalingFactor()
   111→   111→
   112→   112→  // Calculate pip risk (price distance to stop loss)
   113→   113→  let pipRisk = 0
   114→   114→  if (sl) {
   115→   115→    pipRisk = params.direction === "BUY"
   116→   116→      ? params.entryPrice - sl
   117→   117→      : sl - params.entryPrice
   118→   118→    pipRisk = Math.abs(pipRisk)
   119→   119→  }
   120→   120→
   121→   121→  // If no SL, use a default 2% price risk as fallback
   122→   122→  const effectivePipRisk = pipRisk > 0 ? pipRisk : params.entryPrice * 0.02
   123→   123→
   124→   124→  // --- Base Risk Amount by Method ---
   125→   125→  let baseRiskAmount: number
   126→   126→  let methodReasoning: string
   127→   127→
   128→   128→  switch (method) {
   129→   129→    case "FIXED_FRACTIONAL": {
   130→   130→      baseRiskAmount = (config.maxRiskPerTrade / 100) * params.equity
   131→   131→      methodReasoning = `Fixed fractional: ${config.maxRiskPerTrade}% of equity ($${params.equity.toFixed(2)}) = $${baseRiskAmount.toFixed(2)}`
   132→   132→      break
   133→   133→    }
   134→   134→    case "KELLY": {
   135→   135→      const kellyFraction = params.kellyFraction || 0.5
   136→   136→      const kellyResult = await calculateKelly(params.equity, kellyFraction)
   137→   137→      baseRiskAmount = kellyResult.suggestedRisk
   138→   138→      methodReasoning = `Half-Kelly criterion: Kelly suggests ${kellyResult.kellyPercent.toFixed(1)}%, using ${kellyFraction * 100}% = $${baseRiskAmount.toFixed(2)}`
   139→   139→      break
   140→   140→    }
   141→   141→    case "FIXED_DOLLAR": {
   142→   142→      baseRiskAmount = params.fixedDollarRisk || 100
   143→   143→      methodReasoning = `Fixed dollar risk: $${baseRiskAmount.toFixed(2)} per trade`
   144→   144→      break
   145→   145→    }
   146→   146→    case "ANTI_MARTINGALE": {
   147→   147→      const recentTrades = await getRecentTradeResults(5)
   148→   148→      let multiplier = 1
   149→   149→      for (const t of recentTrades) {
   150→   150→        if (t > 0) multiplier = Math.min(multiplier + 0.5, 3)
   151→   151→        else break
   152→   152→      }
   153→   153→      baseRiskAmount = (config.maxRiskPerTrade / 100) * params.equity * multiplier
   154→   154→      methodReasoning = `Anti-Martingale: base ${config.maxRiskPerTrade}% * ${multiplier.toFixed(1)}x (consecutive wins) = $${baseRiskAmount.toFixed(2)}`
   155→   155→      break
   156→   156→    }
   157→   157→    default:
   158→   158→      baseRiskAmount = (config.maxRiskPerTrade / 100) * params.equity
   159→   159→      methodReasoning = `Default fixed fractional: $${baseRiskAmount.toFixed(2)}`
   160→   160→  }
   161→   161→
   162→   162→  // --- Apply Scaling Factor ---
   163→   163→  const scaledRiskAmount = baseRiskAmount * scalingFactor
   164→   164→  let reasoning = methodReasoning
   165→   165→  if (scalingFactor !== 1.0) {
   166→   166→    reasoning += ` | Scaled ${scalingFactor}x -> $${scaledRiskAmount.toFixed(2)}`
   167→   167→  }
   168→   168→
   169→   169→  // --- Commission-Aware Lot Sizing ---
   170→   170→  // Risk budget must cover both SL risk AND commission cost.
   171→   171→  // totalRisk = SL_risk + commission
   172→   172→  // We need to find a lot size L where:
   173→   173→  //   L * effectivePipRisk * PIP_VALUE_PER_LOT + L * commissionPerLot <= scaledRiskAmount
   174→   174→  //   L * (effectivePipRisk * PIP_VALUE_PER_LOT + commissionPerLot) <= scaledRiskAmount
   175→   175→  //   L <= scaledRiskAmount / (effectivePipRisk * PIP_VALUE_PER_LOT + commissionPerLot)
   176→   176→  const costPerLot = effectivePipRisk * PIP_VALUE_PER_LOT + commissionPerLot
   177→   177→  const rawLotSize = costPerLot > 0 ? scaledRiskAmount / costPerLot : 0
   178→   178→  let suggestedLotSize = Math.floor(rawLotSize / LOT_STEP) * LOT_STEP
   179→   179→  suggestedLotSize = Math.max(MIN_LOT, suggestedLotSize)
   180→   180→
   181→   181→  // Calculate commission cost for the suggested lot
   182→   182→  const commissionCost = Math.round(suggestedLotSize * commissionPerLot * 100) / 100
   183→   183→  const slRiskAmount = Math.round(suggestedLotSize * effectivePipRisk * PIP_VALUE_PER_LOT * 100) / 100
   184→   184→  const netRiskAfterCommission = Math.round((slRiskAmount + commissionCost) * 100) / 100
   185→   185→
   186→   186→  reasoning += ` | Commission: $${commissionCost.toFixed(2)} (${suggestedLotSize} lot x $${commissionPerLot}/lot)`
   187→   187→  reasoning += ` | Net risk (SL + commission): $${netRiskAfterCommission.toFixed(2)}`
   188→   188→
   189→   189→  // --- Reserve Capital Enforcement ---
   190→   190→  const maxDeployable = params.equity * (1 - reserveCapitalPct / 100)
   191→   191→  const contractValue = suggestedLotSize * params.entryPrice * 100000
   192→   192→  const marginRequired = contractValue / LEVERAGE
   193→   193→  let reserveCheckApplied = false
   194→   194→
   195→   195→  if (marginRequired > maxDeployable) {
   196→   196→    // Reduce lot size so margin fits within deployable capital
   197→   197→    const maxLotForReserve = (maxDeployable * LEVERAGE) / (params.entryPrice * 100000)
   198→   198→    const reducedLot = Math.floor(maxLotForReserve / LOT_STEP) * LOT_STEP
   199→   199→    if (reducedLot < suggestedLotSize) {
   200→   200→      suggestedLotSize = Math.max(MIN_LOT, reducedLot)
   201→   201→      reserveCheckApplied = true
   202→   202→      // Recalculate values after reserve reduction
   203→   203→    }
   204→   204→    reasoning += ` | Reserve ${reserveCapitalPct}% enforced: max deployable $${maxDeployable.toFixed(2)}, reduced lot to ${suggestedLotSize}`
   205→   205→  }
   206→   206→
   207→   207→  // --- Max Capital Deployment (Open Positions Check) ---
   208→   208→  const openPositions = await db.trade.findMany({ where: { status: "OPEN" } })
   209→   209→  const existingDeployedMargin = openPositions.reduce((sum, t) => sum + (t.margin || 0), 0)
   210→   210→  const newMarginRequired = (suggestedLotSize * params.entryPrice * 100000) / LEVERAGE
   211→   211→  let deployedMarginCheckApplied = false
   212→   212→
   213→   213→  if (existingDeployedMargin + newMarginRequired > maxDeployable) {
   214→   214→    const availableForNew = maxDeployable - existingDeployedMargin
   215→   215→    if (availableForNew > 0) {
   216→   216→      const maxLotForDeployed = (availableForNew * LEVERAGE) / (params.entryPrice * 100000)
   217→   217→      const reducedLot = Math.floor(maxLotForDeployed / LOT_STEP) * LOT_STEP
   218→   218→      if (reducedLot < suggestedLotSize && reducedLot >= MIN_LOT) {
   219→   219→        suggestedLotSize = reducedLot
   220→   220→        deployedMarginCheckApplied = true
   221→   221→        reasoning += ` | Deployed margin check: $${existingDeployedMargin.toFixed(2)} already deployed, reduced lot to ${suggestedLotSize}`
   222→   222→      } else if (reducedLot < MIN_LOT) {
   223→   223→        // Not enough deployable capital for even minimum lot
   224→   224→        suggestedLotSize = 0
   225→   225→        reasoning += ` | REJECTED: Not enough deployable capital ($${availableForNew.toFixed(2)} available, $${newMarginRequired.toFixed(2)} needed)`
   226→   226→      }
   227→   227→    } else {
   228→   228→      suggestedLotSize = 0
   229→   229→      reasoning += ` | REJECTED: All deployable capital ($${maxDeployable.toFixed(2)}) already deployed ($${existingDeployedMargin.toFixed(2)})`
   230→   230→    }
   231→   231→  }
   232→   232→
   233→   233→  // Cap at max lot per trade
   234→   234→  if (suggestedLotSize > 0) {
   235→   235→    suggestedLotSize = Math.min(suggestedLotSize, config.maxLotPerTrade)
   236→   236→  }
   237→   237→
   238→   238→  // --- Final Derived Values ---
   239→   239→  const finalContractValue = suggestedLotSize * params.entryPrice * 100000
   240→   240→  const finalMarginRequired = finalContractValue / LEVERAGE
   241→   241→  const finalCommissionCost = Math.round(suggestedLotSize * commissionPerLot * 100) / 100
   242→   242→  const finalSlRiskAmount = Math.round(suggestedLotSize * effectivePipRisk * PIP_VALUE_PER_LOT * 100) / 100
   243→   243→  const finalNetRisk = Math.round((finalSlRiskAmount + finalCommissionCost) * 100) / 100
   244→   244→  const riskPercent = params.equity > 0 ? (finalNetRisk / params.equity) * 100 : 0
   245→   245→
   246→   246→  return {
   247→   247→    suggestedLotSize: Math.round(suggestedLotSize * 100) / 100,
   248→   248→    riskAmount: Math.round((baseRiskAmount * scalingFactor) * 100) / 100,
   249→   249→    riskPercent: Math.round(riskPercent * 100) / 100,
   250→   250→    method,
   251→   251→    pipRisk: Math.round(effectivePipRisk * 100) / 100,
   252→   252→    contractValue: Math.round(finalContractValue * 100) / 100,
   253→   253→    marginRequired: Math.round(finalMarginRequired * 100) / 100,
   254→   254→    reasoning,
   255→   255→    commissionCost: finalCommissionCost,
   256→   256→    netRiskAfterCommission: finalNetRisk,
   257→   257→    scalingFactor,
   258→   258→    reserveCheckApplied,
   259→   259→    deployedMarginCheckApplied,
   260→   260→  }
   261→   261→}

   263→   263→/**
   264→   264→ * Kelly Criterion position sizing.
   265→   265→ */
   266→   266→async function calculateKelly(equity: number, fraction: number): Promise<{
   267→   267→  kellyPercent: number
   268→   268→  suggestedRisk: number
   269→   269→}> {
   270→   270→  const closedTrades = await db.trade.findMany({
   271→   271→    where: { status: "CLOSED" },
   272→   272→    orderBy: { closeTime: "desc" },
   273→   273→    take: 100,
   274→   274→  })
   275→   275→
   276→   276→  if (closedTrades.length < 10) {
   277→   277→    return { kellyPercent: 1.0, suggestedRisk: (fraction / 100) * equity }
   278→   278→  }
   279→   279→
   280→   280→  const wins = closedTrades.filter((t) => t.pnl > 0)
   281→   281→  const losses = closedTrades.filter((t) => t.pnl < 0)
   282→   282→  const winRate = wins.length / closedTrades.length
   283→   283→  const avgWin = wins.length > 0 ? wins.reduce((s, t) => s + t.pnl, 0) / wins.length : 0
   284→   284→  const avgLoss = losses.length > 0 ? Math.abs(losses.reduce((s, t) => s + t.pnl, 0) / losses.length) : 1
   285→   285→
   286→   286→  const R = avgLoss > 0 ? avgWin / avgLoss : 0
   287→   287→  const kellyPercent = R > 0 ? (winRate - (1 - winRate) / R) * 100 : 0
   288→   288→  const clampedKelly = Math.max(-5, Math.min(25, kellyPercent))
   289→   289→
   290→   290→  return {
   291→   291→    kellyPercent: clampedKelly,
   292→   292→    suggestedRisk: (Math.max(0, clampedKelly) * fraction / 100) * equity,
   293→   293→  }
   294→   294→}

   296→   296→/**
   297→   297→ * Get recent trade results as array of PnL values (newest first).
   298→   298→ */
   299→   299→async function getRecentTradeResults(count: number): Promise<number[]> {
   300→   300→  const trades = await db.trade.findMany({
   301→   301→    where: { status: "CLOSED" },
   302→   302→    orderBy: { closeTime: "desc" },
   303→   303→    take: count,
   304→   304→  })
   305→   305→  return trades.map((t) => t.pnl)
   306→   306→}
   307→   307→
   308→   308→// ---- Risk of Ruin ----
   309→   309→
   310→   310→/**
   311→   311→ * Estimate the probability of ruin (losing X% of account).
   312→   312→ * Uses the formula: P(ruin) = ((1 - edge) / (1 + edge))^units
   313→   313→ * where edge = W - (1-W)*avgLoss/avgWin and units = account / (risk per trade)
   314→   314→ */
   315→   315→export function calculateRiskOfRuin(input: RiskOfRuinInput): {
   316→   316→  probability: number
   317→   317→  interpretation: string
   318→   318→  recommendation: string
   319→   319→} {
   320→   320→  if (input.winRate <= 0 || input.avgLoss <= 0) {
   321→   321→    return {
   322→   322→      probability: 100,
   323→   323→      interpretation: "Cannot calculate - insufficient winning trades or zero avg loss",
   324→   324→      recommendation: "Improve strategy win rate or reduce loss size",
   325→   325→    }
   326→   326→  }
   327→   327→
   328→   328→  const W = input.winRate / 100
   329→   329→  const L = 1 - W
   330→   330→  const R = input.avgWin / input.avgLoss
   331→   331→
   332→   332→  if (R <= 0) {
   333→   333→    return {
   334→   334→      probability: 100,
   335→   335→      interpretation: "Average win is zero or negative",
   336→   336→      recommendation: "Strategy needs improvement - avg win must exceed avg loss",
   337→   337→    }
   338→   338→  }
   339→   339→
   340→   340→  const edge = W - L / R
   341→   341→  const riskFraction = input.riskPerTrade / 100
   342→   342→
   343→   343→  if (edge <= 0) {
   344→   344→    return {
   345→   345→      probability: 100,
   346→   346→      interpretation: `Negative edge (${(edge * 100).toFixed(2)}%) - strategy is unprofitable statistically`,
   347→   347→      recommendation: "Do not trade this strategy. Win rate and/or reward-risk ratio must improve.",
   348→   348→    }
   349→   349→  }
   350→   350→
   351→   351→  const q = (1 - edge) / (1 + edge)
   352→   352→  const units = 1 / riskFraction
   353→   353→  const ror = Math.pow(Math.max(0, q), units) * 100
   354→   354→
   355→   355→  let interpretation: string
   356→   356→  let recommendation: string
   357→   357→
   358→   358→  if (ror < 1) {
   359→   359→    interpretation = "Very Low Risk"
   360→   360→    recommendation = "Strategy has a strong statistical edge. Safe to trade with current parameters."
   361→   361→  } else if (ror < 5) {
   362→   362→    interpretation = "Low Risk"
   363→   363→    recommendation = "Acceptable risk level. Continue monitoring performance metrics."
   364→   364→  } else if (ror < 15) {
   365→   365→    interpretation = "Moderate Risk"
   366→   366→    recommendation = "Consider reducing position sizes or improving win rate to lower risk of ruin."
   367→   367→  } else if (ror < 30) {
   368→   368→    interpretation = "High Risk"
   369→   369→    recommendation = "Significant risk of account drawdown. Reduce risk per trade significantly."
   370→   370→  } else {
   371→   371→    interpretation = "Extreme Risk"
   372→   372→    recommendation = "Do NOT trade this strategy with current parameters. High probability of account destruction."
   373→   373→  }
   374→   374→
   375→   375→  return {
   376→   376→    probability: Math.min(100, Math.round(ror * 100) / 100),
   377→   377→    interpretation,
   378→   378→    recommendation,
   379→   379→  }
   380→   380→}
   381→   381→
   382→   382→// ---- Drawdown Recovery Model ----
   383→   383→
   384→   384→/**
   385→   385→ * Calculate the recovery needed for any drawdown level.
   386→   386→ * Formula: recoveryNeeded = (drawdown / (100 - drawdown)) * 100
   387→   387→ *
   388→   388→ * Examples:
   389→   389→ *   10% DD -> 11.1% recovery needed
   390→   390→ *   20% DD -> 25% recovery needed
   391→   391→ *   50% DD -> 100% recovery needed
   392→   392→ */
   393→   393→export function calculateDrawdownRecovery(drawdownPct: number): DrawdownRecoveryResult {
   394→   394→  const dd = Math.max(0, Math.min(99, drawdownPct)) // Clamp 0-99%
   395→   395→
   396→   396→  const recoveryNeeded = dd >= 100 ? Infinity : (dd / (100 - dd)) * 100
   397→   397→
   398→   398→  // Recovery strategy and risk reduction based on drawdown severity
   399→   399→  let strategy: string
   400→   400→  let riskReductionPct: number
   401→   401→
   402→   402→  if (dd < 5) {
   403→   403→    strategy = "NORMAL: No action needed. Drawdown within acceptable range. Continue normal trading operations."
   404→   404→    riskReductionPct = 0
   405→   405→  } else if (dd < 10) {
   406→   406→    strategy = "CAUTION: Minor drawdown. Review recent trades for patterns. Tighten stop losses by 10%. Avoid high-risk setups."
   407→   407→    riskReductionPct = 10
   408→   408→  } else if (dd < 15) {
   409→   409→    strategy = "ELEVATED: Moderate drawdown. Reduce position sizes by 25%. Focus on highest-confidence setups only. Consider pausing new positions temporarily."
   410→   410→    riskReductionPct = 25
   411→   411→  } else if (dd < 20) {
   412→   412→    strategy = "HIGH: Significant drawdown. Reduce risk to 50% of normal. Trade only A+ setups. Implement stricter entry criteria. Review and potentially pause underperforming strategies."
   413→   413→    riskReductionPct = 50
   414→   414→  } else if (dd < 30) {
   415→   415→    strategy = "CRITICAL: Severe drawdown. Reduce risk to 75% of normal. Consider full trading pause for 24-48 hours. Perform strategy review. Check if market regime has changed."
   416→   416→    riskReductionPct = 75
   417→   417→  } else if (dd < 50) {
   418→   418→    strategy = "EMERGENCY: Extreme drawdown. Stop all new trading immediately. Close weakest positions. Full strategy audit required. Resume only with micro-lots after analysis."
   419→   419→    riskReductionPct = 90
   420→   420→  } else {
   421→   421→    strategy = "CATASTROPHIC: Account severely impaired. ${recoveryNeeded.toFixed(1)}% gain required to recover. Full trading halt. Account preservation is the only priority. Consider account reset or external capital injection after thorough review."
   422→   422→    riskReductionPct = 100
   423→   423→  }
   424→   424→
   425→   425→  // Fix the template literal for 50%+ case
   426→   426→  if (dd >= 50) {
   427→   427→    strategy = `CATASTROPHIC: Account severely impaired. ${recoveryNeeded.toFixed(1)}% gain required to recover. Full trading halt. Account preservation is the only priority. Consider account reset or external capital injection after thorough review.`
   428→   428→  }
   429→   429→
   430→   430→  return {
   431→   431→    drawdownPct: Math.round(dd * 100) / 100,
   432→   432→    recoveryNeeded: Math.round(recoveryNeeded * 100) / 100,
   433→   433→    strategy,
   434→   434→    riskReductionPct,
   435→   435→  }
   436→   436→}

   438→   438→// ---- Performance-Based Dynamic Scaling ----
   439→   439→
   440→   440→/**
   441→   441→ * Calculate a dynamic scaling factor based on recent trading performance.
   442→   442→ * Analyzes the last 30 closed trades (or all if fewer) for:
   443→   443→ *   - Rolling win rate
   444→   444→ *   - Profit factor (gross profit / gross loss)
   445→   445→ *   - Sharpe-like ratio (mean PnL / std dev of PnL)
   446→   446→ *
   447→   447→ * Scaling rules:
   448→   448→ *   WR > 60% AND PF > 1.5  -> scale up to 1.25x (max)
   449→   449→ *   WR > 55% AND PF > 1.2  -> scale up to 1.1x
   450→   450→ *   WR < 40% OR PF < 0.8  -> scale down to 0.5x
   451→   451→ *   WR < 45% OR PF < 1.0  -> scale down to 0.75x
   452→   452→ *   Otherwise              -> 1.0x (neutral)
   453→   453→ */
   454→   454→export async function calculateScalingFactor(): Promise<number> {
   455→   455→  try {
   456→   456→    const closedTrades = await db.trade.findMany({
   457→   457→      where: { status: "CLOSED" },
   458→   458→      orderBy: { closeTime: "desc" },
   459→   459→      take: 30,
   460→   460→    })
   461→   461→
   462→   462→    // Not enough trades to evaluate - use neutral scaling
   463→   463→    if (closedTrades.length < 5) {
   464→   464→      return 1.0
   465→   465→    }
   466→   466→
   467→   467→    // Calculate rolling win rate
   468→   468→    const wins = closedTrades.filter((t) => t.pnl > 0)
   469→   469→    const winRate = (wins.length / closedTrades.length) * 100
   470→   470→
   471→   471→    // Calculate profit factor
   472→   472→    const grossProfit = wins.reduce((sum, t) => sum + t.pnl, 0)
   473→   473→    const losses = closedTrades.filter((t) => t.pnl < 0)
   474→   474→    const grossLoss = Math.abs(losses.reduce((sum, t) => sum + t.pnl, 0))
   475→   475→    const profitFactor = grossLoss > 0 ? grossProfit / grossLoss : 0
   476→   476→
   477→   477→    // Calculate Sharpe-like ratio (mean / stddev of PnL)
   478→   478→    const pnls = closedTrades.map((t) => t.pnl)
   479→   479→    const mean = pnls.reduce((s, v) => s + v, 0) / pnls.length
   480→   480→    const variance = pnls.reduce((s, v) => s + (v - mean) ** 2, 0) / pnls.length
   481→   481→    const stddev = Math.sqrt(variance)
   482→   482→    // We don't use this for scaling decisions directly but log it
   483→   483→    const sharpeLike = stddev > 0 ? mean / stddev : 0
   484→   484→
   485→   485→    // Determine scaling factor based on rules
   486→   486→    let factor: number
   487→   487→    if (winRate > 60 && profitFactor > 1.5) {
   488→   488→      factor = 1.25
   489→   489→    } else if (winRate > 55 && profitFactor > 1.2) {
   490→   490→      factor = 1.1
   491→   491→    } else if (winRate < 40 || profitFactor < 0.8) {
   492→   492→      factor = 0.5
   493→   493→    } else if (winRate < 45 || profitFactor < 1.0) {
   494→   494→      factor = 0.75
   495→   495→    } else {
   496→   496→      factor = 1.0
   497→   497→    }
   498→   498→
   499→   499→    logger.debug("MONEY_MANAGEMENT", `Scaling factor calculated: ${factor}x (WR: ${winRate.toFixed(1)}%, PF: ${profitFactor.toFixed(2)}, Sharpe-like: ${sharpeLike.toFixed(2)})`, {
   500→   500→      metadata: {
   501→   501→        scalingFactor: factor,
   502→   502→        winRate: Math.round(winRate * 100) / 100,
   503→   503→        profitFactor: Math.round(profitFactor * 100) / 100,
   504→   504→        sharpeLike: Math.round(sharpeLike * 100) / 100,
   505→   505→        sampleSize: closedTrades.length,
   506→   506→      },
   507→   507→    })
   508→   508→
   509→   509→    return factor
   510→   510→  } catch (err) {
   511→   511→    logger.error("MONEY_MANAGEMENT", "Error calculating scaling factor, defaulting to 1.0", {
   512→   512→      details: err instanceof Error ? err.stack : undefined,
   513→   513→    })
   514→   514→    return 1.0
   515→   515→  }
   516→   516→}

   547→   547→/**
   548→   548→ * Get or create today's daily performance record.
   549→   549→ */
   550→   550→export async function getDailyPerformance(): Promise<DailyPerformanceData> {
   551→   551→  const todayStr = new Date().toISOString().split("T")[0]
   552→   552→
   553→   553→  const existing = await db.dailyPerformance.findUnique({ where: { date: todayStr } })
   554→   554→  if (existing) {
   555→   555→    return mapDailyPerformanceToData(existing)
   556→   556→  }
   557→   557→
   558→   558→  // Create today's record
   559→   559→  const BASE_BALANCE = 10000
   560→   560→  const allClosed = await db.trade.findMany({ where: { status: "CLOSED" } })
   561→   561→  const totalClosedPnl = allClosed.reduce((s, t) => s + t.pnl, 0)
   562→   562→  const startBalance = Math.round((BASE_BALANCE + totalClosedPnl) * 100) / 100
   563→   563→
   564→   564→  const perf = await db.dailyPerformance.create({
   565→   565→    data: {
   566→   566→      date: todayStr,
   567→   567→      startBalance,
   568→   568→      endBalance: startBalance,
   569→   569→      peakEquity: startBalance,
   570→   570→      troughEquity: startBalance,
   571→   571→    },
   572→   572→  })
   573→   573→
   574→   574→  return mapDailyPerformanceToData(perf)
   575→   575→}

   577→   577→/**
   578→   578→ * Update daily performance after a trade event.
   579→   579→ * Enhanced to track commission, slippage, deployed capital, reserve capital,
   580→   580→ * scaling factor, and sizing method used.
   581→   581→ */
   582→   582→export async function updateDailyPerformance(params: {
   583→   583→  type: "OPEN" | "CLOSE"
   584→   584→  pnl?: number
   585→   585→  isWin?: boolean
   586→   586→  commission?: number
   587→   587→  slippage?: number
   588→   588→  sizingMethod?: string
   589→   589→}): Promise<void> {
   590→   590→  const todayStr = new Date().toISOString().split("T")[0]
   591→   591→
   592→   592→  const existing = await db.dailyPerformance.findUnique({ where: { date: todayStr } })
   593→   593→  if (!existing) return
   594→   594→
   595→   595→  const openTrades = await db.trade.findMany({ where: { status: "OPEN" } })
   596→   596→  const unrealizedPnl = openTrades.reduce((s, t) => s + t.pnl, 0)
   597→   597→
   598→   598→  // Calculate deployed capital = sum of margin for all open positions
   599→   599→  const deployedCapital = openTrades.reduce((s, t) => s + (t.margin || 0), 0)
   600→   600→
   601→   601→  // Reserve capital = equity - deployed capital - unrealized PnL
   602→   602→  const currentEquity = existing.startBalance + existing.realizedPnl + unrealizedPnl
   603→   603→  const reserveCapital = Math.max(0, currentEquity - deployedCapital - unrealizedPnl)
   604→   604→
   605→   605→  // Track commission and slippage from closed trades today
   606→   606→  const todayStart = new Date(todayStr)
   607→   607→  const todayClosedTrades = await db.trade.findMany({
   608→   608→    where: {
   609→   609→      status: "CLOSED",
   610→   610→      closeTime: { gte: todayStart },
   611→   611→    },
   612→   612→  })
   613→   613→  const totalCommissionPaid = todayClosedTrades.reduce((s, t) => s + (t.commission || 0), 0)
   614→   614→  const totalSlippageCost = todayClosedTrades.reduce((s, t) => s + (t.slippage || 0), 0)
   615→   615→
   616→   616→  // Get current scaling factor
   617→   617→  const scalingFactor = await calculateScalingFactor()
   618→   618→
   619→   619→  const updateData: Record<string, unknown> = {
   620→   620→    unrealizedPnl: Math.round(unrealizedPnl * 100) / 100,
   621→   621→    deployedCapital: Math.round(deployedCapital * 100) / 100,
   622→   622→    reserveCapital: Math.round(reserveCapital * 100) / 100,
   623→   623→    commissionPaid: Math.round(totalCommissionPaid * 100) / 100,
   624→   624→    slippageCost: Math.round(totalSlippageCost * 100) / 100,
   625→   625→    scalingFactor,
   626→   626→  }
   627→   627→
   628→   628→  // Track the most common sizing method used today
   629→   629→  if (params.sizingMethod) {
   630→   630→    // Store the latest sizing method used; this is a simplification.
   631→   631→    // A more complex approach would count method occurrences.
   632→   632→    updateData.sizingMethodUsed = params.sizingMethod
   633→   633→  }
   634→   634→
   635→   635→  if (params.type === "CLOSE") {
   636→   636→    const newRealized = Math.round((existing.realizedPnl + (params.pnl || 0)) * 100) / 100
   637→   637→    updateData.realizedPnl = newRealized
   638→   638→    updateData.totalPnl = Math.round((newRealized + unrealizedPnl) * 100) / 100
   639→   639→    updateData.pnlPercent = existing.startBalance > 0 ? Math.round((newRealized / existing.startBalance) * 10000) / 100 : 0
   640→   640→    updateData.tradesClosed = existing.tradesClosed + 1
   641→   641→    if (params.isWin) {
   642→   642→      updateData.winTrades = existing.winTrades + 1
   643→   643→    } else {
   644→   644→      updateData.lossTrades = existing.lossTrades + 1
   645→   645→    }
   646→   646→  } else {
   647→   647→    updateData.tradesOpened = existing.tradesOpened + 1
   648→   648→    updateData.totalPnl = Math.round((existing.realizedPnl + unrealizedPnl) * 100) / 100
   649→   649→    updateData.pnlPercent = existing.startBalance > 0 ? Math.round((existing.realizedPnl / existing.startBalance) * 10000) / 100 : 0
   650→   650→  }
   651→   651→
   652→   652→  // Update peak/trough for drawdown
   653→   653→  const finalEquity = existing.startBalance + (updateData.totalPnl as number)
   654→   654→  updateData.endBalance = Math.round(finalEquity * 100) / 100
   655→   655→
   656→   656→  const newPeak = Math.max(existing.peakEquity, finalEquity)
   657→   657→  const newTrough = Math.min(existing.troughEquity, finalEquity)
   658→   658→  updateData.peakEquity = Math.round(newPeak * 100) / 100
   659→   659→  updateData.troughEquity = Math.round(newTrough * 100) / 100
   660→   660→
   661→   661→  if (newPeak > 0) {
   662→   662→    const dd = ((newPeak - newTrough) / newPeak) * 100
   663→   663→    updateData.maxDrawdown = Math.max(existing.maxDrawdown, Math.round(dd * 100) / 100)
   664→   664→  }
   665→   665→
   666→   666→  await db.dailyPerformance.update({
   667→   667→    where: { date: todayStr },
   668→   668→    data: updateData,
   669→   669→  })
   670→   670→
   671→   671→  logger.debug("MONEY_MANAGEMENT", `Daily performance updated: ${params.type} | PnL: ${params.pnl ?? 0} | Commission: $${totalCommissionPaid.toFixed(2)} | Deployed: $${deployedCapital.toFixed(2)} | Reserve: $${reserveCapital.toFixed(2)}`, {
   672→   672→    metadata: { type: params.type, pnl: params.pnl, date: todayStr, scalingFactor },
   673→   673→  })
   674→   674→}
