    65→    65→// ============================================
    66→    66→// ASYNC MUTEX
    67→    67→// ============================================
    68→    68→
    69→    69→/**
    70→    70→ * Simple async mutex to prevent concurrent MT5 API calls.
    71→    71→ * The MT5 Python module is not thread-safe, so all calls must be serialized.
    72→    72→ */
    73→    73→export class AsyncMutex {
    74→    74→  private _queue: Array<() => void> = []
    75→    75→  private _locked = false
    76→    76→
    77→    77→  /** Acquire the mutex lock. Returns a release function. */
    78→    78→  async acquire(): Promise<() => void> {
    79→    79→    return new Promise<() => void>((resolve) => {
    80→    80→      const tryAcquire = () => {
    81→    81→        if (!this._locked) {
    82→    82→          this._locked = true
    83→    83→          resolve(this._release)
    84→    84→        } else {
    85→    85→          this._queue.push(tryAcquire)
    86→    86→        }
    87→    87→      }
    88→    88→      tryAcquire()
    89→    89→    })
    90→    90→  }
    91→    91→
    92→    92→  /** Run a function exclusively within the mutex lock. */
    93→    93→  async runExclusive<T>(fn: () => Promise<T>): Promise<T> {
    94→    94→    const release = await this.acquire()
    95→    95→    try {
    96→    96→      return await fn()
    97→    97→    } finally {
    98→    98→      release()
    99→    99→    }
   100→   100→  }
   101→   101→
   102→   102→  private _release = (): void => {
   103→   103→    this._locked = false
   104→   104→    const next = this._queue.shift()
   105→   105→    if (next) next()
   106→   106→  }
   107→   107→
   108→   108→  get locked(): boolean {
   109→   109→    return this._locked
   110→   110→  }
   111→   111→
   112→   112→  get queueLength(): number {
   113→   113→    return this._queue.length
   114→   114→  }
   115→   115→}

   572→   572→/** Convert WIB (UTC+7) hours/minutes to UTC hours/minutes */
   573→   573→function wibToUtc(wibHour: number, wibMinute: number = 0): { hour: number; minute: number } {
   574→   574→  let utcHour = wibHour - 7
   575→   575→  let utcMinute = wibMinute
   576→   576→  if (utcHour < 0) utcHour += 24
   577→   577→  return { hour: utcHour, minute: utcMinute }
   578→   578→}
   579→   579→
   580→   580→// Pre-computed UTC boundaries for IDX trading
   581→   581→const PHASE_BOUNDARIES_UTC = {
   582→   582→  preOpenStart: wibToUtc(8, 45),     // 01:45 UTC
   583→   583→  morningOpen: wibToUtc(9, 0),        // 02:00 UTC
   584→   584→  preCloseStart: wibToUtc(11, 30),     // 04:30 UTC
   585→   585→  preCloseEnd: wibToUtc(11, 30, 30),   // 04:30:30 UTC (treated as 04:30 since we only have minutes)
   586→   586→  afternoonOpen: wibToUtc(13, 30),     // 06:30 UTC
   587→   587→  marketClose: wibToUtc(15, 0),        // 08:00 UTC
   588→   588→} as const
   589→   589→
   590→   590→/**
   591→   591→ * Determine the current IDX trading phase based on UTC time.
   592→   592→ * IDX Schedule (WIB / UTC):
   593→   593→ *   PRE_OPEN:     08:45-09:00 WIB  = 01:45-02:00 UTC
   594→   594→ *   OPEN:         09:00-11:30 WIB  = 02:00-04:30 UTC
   595→   595→ *   PRE_CLOSE:    11:30-11:30:30    = ~04:30 UTC (30s window)
   596→   596→ *   CLOSED:       11:30-13:30 WIB  = 04:30-06:30 UTC (lunch break)
   597→   597→ *   OPEN:         13:30-15:00 WIB  = 06:30-08:00 UTC
   598→   598→ *   AFTER_HOURS:  15:00+ WIB       = 08:00+ UTC
   599→   599→ */
   600→   600→export function getTradingPhase(now?: Date): TradingPhase {
   601→   601→  const d = now || new Date()
   602→   602→  // Get UTC hours and minutes as decimal for comparison
   603→   603→  const utcDecimal = d.getUTCHours() + d.getUTCMinutes() / 60 + d.getUTCSeconds() / 3600
   604→   604→
   605→   605→  const { preOpenStart, morningOpen, preCloseStart, preCloseEnd, afternoonOpen, marketClose } = PHASE_BOUNDARIES_UTC
   606→   606→
   607→   607→  // Convert boundaries to decimal hours for comparison
   608→   608→  const preOpenDec = preOpenStart.hour + preOpenStart.minute / 60
   609→   609→  const morningDec = morningOpen.hour + morningOpen.minute / 60
   610→   610→  const preCloseDec = preCloseStart.hour + preCloseStart.minute / 60
   611→   611→  const preCloseEndDec = preCloseStart.hour + preCloseStart.minute / 60 + 30 / 3600 // 30 seconds
   612→   612→  const afternoonDec = afternoonOpen.hour + afternoonOpen.minute / 60
   613→   613→  const closeDec = marketClose.hour + marketClose.minute / 60
   614→   614→
   615→   615→  if (utcDecimal >= preOpenDec && utcDecimal < morningDec) {
   616→   616→    return "PRE_OPEN"
   617→   617→  }
   618→   618→  if (utcDecimal >= morningDec && utcDecimal < preCloseDec) {
   619→   619→    return "OPEN"
   620→   620→  }
   621→   621→  // PRE_CLOSE is a 30-second window; treat it as part of PRE_CLOSE
   622→   622→  if (utcDecimal >= preCloseDec && utcDecimal < preCloseEndDec) {
   623→   623→    return "PRE_CLOSE"
   624→   624→  }
   625→   625→  // Lunch break CLOSED
   626→   626→  if (utcDecimal >= preCloseDec && utcDecimal < afternoonDec) {
   627→   627→    return "CLOSED"
   628→   628→  }
   629→   629→  // Afternoon session OPEN
   630→   630→  if (utcDecimal >= afternoonDec && utcDecimal < closeDec) {
   631→   631→    return "OPEN"
   632→   632→  }
   633→   633→  // After hours
   634→   634→  return "AFTER_HOURS"
   635→   635→}

   637→   637→/** Check if the IDX market is currently open for trading. */
   638→   638→export function isMarketOpen(now?: Date): boolean {
   639→   639→  const phase = getTradingPhase(now)
   640→   640→  return phase === "OPEN" || phase === "PRE_OPEN"
   641→   641→}

   663→   663→          severity: entry.severity,
   664→   664→          category: entry.category,
   665→   665→          remediation: entry.remediation,
   666→   666→          retryable: entry.retryable,
   667→   667→        },
   668→   668→        create: {
   669→   669→          code: entry.code,
   670→   670→          description: entry.description,
   671→   671→          severity: entry.severity,
   672→   672→          category: entry.category,
   673→   673→          remediation: entry.remediation,
   674→   674→          retryable: entry.retryable,
   675→   675→        },

   823→   823→  validateReturn<T>(
   824→   824→    data: T,
   825→   825→    context: string,
   826→   826→    options?: {

   840→   840→    }
   841→   841→
   842→   842→    // Array checks
   843→   843→    if (expectArray && !Array.isArray(data)) {
   844→   844→      const msg = `MT5 API silent failure: ${context} expected array but got ${typeof data}`
   845→   845→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context, type: typeof data } })
   846→   846→      return { valid: false, issue: msg, data }
   847→   847→    }
   848→   848→
   849→   849→    if (expectArray && expectNonEmpty && Array.isArray(data) && data.length === 0) {
   850→   850→      // Empty array is OK if market is closed
   851→   851→      if (!this.metrics.marketIsOpen) {
   852→   852→        logger.debug("MT5_CONNECTION", `${context} returned empty array but market is closed (expected)`, { symbol })
   853→   853→        return { valid: true, data }
   854→   854→      }
   855→   855→      // During trading hours, empty array = silent failure
   856→   856→      const msg = `MT5 API silent failure: ${context} returned empty array during market hours`
   857→   857→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context, tradingPhase: this.metrics.currentTradingPhase } })
   858→   858→      return { valid: false, issue: msg, data }
   859→   859→    }
   860→   860→
   861→   861→    // Zero-tick check for tick data
   862→   862→    if (expectNonNull && (data as unknown) === 0) {
   863→   863→      if (!this.metrics.marketIsOpen) {

   865→   865→      }

   868→   868→      return { valid: false, issue: msg, data }
   869→   869→    }
   870→   870→
   871→   871→    return { valid: true, data }
   872→   872→  }

   982→   982→  private async onConnected(): Promise<void> {
   983→   983→    this.reconnectAttempt = 0
   984→   984→    this.metrics.connectedAt = new Date()
   985→   985→    this.metrics.lastError = null

   988→   988→      this.status === "RECONNECTING" ? this.metrics.reconnectCount : 0
   989→   989→    this.status = "CONNECTED"
   990→   990→
   991→   991→    // Start heartbeat
   992→   992→    this.heartbeatTimer = setInterval(
   993→   993→      () => this.heartbeat(),
   994→   994→      this.config?.heartbeatIntervalMs || 5000

   996→   996→

   998→   998→    this.uptimeTimer = setInterval(() => {
   999→   999→      if (this.metrics.connectedAt) {
  1000→  1000→        this.metrics.uptimeSeconds = Math.floor(
  1001→  1001→          (Date.now() - this.metrics.connectedAt.getTime()) / 1000
  1002→  1002→        )
  1003→  1003→      }
  1004→  1004→    }, 1000)

  1055→  1055→  /**
  1056→  1056→   * Check if we should enter or exit DEGRADED state.
  1057→  1057→   * DEGRADED triggers when:
  1058→  1058→   *   - Latency exceeds threshold (default 200ms)
  1059→  1059→   *   - Consecutive heartbeat failures >= threshold (default 2)
  1060→  1060→   *
  1061→  1061→   * PROACTIVE_MC_70: When in DEGRADED state, log a risk event suggesting
  1062→  1062→   * margin call awareness if margin usage is approaching 70%.
  1063→  1063→   * PROACTIVE_MC_60: Similar at 60% - more urgent.
  1064→  1064→   */
  1065→  1065→  private async evaluateDegradedState(latencyMs: number, heartbeatOk: boolean): Promise<void> {
  1066→  1066→    const threshold = this.config?.degradedLatencyThresholdMs || 200
  1067→  1067→    const hbThreshold = this.config?.degradedHeartbeatFailureThreshold || 2
  1068→  1068→
  1069→  1069→    const wasDegraded = this.status === "DEGRADED"
  1070→  1070→    const wasConnected = this.status === "CONNECTED"
  1071→  1071→
  1072→  1072→    if (heartbeatOk) {
  1073→  1073→      this.metrics.consecutiveHeartbeatFailures = 0
  1074→  1074→    } else {
  1075→  1075→      this.metrics.consecutiveHeartbeatFailures++
  1076→  1076→    }
  1077→  1077→
  1078→  1078→    const shouldDegrade =
  1079→  1079→      latencyMs > threshold || this.metrics.consecutiveHeartbeatFailures >= hbThreshold
  1080→  1080→
  1081→  1081→    const shouldRecover =
  1082→  1082→      !shouldDegrade &&
  1083→  1083→      latencyMs <= threshold * 0.8 &&
  1084→  1084→      this.metrics.consecutiveHeartbeatFailures === 0
  1085→  1085→
  1086→  1086→    if (shouldDegrade && (wasConnected || wasDegraded) && this.status !== "DEGRADED") {
  1087→  1087→      // Transition to DEGRADED
  1088→  1088→      const reason = latencyMs > threshold
  1089→  1089→        ? `High latency: ${latencyMs}ms > ${threshold}ms threshold`
  1090→  1090→        : `${this.metrics.consecutiveHeartbeatFailures} consecutive heartbeat failures >= ${hbThreshold}`
  1091→  1091→
  1092→  1092→      this.status = "DEGRADED"
  1093→  1093→      await this.setStatus("DEGRADED")
  1094→  1094→      await this.logConnectionEvent("DEGRADED", reason)
  1095→  1095→      logger.warn("MT5_CONNECTION", `Connection DEGRADED: ${reason}`, {
  1096→  1096→        metadata: {
  1097→  1097→          latencyMs,
  1098→  1098→          consecutiveFailures: this.metrics.consecutiveHeartbeatFailures,
  1099→  1099→          threshold,
  1100→  1100→        },
  1101→  1101→      })
  1102→  1102→
  1103→  1103→      // PROACTIVE_MC_70 concept: log risk awareness when in DEGRADED state
  1104→  1104→      await this.logProactiveMarginCall("PROACTIVE_MC_70", 70)
  1105→  1105→
  1106→  1106→      emitStatusChange(this.status, this.metrics)
  1107→  1107→    } else if (shouldRecover && this.status === "DEGRADED") {
  1108→  1108→      // Recover from DEGRADED back to CONNECTED
  1109→  1109→      this.status = "CONNECTED"
  1110→  1110→      await this.setStatus("CONNECTED")
  1111→  1111→      await this.logConnectionEvent("RECOVERED", `Recovered from DEGRADED (latency: ${latencyMs}ms)`)
  1112→  1112→      logger.info("MT5_CONNECTION", `Connection recovered from DEGRADED (latency: ${latencyMs}ms)`, {
  1113→  1113→        metadata: { latencyMs },
  1114→  1114→      })
  1115→  1115→      emitStatusChange(this.status, this.metrics)
  1116→  1116→    } else if (shouldDegrade && this.status === "DEGRADED") {
  1117→  1117→      // Still in DEGRADED state - check for PROACTIVE_MC_60 if failures increasing
  1118→  1118→      if (this.metrics.consecutiveHeartbeatFailures >= hbThreshold + 1) {
  1119→  1119→        await this.logProactiveMarginCall("PROACTIVE_MC_60", 60)
  1120→  1120→      }
  1121→  1121→    }
  1122→  1122→  }

  1222→  1222→    const delay = Math.min(
  1223→  1223→      this.config.baseReconnectDelayMs * Math.pow(2, this.reconnectAttempt - 1),

  1236→  1236→      }

  1243→  1243→        if (result.success) {

  1245→  1245→        } else {
  1246→  1246→          this.metrics.reconnectCount++

  1251→  1251→        const msg = err instanceof Error ? err.message : String(err)

  1253→  1253→      }

  1259→  1259→  private async heartbeat(): Promise<void> {

  1261→  1261→

  1264→  1264→      await new Promise((r) => setTimeout(r, 5 + Math.random() * 15))
  1265→  1265→      const latency = Date.now() - start
  1266→  1266→      this.metrics.latencyMs = latency
  1267→  1267→      this.metrics.lastHeartbeat = new Date()
  1268→  1268→

  1273→  1273→      this._heartbeatCountSinceLastPersist++
  1274→  1274→      if (this._heartbeatCountSinceLastPersist >= 6) {
  1275→  1275→        this._heartbeatCountSinceLastPersist = 0

  1277→  1277→      }
  1278→  1278→
  1279→  1279→      // Evaluate DEGRADED state

  1286→  1286→      // Evaluate DEGRADED state with failure

  1289→  1289→      // If fully disconnected (not just degraded), go to DISCONNECTED

  1291→  1291→        this.status = "DISCONNECTED"
  1292→  1292→        await this.setStatus("DISCONNECTED")
  1293→  1293→        emitStatusChange(this.status, this.metrics)
  1294→  1294→        this.scheduleReconnect()
  1295→  1295→      }
  1296→  1296→    }
  1297→  1297→  }

  1397→  1397→export type CircuitBreakerState = "CLOSED" | "OPEN" | "HALF_OPEN"

  1399→  1399→/**
  1400→  1400→ * Error thrown when the circuit breaker is OPEN and rejects a call.
  1401→  1401→ */
  1402→  1402→export class CircuitBreakerOpenError extends Error {
  1403→  1403→  public readonly state: CircuitBreakerState
  1404→  1404→  public readonly failureCount: number
  1405→  1405→  public readonly nextRetryAt: Date
  1406→  1406→
  1407→  1407→  constructor(failureCount: number, nextRetryAt: Date) {
  1408→  1408→    super(
  1409→  1409→      `Circuit breaker is OPEN (${failureCount} failures). Next retry allowed at ${nextRetryAt.toISOString()}`
  1410→  1410→    )
  1411→  1411→    this.name = "CircuitBreakerOpenError"
  1412→  1412→    this.state = "OPEN"
  1413→  1413→    this.failureCount = failureCount
  1414→  1414→    this.nextRetryAt = nextRetryAt
  1415→  1415→  }
  1416→  1416→}
  1417→  1417→
  1418→  1418→interface CircuitBreakerConfig {
  1419→  1419→  /** Number of consecutive failures before tripping to OPEN */
  1420→  1420→  failureThreshold: number
  1421→  1421→  /** Milliseconds to wait before transitioning OPEN → HALF_OPEN */
  1422→  1422→  recoveryTimeoutMs: number
  1423→  1423→  /** Maximum calls allowed during HALF_OPEN probe */
  1424→  1424→  halfOpenMaxAttempts: number
  1425→  1425→}
  1426→  1426→
  1427→  1427→/**
  1428→  1428→ * Circuit breaker that wraps MT5 API calls to prevent cascading failures.
  1429→  1429→ *
  1430→  1430→ * States:
  1431→  1431→ *   CLOSED  — Normal operation. Failures are counted; successes reset the counter.
  1432→  1432→ *   OPEN    — Tripped. All calls are rejected immediately with CircuitBreakerOpenError.
  1433→  1433→ *   HALF_OPEN — Recovery probe. A limited number of calls are allowed to test the service.
  1434→  1434→ */
  1435→  1435→export class CircuitBreaker {
  1436→  1436→  private _state: CircuitBreakerState = "CLOSED"
  1437→  1437→  private _failureCount = 0
  1438→  1438→  private _successCount = 0
  1439→  1439→  private _halfOpenAttempts = 0
  1440→  1440→  private _openedAt: Date | null = null
  1441→  1441→  private readonly _config: CircuitBreakerConfig
  1442→  1442→
  1443→  1443→  constructor(config?: Partial<CircuitBreakerConfig>) {
  1444→  1444→    this._config = {
  1445→  1445→      failureThreshold: config?.failureThreshold ?? 5,
  1446→  1446→      recoveryTimeoutMs: config?.recoveryTimeoutMs ?? 30_000,
  1447→  1447→      halfOpenMaxAttempts: config?.halfOpenMaxAttempts ?? 1,
  1448→  1448→    }
  1449→  1449→  }
  1450→  1450→
  1451→  1451→  get state(): CircuitBreakerState {
  1452→  1452→    this._maybeTransitionToHalfOpen()
  1453→  1453→    return this._state
  1454→  1454→  }
  1455→  1455→
  1456→  1456→  get failureCount(): number {
  1457→  1457→    return this._failureCount
  1458→  1458→  }
  1459→  1459→
  1460→  1460→  get successCount(): number {
  1461→  1461→    return this._successCount
  1462→  1462→  }
  1463→  1463→
  1464→  1464→  /** Execute a function through the circuit breaker. */
  1465→  1465→  async execute<T>(fn: () => Promise<T>): Promise<T> {
  1466→  1466→    this._maybeTransitionToHalfOpen()
  1467→  1467→
  1468→  1468→    if (this._state === "OPEN") {
  1469→  1469→      throw new CircuitBreakerOpenError(this._failureCount, this._nextRetryAt())
  1470→  1470→    }
  1471→  1471→
  1472→  1472→    if (this._state === "HALF_OPEN") {
  1473→  1473→      if (this._halfOpenAttempts >= this._config.halfOpenMaxAttempts) {
  1474→  1474→        throw new CircuitBreakerOpenError(this._failureCount, this._nextRetryAt())
  1475→  1475→      }
  1476→  1476→      this._halfOpenAttempts++
  1477→  1477→    }
  1478→  1478→
  1479→  1479→    try {
  1480→  1480→      const result = await fn()
  1481→  1481→      this._onSuccess()
  1482→  1482→      return result
  1483→  1483→    } catch (err) {
  1484→  1484→      this._onFailure()
  1485→  1485→      throw err
  1486→  1486→    }
  1487→  1487→  }
  1488→  1488→
  1489→  1489→  /** Manually reset the circuit breaker to CLOSED. */
  1490→  1490→  reset(): void {
  1491→  1491→    this._state = "CLOSED"
  1492→  1492→    this._failureCount = 0
  1493→  1493→    this._successCount = 0
  1494→  1494→    this._halfOpenAttempts = 0
  1495→  1495→    this._openedAt = null
  1496→  1496→    logger.info("MT5_CONNECTION", "Circuit breaker manually reset to CLOSED")
  1497→  1497→  }
  1498→  1498→
  1499→  1499→  /** Manually trip the circuit breaker to OPEN. */
  1500→  1500→  trip(): void {
  1501→  1501→    this._state = "OPEN"
  1502→  1502→    this._openedAt = new Date()
  1503→  1503→    logger.warn("MT5_CONNECTION", `Circuit breaker manually tripped to OPEN (${this._failureCount} failures)`)
  1504→  1504→  }
  1505→  1505→
  1506→  1506→  private _onSuccess(): void {
  1507→  1507→    this._successCount++
  1508→  1508→    if (this._state === "HALF_OPEN") {
  1509→  1509→      // Recovery probe succeeded → close the circuit
  1510→  1510→      this._state = "CLOSED"
  1511→  1511→      this._failureCount = 0
  1512→  1512→      this._halfOpenAttempts = 0
  1513→  1513→      this._openedAt = null
  1514→  1514→      logger.info("MT5_CONNECTION", "HALF_OPEN probe succeeded → CLOSED")
  1515→  1515→    } else {
  1516→  1516→      // In CLOSED state, reset failure counter on success
  1517→  1517→      this._failureCount = 0
  1518→  1518→    }
  1519→  1519→  }
  1520→  1520→
  1521→  1521→  private _onFailure(): void {
  1522→  1522→    this._failureCount++
  1523→  1523→    if (this._state === "HALF_OPEN") {
  1524→  1524→      // Recovery probe failed → back to OPEN
  1525→  1525→      this._state = "OPEN"
  1526→  1526→      this._openedAt = new Date()
  1527→  1527→      this._halfOpenAttempts = 0
  1528→  1528→      logger.warn(
  1529→  1529→        "MT5_CONNECTION",
  1530→  1530→        `HALF_OPEN probe failed → OPEN (failures: ${this._failureCount})`
  1531→  1531→      )
  1532→  1532→    } else if (this._failureCount >= this._config.failureThreshold) {
  1533→  1533→      // Threshold reached → trip to OPEN
  1534→  1534→      this._state = "OPEN"
  1535→  1535→      this._openedAt = new Date()
  1536→  1536→      logger.warn(
  1537→  1537→        "MT5_CONNECTION",
  1538→  1538→        `CLOSED → OPEN (failures: ${this._failureCount}/${this._config.failureThreshold})`
  1539→  1539→      )
  1540→  1540→    }
  1541→  1541→  }
  1542→  1542→
  1543→  1543→  private _maybeTransitionToHalfOpen(): void {
  1544→  1544→    if (this._state === "OPEN" && this._openedAt) {
  1545→  1545→      const elapsed = Date.now() - this._openedAt.getTime()
  1546→  1546→      if (elapsed >= this._config.recoveryTimeoutMs) {
  1547→  1547→        this._state = "HALF_OPEN"
  1548→  1548→        this._halfOpenAttempts = 0
  1549→  1549→        logger.info("MT5_CONNECTION", "OPEN → HALF_OPEN (recovery timeout elapsed)")
  1550→  1550→      }
  1551→  1551→    }
  1552→  1552→  }
  1553→  1553→
  1554→  1554→  private _nextRetryAt(): Date {
  1555→  1555→    if (!this._openedAt) return new Date()
  1556→  1556→    return new Date(this._openedAt.getTime() + this._config.recoveryTimeoutMs)
  1557→  1557→  }
  1558→  1558→}
  1559→  1559→
  1560→  1560→// ============================================
  1561→  1561→// CONNECTION QUALITY SCORE
  1562→  1562→// ============================================
  1563→  1563→
  1564→  1564→interface ConnectionQualityParams {
  1565→  1565→  latencyMs: number
  1566→  1566→  successRate: number       // 0.0 – 1.0
  1567→  1567→  consecutiveFailures: number
  1568→  1568→  uptimeSeconds: number
  1569→  1569→}
  1570→  1570→
  1571→  1571→/**
  1572→  1572→ * Calculate a single 0–100 connection quality score from multiple metrics.
  1573→  1573→ *
  1574→  1574→ * Components:
  1575→  1575→ *   - Latency score:   <50ms → 100, 50–100 → 90, 100–200 → 70, 200–500 → 40, >500 → 10
  1576→  1576→ *   - Success rate:    weighted 40%
  1577→  1577→ *   - Consecutive failures: 0 → 100, 1 → 80, 2 → 50, 3+ → 10
  1578→  1578→ *   - Uptime:          <1min → 50, 1–10min → 70, >10min → 90, >1h → 100
  1579→  1579→ *
  1580→  1580→ * Final = latency*30% + successRate*40% + consecutiveFailures*15% + uptime*15%
  1581→  1581→ */
  1582→  1582→export function calculateConnectionQuality(params: ConnectionQualityParams): number {
  1583→  1583→  const { latencyMs, successRate, consecutiveFailures, uptimeSeconds } = params
  1584→  1584→
  1585→  1585→  // --- Latency score (30% weight) ---
  1586→  1586→  let latencyScore: number
  1587→  1587→  if (latencyMs < 50) {
  1588→  1588→    latencyScore = 100
  1589→  1589→  } else if (latencyMs <= 100) {
  1590→  1590→    latencyScore = 90
  1591→  1591→  } else if (latencyMs <= 200) {
  1592→  1592→    latencyScore = 70
  1593→  1593→  } else if (latencyMs <= 500) {
  1594→  1594→    latencyScore = 40
  1595→  1595→  } else {
  1596→  1596→    latencyScore = 10
  1597→  1597→  }
  1598→  1598→
  1599→  1599→  // --- Success rate score (40% weight, already 0-100 when multiplied) ---
  1600→  1600→  const successRateScore = successRate * 100
  1601→  1601→
  1602→  1602→  // --- Consecutive failures score (15% weight) ---
  1603→  1603→  let failureScore: number
  1604→  1604→  if (consecutiveFailures === 0) {
  1605→  1605→    failureScore = 100
  1606→  1606→  } else if (consecutiveFailures === 1) {
  1607→  1607→    failureScore = 80
  1608→  1608→  } else if (consecutiveFailures === 2) {
  1609→  1609→    failureScore = 50
  1610→  1610→  } else {
  1611→  1611→    failureScore = 10
  1612→  1612→  }
  1613→  1613→
  1614→  1614→  // --- Uptime score (15% weight) ---
  1615→  1615→  let uptimeScore: number
  1616→  1616→  if (uptimeSeconds < 60) {
  1617→  1617→    uptimeScore = 50
  1618→  1618→  } else if (uptimeSeconds <= 600) {
  1619→  1619→    uptimeScore = 70
  1620→  1620→  } else if (uptimeSeconds <= 3600) {
  1621→  1621→    uptimeScore = 90
  1622→  1622→  } else {
  1623→  1623→    uptimeScore = 100
  1624→  1624→  }
  1625→  1625→
  1626→  1626→  const total =
  1627→  1627→    latencyScore * 0.30 +
  1628→  1628→    successRateScore * 0.40 +
  1629→  1629→    failureScore * 0.15 +
  1630→  1630→    uptimeScore * 0.15
  1631→  1631→
  1632→  1632→  // Clamp to 0-100
  1633→  1633→  return Math.round(Math.max(0, Math.min(100, total)))
  1634→  1634→}
