    74→// Simple String Hash (non-crypto, for dedup fingerprint)
    75→
    76→function simpleHash(str: string): string {
    77→  let hash = 5381
    78→  for (let i = 0; i < str.length; i++) {
    79→    hash = ((hash << 5) + hash + str.charCodeAt(i)) & 0x7fffffff
    80→  }
    81→  return hash.toString(36)
    82→}

    84→// 1. Dedup Cache
    85→
    86→interface DedupEntry {
    87→  count: number
    88→  lastSeen: Date
    89→  firstSeen: Date
    90→}
    91→
    92→class DedupCache {
    93→  private cache = new Map<string, DedupEntry>()
    94→  private windowMs: number
    95→  private flushTimer: ReturnType<typeof setInterval> | null = null
    96→
    97→  constructor(windowMs: number) {
    98→    this.windowMs = windowMs
    99→    this.flushTimer = setInterval(() => {
   100→      const expired = this.sweepExpired()
   101→      for (const e of expired) {
   102→        console.log(`[Logger-Dedup] Flushed fingerprint=${e.fingerprint} count=${e.count}`)
   103→      }
   104→    }, 5000)
   105→  }
   106→
   107→  checkAndIncrement(fingerprint: string): { isDuplicate: boolean; entry: DedupEntry | null } {
   108→    const now = new Date()
   109→    const existing = this.cache.get(fingerprint)
   110→
   111→    if (existing && now.getTime() - existing.firstSeen.getTime() < this.windowMs) {
   112→      existing.count++
   113→      existing.lastSeen = now
   114→      return { isDuplicate: true, entry: existing }
   115→    }
   116→
   117→    if (existing) this.cache.delete(fingerprint)
   118→
   119→    const entry: DedupEntry = { count: 1, lastSeen: now, firstSeen: now }
   120→    this.cache.set(fingerprint, entry)
   121→    return { isDuplicate: false, entry: null }
   122→  }
   123→
   124→  sweepExpired(): Array<{ fingerprint: string; count: number; firstSeen: Date }> {
   125→    const now = new Date()
   126→    const expired: Array<{ fingerprint: string; count: number; firstSeen: Date }> = []
   127→    for (const [fp, entry] of this.cache) {
   128→      if (now.getTime() - entry.firstSeen.getTime() >= this.windowMs) {
   129→        if (entry.count > 1) expired.push({ fingerprint: fp, count: entry.count, firstSeen: entry.firstSeen })
   130→        this.cache.delete(fp)
   131→      }
   132→    }
   133→    return expired
   134→  }
   135→
   136→  shutdown() { if (this.flushTimer) clearInterval(this.flushTimer) }
   137→}

   148→export async function cleanupOldLogs(): Promise<{ tradingLogsDeleted: number; mt5LogsDeleted: number }> {
   149→  const tradingCutoff = new Date(Date.now() - retentionDays * 24 * 60 * 60 * 1000)
   150→  const mt5Cutoff = new Date(Date.now() - mt5LogRetentionDays * 24 * 60 * 60 * 1000)
   151→  let tradingLogsDeleted = 0
   152→  let mt5LogsDeleted = 0
   153→
   154→  try {
   155→    const result = await db.tradingLog.deleteMany({ where: { createdAt: { lt: tradingCutoff } } })
   156→    tradingLogsDeleted = result.count
   157→  } catch (err) {
   158→    console.error("[Logger-Cleanup] Failed to delete old TradingLogs:", err)
   159→  }
   160→
   161→  try {
   162→    const result = await db.mt5ConnectionLog.deleteMany({ where: { createdAt: { lt: mt5Cutoff } } })
   163→    mt5LogsDeleted = result.count
   164→  } catch (err) {
   165→    console.error("[Logger-Cleanup] Failed to delete old Mt5ConnectionLogs:", err)
   166→  }
   167→
   168→  if (tradingLogsDeleted > 0 || mt5LogsDeleted > 0) {
   169→    console.log(`[Logger-Cleanup] Deleted ${tradingLogsDeleted} TradingLogs (>${retentionDays}d) + ${mt5LogsDeleted} Mt5ConnectionLogs (>${mt5LogRetentionDays}d)`)
   170→  }
   171→  return { tradingLogsDeleted, mt5LogsDeleted }
   172→}

   180→// 3. API Rate Limit Tracking
   181→
   182→type ApiService = "FINNHUB" | "MARKETAUX" | "MT5"
   183→
   184→const RATE_LIMITS: Record<ApiService, { limitPerMinute: number; limitPerHour: number }> = {
   185→  FINNHUB: { limitPerMinute: 60, limitPerHour: 3600 },
   186→  MARKETAUX: { limitPerMinute: 100, limitPerHour: 6000 },
   187→  MT5: { limitPerMinute: 120, limitPerHour: 7200 },
   188→}
   189→
   190→interface RateCounter {
   191→  callsThisMinute: number
   192→  callsThisHour: number
   193→  minuteStart: number
   194→  hourStart: number
   195→}
   196→
   197→class RateLimitTracker {
   198→  private counters = new Map<ApiService, RateCounter>()
   199→  private persistTimer: ReturnType<typeof setInterval> | null = null
   200→  private initialized = false
   201→
   202→  init() {
   203→    if (this.initialized) return
   204→    this.initialized = true
   205→    const now = Date.now()
   206→    for (const svc of ["FINNHUB", "MARKETAUX", "MT5"] as ApiService[]) {
   207→      this.counters.set(svc, { callsThisMinute: 0, callsThisHour: 0, minuteStart: now, hourStart: now })
   208→    }
   209→    this.persistTimer = setInterval(() => { void this.persistToDb() }, 30_000)
   210→  }
   211→
   212→  private getCounter(service: ApiService): RateCounter {
   213→    this.init()
   214→    return this.counters.get(service)!
   215→  }
   216→
   217→  private resetIfNeeded(counter: RateCounter, now: number) {
   218→    if (now - counter.minuteStart >= 60_000) { counter.callsThisMinute = 0; counter.minuteStart = now }
   219→    if (now - counter.hourStart >= 3_600_000) { counter.callsThisHour = 0; counter.hourStart = now }
   220→  }
   221→
   222→  async track(service: ApiService): Promise<boolean> {
   223→    this.init()
   224→    const counter = this.getCounter(service)
   225→    const config = RATE_LIMITS[service]
   226→    const now = Date.now()
   227→    this.resetIfNeeded(counter, now)
   228→
   229→    const minuteUsage = counter.callsThisMinute / config.limitPerMinute
   230→    const hourUsage = counter.callsThisHour / config.limitPerHour
   231→    const usage = Math.max(minuteUsage, hourUsage)
   232→
   233→    if (minuteUsage >= 1.0 || hourUsage >= 1.0) {
   234→      logger.critical("API_RATE_LIMIT", `${service} rate limit EXCEEDED: ${counter.callsThisMinute}/${config.limitPerMinute}/min, ${counter.callsThisHour}/${config.limitPerHour}/hour`, {
   235→        metadata: { service, callsThisMinute: counter.callsThisMinute, limitPerMinute: config.limitPerMinute },
   236→      })
   237→      return false
   238→    }
   239→
   240→    if (usage >= 0.95) {
   241→      logger.error("API_RATE_LIMIT", `${service} rate limit CRITICAL (${(usage * 100).toFixed(1)}%)`, {
   242→        metadata: { service, usagePct: Math.round(usage * 100) },
   243→      })
   244→      await new Promise((r) => setTimeout(r, 1000))
   245→    } else if (usage >= 0.8) {
   246→      logger.warn("API_RATE_LIMIT", `${service} rate limit approaching (${(usage * 100).toFixed(1)}%)`, {
   247→        metadata: { service, usagePct: Math.round(usage * 100) },
   248→      })
   249→    }
   250→
   251→    counter.callsThisMinute++
   252→    counter.callsThisHour++
   253→    return true
   254→  }
   255→
   256→  async persistToDb() {
   257→    const now = Date.now()
   258→    for (const [service, counter] of this.counters) {
   259→      this.resetIfNeeded(counter, now)
   260→      try {
   261→        await db.apiRateLimit.upsert({
   262→          where: { id: `rate-${service}` },
   263→          create: { id: `rate-${service}`, service, callsThisMinute: counter.callsThisMinute, callsThisHour: counter.callsThisHour, limitPerMinute: RATE_LIMITS[service].limitPerMinute, limitPerHour: RATE_LIMITS[service].limitPerHour, lastCallAt: new Date(now) },
   264→          update: { callsThisMinute: counter.callsThisMinute, callsThisHour: counter.callsThisHour, lastCallAt: new Date(now) },
   265→        })
   266→      } catch { /* persist failure is non-critical */ }
   267→    }
   268→  }
   269→
   270→  shutdown() { if (this.persistTimer) clearInterval(this.persistTimer) }
   271→}

   320→export function handleMt5Error(
   321→  code: number,
   322→  context?: { symbol?: string; tradeId?: string; orderType?: string }
   323→): Mt5ErrorResult {
   324→  const def = MT5_ERROR_MAP.get(code)
   325→  const description = def?.description ?? `Unknown MT5 error code ${code}`
   326→  const severity = def?.severity ?? "ERROR"
   327→  const category = def?.category ?? "TRADE_EXECUTION"
   328→  const shouldRetry = def?.retryable ?? false
   329→  const delayMs = def?.delayMs ?? 0
   330→  const action = def?.action ?? "No auto-remediation available"
   331→
   332→  logger[severity === "INFO" ? "info" : severity === "WARN" ? "warn" : severity === "ERROR" ? "error" : severity === "CRITICAL" ? "critical" : "fatal"](
   333→    category,
   334→    `MT5 Error ${code}: ${description}${context?.symbol ? ` [${context.symbol}]` : ""}${context?.tradeId ? ` trade=${context.tradeId}` : ""}`,
   335→    {
   336→      symbol: context?.symbol,
   337→      tradeId: context?.tradeId,
   338→      metadata: { code, description, shouldRetry, delayMs, action, orderType: context?.orderType },
   339→    }
   340→  )
   341→
   342→  return { shouldRetry, delayMs, action, description, severity }
   343→}

   345→// 5. Silent Failure Detection
   346→
   347→export function validateData<T>(
   348→  data: T | null | undefined,
   349→  context: string,
   350→  category: LogCategory
   351→): asserts data is T {
   352→  if (data === null || data === undefined) {
   353→    const msg = `Silent failure detected: ${context} returned null/undefined`
   354→    logger.error(category, msg, { details: "Expected non-null data but received null or undefined" })
   355→    throw new Error(msg)
   356→  }
   357→  if (Array.isArray(data) && data.length === 0) {
   358→    const msg = `Silent failure detected: ${context} returned empty array`
   359→    logger.warn(category, msg, { details: "Expected non-empty array but received []" })
   360→    throw new Error(msg)
   361→  }
   362→}

   364→// 6. Log Buffer with Dedup
   365→
   366→class LogBuffer {
   367→  private buffer: Array<{
   368→    level: LogLevel
   369→    category: LogCategory
   370→    message: string
   371→    source: string | null
   372→    details: string | null
   373→    stackTrace: string | null
   374→    tradeId: string | null
   375→    symbol: string | null
   376→    metadata: string
   377→    fingerprint: string | null
   378→  }> = []
   379→  private flushTimer: ReturnType<typeof setInterval> | null = null
   380→  private isFlushing = false
   381→  private readonly MAX_BUFFER = 50
   382→  private readonly FLUSH_INTERVAL_MS = 2000
   383→
   384→  constructor() {
   385→    this.flushTimer = setInterval(() => this.flush(), this.FLUSH_INTERVAL_MS)
   386→  }
   387→
   388→  async push(entry: Omit<typeof this.buffer[number], "fingerprint">) {
   389→    const ts = new Date().toISOString()
   390→    const color = SEVERITY_COLORS[entry.level]
   391→    console.log(`${color}[${entry.level}]${RESET} ${ts} [${entry.category}] ${entry.message}`)
   392→    if (entry.level === "ERROR" || entry.level === "CRITICAL" || entry.level === "FATAL") {
   393→      if (entry.details) console.error(`  Details: ${entry.details}`)
   394→      if (entry.stackTrace) console.error(`  Stack: ${entry.stackTrace}`)
   395→    }
   396→
   397→    // Dedup check
   398→    const fp = `${entry.category}:${entry.message.substring(0, 80)}`
   399→    const fingerprint = simpleHash(fp)
   400→    const { isDuplicate } = dedupCache.checkAndIncrement(fingerprint)
   401→    if (isDuplicate) return // Skip DB write for duplicates within window
   402→
   403→    this.buffer.push({ ...entry, fingerprint })
   404→
   405→    if (this.buffer.length >= this.MAX_BUFFER) this.flush()
   406→
   407→    // Lazy cleanup scheduling
   408→    ensureCleanupScheduled()
   409→  }
   410→
   411→  async flush() {
   412→    if (this.isFlushing || this.buffer.length === 0) return
   413→    this.isFlushing = true
   414→    const batch = [...this.buffer]
   415→    this.buffer = []
   416→
   417→    try {
   418→      await db.tradingLog.createMany({
   419→        data: batch.map((entry) => ({
   420→          level: entry.level,
   421→          category: entry.category,
   422→          message: entry.message,
   423→          source: entry.source,
   424→          details: entry.details,
   425→          stackTrace: entry.stackTrace,
   426→          tradeId: entry.tradeId,
   427→          symbol: entry.symbol,
   428→          metadata: entry.metadata,
   429→          fingerprint: entry.fingerprint,
   430→        })),
   431→      })
   432→      logHealthMonitor.recordFlushSuccess()
   433→    } catch (err) {
   434→      const errMsg = err instanceof Error ? err.message : String(err)
   435→      console.error("[Logger] FAILED TO FLUSH LOG BUFFER:", err)
   436→      logHealthMonitor.recordFlushFailure(errMsg)
   437→      this.buffer.unshift(...batch.slice(-20))
   438→    } finally {
   439→      this.isFlushing = false
   440→    }
   441→  }
   442→
   443→  async shutdown() {
   444→    if (this.flushTimer) clearInterval(this.flushTimer)
   445→    await this.flush()
   446→  }
   447→
   448→  size(): number {
   449→    return this.buffer.length
   450→  }
   451→}

   463→// Minimum log level filter
   464→let minLevel: LogLevel = (process.env.LOG_LEVEL as LogLevel) || "DEBUG"
   465→
   466→export function setMinLevel(level: LogLevel) { minLevel = level }
   467→export function getMinLevel(): LogLevel { return minLevel }
   468→
   469→// Core log function
   470→async function log(level: LogLevel, category: LogCategory, message: string, ctx: LogContext = {}) {
   471→  if (SEVERITY_PRIORITY[level] < SEVERITY_PRIORITY[minLevel]) return
   472→
   473→  const stackTrace =
   474→    level === "ERROR" || level === "CRITICAL" || level === "FATAL"
   475→      ? new Error().stack?.split("\n").slice(1, 4).join("\n") || null
   476→      : null
   477→
   478→  await getBuffer().push({
   479→    level, category, message,
   480→    source: ctx.source || null,
   481→    details: ctx.details || null,
   482→    stackTrace,
   483→    tradeId: ctx.tradeId || null,
   484→    symbol: ctx.symbol || null,
   485→    metadata: ctx.metadata ? JSON.stringify(ctx.metadata) : "{}",
   486→  })
   487→
   488→  // Auto-escalate ERROR, CRITICAL, FATAL
   489→  if (SEVERITY_PRIORITY[level] >= SEVERITY_PRIORITY["ERROR"]) {
   490→    try {
   491→      await escalationManager.escalate(level, category, message, ctx)
   492→    } catch { /* escalation failure is non-critical, don't recurse */ }
   493→  }
   494→}

   496→// Convenience methods
   497→export const logger = {
   498→  debug: (category: LogCategory, message: string, ctx?: LogContext) => log("DEBUG", category, message, ctx),
   499→  info: (category: LogCategory, message: string, ctx?: LogContext) => log("INFO", category, message, ctx),
   500→  warn: (category: LogCategory, message: string, ctx?: LogContext) => log("WARN", category, message, ctx),
   501→  error: (category: LogCategory, message: string, ctx?: LogContext) => log("ERROR", category, message, ctx),
   502→  critical: (category: LogCategory, message: string, ctx?: LogContext) => log("CRITICAL", category, message, ctx),
   503→  fatal: (category: LogCategory, message: string, ctx?: LogContext) => log("FATAL", category, message, ctx),
   504→
   505→  extractError: (err: unknown): { message: string; stack?: string } => {
   506→    if (err instanceof Error) return { message: err.message, stack: err.stack }
   507→    return { message: String(err) }
   508→  },
   509→
   510→  wrapAsync: <T extends (...args: unknown[]) => Promise<unknown>>(fn: T, category: LogCategory, context: string): T => {
   511→    return ((...args: unknown[]) => {
   512→      return fn(...args).catch((err: unknown) => {
   513→        const { message, stack } = logger.extractError(err)
   514→        logger.error(category, `${context} failed: ${message}`, { details: stack, metadata: { args: JSON.stringify(args).slice(0, 500) } })
   515→        throw err
   516→      })
   517→    }) as T
   518→  },
   519→
   520→  shutdown: () => getBuffer().shutdown(),
   521→}

   531→export async function getLogAnalytics(): Promise<LogAnalytics> {
   532→  const now = new Date()
   533→  const lastHour = new Date(now.getTime() - 60 * 60 * 1000)
   534→  const prevHourStart = new Date(now.getTime() - 2 * 60 * 60 * 1000)
   535→
   536→  const [lastHourErrors, prevHourErrors, topCategories, topMessages] = await Promise.all([
   537→    db.tradingLog.count({ where: { level: { in: ["ERROR", "CRITICAL", "FATAL"] }, createdAt: { gte: lastHour } } }),
   538→    db.tradingLog.count({ where: { level: { in: ["ERROR", "CRITICAL", "FATAL"] }, createdAt: { gte: prevHourStart, lt: lastHour } } }),
   539→    db.tradingLog.groupBy({ by: ["category"], _count: true, orderBy: { _count: { category: "desc" } }, take: 5 }),
   540→    db.tradingLog.groupBy({ by: ["message"], where: { level: { in: ["ERROR", "CRITICAL", "FATAL"] } }, _count: true, orderBy: { _count: { message: "desc" } }, take: 5 }),
   541→  ])
   542→
   543→  // Error burst detection: >10 errors in 5 min
   544→  const burstStart = new Date(now.getTime() - 5 * 60 * 1000)
   545→  const burstLogs = await db.tradingLog.findMany({
   546→    where: { level: { in: ["ERROR", "CRITICAL", "FATAL"] }, createdAt: { gte: burstStart } },
   547→    orderBy: { createdAt: "asc" },
   548→  })
   549→
   550→  const bursts: LogAnalytics["bursts"] = []
   551→  if (burstLogs.length > 10) {
   552→    const msgCounts = new Map<string, number>()
   553→    for (const l of burstLogs) msgCounts.set(l.message, (msgCounts.get(l.message) || 0) + 1)
   554→    let topMsg = ""
   555→    let topCount = 0
   556→    for (const [msg, count] of msgCounts) { if (count > topCount) { topMsg = msg; topCount = count } }
   557→    bursts.push({ startTime: burstLogs[0].createdAt, count: burstLogs.length, topMessage: topMsg })
   558→  }
   559→
   560→  const direction: LogAnalytics["errorRateTrend"]["direction"] =
   561→    lastHourErrors < prevHourErrors * 0.8 ? "improving" :
   562→    lastHourErrors > prevHourErrors * 1.2 ? "degrading" : "stable"
   563→
   564→  return {
   565→    errorRateTrend: { lastHour: lastHourErrors, previousHour: prevHourErrors, direction },
   566→    topCategories: topCategories.map((g) => ({ category: g.category, count: g._count })),
   567→    bursts,
   568→    topMessages: topMessages.map((g) => ({ message: g.message, count: g._count })),
   569→  }
   570→}

   706→// PHASE 3: LOG HEALTH MONITORING
   707→// ============================================
   708→
   709→export interface LogHealthResult {
   710→  isHealthy: boolean
   711→  flushSuccessRate: number
   712→  totalFlushes: number
   713→  failedFlushes: number
   714→  lastFlushTime: Date | null
   715→  lastFailureTime: Date | null
   716→  lastFailureReason: string | null
   717→  bufferBacklog: number
   718→}
   719→
   720→class LogHealthMonitor {
   721→  private recentFlushes: Array<{ success: boolean; time: Date }> = []
   722→  private readonly WINDOW_SIZE = 20
   723→  private _lastFlushTime: Date | null = null
   724→  private _lastFailureTime: Date | null = null
   725→  private _lastFailureReason: string | null = null
   726→  private _totalFlushes = 0
   727→  private _failedFlushes = 0
   728→
   729→  recordFlushSuccess(): void {
   730→    const now = new Date()
   731→    this.recentFlushes.push({ success: true, time: now })
   732→    if (this.recentFlushes.length > this.WINDOW_SIZE) {
   733→      this.recentFlushes.shift()
   734→    }
   735→    this._lastFlushTime = now
   736→    this._totalFlushes++
   737→  }
   738→
   739→  recordFlushFailure(error: string): void {
   740→    const now = new Date()
   741→    this.recentFlushes.push({ success: false, time: now })
   742→    if (this.recentFlushes.length > this.WINDOW_SIZE) {
   743→      this.recentFlushes.shift()
   744→    }
   745→    this._lastFlushTime = now
   746→    this._lastFailureTime = now
   747→    this._lastFailureReason = error
   748→    this._totalFlushes++
   749→    this._failedFlushes++
   750→  }
   751→
   752→  getHealth(): LogHealthResult {
   753→    const successCount = this.recentFlushes.filter(f => f.success).length
   754→    const flushSuccessRate = this.recentFlushes.length > 0
   755→      ? (successCount / this.recentFlushes.length) * 100
   756→      : 100
   757→
   758→    const isHealthy = this.recentFlushes.length === 0 || flushSuccessRate >= 90
   759→
   760→    return {
   761→      isHealthy,
   762→      flushSuccessRate: Math.round(flushSuccessRate * 100) / 100,
   763→      totalFlushes: this._totalFlushes,
   764→      failedFlushes: this._failedFlushes,
   765→      lastFlushTime: this._lastFlushTime,
   766→      lastFailureTime: this._lastFailureTime,
   767→      lastFailureReason: this._lastFailureReason,
   768→      bufferBacklog: getBuffer().size(),
   769→    }
   770→  }
   771→}

   779→interface EscalationEventRecord {
   780→  id: string
   781→  triggerLevel: LogLevel
   782→  triggerCategory: LogCategory
   783→  triggerMessage: string
   784→  escalationLevel: number
   785→  actions: string[]
   786→  resolved: boolean
   787→  createdAt: Date
   788→}
   789→
   790→class EscalationManager {
   791→  private recentEvents: EscalationEventRecord[] = []
   792→  private readonly MAX_EVENTS = 100
   793→
   794→  async escalate(
   795→    level: LogLevel,
   796→    category: LogCategory,
   797→    message: string,
   798→    ctx?: LogContext
   799→  ): Promise<void> {
   800→    let escalationLevel: number
   801→    let actions: string[] = []
   802→
   803→    switch (level) {
   804→      case 'ERROR':
   805→        escalationLevel = 1
   806→        break
   807→      case 'CRITICAL':
   808→        escalationLevel = 2
   809→        // Create EscalationEvent in DB for CRITICAL
   810→        try {
   811→          const record = await db.escalationEvent.create({
   812→            data: {
   813→              triggerLevel: level,
   814→              triggerCategory: category,
   815→              triggerMessage: message.substring(0, 500),
   816→              escalationLevel: 2,
   817→              actions: JSON.stringify(['LOG_AND_TRACK']),
   818→              resolved: false,
   819→            },
   820→          })
   821→          this.trackEvent({
   822→            id: record.id,
   823→            triggerLevel: level,
   824→            triggerCategory: category,
   825→            triggerMessage: message,
   826→            escalationLevel: 2,
   827→            actions: ['LOG_AND_TRACK'],
   828→            resolved: false,
   829→            createdAt: record.createdAt,
   830→          })
   831→        } catch (err) {
   832→          console.error('[EscalationManager] Failed to persist CRITICAL escalation event:', err)
   833→        }
   834→        actions.push('LOG_AND_TRACK')
   835→        return
   836→
   837→      case 'FATAL':
   838→        escalationLevel = 3
   839→        // Create EscalationEvent in DB for FATAL
   840→        const fatalActions: string[] = []
   841→
   842→        // Attempt auto-recovery based on category
   843→        const recoveryAction = this.getAutoRecoveryAction(category)
   844→        fatalActions.push(recoveryAction)
   845→        actions.push(recoveryAction)
   846→
   847→        try {
   848→          const record = await db.escalationEvent.create({
   849→            data: {
   850→              triggerLevel: level,
   851→              triggerCategory: category,
   852→              triggerMessage: message.substring(0, 500),
   853→              escalationLevel: 3,
   854→              actions: JSON.stringify(fatalActions),
   855→              resolved: false,
   856→            },
   857→          })
   858→          this.trackEvent({
   859→            id: record.id,
   860→            triggerLevel: level,
   861→            triggerCategory: category,
   862→            triggerMessage: message,
   863→            escalationLevel: 3,
   864→            actions: fatalActions,
   865→            resolved: false,
   866→            createdAt: record.createdAt,
   867→          })
   868→        } catch (err) {
   869→          console.error('[EscalationManager] Failed to persist FATAL escalation event:', err)
   870→        }
   871→        return
   872→
   873→      default:
   874→        // DEBUG, INFO, WARN — no escalation
   875→        return
   876→    }
   877→  }
   878→
   879→  private getAutoRecoveryAction(category: LogCategory): string {
   880→    switch (category) {
   881→      case 'MT5_CONNECTION':
   882→        return 'suggest_reconnect'
   883→      case 'TRADE_EXECUTION':
   884→        return 'verify_all_open_positions'
   885→      case 'SYSTEM':
   886→        return 'check_system_health'
   887→      default:
   888→        return 'manual_investigation_required'
   889→    }
   890→  }
   891→
   892→  private trackEvent(event: EscalationEventRecord): void {
   893→    this.recentEvents.push(event)
   894→    if (this.recentEvents.length > this.MAX_EVENTS) {
   895→      this.recentEvents.shift()
   896→    }
   897→  }
   898→
   899→  async getPendingEscalations(): Promise<EscalationEventRecord[]> {
   900→    try {
   901→      const dbEvents = await db.escalationEvent.findMany({
   902→        where: { resolved: false },
   903→        orderBy: { createdAt: 'desc' },
   904→      })
   905→
   906→      // Merge in-memory events that may not yet be in DB
   907→      const merged = new Map<string, EscalationEventRecord>()
   908→      for (const mem of this.recentEvents) {
   909→        if (!mem.resolved) merged.set(mem.id, mem)
   910→      }
   911→      for (const dbEvt of dbEvents) {
   912→        merged.set(dbEvt.id, {
   913→          id: dbEvt.id,
   914→          triggerLevel: dbEvt.triggerLevel as LogLevel,
   915→          triggerCategory: dbEvt.triggerCategory as LogCategory,
   916→          triggerMessage: dbEvt.triggerMessage,
   917→          escalationLevel: dbEvt.escalationLevel,
   918→          actions: JSON.parse(dbEvt.actions),
   919→          resolved: dbEvt.resolved,
   920→          createdAt: dbEvt.createdAt,
   921→        })
   922→      }
   923→      return Array.from(merged.values()).sort(
   924→        (a, b) => b.createdAt.getTime() - a.createdAt.getTime()
   925→      )
   926→    } catch (err) {
   927→      console.error('[EscalationManager] Failed to query pending escalations:', err)
   928→      // Fall back to in-memory events
   929→      return this.recentEvents.filter(e => !e.resolved)
   930→    }
   931→  }
   932→
   933→  async resolveEscalation(id: string): Promise<void> {
   934→    // Update in-memory
   935→    const memEvent = this.recentEvents.find(e => e.id === id)
   936→    if (memEvent) memEvent.resolved = true
   937→
   938→    // Update DB
   939→    try {
   940→      await db.escalationEvent.update({
   941→        where: { id },
   942→        data: { resolved: true, resolvedAt: new Date() },
   943→      })
   944→    } catch (err) {
   945→      console.error(`[EscalationManager] Failed to resolve escalation ${id}:`, err)
   946→      throw err
   947→    }
   948→  }
   949→}

   958→ * Export logs with optional filters in JSON or CSV format.

   960→export async function exportLogs(params: {
   961→  level?: LogLevel
   962→  category?: LogCategory
   963→  startDate?: Date
   964→  endDate?: Date
   965→  format?: 'json' | 'csv'
   966→}): Promise<string> {
   967→  const { level, category, startDate, endDate, format = 'json' } = params
   968→
   969→  const where: Record<string, unknown> = {}
   970→  if (level) where.level = level
   971→  if (category) where.category = category
   972→  if (startDate || endDate) {
   973→    const createdAtFilter: Record<string, unknown> = {}
   974→    if (startDate) createdAtFilter.gte = startDate
   975→    if (endDate) createdAtFilter.lte = endDate
   976→    where.createdAt = createdAtFilter
   977→  }
   978→
   979→  const logs = await db.tradingLog.findMany({
   980→    where,
   981→    orderBy: { createdAt: 'desc' },
   982→    take: 10000,
   983→  })
   984→
   985→  if (format === 'csv') {
   986→    const headers = 'id,timestamp,level,category,message,source,symbol,tradeId'
   987→    const rows = logs.map(l => {
   988→      const id = l.id
   989→      const timestamp = l.createdAt.toISOString()
   990→      const lvl = l.level
   991→      const cat = l.category
   992→      // Escape double quotes in message
   993→      const msg = `"${(l.message ?? '').replace(/"/g, '""')}"`
   994→      const src = l.source ? `"${l.source.replace(/"/g, '""')}"` : ''
   995→      const sym = l.symbol ?? ''
   996→      const tid = l.tradeId ?? ''
   997→      return `${id},${timestamp},${lvl},${cat},${msg},${src},${sym},${tid}`
   998→    })
   999→    return [headers, ...rows].join('\n')
  1000→  }

  1002→  // Default: JSON format
  1003→  return JSON.stringify(logs.map(l => ({
  1004→    id: l.id,

  1010→    symbol: l.symbol,
