    74→    74→// Simple String Hash (non-crypto, for dedup fingerprint)
    75→    75→
    76→    76→function simpleHash(str: string): string {
    77→    77→  let hash = 5381
    78→    78→  for (let i = 0; i < str.length; i++) {
    79→    79→    hash = ((hash << 5) + hash + str.charCodeAt(i)) & 0x7fffffff
    80→    80→  }
    81→    81→  return hash.toString(36)
    82→    82→}

    84→    84→// 1. Dedup Cache
    85→    85→
    86→    86→interface DedupEntry {
    87→    87→  count: number
    88→    88→  lastSeen: Date
    89→    89→  firstSeen: Date
    90→    90→}
    91→    91→
    92→    92→class DedupCache {
    93→    93→  private cache = new Map<string, DedupEntry>()
    94→    94→  private windowMs: number
    95→    95→  private flushTimer: ReturnType<typeof setInterval> | null = null
    96→    96→
    97→    97→  constructor(windowMs: number) {
    98→    98→    this.windowMs = windowMs
    99→    99→    this.flushTimer = setInterval(() => {
   100→   100→      const expired = this.sweepExpired()
   101→   101→      for (const e of expired) {
   102→   102→        console.log(`[Logger-Dedup] Flushed fingerprint=${e.fingerprint} count=${e.count}`)
   103→   103→      }
   104→   104→    }, 5000)
   105→   105→  }
   106→   106→
   107→   107→  checkAndIncrement(fingerprint: string): { isDuplicate: boolean; entry: DedupEntry | null } {
   108→   108→    const now = new Date()
   109→   109→    const existing = this.cache.get(fingerprint)
   110→   110→
   111→   111→    if (existing && now.getTime() - existing.firstSeen.getTime() < this.windowMs) {
   112→   112→      existing.count++
   113→   113→      existing.lastSeen = now
   114→   114→      return { isDuplicate: true, entry: existing }
   115→   115→    }
   116→   116→
   117→   117→    if (existing) this.cache.delete(fingerprint)
   118→   118→
   119→   119→    const entry: DedupEntry = { count: 1, lastSeen: now, firstSeen: now }
   120→   120→    this.cache.set(fingerprint, entry)
   121→   121→    return { isDuplicate: false, entry: null }
   122→   122→  }
   123→   123→
   124→   124→  sweepExpired(): Array<{ fingerprint: string; count: number; firstSeen: Date }> {
   125→   125→    const now = new Date()
   126→   126→    const expired: Array<{ fingerprint: string; count: number; firstSeen: Date }> = []
   127→   127→    for (const [fp, entry] of this.cache) {
   128→   128→      if (now.getTime() - entry.firstSeen.getTime() >= this.windowMs) {
   129→   129→        if (entry.count > 1) expired.push({ fingerprint: fp, count: entry.count, firstSeen: entry.firstSeen })
   130→   130→        this.cache.delete(fp)
   131→   131→      }
   132→   132→    }
   133→   133→    return expired
   134→   134→  }
   135→   135→
   136→   136→  shutdown() { if (this.flushTimer) clearInterval(this.flushTimer) }
   137→   137→}

   180→   180→// 3. API Rate Limit Tracking
   181→   181→
   182→   182→type ApiService = "FINNHUB" | "MARKETAUX" | "MT5"
   183→   183→
   184→   184→const RATE_LIMITS: Record<ApiService, { limitPerMinute: number; limitPerHour: number }> = {
   185→   185→  FINNHUB: { limitPerMinute: 60, limitPerHour: 3600 },
   186→   186→  MARKETAUX: { limitPerMinute: 100, limitPerHour: 6000 },
   187→   187→  MT5: { limitPerMinute: 120, limitPerHour: 7200 },
   188→   188→}
   189→   189→
   190→   190→interface RateCounter {
   191→   191→  callsThisMinute: number
   192→   192→  callsThisHour: number
   193→   193→  minuteStart: number
   194→   194→  hourStart: number
   195→   195→}
   196→   196→
   197→   197→class RateLimitTracker {
   198→   198→  private counters = new Map<ApiService, RateCounter>()
   199→   199→  private persistTimer: ReturnType<typeof setInterval> | null = null
   200→   200→  private initialized = false
   201→   201→
   202→   202→  init() {
   203→   203→    if (this.initialized) return
   204→   204→    this.initialized = true
   205→   205→    const now = Date.now()
   206→   206→    for (const svc of ["FINNHUB", "MARKETAUX", "MT5"] as ApiService[]) {
   207→   207→      this.counters.set(svc, { callsThisMinute: 0, callsThisHour: 0, minuteStart: now, hourStart: now })
   208→   208→    }
   209→   209→    this.persistTimer = setInterval(() => { void this.persistToDb() }, 30_000)
   210→   210→  }
   211→   211→
   212→   212→  private getCounter(service: ApiService): RateCounter {
   213→   213→    this.init()
   214→   214→    return this.counters.get(service)!
   215→   215→  }
   216→   216→
   217→   217→  private resetIfNeeded(counter: RateCounter, now: number) {
   218→   218→    if (now - counter.minuteStart >= 60_000) { counter.callsThisMinute = 0; counter.minuteStart = now }
   219→   219→    if (now - counter.hourStart >= 3_600_000) { counter.callsThisHour = 0; counter.hourStart = now }
   220→   220→  }
   221→   221→
   222→   222→  async track(service: ApiService): Promise<boolean> {
   223→   223→    this.init()
   224→   224→    const counter = this.getCounter(service)
   225→   225→    const config = RATE_LIMITS[service]
   226→   226→    const now = Date.now()
   227→   227→    this.resetIfNeeded(counter, now)
   228→   228→
   229→   229→    const minuteUsage = counter.callsThisMinute / config.limitPerMinute
   230→   230→    const hourUsage = counter.callsThisHour / config.limitPerHour
   231→   231→    const usage = Math.max(minuteUsage, hourUsage)
   232→   232→
   233→   233→    if (minuteUsage >= 1.0 || hourUsage >= 1.0) {
   234→   234→      logger.critical("API_RATE_LIMIT", `${service} rate limit EXCEEDED: ${counter.callsThisMinute}/${config.limitPerMinute}/min, ${counter.callsThisHour}/${config.limitPerHour}/hour`, {
   235→   235→        metadata: { service, callsThisMinute: counter.callsThisMinute, limitPerMinute: config.limitPerMinute },
   236→   236→      })
   237→   237→      return false
   238→   238→    }
   239→   239→
   240→   240→    if (usage >= 0.95) {
   241→   241→      logger.error("API_RATE_LIMIT", `${service} rate limit CRITICAL (${(usage * 100).toFixed(1)}%)`, {
   242→   242→        metadata: { service, usagePct: Math.round(usage * 100) },
   243→   243→      })
   244→   244→      await new Promise((r) => setTimeout(r, 1000))
   245→   245→    } else if (usage >= 0.8) {
   246→   246→      logger.warn("API_RATE_LIMIT", `${service} rate limit approaching (${(usage * 100).toFixed(1)}%)`, {
   247→   247→        metadata: { service, usagePct: Math.round(usage * 100) },
   248→   248→      })
   249→   249→    }
   250→   250→
   251→   251→    counter.callsThisMinute++
   252→   252→    counter.callsThisHour++
   253→   253→    return true
   254→   254→  }
   255→   255→
   256→   256→  async persistToDb() {
   257→   257→    const now = Date.now()
   258→   258→    for (const [service, counter] of this.counters) {
   259→   259→      this.resetIfNeeded(counter, now)
   260→   260→      try {
   261→   261→        await db.apiRateLimit.upsert({
   262→   262→          where: { id: `rate-${service}` },
   263→   263→          create: { id: `rate-${service}`, service, callsThisMinute: counter.callsThisMinute, callsThisHour: counter.callsThisHour, limitPerMinute: RATE_LIMITS[service].limitPerMinute, limitPerHour: RATE_LIMITS[service].limitPerHour, lastCallAt: new Date(now) },
   264→   264→          update: { callsThisMinute: counter.callsThisMinute, callsThisHour: counter.callsThisHour, lastCallAt: new Date(now) },
   265→   265→        })
   266→   266→      } catch { /* persist failure is non-critical */ }
   267→   267→    }
   268→   268→  }
   269→   269→
   270→   270→  shutdown() { if (this.persistTimer) clearInterval(this.persistTimer) }
   271→   271→}

   320→   320→export function handleMt5Error(
   321→   321→  code: number,
   322→   322→  context?: { symbol?: string; tradeId?: string; orderType?: string }
   323→   323→): Mt5ErrorResult {
   324→   324→  const def = MT5_ERROR_MAP.get(code)
   325→   325→  const description = def?.description ?? `Unknown MT5 error code ${code}`
   326→   326→  const severity = def?.severity ?? "ERROR"
   327→   327→  const category = def?.category ?? "TRADE_EXECUTION"
   328→   328→  const shouldRetry = def?.retryable ?? false
   329→   329→  const delayMs = def?.delayMs ?? 0
   330→   330→  const action = def?.action ?? "No auto-remediation available"
   331→   331→
   332→   332→  logger[severity === "INFO" ? "info" : severity === "WARN" ? "warn" : severity === "ERROR" ? "error" : severity === "CRITICAL" ? "critical" : "fatal"](
   333→   333→    category,
   334→   334→    `MT5 Error ${code}: ${description}${context?.symbol ? ` [${context.symbol}]` : ""}${context?.tradeId ? ` trade=${context.tradeId}` : ""}`,
   335→   335→    {
   336→   336→      symbol: context?.symbol,
   337→   337→      tradeId: context?.tradeId,
   338→   338→      metadata: { code, description, shouldRetry, delayMs, action, orderType: context?.orderType },
   339→   339→    }
   340→   340→  )
   341→   341→
   342→   342→  return { shouldRetry, delayMs, action, description, severity }
   343→   343→}

   347→   347→export function validateData<T>(
   348→   348→  data: T | null | undefined,
   349→   349→  context: string,
   350→   350→  category: LogCategory
   351→   351→): asserts data is T {
   352→   352→  if (data === null || data === undefined) {
   353→   353→    const msg = `Silent failure detected: ${context} returned null/undefined`
   354→   354→    logger.error(category, msg, { details: "Expected non-null data but received null or undefined" })
   355→   355→    throw new Error(msg)
   356→   356→  }
   357→   357→  if (Array.isArray(data) && data.length === 0) {
   358→   358→    const msg = `Silent failure detected: ${context} returned empty array`
   359→   359→    logger.warn(category, msg, { details: "Expected non-empty array but received []" })
   360→   360→    throw new Error(msg)
   361→   361→  }

   390→   390→    const color = SEVERITY_COLORS[entry.level]
   391→   391→    console.log(`${color}[${entry.level}]${RESET} ${ts} [${entry.category}] ${entry.message}`)
   392→   392→    if (entry.level === "ERROR" || entry.level === "CRITICAL" || entry.level === "FATAL") {
   393→   393→      if (entry.details) console.error(`  Details: ${entry.details}`)
   394→   394→      if (entry.stackTrace) console.error(`  Stack: ${entry.stackTrace}`)
   395→   395→    }
   396→   396→
   397→   397→    // Dedup check
   398→   398→    const fp = `${entry.category}:${entry.message.substring(0, 80)}`
   399→   399→    const fingerprint = simpleHash(fp)
   400→   400→    const { isDuplicate } = dedupCache.checkAndIncrement(fingerprint)
   401→   401→    if (isDuplicate) return // Skip DB write for duplicates within window
   402→   402→
   403→   403→    this.buffer.push({ ...entry, fingerprint })
   404→   404→
   405→   405→    if (this.buffer.length >= this.MAX_BUFFER) this.flush()
   406→   406→
   407→   407→    // Lazy cleanup scheduling
   408→   408→    ensureCleanupScheduled()
   409→   409→  }
   410→   410→
   411→   411→  async flush() {
   412→   412→    if (this.isFlushing || this.buffer.length === 0) return
   413→   413→    this.isFlushing = true
   414→   414→    const batch = [...this.buffer]
   415→   415→    this.buffer = []
   416→   416→
   417→   417→    try {
   418→   418→      await db.tradingLog.createMany({
   419→   419→        data: batch.map((entry) => ({
   420→   420→          level: entry.level,
   421→   421→          category: entry.category,
   422→   422→          message: entry.message,
   423→   423→          source: entry.source,
   424→   424→          details: entry.details,
   425→   425→          stackTrace: entry.stackTrace,
   426→   426→          tradeId: entry.tradeId,
   427→   427→          symbol: entry.symbol,
   428→   428→          metadata: entry.metadata,
   429→   429→          fingerprint: entry.fingerprint,
   430→   430→        })),
   431→   431→      })
   432→   432→      logHealthMonitor.recordFlushSuccess()
   433→   433→    } catch (err) {
   434→   434→      const errMsg = err instanceof Error ? err.message : String(err)
   435→   435→      console.error("[Logger] FAILED TO FLUSH LOG BUFFER:", err)
   436→   436→      logHealthMonitor.recordFlushFailure(errMsg)
   437→   437→      this.buffer.unshift(...batch.slice(-20))
   438→   438→    } finally {
   439→   439→      this.isFlushing = false
   440→   440→    }
   441→   441→  }
   442→   442→
   443→   443→  async shutdown() {
   444→   444→    if (this.flushTimer) clearInterval(this.flushTimer)
   445→   445→    await this.flush()
   446→   446→  }
   447→   447→
   448→   448→  size(): number {
   449→   449→    return this.buffer.length
   450→   450→  }
   451→   451→}

   489→   489→  if (SEVERITY_PRIORITY[level] >= SEVERITY_PRIORITY["ERROR"]) {
   490→   490→    try {
   491→   491→      await escalationManager.escalate(level, category, message, ctx)
   492→   492→    } catch { /* escalation failure is non-critical, don't recurse */ }

   517→   517→    }) as T
   518→   518→  },
   519→   519→

   521→   521→}

   540→   540→    db.tradingLog.groupBy({ by: ["message"], where: { level: { in: ["ERROR", "CRITICAL", "FATAL"] } }, _count: true, orderBy: { _count: { message: "desc" } }, take: 5 }),
   541→   541→  ])

   556→   556→    for (const [msg, count] of msgCounts) { if (count > topCount) { topMsg = msg; topCount = count } }

   558→   558→  }
   559→   559→
   560→   560→  const direction: LogAnalytics["errorRateTrend"]["direction"] =
   561→   561→    lastHourErrors < prevHourErrors * 0.8 ? "improving" :
   562→   562→    lastHourErrors > prevHourErrors * 1.2 ? "degrading" : "stable"
   563→   563→
   564→   564→  return {
   565→   565→    errorRateTrend: { lastHour: lastHourErrors, previousHour: prevHourErrors, direction },
   566→   566→    topCategories: topCategories.map((g) => ({ category: g.category, count: g._count })),
   567→   567→    bursts,
   568→   568→    topMessages: topMessages.map((g) => ({ message: g.message, count: g._count })),
   569→   569→  }

   709→   709→export interface LogHealthResult {
   710→   710→  isHealthy: boolean
   711→   711→  flushSuccessRate: number
   712→   712→  totalFlushes: number
   713→   713→  failedFlushes: number
   714→   714→  lastFlushTime: Date | null
   715→   715→  lastFailureTime: Date | null
   716→   716→  lastFailureReason: string | null
   717→   717→  bufferBacklog: number
   718→   718→}
   719→   719→
   720→   720→class LogHealthMonitor {
   721→   721→  private recentFlushes: Array<{ success: boolean; time: Date }> = []
   722→   722→  private readonly WINDOW_SIZE = 20
   723→   723→  private _lastFlushTime: Date | null = null
   724→   724→  private _lastFailureTime: Date | null = null
   725→   725→  private _lastFailureReason: string | null = null
   726→   726→  private _totalFlushes = 0
   727→   727→  private _failedFlushes = 0
   728→   728→
   729→   729→  recordFlushSuccess(): void {
   730→   730→    const now = new Date()
   731→   731→    this.recentFlushes.push({ success: true, time: now })
   732→   732→    if (this.recentFlushes.length > this.WINDOW_SIZE) {
   733→   733→      this.recentFlushes.shift()
   734→   734→    }
   735→   735→    this._lastFlushTime = now
   736→   736→    this._totalFlushes++
   737→   737→  }
   738→   738→
   739→   739→  recordFlushFailure(error: string): void {
   740→   740→    const now = new Date()
   741→   741→    this.recentFlushes.push({ success: false, time: now })
   742→   742→    if (this.recentFlushes.length > this.WINDOW_SIZE) {
   743→   743→      this.recentFlushes.shift()
   744→   744→    }
   745→   745→    this._lastFlushTime = now
   746→   746→    this._lastFailureTime = now
   747→   747→    this._lastFailureReason = error
   748→   748→    this._totalFlushes++
   749→   749→    this._failedFlushes++
   750→   750→  }
   751→   751→
   752→   752→  getHealth(): LogHealthResult {
   753→   753→    const successCount = this.recentFlushes.filter(f => f.success).length
   754→   754→    const flushSuccessRate = this.recentFlushes.length > 0
   755→   755→      ? (successCount / this.recentFlushes.length) * 100
   756→   756→      : 100
   757→   757→
   758→   758→    const isHealthy = this.recentFlushes.length === 0 || flushSuccessRate >= 90
   759→   759→
   760→   760→    return {
   761→   761→      isHealthy,
   762→   762→      flushSuccessRate: Math.round(flushSuccessRate * 100) / 100,
   763→   763→      totalFlushes: this._totalFlushes,
   764→   764→      failedFlushes: this._failedFlushes,
   765→   765→      lastFlushTime: this._lastFlushTime,
   766→   766→      lastFailureTime: this._lastFailureTime,
   767→   767→      lastFailureReason: this._lastFailureReason,
   768→   768→      bufferBacklog: getBuffer().size(),
   769→   769→    }
   770→   770→  }
   771→   771→}

   790→   790→class EscalationManager {
   791→   791→  private recentEvents: EscalationEventRecord[] = []

   793→   793→
   794→   794→  async escalate(
   795→   795→    level: LogLevel,
   796→   796→    category: LogCategory,

   800→   800→    let escalationLevel: number
   801→   801→    let actions: string[] = []
   802→   802→
   803→   803→    switch (level) {
   804→   804→      case 'ERROR':
   805→   805→        escalationLevel = 1
   806→   806→        break
   807→   807→      case 'CRITICAL':
   808→   808→        escalationLevel = 2
   809→   809→        // Create EscalationEvent in DB for CRITICAL
   810→   810→        try {
   811→   811→          const record = await db.escalationEvent.create({
   812→   812→            data: {
   813→   813→              triggerLevel: level,
   814→   814→              triggerCategory: category,
   815→   815→              triggerMessage: message.substring(0, 500),
   816→   816→              escalationLevel: 2,
   817→   817→              actions: JSON.stringify(['LOG_AND_TRACK']),
   818→   818→              resolved: false,
   819→   819→            },
   820→   820→          })
   821→   821→          this.trackEvent({
   822→   822→            id: record.id,
   823→   823→            triggerLevel: level,
   824→   824→            triggerCategory: category,
   825→   825→            triggerMessage: message,
   826→   826→            escalationLevel: 2,
   827→   827→            actions: ['LOG_AND_TRACK'],
   828→   828→            resolved: false,
   829→   829→            createdAt: record.createdAt,
   830→   830→          })
   831→   831→        } catch (err) {
   832→   832→          console.error('[EscalationManager] Failed to persist CRITICAL escalation event:', err)
   833→   833→        }
   834→   834→        actions.push('LOG_AND_TRACK')
   835→   835→        return
   836→   836→
   837→   837→      case 'FATAL':
   838→   838→        escalationLevel = 3
   839→   839→        // Create EscalationEvent in DB for FATAL
   840→   840→        const fatalActions: string[] = []
   841→   841→
   842→   842→        // Attempt auto-recovery based on category
   843→   843→        const recoveryAction = this.getAutoRecoveryAction(category)
   844→   844→        fatalActions.push(recoveryAction)
   845→   845→        actions.push(recoveryAction)
   846→   846→
   847→   847→        try {
   848→   848→          const record = await db.escalationEvent.create({
   849→   849→            data: {
   850→   850→              triggerLevel: level,
   851→   851→              triggerCategory: category,
   852→   852→              triggerMessage: message.substring(0, 500),
   853→   853→              escalationLevel: 3,
   854→   854→              actions: JSON.stringify(fatalActions),
   855→   855→              resolved: false,
   856→   856→            },
   857→   857→          })
   858→   858→          this.trackEvent({
   859→   859→            id: record.id,
   860→   860→            triggerLevel: level,
   861→   861→            triggerCategory: category,
   862→   862→            triggerMessage: message,
   863→   863→            escalationLevel: 3,
   864→   864→            actions: fatalActions,
   865→   865→            resolved: false,
   866→   866→            createdAt: record.createdAt,
   867→   867→          })
   868→   868→        } catch (err) {
   869→   869→          console.error('[EscalationManager] Failed to persist FATAL escalation event:', err)
   870→   870→        }
   871→   871→        return
   872→   872→
   873→   873→      default:
   874→   874→        // DEBUG, INFO, WARN — no escalation
   875→   875→        return
   876→   876→    }
   877→   877→  }
   878→   878→
   879→   879→  private getAutoRecoveryAction(category: LogCategory): string {
   880→   880→    switch (category) {
   881→   881→      case 'MT5_CONNECTION':
   882→   882→        return 'suggest_reconnect'
   883→   883→      case 'TRADE_EXECUTION':
   884→   884→        return 'verify_all_open_positions'
   885→   885→      case 'SYSTEM':
   886→   886→        return 'check_system_health'
   887→   887→      default:
   888→   888→        return 'manual_investigation_required'
   889→   889→    }
   890→   890→  }
   891→   891→
   892→   892→  private trackEvent(event: EscalationEventRecord): void {
   893→   893→    this.recentEvents.push(event)
   894→   894→    if (this.recentEvents.length > this.MAX_EVENTS) {
   895→   895→      this.recentEvents.shift()
   896→   896→    }
   897→   897→  }
   898→   898→
   899→   899→  async getPendingEscalations(): Promise<EscalationEventRecord[]> {
   900→   900→    try {
   901→   901→      const dbEvents = await db.escalationEvent.findMany({
   902→   902→        where: { resolved: false },
   903→   903→        orderBy: { createdAt: 'desc' },
   904→   904→      })
   905→   905→
   906→   906→      // Merge in-memory events that may not yet be in DB
   907→   907→      const merged = new Map<string, EscalationEventRecord>()
   908→   908→      for (const mem of this.recentEvents) {
   909→   909→        if (!mem.resolved) merged.set(mem.id, mem)
   910→   910→      }
   911→   911→      for (const dbEvt of dbEvents) {
   912→   912→        merged.set(dbEvt.id, {
   913→   913→          id: dbEvt.id,
   914→   914→          triggerLevel: dbEvt.triggerLevel as LogLevel,
   915→   915→          triggerCategory: dbEvt.triggerCategory as LogCategory,
   916→   916→          triggerMessage: dbEvt.triggerMessage,
   917→   917→          escalationLevel: dbEvt.escalationLevel,
   918→   918→          actions: JSON.parse(dbEvt.actions),
   919→   919→          resolved: dbEvt.resolved,
   920→   920→          createdAt: dbEvt.createdAt,
   921→   921→        })
   922→   922→      }
   923→   923→      return Array.from(merged.values()).sort(
   924→   924→        (a, b) => b.createdAt.getTime() - a.createdAt.getTime()
   925→   925→      )
   926→   926→    } catch (err) {
   927→   927→      console.error('[EscalationManager] Failed to query pending escalations:', err)
   928→   928→      // Fall back to in-memory events
   929→   929→      return this.recentEvents.filter(e => !e.resolved)
   930→   930→    }
   931→   931→  }
   932→   932→
   933→   933→  async resolveEscalation(id: string): Promise<void> {
   934→   934→    // Update in-memory
   935→   935→    const memEvent = this.recentEvents.find(e => e.id === id)
   936→   936→    if (memEvent) memEvent.resolved = true
   937→   937→
   938→   938→    // Update DB
   939→   939→    try {
   940→   940→      await db.escalationEvent.update({
   941→   941→        where: { id },
   942→   942→        data: { resolved: true, resolvedAt: new Date() },
   943→   943→      })
   944→   944→    } catch (err) {
   945→   945→      console.error(`[EscalationManager] Failed to resolve escalation ${id}:`, err)
   946→   946→      throw err
   947→   947→    }
   948→   948→  }
   949→   949→}

   957→   957→/**
   958→   958→ * Export logs with optional filters in JSON or CSV format.
   959→   959→ */
   960→   960→export async function exportLogs(params: {
   961→   961→  level?: LogLevel
   962→   962→  category?: LogCategory
   963→   963→  startDate?: Date
   964→   964→  endDate?: Date
   965→   965→  format?: 'json' | 'csv'
   966→   966→}): Promise<string> {
   967→   967→  const { level, category, startDate, endDate, format = 'json' } = params
   968→   968→
   969→   969→  const where: Record<string, unknown> = {}
   970→   970→  if (level) where.level = level
   971→   971→  if (category) where.category = category
   972→   972→  if (startDate || endDate) {
   973→   973→    const createdAtFilter: Record<string, unknown> = {}
   974→   974→    if (startDate) createdAtFilter.gte = startDate
   975→   975→    if (endDate) createdAtFilter.lte = endDate
   976→   976→    where.createdAt = createdAtFilter
   977→   977→  }
   978→   978→
   979→   979→  const logs = await db.tradingLog.findMany({
   980→   980→    where,
   981→   981→    orderBy: { createdAt: 'desc' },
   982→   982→    take: 10000,
   983→   983→  })
   984→   984→
   985→   985→  if (format === 'csv') {
   986→   986→    const headers = 'id,timestamp,level,category,message,source,symbol,tradeId'
   987→   987→    const rows = logs.map(l => {
   988→   988→      const id = l.id
   989→   989→      const timestamp = l.createdAt.toISOString()
   990→   990→      const lvl = l.level
   991→   991→      const cat = l.category
   992→   992→      // Escape double quotes in message
   993→   993→      const msg = `"${(l.message ?? '').replace(/"/g, '""')}"`
   994→   994→      const src = l.source ? `"${l.source.replace(/"/g, '""')}"` : ''
   995→   995→      const sym = l.symbol ?? ''
   996→   996→      const tid = l.tradeId ?? ''
   997→   997→      return `${id},${timestamp},${lvl},${cat},${msg},${src},${sym},${tid}`
   998→   998→    })
   999→   999→    return [headers, ...rows].join('\n')
  1000→  1000→  }
  1001→  1001→
  1002→  1002→  // Default: JSON format
  1003→  1003→  return JSON.stringify(logs.map(l => ({
  1004→  1004→    id: l.id,
  1005→  1005→    timestamp: l.createdAt.toISOString(),
  1006→  1006→    level: l.level,
  1007→  1007→    category: l.category,
  1008→  1008→    message: l.message,
  1009→  1009→    source: l.source,
  1010→  1010→    symbol: l.symbol,
  1011→  1011→    tradeId: l.tradeId,
  1012→  1012→  })), null, 2)
  1013→  1013→}
