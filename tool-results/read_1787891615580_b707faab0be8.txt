    86→// ---- Position Sizing ----
    87→
    88→/**
    89→ * Calculate the optimal lot size for a trade.
    90→ * Enhanced with commission-aware sizing, reserve capital enforcement,
    91→ * deployed margin cap, and performance-based dynamic scaling.
    92→ */
    93→export async function calculatePositionSize(params: {
    94→  symbol: string
    95→  direction: "BUY" | "SELL"
    96→  entryPrice: number
    97→  sl?: number | null
    98→  equity: number
    99→  method?: SizingMethod
   100→  fixedDollarRisk?: number
   101→  kellyFraction?: number // default 0.5 (half-Kelly)
   102→  commissionPerLot?: number // default $1 (FINEX)
   103→  reserveCapitalPct?: number // default from RiskConfig
   104→}): Promise<PositionSizeResult> {
   105→  const config = await getRiskConfig()
   106→  const method = params.method || "FIXED_FRACTIONAL"
   107→  const sl = params.sl
   108→  const commissionPerLot = params.commissionPerLot ?? DEFAULT_COMMISSION_PER_LOT
   109→  const reserveCapitalPct = params.reserveCapitalPct ?? config.reserveCapitalPct
   110→
   111→  // --- Performance-Based Dynamic Scaling ---
   112→  const scalingFactor = await calculateScalingFactor()
   113→
   114→  // Phase 4: Integrate volatility regime with scaling factor
   115→  let volMultiplier = 1.0
   116→  try {
   117→    const { detectVolatilityRegime } = await import('./risk-engine')
   118→    const volResult = detectVolatilityRegime({
   119→      recentVolatility: 0.015, // default if no data
   120→      avgVolatility: 0.015,
   121→    })
   122→    volMultiplier = volResult.riskMultiplier
   123→  } catch { /* non-critical */ }
   124→
   125→  // Combine scaling factor with volatility regime
   126→  const effectiveScaling = (scalingFactor ?? 1.0) * volMultiplier
   127→
   128→  // Calculate pip risk (price distance to stop loss)
   129→  let pipRisk = 0
   130→  if (sl) {
   131→    pipRisk = params.direction === "BUY"
   132→      ? params.entryPrice - sl
   133→      : sl - params.entryPrice
   134→    pipRisk = Math.abs(pipRisk)
   135→  }
   136→
   137→  // If no SL, use a default 2% price risk as fallback
   138→  const effectivePipRisk = pipRisk > 0 ? pipRisk : params.entryPrice * 0.02
   139→
   140→  // --- Base Risk Amount by Method ---
   141→  let baseRiskAmount: number
   142→  let methodReasoning: string
   143→
   144→  switch (method) {
   145→    case "FIXED_FRACTIONAL": {
   146→      baseRiskAmount = (config.maxRiskPerTrade / 100) * params.equity
   147→      methodReasoning = `Fixed fractional: ${config.maxRiskPerTrade}% of equity ($${params.equity.toFixed(2)}) = $${baseRiskAmount.toFixed(2)}`
   148→      break
   149→    }
   150→    case "KELLY": {
   151→      const kellyFraction = params.kellyFraction || 0.5
   152→      const kellyResult = await calculateKelly(params.equity, kellyFraction)
   153→      baseRiskAmount = kellyResult.suggestedRisk
   154→      methodReasoning = `Half-Kelly criterion: Kelly suggests ${kellyResult.kellyPercent.toFixed(1)}%, using ${kellyFraction * 100}% = $${baseRiskAmount.toFixed(2)}`
   155→      break
   156→    }
   157→    case "FIXED_DOLLAR": {
   158→      baseRiskAmount = params.fixedDollarRisk || 100
   159→      methodReasoning = `Fixed dollar risk: $${baseRiskAmount.toFixed(2)} per trade`
   160→      break
   161→    }
   162→    case "ANTI_MARTINGALE": {
   163→      const recentTrades = await getRecentTradeResults(5)
   164→      let multiplier = 1
   165→      for (const t of recentTrades) {
   166→        if (t > 0) multiplier = Math.min(multiplier + 0.5, 3)
   167→        else break
   168→      }
   169→      baseRiskAmount = (config.maxRiskPerTrade / 100) * params.equity * multiplier
   170→      methodReasoning = `Anti-Martingale: base ${config.maxRiskPerTrade}% * ${multiplier.toFixed(1)}x (consecutive wins) = $${baseRiskAmount.toFixed(2)}`
   171→      break
   172→    }
   173→    default:
   174→      baseRiskAmount = (config.maxRiskPerTrade / 100) * params.equity
   175→      methodReasoning = `Default fixed fractional: $${baseRiskAmount.toFixed(2)}`
   176→  }
   177→
   178→  // --- Apply Scaling Factor (combined with volatility regime) ---
   179→  const scaledRiskAmount = baseRiskAmount * effectiveScaling
   180→  let reasoning = methodReasoning
   181→  if (effectiveScaling !== 1.0) {
   182→    reasoning += ` | Scaled ${effectiveScaling.toFixed(3)}x (perf=${scalingFactor}, vol=${volMultiplier.toFixed(3)}) -> $${scaledRiskAmount.toFixed(2)}`
   183→  }
   184→
   185→  // --- Commission-Aware Lot Sizing ---
   186→  // Risk budget must cover both SL risk AND commission cost.
   187→  // totalRisk = SL_risk + commission
   188→  // We need to find a lot size L where:
   189→  //   L * effectivePipRisk * PIP_VALUE_PER_LOT + L * commissionPerLot <= scaledRiskAmount
   190→  //   L * (effectivePipRisk * PIP_VALUE_PER_LOT + commissionPerLot) <= scaledRiskAmount
   191→  //   L <= scaledRiskAmount / (effectivePipRisk * PIP_VALUE_PER_LOT + commissionPerLot)
   192→  const costPerLot = effectivePipRisk * PIP_VALUE_PER_LOT + commissionPerLot
   193→  const rawLotSize = costPerLot > 0 ? scaledRiskAmount / costPerLot : 0
   194→  let suggestedLotSize = Math.floor(rawLotSize / LOT_STEP) * LOT_STEP
   195→  suggestedLotSize = Math.max(MIN_LOT, suggestedLotSize)
   196→
   197→  // Calculate commission cost for the suggested lot
   198→  const commissionCost = Math.round(suggestedLotSize * commissionPerLot * 100) / 100
   199→  const slRiskAmount = Math.round(suggestedLotSize * effectivePipRisk * PIP_VALUE_PER_LOT * 100) / 100
   200→  const netRiskAfterCommission = Math.round((slRiskAmount + commissionCost) * 100) / 100
   201→
   202→  reasoning += ` | Commission: $${commissionCost.toFixed(2)} (${suggestedLotSize} lot x $${commissionPerLot}/lot)`
   203→  reasoning += ` | Net risk (SL + commission): $${netRiskAfterCommission.toFixed(2)}`
   204→
   205→  // --- Reserve Capital Enforcement ---
   206→  const maxDeployable = params.equity * (1 - reserveCapitalPct / 100)
   207→  const contractValue = suggestedLotSize * params.entryPrice * 100000
   208→  const marginRequired = contractValue / LEVERAGE
   209→  let reserveCheckApplied = false
   210→
   211→  if (marginRequired > maxDeployable) {
   212→    // Reduce lot size so margin fits within deployable capital
   213→    const maxLotForReserve = (maxDeployable * LEVERAGE) / (params.entryPrice * 100000)
   214→    const reducedLot = Math.floor(maxLotForReserve / LOT_STEP) * LOT_STEP
   215→    if (reducedLot < suggestedLotSize) {
   216→      suggestedLotSize = Math.max(MIN_LOT, reducedLot)
   217→      reserveCheckApplied = true
   218→      // Recalculate values after reserve reduction
   219→    }
   220→    reasoning += ` | Reserve ${reserveCapitalPct}% enforced: max deployable $${maxDeployable.toFixed(2)}, reduced lot to ${suggestedLotSize}`
   221→  }
   222→
   223→  // --- Max Capital Deployment (Open Positions Check) ---
   224→  const openPositions = await db.trade.findMany({ where: { status: "OPEN" } })
   225→  const existingDeployedMargin = openPositions.reduce((sum, t) => sum + (t.margin || 0), 0)
   226→  const newMarginRequired = (suggestedLotSize * params.entryPrice * 100000) / LEVERAGE
   227→  let deployedMarginCheckApplied = false
   228→
   229→  if (existingDeployedMargin + newMarginRequired > maxDeployable) {
   230→    const availableForNew = maxDeployable - existingDeployedMargin
   231→    if (availableForNew > 0) {
   232→      const maxLotForDeployed = (availableForNew * LEVERAGE) / (params.entryPrice * 100000)
   233→      const reducedLot = Math.floor(maxLotForDeployed / LOT_STEP) * LOT_STEP
   234→      if (reducedLot < suggestedLotSize && reducedLot >= MIN_LOT) {
   235→        suggestedLotSize = reducedLot
   236→        deployedMarginCheckApplied = true
   237→        reasoning += ` | Deployed margin check: $${existingDeployedMargin.toFixed(2)} already deployed, reduced lot to ${suggestedLotSize}`
   238→      } else if (reducedLot < MIN_LOT) {
   239→        // Not enough deployable capital for even minimum lot
   240→        suggestedLotSize = 0
   241→        reasoning += ` | REJECTED: Not enough deployable capital ($${availableForNew.toFixed(2)} available, $${newMarginRequired.toFixed(2)} needed)`
   242→      }
   243→    } else {
   244→      suggestedLotSize = 0
   245→      reasoning += ` | REJECTED: All deployable capital ($${maxDeployable.toFixed(2)}) already deployed ($${existingDeployedMargin.toFixed(2)})`
   246→    }
   247→  }
   248→
   249→  // Cap at max lot per trade
   250→  if (suggestedLotSize > 0) {
   251→    suggestedLotSize = Math.min(suggestedLotSize, config.maxLotPerTrade)
   252→  }
   253→
   254→  // --- Final Derived Values ---
   255→  const finalContractValue = suggestedLotSize * params.entryPrice * 100000
   256→  const finalMarginRequired = finalContractValue / LEVERAGE
   257→  const finalCommissionCost = Math.round(suggestedLotSize * commissionPerLot * 100) / 100
   258→  const finalSlRiskAmount = Math.round(suggestedLotSize * effectivePipRisk * PIP_VALUE_PER_LOT * 100) / 100
   259→  const finalNetRisk = Math.round((finalSlRiskAmount + finalCommissionCost) * 100) / 100
   260→  const riskPercent = params.equity > 0 ? (finalNetRisk / params.equity) * 100 : 0
   261→
   262→  return {
   263→    suggestedLotSize: Math.round(suggestedLotSize * 100) / 100,
   264→    riskAmount: Math.round((baseRiskAmount * effectiveScaling) * 100) / 100,
   265→    riskPercent: Math.round(riskPercent * 100) / 100,
   266→    method,
   267→    pipRisk: Math.round(effectivePipRisk * 100) / 100,
   268→    contractValue: Math.round(finalContractValue * 100) / 100,
   269→    marginRequired: Math.round(finalMarginRequired * 100) / 100,
   270→    reasoning,
   271→    commissionCost: finalCommissionCost,
   272→    netRiskAfterCommission: finalNetRisk,
   273→    scalingFactor,
   274→    volatilityRegimeMultiplier: volMultiplier,
   275→    reserveCheckApplied,
   276→    deployedMarginCheckApplied,
   277→  }
   278→}

   280→/**
   281→ * Kelly Criterion position sizing.
   282→ */
   283→async function calculateKelly(equity: number, fraction: number): Promise<{
   284→  kellyPercent: number
   285→  suggestedRisk: number
   286→}> {
   287→  const closedTrades = await db.trade.findMany({
   288→    where: { status: "CLOSED" },
   289→    orderBy: { closeTime: "desc" },
   290→    take: 100,
   291→  })
   292→
   293→  if (closedTrades.length < 10) {
   294→    return { kellyPercent: 1.0, suggestedRisk: (fraction / 100) * equity }
   295→  }
   296→
   297→  const wins = closedTrades.filter((t) => t.pnl > 0)
   298→  const losses = closedTrades.filter((t) => t.pnl < 0)
   299→  const winRate = wins.length / closedTrades.length
   300→  const avgWin = wins.length > 0 ? wins.reduce((s, t) => s + t.pnl, 0) / wins.length : 0
   301→  const avgLoss = losses.length > 0 ? Math.abs(losses.reduce((s, t) => s + t.pnl, 0) / losses.length) : 1
   302→
   303→  const R = avgLoss > 0 ? avgWin / avgLoss : 0
   304→  const kellyPercent = R > 0 ? (winRate - (1 - winRate) / R) * 100 : 0
   305→  const clampedKelly = Math.max(-5, Math.min(25, kellyPercent))
   306→
   307→  return {
   308→    kellyPercent: clampedKelly,
   309→    suggestedRisk: (Math.max(0, clampedKelly) * fraction / 100) * equity,
   310→  }
   311→}

   325→// ---- Risk of Ruin ----
   326→
   327→/**
   328→ * Estimate the probability of ruin (losing X% of account).
   329→ * Uses the formula: P(ruin) = ((1 - edge) / (1 + edge))^units
   330→ * where edge = W - (1-W)*avgLoss/avgWin and units = account / (risk per trade)
   331→ */
   332→export function calculateRiskOfRuin(input: RiskOfRuinInput): {
   333→  probability: number
   334→  interpretation: string
   335→  recommendation: string
   336→} {
   337→  if (input.winRate <= 0 || input.avgLoss <= 0) {
   338→    return {
   339→      probability: 100,
   340→      interpretation: "Cannot calculate - insufficient winning trades or zero avg loss",
   341→      recommendation: "Improve strategy win rate or reduce loss size",
   342→    }
   343→  }
   344→
   345→  const W = input.winRate / 100
   346→  const L = 1 - W
   347→  const R = input.avgWin / input.avgLoss
   348→
   349→  if (R <= 0) {
   350→    return {
   351→      probability: 100,
   352→      interpretation: "Average win is zero or negative",
   353→      recommendation: "Strategy needs improvement - avg win must exceed avg loss",
   354→    }
   355→  }
   356→
   357→  const edge = W - L / R
   358→  const riskFraction = input.riskPerTrade / 100
   359→
   360→  if (edge <= 0) {
   361→    return {
   362→      probability: 100,
   363→      interpretation: `Negative edge (${(edge * 100).toFixed(2)}%) - strategy is unprofitable statistically`,
   364→      recommendation: "Do not trade this strategy. Win rate and/or reward-risk ratio must improve.",
   365→    }
   366→  }
   367→
   368→  const q = (1 - edge) / (1 + edge)
   369→  const units = 1 / riskFraction
   370→  const ror = Math.pow(Math.max(0, q), units) * 100
   371→
   372→  let interpretation: string
   373→  let recommendation: string
   374→
   375→  if (ror < 1) {
   376→    interpretation = "Very Low Risk"
   377→    recommendation = "Strategy has a strong statistical edge. Safe to trade with current parameters."
   378→  } else if (ror < 5) {
   379→    interpretation = "Low Risk"
   380→    recommendation = "Acceptable risk level. Continue monitoring performance metrics."
   381→  } else if (ror < 15) {
   382→    interpretation = "Moderate Risk"
   383→    recommendation = "Consider reducing position sizes or improving win rate to lower risk of ruin."
   384→  } else if (ror < 30) {
   385→    interpretation = "High Risk"
   386→    recommendation = "Significant risk of account drawdown. Reduce risk per trade significantly."
   387→  } else {
   388→    interpretation = "Extreme Risk"
   389→    recommendation = "Do NOT trade this strategy with current parameters. High probability of account destruction."
   390→  }
   391→
   392→  return {
   393→    probability: Math.min(100, Math.round(ror * 100) / 100),
   394→    interpretation,
   395→    recommendation,
   396→  }
   397→}

   399→// ---- Drawdown Recovery Model ----
   400→
   401→/**
   402→ * Calculate the recovery needed for any drawdown level.
   403→ * Formula: recoveryNeeded = (drawdown / (100 - drawdown)) * 100
   404→ *
   405→ * Examples:
   406→ *   10% DD -> 11.1% recovery needed
   407→ *   20% DD -> 25% recovery needed
   408→ *   50% DD -> 100% recovery needed
   409→ */
   410→export function calculateDrawdownRecovery(drawdownPct: number): DrawdownRecoveryResult {
   411→  const dd = Math.max(0, Math.min(99, drawdownPct)) // Clamp 0-99%
   412→
   413→  const recoveryNeeded = dd >= 100 ? Infinity : (dd / (100 - dd)) * 100
   414→
   415→  // Recovery strategy and risk reduction based on drawdown severity
   416→  let strategy: string
   417→  let riskReductionPct: number
   418→
   419→  if (dd < 5) {
   420→    strategy = "NORMAL: No action needed. Drawdown within acceptable range. Continue normal trading operations."
   421→    riskReductionPct = 0
   422→  } else if (dd < 10) {
   423→    strategy = "CAUTION: Minor drawdown. Review recent trades for patterns. Tighten stop losses by 10%. Avoid high-risk setups."
   424→    riskReductionPct = 10
   425→  } else if (dd < 15) {
   426→    strategy = "ELEVATED: Moderate drawdown. Reduce position sizes by 25%. Focus on highest-confidence setups only. Consider pausing new positions temporarily."
   427→    riskReductionPct = 25
   428→  } else if (dd < 20) {
   429→    strategy = "HIGH: Significant drawdown. Reduce risk to 50% of normal. Trade only A+ setups. Implement stricter entry criteria. Review and potentially pause underperforming strategies."
   430→    riskReductionPct = 50
   431→  } else if (dd < 30) {
   432→    strategy = "CRITICAL: Severe drawdown. Reduce risk to 75% of normal. Consider full trading pause for 24-48 hours. Perform strategy review. Check if market regime has changed."
   433→    riskReductionPct = 75
   434→  } else if (dd < 50) {
   435→    strategy = "EMERGENCY: Extreme drawdown. Stop all new trading immediately. Close weakest positions. Full strategy audit required. Resume only with micro-lots after analysis."
   436→    riskReductionPct = 90
   437→  } else {
   438→    strategy = "CATASTROPHIC: Account severely impaired. ${recoveryNeeded.toFixed(1)}% gain required to recover. Full trading halt. Account preservation is the only priority. Consider account reset or external capital injection after thorough review."
   439→    riskReductionPct = 100
   440→  }
   441→
   442→  // Fix the template literal for 50%+ case
   443→  if (dd >= 50) {
   444→    strategy = `CATASTROPHIC: Account severely impaired. ${recoveryNeeded.toFixed(1)}% gain required to recover. Full trading halt. Account preservation is the only priority. Consider account reset or external capital injection after thorough review.`
   445→  }
   446→
   447→  return {
   448→    drawdownPct: Math.round(dd * 100) / 100,
   449→    recoveryNeeded: Math.round(recoveryNeeded * 100) / 100,
   450→    strategy,
   451→    riskReductionPct,
   452→  }
   453→}

   455→// ---- Performance-Based Dynamic Scaling ----
   456→
   457→/**
   458→ * Calculate a dynamic scaling factor based on recent trading performance.
   459→ * Analyzes the last 30 closed trades (or all if fewer) for:
   460→ *   - Rolling win rate
   461→ *   - Profit factor (gross profit / gross loss)
   462→ *   - Sharpe-like ratio (mean PnL / std dev of PnL)
   463→ *
   464→ * Scaling rules:
   465→ *   WR > 60% AND PF > 1.5  -> scale up to 1.25x (max)
   466→ *   WR > 55% AND PF > 1.2  -> scale up to 1.1x
   467→ *   WR < 40% OR PF < 0.8  -> scale down to 0.5x
   468→ *   WR < 45% OR PF < 1.0  -> scale down to 0.75x
   469→ *   Otherwise              -> 1.0x (neutral)
   470→ */
   471→export async function calculateScalingFactor(): Promise<number> {
   472→  try {
   473→    const closedTrades = await db.trade.findMany({
   474→      where: { status: "CLOSED" },
   475→      orderBy: { closeTime: "desc" },
   476→      take: 30,
   477→    })
   478→
   479→    // Not enough trades to evaluate - use neutral scaling
   480→    if (closedTrades.length < 5) {
   481→      return 1.0
   482→    }
   483→
   484→    // Calculate rolling win rate
   485→    const wins = closedTrades.filter((t) => t.pnl > 0)
   486→    const winRate = (wins.length / closedTrades.length) * 100
   487→
   488→    // Calculate profit factor
   489→    const grossProfit = wins.reduce((sum, t) => sum + t.pnl, 0)
   490→    const losses = closedTrades.filter((t) => t.pnl < 0)
   491→    const grossLoss = Math.abs(losses.reduce((sum, t) => sum + t.pnl, 0))
   492→    const profitFactor = grossLoss > 0 ? grossProfit / grossLoss : 0
   493→
   494→    // Calculate Sharpe-like ratio (mean / stddev of PnL)
   495→    const pnls = closedTrades.map((t) => t.pnl)
   496→    const mean = pnls.reduce((s, v) => s + v, 0) / pnls.length
   497→    const variance = pnls.reduce((s, v) => s + (v - mean) ** 2, 0) / pnls.length
   498→    const stddev = Math.sqrt(variance)
   499→    // We don't use this for scaling decisions directly but log it
   500→    const sharpeLike = stddev > 0 ? mean / stddev : 0
   501→
   502→    // Determine scaling factor based on rules
   503→    let factor: number
   504→    if (winRate > 60 && profitFactor > 1.5) {
   505→      factor = 1.25
   506→    } else if (winRate > 55 && profitFactor > 1.2) {
   507→      factor = 1.1
   508→    } else if (winRate < 40 || profitFactor < 0.8) {
   509→      factor = 0.5
   510→    } else if (winRate < 45 || profitFactor < 1.0) {
   511→      factor = 0.75
   512→    } else {
   513→      factor = 1.0
   514→    }
   515→
   516→    logger.debug("MONEY_MANAGEMENT", `Scaling factor calculated: ${factor}x (WR: ${winRate.toFixed(1)}%, PF: ${profitFactor.toFixed(2)}, Sharpe-like: ${sharpeLike.toFixed(2)})`, {
   517→      metadata: {
   518→        scalingFactor: factor,
   519→        winRate: Math.round(winRate * 100) / 100,
   520→        profitFactor: Math.round(profitFactor * 100) / 100,
   521→        sharpeLike: Math.round(sharpeLike * 100) / 100,
   522→        sampleSize: closedTrades.length,
   523→      },
   524→    })
   525→
   526→    return factor
   527→  } catch (err) {
   528→    logger.error("MONEY_MANAGEMENT", "Error calculating scaling factor, defaulting to 1.0", {
   529→      details: err instanceof Error ? err.stack : undefined,
   530→    })
   531→    return 1.0
   532→  }
   533→}

   567→export async function getDailyPerformance(): Promise<DailyPerformanceData> {
   568→  const todayStr = new Date().toISOString().split("T")[0]
   569→

   594→/**
   595→ * Update daily performance after a trade event.
   596→ * Enhanced to track commission, slippage, deployed capital, reserve capital,
   597→ * scaling factor, and sizing method used.
   598→ */
   599→export async function updateDailyPerformance(params: {
   600→  type: "OPEN" | "CLOSE"
   601→  pnl?: number
   602→  isWin?: boolean
   603→  commission?: number
   604→  slippage?: number
   605→  sizingMethod?: string
   606→}): Promise<void> {
   607→  const todayStr = new Date().toISOString().split("T")[0]
   608→
   609→  const existing = await db.dailyPerformance.findUnique({ where: { date: todayStr } })
   610→  if (!existing) return
   611→
   612→  const openTrades = await db.trade.findMany({ where: { status: "OPEN" } })
   613→  const unrealizedPnl = openTrades.reduce((s, t) => s + t.pnl, 0)
   614→
   615→  // Calculate deployed capital = sum of margin for all open positions
   616→  const deployedCapital = openTrades.reduce((s, t) => s + (t.margin || 0), 0)
   617→
   618→  // Reserve capital = equity - deployed capital - unrealized PnL
   619→  const currentEquity = existing.startBalance + existing.realizedPnl + unrealizedPnl
   620→  const reserveCapital = Math.max(0, currentEquity - deployedCapital - unrealizedPnl)
   621→
   622→  // Track commission and slippage from closed trades today
   623→  const todayStart = new Date(todayStr)
   624→  const todayClosedTrades = await db.trade.findMany({
   625→    where: {
   626→      status: "CLOSED",
   627→      closeTime: { gte: todayStart },
   628→    },
   629→  })
   630→  const totalCommissionPaid = todayClosedTrades.reduce((s, t) => s + (t.commission || 0), 0)
   631→  const totalSlippageCost = todayClosedTrades.reduce((s, t) => s + (t.slippage || 0), 0)
   632→
   633→  // Get current scaling factor
   634→  const scalingFactor = await calculateScalingFactor()
   635→
   636→  const updateData: Record<string, unknown> = {
   637→    unrealizedPnl: Math.round(unrealizedPnl * 100) / 100,
   638→    deployedCapital: Math.round(deployedCapital * 100) / 100,
   639→    reserveCapital: Math.round(reserveCapital * 100) / 100,
   640→    commissionPaid: Math.round(totalCommissionPaid * 100) / 100,
   641→    slippageCost: Math.round(totalSlippageCost * 100) / 100,
   642→    scalingFactor,
   643→  }
   644→
   645→  // Track the most common sizing method used today
   646→  if (params.sizingMethod) {
   647→    // Store the latest sizing method used; this is a simplification.
   648→    // A more complex approach would count method occurrences.
   649→    updateData.sizingMethodUsed = params.sizingMethod
   650→  }
   651→
   652→  if (params.type === "CLOSE") {
   653→    const newRealized = Math.round((existing.realizedPnl + (params.pnl || 0)) * 100) / 100
   654→    updateData.realizedPnl = newRealized
   655→    updateData.totalPnl = Math.round((newRealized + unrealizedPnl) * 100) / 100
   656→    updateData.pnlPercent = existing.startBalance > 0 ? Math.round((newRealized / existing.startBalance) * 10000) / 100 : 0
   657→    updateData.tradesClosed = existing.tradesClosed + 1
   658→    if (params.isWin) {
   659→      updateData.winTrades = existing.winTrades + 1
   660→    } else {
   661→      updateData.lossTrades = existing.lossTrades + 1
   662→    }
   663→  } else {
   664→    updateData.tradesOpened = existing.tradesOpened + 1
   665→    updateData.totalPnl = Math.round((existing.realizedPnl + unrealizedPnl) * 100) / 100
   666→    updateData.pnlPercent = existing.startBalance > 0 ? Math.round((existing.realizedPnl / existing.startBalance) * 10000) / 100 : 0
   667→  }
   668→
   669→  // Update peak/trough for drawdown
   670→  const finalEquity = existing.startBalance + (updateData.totalPnl as number)
   671→  updateData.endBalance = Math.round(finalEquity * 100) / 100
   672→
   673→  const newPeak = Math.max(existing.peakEquity, finalEquity)
   674→  const newTrough = Math.min(existing.troughEquity, finalEquity)
   675→  updateData.peakEquity = Math.round(newPeak * 100) / 100
   676→  updateData.troughEquity = Math.round(newTrough * 100) / 100
   677→
   678→  if (newPeak > 0) {
   679→    const dd = ((newPeak - newTrough) / newPeak) * 100
   680→    updateData.maxDrawdown = Math.max(existing.maxDrawdown, Math.round(dd * 100) / 100)
   681→  }
   682→
   683→  await db.dailyPerformance.update({
   684→    where: { date: todayStr },
   685→    data: updateData,
   686→  })
   687→
   688→  logger.debug("MONEY_MANAGEMENT", `Daily performance updated: ${params.type} | PnL: ${params.pnl ?? 0} | Commission: $${totalCommissionPaid.toFixed(2)} | Deployed: $${deployedCapital.toFixed(2)} | Reserve: $${reserveCapital.toFixed(2)}`, {
   689→    metadata: { type: params.type, pnl: params.pnl, date: todayStr, scalingFactor },
   690→  })
   691→}

   745→// ---- 1. Maximum Consecutive Loss Protection ----
   746→
   747→export interface ConsecutiveLossResult {
   748→  isHalted: boolean
   749→  consecutiveLosses: number
   750→  maxAllowed: number
   751→  haltReason: string | null
   752→  cooldownRemainingMinutes: number
   753→  lastLossTime: Date | null
   754→}
   755→
   756→/**
   757→ * Check whether trading should be halted due to consecutive losses.
   758→ *
   759→ * Queries recent CLOSED trades ordered by closeTime desc, counts consecutive
   760→ * losses (pnl < 0) from the most recent trade. If the count reaches or exceeds
   761→ * maxConsecutiveLosses (default 5), trading is halted.
   762→ *
   763→ * A cooldown period (default 60 minutes) must elapse before trading resumes.
   764→ * Once the cooldown elapses, the counter is effectively reset and trading is allowed.
   765→ */
   766→export async function checkConsecutiveLossHalt(): Promise<ConsecutiveLossResult> {
   767→  // Defaults (these fields exist in the DB RiskConfig model but are not yet
   768→  // exposed through the TypeScript RiskConfigData interface).
   769→  const maxAllowed = 5
   770→  const cooldownMinutes = 60
   771→
   772→  const closedTrades = await db.trade.findMany({
   773→    where: { status: "CLOSED", closeTime: { not: null } },
   774→    orderBy: { closeTime: "desc" },
   775→    select: { pnl: true, closeTime: true },
   776→    take: maxAllowed + 1,
   777→  })
   778→
   779→  // Count consecutive losses from the most recent trade
   780→  let consecutiveLosses = 0
   781→  let lastLossTime: Date | null = null
   782→
   783→  for (const trade of closedTrades) {
   784→    if (trade.pnl < 0) {
   785→      consecutiveLosses++
   786→      if (lastLossTime === null && trade.closeTime) {
   787→        lastLossTime = trade.closeTime
   788→      }
   789→    } else {
   790→      break // Stop counting at first non-loss
   791→    }
   792→  }
   793→
   794→  const result: ConsecutiveLossResult = {
   795→    isHalted: false,
   796→    consecutiveLosses,
   797→    maxAllowed,
   798→    haltReason: null,
   799→    cooldownRemainingMinutes: 0,
   800→    lastLossTime,
   801→  }
   802→
   803→  if (consecutiveLosses >= maxAllowed && lastLossTime) {
   804→    const elapsedMs = Date.now() - lastLossTime.getTime()
   805→    const cooldownMs = cooldownMinutes * 60 * 1000
   806→    const remainingMs = cooldownMs - elapsedMs
   807→
   808→    if (remainingMs > 0) {
   809→      const remainingMinutes = Math.ceil(remainingMs / (60 * 1000))
   810→      result.isHalted = true
   811→      result.cooldownRemainingMinutes = remainingMinutes
   812→      result.haltReason = `Consecutive loss halt: ${consecutiveLosses} consecutive losses (max ${maxAllowed}). Cooldown: ${remainingMinutes} minutes remaining.`
   813→
   814→      logger.warn(
   815→        "MONEY_MANAGEMENT",
   816→        result.haltReason,
   817→        {
   818→          metadata: {
   819→            consecutiveLosses,
   820→            maxAllowed,
   821→            cooldownMinutes,
   822→            cooldownRemainingMinutes: remainingMinutes,
   823→            lastLossTime: lastLossTime.toISOString(),
   824→          },
   825→        },
   826→      )
   827→    }
   828→    // If cooldown has elapsed, isHalted stays false — counter is effectively reset
   829→  }
   830→
   831→  return result
   832→}

   834→// ---- 2. Equity Curve Trading ----
   835→
   836→export type EquityCurveStatus = "NORMAL" | "BELOW_MA" | "RECOVERING"
   837→
   838→export interface EquityCurveResult {
   839→  status: EquityCurveStatus
   840→  currentEquity: number
   841→  equityCurveMa: number
   842→  maPeriod: number
   843→}
   844→
   845→/**
   846→ * Check equity curve status against its simple moving average.
   847→ *
   848→ * Queries DailyPerformance for the last N days (maPeriod, default 20),
   849→ * calculates the SMA of endBalance, and compares the current equity.
   850→ *
   851→ * - BELOW_MA: current equity is below the SMA → trading should be disabled/reduced
   852→ * - RECOVERING: current equity was below MA recently but now above
   853→ * - NORMAL: current equity is above the SMA
   854→ */
   855→export async function checkEquityCurveStatus(): Promise<EquityCurveResult> {
   856→  const maPeriod = 20
   857→
   858→  const recentPerf = await db.dailyPerformance.findMany({
   859→    orderBy: { date: "desc" },
   860→    take: maPeriod + 1, // +1 to detect RECOVERING (was below, now above)
   861→  })
   862→
   863→  // Need at least a few days of data to compute a meaningful MA
   864→  if (recentPerf.length < 3) {
   865→    const currentEquity = recentPerf.length > 0 ? recentPerf[0].endBalance : 0
   866→    return {
   867→      status: "NORMAL",
   868→      currentEquity,
   869→      equityCurveMa: currentEquity,
   870→      maPeriod,
   871→    }
   872→  }
   873→
   874→  // Calculate SMA of endBalance over the available period (up to maPeriod)
   875→  const effectivePeriod = Math.min(maPeriod, recentPerf.length)
   876→  const balancesForMa = recentPerf.slice(0, effectivePeriod).map((p) => p.endBalance)
   877→  const equityCurveMa = balancesForMa.reduce((s, v) => s + v, 0) / balancesForMa.length
   878→
   879→  const currentEquity = recentPerf[0].endBalance
   880→
   881→  // Determine if the previous day was below MA (for RECOVERING detection)
   882→  const prevEquity = recentPerf.length > 1 ? recentPerf[1].endBalance : currentEquity
   883→  const prevBelowMa = prevEquity < equityCurveMa
   884→  const currentBelowMa = currentEquity < equityCurveMa
   885→
   886→  let status: EquityCurveStatus
   887→  if (currentBelowMa) {
   888→    status = "BELOW_MA"
   889→    logger.warn(
   890→      "MONEY_MANAGEMENT",
   891→      `[Equity Curve] BELOW moving average. Current: ${currentEquity.toFixed(2)}, MA(${effectivePeriod}): ${equityCurveMa.toFixed(2)}. Consider reducing or halting trading.`,
   892→      {
   893→        metadata: {
   894→          currentEquity: Math.round(currentEquity * 100) / 100,
   895→          equityCurveMa: Math.round(equityCurveMa * 100) / 100,
   896→          maPeriod: effectivePeriod,
   897→          deviationPct: equityCurveMa > 0
   898→            ? Math.round(((currentEquity - equityCurveMa) / equityCurveMa) * 10000) / 100
   899→            : 0,
   900→        },
   901→      },
   902→    )
   903→  } else if (prevBelowMa && !currentBelowMa) {
   904→    status = "RECOVERING"
   905→    logger.info(
   906→      "MONEY_MANAGEMENT",
   907→      `[Equity Curve] RECOVERING — above MA after being below. Current: ${currentEquity.toFixed(2)}, MA(${effectivePeriod}): ${equityCurveMa.toFixed(2)}.`,
   908→      {
   909→        metadata: {
   910→          currentEquity: Math.round(currentEquity * 100) / 100,
   911→          equityCurveMa: Math.round(equityCurveMa * 100) / 100,
   912→          maPeriod: effectivePeriod,
   913→        },
   914→      },
   915→    )
   916→  } else {
   917→    status = "NORMAL"
   918→  }
   919→
   920→  return {
   921→    status,
   922→    currentEquity: Math.round(currentEquity * 100) / 100,
   923→    equityCurveMa: Math.round(equityCurveMa * 100) / 100,
   924→    maPeriod: effectivePeriod,
   925→  }
   926→}

   928→// ---- 3. Session-Based Risk Limits ----
   929→
   930→export interface SessionRiskResult {
   931→  isLimitReached: boolean
   932→  sessionPnl: number
   933→  sessionLimit: number
   934→  sessionTrades: number
   935→  remainingRiskBudget: number
   936→}
   937→
   938→/**
   939→ * Check whether the intraday session risk limit has been reached.
   940→ *
   941→ * The trading session window is 09:00–15:00 WIB (UTC+7).
   942→ * Uses sessionRiskLimitPct from RiskConfig (default 1.0% of equity).
   943→ * Calculates the session's realized P&L by summing PnL of all trades
   944→ * closed within the current session window today.
   945→ */
   946→export async function checkSessionRiskLimit(): Promise<SessionRiskResult> {
   947→  const sessionRiskLimitPct = 1.0 // default 1.0% of equity
   948→
   949→  const now = new Date()
   950→  const wibOffsetMs = 7 * 60 * 60 * 1000
   951→  const nowWib = new Date(now.getTime() + wibOffsetMs)
   952→  const wibHours = nowWib.getUTCHours()
   953→  const wibMinutes = nowWib.getUTCMinutes()
   954→  const currentWibMinutes = wibHours * 60 + wibMinutes
   955→
   956→  // Session window: 09:00 (540) - 15:00 (900) WIB
   957→  const SESSION_START = 540
   958→  const SESSION_END = 900
   959→
   960→  // If outside session, return clean state
   961→  if (currentWibMinutes < SESSION_START || currentWibMinutes > SESSION_END) {
   962→    return {
   963→      isLimitReached: false,
   964→      sessionPnl: 0,
   965→      sessionLimit: 0,
   966→      sessionTrades: 0,
   967→      remainingRiskBudget: 0,
   968→    }
   969→  }
   970→
   971→  // Get today's start balance from DailyPerformance
   972→  const todayStr = now.toISOString().split("T")[0]
   973→  const todayPerf = await db.dailyPerformance.findUnique({ where: { date: todayStr } })
   974→  const equity = todayPerf ? todayPerf.startBalance : 10000
   975→  const sessionLimit = (sessionRiskLimitPct / 100) * equity
   976→
   977→  // Calculate session start/end in UTC for DB query
   978→  // WIB 09:00 today = UTC 02:00 today
   979→  // WIB 15:00 today = UTC 08:00 today
   980→  const sessionStartUtc = new Date(now)
   981→  sessionStartUtc.setUTCHours(2, 0, 0, 0) // 09:00 WIB
   982→  const sessionEndUtc = new Date(now)
   983→  sessionEndUtc.setUTCHours(8, 0, 0, 0) // 15:00 WIB
   984→
   985→  // Query trades closed within today's session
   986→  const sessionClosedTrades = await db.trade.findMany({
   987→    where: {
   988→      status: "CLOSED",
   989→      closeTime: {
   990→        gte: sessionStartUtc,
   991→        lte: sessionEndUtc,
   992→      },
   993→    },
   994→    select: { pnl: true },
   995→  })
   996→
   997→  // Session P&L only counts losses (negative PnL) against the limit
   998→  const sessionPnl = sessionClosedTrades.reduce((sum, t) => sum + t.pnl, 0)
   999→  const sessionLosses = sessionClosedTrades.filter((t) => t.pnl < 0).reduce((sum, t) => sum + t.pnl, 0)
  1000→  const sessionTrades = sessionClosedTrades.length
  1001→
  1002→  // Remaining risk budget is based on how much loss capacity is left
  1003→  const remainingRiskBudget = Math.max(0, Math.round((sessionLimit + sessionLosses) * 100) / 100)
  1004→  const isLimitReached = sessionLosses <= -sessionLimit
  1005→
  1006→  if (isLimitReached) {
  1007→    logger.warn(
  1008→      "MONEY_MANAGEMENT",
  1009→      `Session risk limit reached. Session loss: ${sessionLosses.toFixed(2)}, Limit: ${sessionLimit.toFixed(2)}, Trades: ${sessionTrades}`,
  1010→      {
  1011→        metadata: {
  1012→          sessionPnl: Math.round(sessionPnl * 100) / 100,
  1013→          sessionLoss: Math.round(sessionLosses * 100) / 100,
  1014→          sessionLimit: Math.round(sessionLimit * 100) / 100,
  1015→          sessionTrades,
  1016→          remainingRiskBudget: 0,
  1017→        },
  1018→      },
  1019→    )
  1020→  }
  1021→
  1022→  return {
  1023→    isLimitReached,
  1024→    sessionPnl: Math.round(sessionPnl * 100) / 100,
  1025→    sessionLimit: Math.round(sessionLimit * 100) / 100,
  1026→    sessionTrades,
  1027→    remainingRiskBudget,
  1028→  }
  1029→}

  1031→// ---- 4. Partial Profit Taking Model ----
  1032→
  1033→export interface PartialProfitLevel {
  1034→  level: number        // 1, 2, 3
  1035→  price: number        // Target price for this level
  1036→  closePercent: number // % of position to close at this level (0-100)
  1037→  reason: string       // Human-readable explanation
  1038→}
  1039→
  1040→export interface PartialProfitResult {
  1041→  levels: PartialProfitLevel[]
  1042→}
  1043→
  1044→/**
  1045→ * Calculate systematic partial profit-taking levels for a trade.
  1046→ *
  1047→ * Default: 3 levels at R:R ratios of 1:1, 1:2, 1:3 with 30%, 30%, 40% closes.
  1048→ * The levels are distributed evenly along the path from entry to TP.
  1049→ *
  1050→ * For BUY:  price = entry + (tp - entry) * level / 3
  1051→ * For SELL: price = entry - (entry - tp) * level / 3
  1052→ *
  1053→ * @param entryPrice  - Trade entry price
  1054→ * @param direction   - BUY or SELL
  1055→ * @param tp          - Take-profit price
  1056→ * @param riskRewardRatio - Optional override; if provided, adjusts level spacing
  1057→ */
  1058→export function calculatePartialProfitLevels(params: {
  1059→  entryPrice: number
  1060→  direction: "BUY" | "SELL"
  1061→  tp: number
  1062→  riskRewardRatio?: number
  1063→}): PartialProfitResult {
  1064→  const { entryPrice, direction, tp, riskRewardRatio } = params
  1065→
  1066→  const isBuy = direction === "BUY"
  1067→  const totalRange = isBuy ? tp - entryPrice : entryPrice - tp
  1068→
  1069→  // Validate inputs
  1070→  if (totalRange <= 0) {
  1071→    return { levels: [] }
  1072→  }
  1073→
  1074→  // Default: 3 levels at 1:1, 1:2, 1:3 R:R with 30%/30%/40% allocation
  1075→  const levels: PartialProfitLevel[] = [
  1076→    {
  1077→      level: 1,
  1078→      price: isBuy
  1079→        ? Math.round((entryPrice + totalRange * (1 / 3)) * 100) / 100
  1080→        : Math.round((entryPrice - totalRange * (1 / 3)) * 100) / 100,
  1081→      closePercent: 30,
  1082→      reason: "R:R 1:1 — Lock in first profit, reduce risk exposure",
  1083→    },
  1084→    {
  1085→      level: 2,
  1086→      price: isBuy
  1087→        ? Math.round((entryPrice + totalRange * (2 / 3)) * 100) / 100
  1088→        : Math.round((entryPrice - totalRange * (2 / 3)) * 100) / 100,
  1089→      closePercent: 30,
  1090→      reason: "R:R 1:2 — Second profit target, trail remaining stop",
  1091→    },
  1092→    {
  1093→      level: 3,
  1094→      price: isBuy
  1095→        ? Math.round(tp * 100) / 100
  1096→        : Math.round(tp * 100) / 100,
  1097→      closePercent: 40,
  1098→      reason: "R:R 1:3 — Final profit target, close remainder",
  1099→    },
  1100→  ]
  1101→
  1102→  // If a custom risk-reward ratio is provided, adjust the level distances
  1103→  // while keeping the same close percentages
  1104→  if (riskRewardRatio && riskRewardRatio > 0) {
  1105→    const scaledRange = totalRange * (riskRewardRatio / 3)
  1106→    levels[0].price = isBuy
  1107→      ? Math.round((entryPrice + scaledRange) * 100) / 100
  1108→      : Math.round((entryPrice - scaledRange) * 100) / 100
  1109→    levels[0].reason = `R:R ${riskRewardRatio.toFixed(1)}:1 — First scaled profit target (30%)`
  1110→
  1111→    levels[1].price = isBuy
  1112→      ? Math.round((entryPrice + scaledRange * 2) * 100) / 100
  1113→      : Math.round((entryPrice - scaledRange * 2) * 100) / 100
  1114→    levels[1].reason = `R:R ${(riskRewardRatio * 2).toFixed(1)}:1 — Second scaled profit target (30%)`
  1115→
  1116→    levels[2].price = isBuy
  1117→      ? Math.round((entryPrice + scaledRange * 3) * 100) / 100
  1118→      : Math.round((entryPrice - scaledRange * 3) * 100) / 100
  1119→    levels[2].reason = `R:R ${(riskRewardRatio * 3).toFixed(1)}:1 — Final scaled profit target (40%)`
  1120→  }
  1121→
  1122→  logger.debug(
  1123→    "MONEY_MANAGEMENT",
  1124→    `Calculated ${levels.length} partial profit levels for ${direction} @ ${entryPrice} → TP ${tp}`,
  1125→    {
  1126→      metadata: { direction, entryPrice, tp, levels: levels.length, riskRewardRatio },
  1127→    },
  1128→  )
  1129→
  1130→  return { levels }
  1131→}

  1135→export interface PreTradeHaltStatus {
  1136→  canTrade: boolean
  1137→  haltReasons: string[]
  1138→  consecutiveLossHalted: boolean
  1139→  equityCurveHalted: boolean
  1140→  sessionLimitReached: boolean
  1141→  marketClosed: boolean
  1142→}
  1143→
  1144→/**
  1145→ * Combine all pre-trade halt checks into a single call.
  1146→ *
  1147→ * Checks:
  1148→ *  1. Consecutive loss halt
  1149→ *  2. Equity curve status (BELOW_MA → halted)
  1150→ *  3. Session risk limit
  1151→ *  4. Market hours (uses isMarketOpen from mt5-connection)
  1152→ *
  1153→ * If ANY halt condition is true, canTrade is set to false.
  1154→ */
  1155→export async function getPreTradeHaltStatus(): Promise<PreTradeHaltStatus> {
  1156→  const haltReasons: string[] = []
  1157→  let consecutiveLossHalted = false
  1158→  let equityCurveHalted = false
  1159→  let sessionLimitReached = false
  1160→  let marketClosed = false
  1161→
  1162→  // 1. Consecutive loss check
  1163→  try {
  1164→    const clResult = await checkConsecutiveLossHalt()
  1165→    if (clResult.isHalted) {
  1166→      consecutiveLossHalted = true
  1167→      haltReasons.push(clResult.haltReason || "Consecutive loss halt active")
  1168→    }
  1169→  } catch (err) {
  1170→    logger.error("MONEY_MANAGEMENT", "[Pre-Trade Halt] Error checking consecutive loss halt", {
  1171→      details: err instanceof Error ? err.stack : undefined,
  1172→    })
  1173→  }
  1174→
  1175→  // 2. Equity curve check
  1176→  try {
  1177→    const ecResult = await checkEquityCurveStatus()
  1178→    if (ecResult.status === "BELOW_MA") {
  1179→      equityCurveHalted = true
  1180→      haltReasons.push(
  1181→        `Equity curve below MA: current ${ecResult.currentEquity} < MA(${ecResult.maPeriod}) ${ecResult.equityCurveMa}`,
  1182→      )
  1183→    }
  1184→  } catch (err) {
  1185→    logger.error("MONEY_MANAGEMENT", "[Pre-Trade Halt] Error checking equity curve status", {
  1186→      details: err instanceof Error ? err.stack : undefined,
  1187→    })
  1188→  }
  1189→
  1190→  // 3. Session risk limit check
  1191→  try {
  1192→    const srResult = await checkSessionRiskLimit()
  1193→    if (srResult.isLimitReached) {
  1194→      sessionLimitReached = true
  1195→      haltReasons.push(
  1196→        `Session risk limit reached: session P&L ${srResult.sessionPnl}, limit ${srResult.sessionLimit}`,
  1197→      )
  1198→    }
  1199→  } catch (err) {
  1200→    logger.error("MONEY_MANAGEMENT", "[Pre-Trade Halt] Error checking session risk limit", {
  1201→      details: err instanceof Error ? err.stack : undefined,
  1202→    })
  1203→  }
  1204→
  1205→  // 4. Market hours check
  1206→  try {
  1207→    marketClosed = !isMarketOpen()
  1208→    if (marketClosed) {
  1209→      haltReasons.push("Market is currently closed (outside 09:00-15:00 WIB trading hours)")
  1210→    }
  1211→  } catch (err) {
  1212→    logger.error("MONEY_MANAGEMENT", "[Pre-Trade Halt] Error checking market hours", {
  1213→      details: err instanceof Error ? err.stack : undefined,
  1214→    })
  1215→  }
  1216→
  1217→  const canTrade = haltReasons.length === 0
  1218→
  1219→  if (!canTrade) {
  1220→    logger.warn(
  1221→      "MONEY_MANAGEMENT",
  1222→      `[Pre-Trade Halt] Trading HALTED: ${haltReasons.length} condition(s) active`,
  1223→      {
  1224→        metadata: {
  1225→          canTrade: false,
  1226→          haltReasons,
  1227→          consecutiveLossHalted,
  1228→          equityCurveHalted,
  1229→          sessionLimitReached,
  1230→          marketClosed,
  1231→        },
  1232→      },
  1233→    )
  1234→  }
  1235→
  1236→  return {
  1237→    canTrade,
  1238→    haltReasons,
  1239→    consecutiveLossHalted,
  1240→    equityCurveHalted,
  1241→    sessionLimitReached,
  1242→    marketClosed,
  1243→  }
  1244→}

  1250→// ---- Fix 10: Progressive Drawdown Risk Reduction ----
  1251→
  1252→/**
  1253→ * Calculate progressive drawdown risk reduction.
  1254→ * Instead of binary 0.25x at 80%, apply a smooth curve:
  1255→ *   0-50% of max drawdown: 1.0x (no reduction)
  1256→ *   50-70%: linear from 1.0 to 0.75
  1257→ *   70-85%: linear from 0.75 to 0.5
  1258→ *   85-95%: linear from 0.5 to 0.25
  1259→ *   95-100%: 0.25x (minimum)
  1260→ */
  1261→export function calculateProgressiveDrawdownFactor(
  1262→  currentDrawdown: number,
  1263→  maxDrawdown: number,
  1264→): number {
  1265→  if (maxDrawdown <= 0) return 1.0
  1266→  const ratio = Math.min(currentDrawdown / maxDrawdown, 1.0)
  1267→
  1268→  if (ratio <= 0.50) return 1.0
  1269→  if (ratio <= 0.70) return 1.0 - (ratio - 0.50) / 0.20 * 0.25  // 1.0 → 0.75
  1270→  if (ratio <= 0.85) return 0.75 - (ratio - 0.70) / 0.15 * 0.25  // 0.75 → 0.50
  1271→  if (ratio <= 0.95) return 0.50 - (ratio - 0.85) / 0.10 * 0.25  // 0.50 → 0.25
  1272→  return 0.25
  1273→}

  1277→export interface WinRateAdjustmentResult {
  1278→  adjustedMultiplier: number
  1279→  reason: string
  1280→  currentWinRate: number
  1281→  historicalWinRate: number
  1282→  sampleSize: number
  1283→}
  1284→
  1285→/**
  1286→ * Adjust position sizing based on recent vs historical win rate.
  1287→ * If recent win rate drops below 80% of historical, reduce sizing proportionally.
  1288→ */
  1289→export async function calculateWinRateAdjustment(): Promise<WinRateAdjustmentResult> {
  1290→  const closedTrades = await db.trade.findMany({
  1291→    where: { status: 'CLOSED' },
  1292→    orderBy: { closeTime: 'desc' },
  1293→    take: 100,
  1294→  })
  1295→
  1296→  if (closedTrades.length < 20) {
  1297→    return { adjustedMultiplier: 1.0, reason: 'Insufficient trade history (need 20+ trades)', currentWinRate: 0, historicalWinRate: 0, sampleSize: closedTrades.length }
  1298→  }
  1299→
  1300→  // Historical = all 100 trades
  1301→  const historicalWins = closedTrades.filter(t => t.pnl > 0).length
  1302→  const historicalWinRate = historicalWins / closedTrades.length
  1303→
  1304→  // Recent = last 20 trades
  1305→  const recentTrades = closedTrades.slice(0, 20)
  1306→  const recentWins = recentTrades.filter(t => t.pnl > 0).length
  1307→  const recentWinRate = recentWins / recentTrades.length
  1308→
  1309→  if (historicalWinRate <= 0) {
  1310→    return { adjustedMultiplier: 1.0, reason: 'Zero historical win rate', currentWinRate: recentWinRate, historicalWinRate, sampleSize: closedTrades.length }
  1311→  }
  1312→
  1313→  // If recent win rate < 80% of historical, reduce proportionally
  1314→  const threshold = historicalWinRate * 0.80
  1315→  if (recentWinRate < threshold) {
  1316→    const ratio = recentWinRate / threshold
  1317→    const multiplier = Math.max(0.5, ratio) // Never reduce below 0.5x
  1318→    return {
  1319→      adjustedMultiplier: Math.round(multiplier * 100) / 100,
  1320→      reason: `Recent win rate (${(recentWinRate * 100).toFixed(1)}%) below 80% of historical (${(historicalWinRate * 100).toFixed(1)}%). Reducing to ${multiplier.toFixed(2)}x`,
  1321→      currentWinRate: recentWinRate,
  1322→      historicalWinRate,
  1323→      sampleSize: closedTrades.length,
  1324→    }
  1325→  }
  1326→
  1327→  return { adjustedMultiplier: 1.0, reason: 'Win rate within normal range', currentWinRate: recentWinRate, historicalWinRate, sampleSize: closedTrades.length }
  1328→}
