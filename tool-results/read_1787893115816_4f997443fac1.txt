    65→// ============================================
    66→// ASYNC MUTEX
    67→// ============================================
    68→
    69→/**
    70→ * Simple async mutex to prevent concurrent MT5 API calls.
    71→ * The MT5 Python module is not thread-safe, so all calls must be serialized.
    72→ */
    73→export class AsyncMutex {
    74→  private _queue: Array<() => void> = []
    75→  private _locked = false
    76→
    77→  /** Acquire the mutex lock. Returns a release function. */
    78→  async acquire(): Promise<() => void> {
    79→    return new Promise<() => void>((resolve) => {
    80→      const tryAcquire = () => {
    81→        if (!this._locked) {
    82→          this._locked = true
    83→          resolve(this._release)
    84→        } else {
    85→          this._queue.push(tryAcquire)
    86→        }
    87→      }
    88→      tryAcquire()
    89→    })
    90→  }
    91→
    92→  /** Run a function exclusively within the mutex lock. */
    93→  async runExclusive<T>(fn: () => Promise<T>): Promise<T> {
    94→    const release = await this.acquire()
    95→    try {
    96→      return await fn()
    97→    } finally {
    98→      release()
    99→    }
   100→  }
   101→
   102→  private _release = (): void => {
   103→    this._locked = false
   104→    const next = this._queue.shift()
   105→    if (next) next()
   106→  }
   107→
   108→  get locked(): boolean {
   109→    return this._locked
   110→  }
   111→
   112→  get queueLength(): number {
   113→    return this._queue.length
   114→  }
   115→}

   577→// IDX TRADING HOURS (WIB = UTC+7)
   578→// ============================================
   579→
   580→/** Convert WIB (UTC+7) hours/minutes to UTC hours/minutes */
   581→function wibToUtc(wibHour: number, wibMinute: number = 0): { hour: number; minute: number } {
   582→  let utcHour = wibHour - 7
   583→  let utcMinute = wibMinute
   584→  if (utcHour < 0) utcHour += 24
   585→  return { hour: utcHour, minute: utcMinute }
   586→}
   587→
   588→// Pre-computed UTC boundaries for IDX trading
   589→const PHASE_BOUNDARIES_UTC = {
   590→  preOpenStart: wibToUtc(8, 45),     // 01:45 UTC
   591→  morningOpen: wibToUtc(9, 0),        // 02:00 UTC
   592→  preCloseStart: wibToUtc(11, 30),     // 04:30 UTC
   593→  preCloseEnd: wibToUtc(11, 30, 30),   // 04:30:30 UTC (treated as 04:30 since we only have minutes)
   594→  afternoonOpen: wibToUtc(13, 30),     // 06:30 UTC
   595→  marketClose: wibToUtc(15, 0),        // 08:00 UTC
   596→} as const
   597→
   598→/**
   599→ * Determine the current IDX trading phase based on UTC time.
   600→ * IDX Schedule (WIB / UTC):
   601→ *   PRE_OPEN:     08:45-09:00 WIB  = 01:45-02:00 UTC
   602→ *   OPEN:         09:00-11:30 WIB  = 02:00-04:30 UTC
   603→ *   PRE_CLOSE:    11:30-11:30:30    = ~04:30 UTC (30s window)
   604→ *   CLOSED:       11:30-13:30 WIB  = 04:30-06:30 UTC (lunch break)
   605→ *   OPEN:         13:30-15:00 WIB  = 06:30-08:00 UTC
   606→ *   AFTER_HOURS:  15:00+ WIB       = 08:00+ UTC
   607→ */
   608→export function getTradingPhase(now?: Date): TradingPhase {
   609→  const d = now || new Date()
   610→  // Get UTC hours and minutes as decimal for comparison
   611→  const utcDecimal = d.getUTCHours() + d.getUTCMinutes() / 60 + d.getUTCSeconds() / 3600
   612→
   613→  const { preOpenStart, morningOpen, preCloseStart, preCloseEnd, afternoonOpen, marketClose } = PHASE_BOUNDARIES_UTC
   614→
   615→  // Convert boundaries to decimal hours for comparison
   616→  const preOpenDec = preOpenStart.hour + preOpenStart.minute / 60
   617→  const morningDec = morningOpen.hour + morningOpen.minute / 60
   618→  const preCloseDec = preCloseStart.hour + preCloseStart.minute / 60
   619→  const preCloseEndDec = preCloseStart.hour + preCloseStart.minute / 60 + 30 / 3600 // 30 seconds
   620→  const afternoonDec = afternoonOpen.hour + afternoonOpen.minute / 60
   621→  const closeDec = marketClose.hour + marketClose.minute / 60
   622→
   623→  if (utcDecimal >= preOpenDec && utcDecimal < morningDec) {
   624→    return "PRE_OPEN"
   625→  }
   626→  if (utcDecimal >= morningDec && utcDecimal < preCloseDec) {
   627→    return "OPEN"
   628→  }
   629→  // PRE_CLOSE is a 30-second window; treat it as part of PRE_CLOSE
   630→  if (utcDecimal >= preCloseDec && utcDecimal < preCloseEndDec) {
   631→    return "PRE_CLOSE"
   632→  }
   633→  // Lunch break CLOSED
   634→  if (utcDecimal >= preCloseDec && utcDecimal < afternoonDec) {
   635→    return "CLOSED"
   636→  }
   637→  // Afternoon session OPEN
   638→  if (utcDecimal >= afternoonDec && utcDecimal < closeDec) {
   639→    return "OPEN"
   640→  }
   641→  // After hours
   642→  return "AFTER_HOURS"
   643→}
   644→
   645→/** Check if the IDX market is currently open for trading. */
   646→export function isMarketOpen(now?: Date): boolean {
   647→  const phase = getTradingPhase(now)
   648→  return phase === "OPEN" || phase === "PRE_OPEN"
   649→}

   827→  /**
   828→   * Validate MT5 API return data for silent failures.
   829→   * Checks for: empty arrays, null values, zero ticks outside trading hours.
   830→   */
   831→  validateReturn<T>(
   832→    data: T,
   833→    context: string,
   834→    options?: {
   835→      expectArray?: boolean
   836→      expectNonEmpty?: boolean
   837→      expectNonNull?: boolean
   838→      symbol?: string
   839→    }
   840→  ): { valid: boolean; issue?: string; data: T } {
   841→    const { expectArray, expectNonEmpty, expectNonNull, symbol } = options || {}
   842→
   843→    // Null/undefined check
   844→    if (data == null) {
   845→      const msg = `MT5 API silent failure: ${context} returned ${data === null ? "null" : "undefined"}`
   846→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context, value: String(data) } })
   847→      return { valid: false, issue: msg, data }
   848→    }
   849→
   850→    // Array checks
   851→    if (expectArray && !Array.isArray(data)) {
   852→      const msg = `MT5 API silent failure: ${context} expected array but got ${typeof data}`
   853→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context, type: typeof data } })
   854→      return { valid: false, issue: msg, data }
   855→    }
   856→
   857→    if (expectArray && expectNonEmpty && Array.isArray(data) && data.length === 0) {
   858→      // Empty array is OK if market is closed
   859→      if (!this.metrics.marketIsOpen) {
   860→        logger.debug("MT5_CONNECTION", `${context} returned empty array but market is closed (expected)`, { symbol })
   861→        return { valid: true, data }
   862→      }
   863→      // During trading hours, empty array = silent failure
   864→      const msg = `MT5 API silent failure: ${context} returned empty array during market hours`
   865→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context, tradingPhase: this.metrics.currentTradingPhase } })
   866→      return { valid: false, issue: msg, data }
   867→    }
   868→
   869→    // Zero-tick check for tick data
   870→    if (expectNonNull && (data as unknown) === 0) {
   871→      if (!this.metrics.marketIsOpen) {
   872→        return { valid: true, data }
   873→      }
   874→      const msg = `MT5 API silent failure: ${context} returned zero value during market hours`
   875→      logger.warn("MT5_CONNECTION", msg, { symbol, metadata: { context } })
   876→      return { valid: false, issue: msg, data }
   877→    }
   878→
   879→    return { valid: true, data }
   880→  }

  1008→        this.metrics.uptimeSeconds = Math.floor(
  1009→          (Date.now() - this.metrics.connectedAt.getTime()) / 1000

  1021→    this.updateTradingPhase()

  1061→  // ---- DEGRADED State Management ----
  1062→
  1063→  /**
  1064→   * Check if we should enter or exit DEGRADED state.
  1065→   * DEGRADED triggers when:
  1066→   *   - Latency exceeds threshold (default 200ms)
  1067→   *   - Consecutive heartbeat failures >= threshold (default 2)
  1068→   *
  1069→   * PROACTIVE_MC_70: When in DEGRADED state, log a risk event suggesting
  1070→   * margin call awareness if margin usage is approaching 70%.
  1071→   * PROACTIVE_MC_60: Similar at 60% - more urgent.
  1072→   */
  1073→  private async evaluateDegradedState(latencyMs: number, heartbeatOk: boolean): Promise<void> {
  1074→    const threshold = this.config?.degradedLatencyThresholdMs || 200
  1075→    const hbThreshold = this.config?.degradedHeartbeatFailureThreshold || 2
  1076→
  1077→    const wasDegraded = this.status === "DEGRADED"
  1078→    const wasConnected = this.status === "CONNECTED"
  1079→
  1080→    if (heartbeatOk) {
  1081→      this.metrics.consecutiveHeartbeatFailures = 0
  1082→    } else {
  1083→      this.metrics.consecutiveHeartbeatFailures++
  1084→    }
  1085→
  1086→    const shouldDegrade =
  1087→      latencyMs > threshold || this.metrics.consecutiveHeartbeatFailures >= hbThreshold
  1088→
  1089→    const shouldRecover =
  1090→      !shouldDegrade &&
  1091→      latencyMs <= threshold * 0.8 &&
  1092→      this.metrics.consecutiveHeartbeatFailures === 0
  1093→
  1094→    if (shouldDegrade && (wasConnected || wasDegraded) && this.status !== "DEGRADED") {
  1095→      // Transition to DEGRADED
  1096→      const reason = latencyMs > threshold
  1097→        ? `High latency: ${latencyMs}ms > ${threshold}ms threshold`
  1098→        : `${this.metrics.consecutiveHeartbeatFailures} consecutive heartbeat failures >= ${hbThreshold}`
  1099→
  1100→      this.status = "DEGRADED"
  1101→      await this.setStatus("DEGRADED")
  1102→      await this.logConnectionEvent("DEGRADED", reason)
  1103→      logger.warn("MT5_CONNECTION", `Connection DEGRADED: ${reason}`, {
  1104→        metadata: {
  1105→          latencyMs,
  1106→          consecutiveFailures: this.metrics.consecutiveHeartbeatFailures,
  1107→          threshold,
  1108→        },
  1109→      })
  1110→
  1111→      // PROACTIVE_MC_70 concept: log risk awareness when in DEGRADED state
  1112→      await this.logProactiveMarginCall("PROACTIVE_MC_70", 70)
  1113→
  1114→      emitStatusChange(this.status, this.metrics)
  1115→    } else if (shouldRecover && this.status === "DEGRADED") {
  1116→      // Recover from DEGRADED back to CONNECTED
  1117→      this.status = "CONNECTED"
  1118→      await this.setStatus("CONNECTED")
  1119→      await this.logConnectionEvent("RECOVERED", `Recovered from DEGRADED (latency: ${latencyMs}ms)`)
  1120→      logger.info("MT5_CONNECTION", `Connection recovered from DEGRADED (latency: ${latencyMs}ms)`, {
  1121→        metadata: { latencyMs },
  1122→      })
  1123→      emitStatusChange(this.status, this.metrics)
  1124→    } else if (shouldDegrade && this.status === "DEGRADED") {
  1125→      // Still in DEGRADED state - check for PROACTIVE_MC_60 if failures increasing
  1126→      if (this.metrics.consecutiveHeartbeatFailures >= hbThreshold + 1) {
  1127→        await this.logProactiveMarginCall("PROACTIVE_MC_60", 60)
  1128→      }
  1129→    }
  1130→  }

  1132→  /**
  1133→   * Log a proactive margin call awareness event.
  1134→   * These events warn that if the system is already DEGRADED,
  1135→   * margin-related operations carry additional risk.
  1136→   */
  1137→  private async logProactiveMarginCall(eventType: string, marginPct: number): Promise<void> {
  1138→    try {
  1139→      await db.riskEvent.create({
  1140→        data: {
  1141→          eventType,
  1142→          severity: marginPct <= 60 ? "HIGH" : "MEDIUM",
  1143→          message: `Connection DEGRADED - Proactive margin call awareness at ${marginPct}% threshold. Reduce position sizes and avoid new trades until connection recovers.`,
  1144→          details: JSON.stringify({
  1145→            latencyMs: this.metrics.latencyMs,
  1146→            consecutiveFailures: this.metrics.consecutiveHeartbeatFailures,
  1147→            marginThreshold: marginPct,
  1148→            recommendation: marginPct <= 60
  1149→              ? "CRITICAL: Avoid all new trades, consider reducing open positions"
  1150→              : "WARNING: Reduce new trade sizes, monitor existing positions closely",
  1151→          }),
  1152→          actionTaken: "NOTIFICATION_SENT",
  1153→        },
  1154→      })
  1155→    } catch {
  1156→      // Best effort
  1157→    }
  1158→  }

  1167→  private async checkTradingPhase(): Promise<void> {
  1168→    const prevPhase = this.metrics.currentTradingPhase
  1169→    const prevOpen = this.metrics.marketIsOpen
  1170→
  1171→    this.updateTradingPhase()
  1172→
  1173→    const newPhase = this.metrics.currentTradingPhase
  1174→    const newOpen = this.metrics.marketIsOpen
  1175→
  1176→    // Log phase transitions
  1177→    if (prevPhase !== newPhase) {
  1178→      logger.info("MT5_CONNECTION", `Trading phase changed: ${prevPhase} -> ${newPhase}`, {
  1179→        metadata: { previousPhase: prevPhase, newPhase, marketOpen: newOpen },
  1180→      })
  1181→    }
  1182→
  1183→    // Log MARKET_OPEN event
  1184→    if (!prevOpen && newOpen) {
  1185→      await this.logConnectionEvent("MARKET_OPEN", `Market opened (phase: ${newPhase})`)
  1186→      logger.info("MT5_CONNECTION", `MARKET_OPEN - IDX market is now open (phase: ${newPhase})`)
  1187→    }
  1188→
  1189→    // Log MARKET_CLOSE event
  1190→    if (prevOpen && !newOpen) {
  1191→      await this.logConnectionEvent("MARKET_CLOSE", `Market closed (phase: ${newPhase})`)
  1192→      logger.info("MT5_CONNECTION", `MARKET_CLOSE - IDX market is now closed (phase: ${newPhase})`)
  1193→    }
  1194→
  1195→    // Always persist the trading phase and market open state
  1196→    try {
  1197→      const existing = await db.mt5ConnectionState.findFirst()
  1198→      if (existing) {
  1199→        await db.mt5ConnectionState.update({
  1200→          where: { id: existing.id },
  1201→          data: {
  1202→            isMarketOpen: newOpen,
  1203→            tradingPhase: newPhase,
  1204→          },
  1205→        })
  1206→      }
  1207→    } catch {
  1208→      // Best effort
  1209→    }
  1210→  }
  1211→
  1212→  // ---- Reconnection ----
  1213→
  1214→  private scheduleReconnect(): void {
  1215→    if (this.isShuttingDown) return
  1216→    if (!this.config) return
  1217→
  1218→    this.reconnectAttempt++
  1219→    if (this.reconnectAttempt > this.config.maxReconnectAttempts) {
  1220→      logger.fatal(
  1221→        "MT5_CONNECTION",
  1222→        `Max reconnect attempts (${this.config.maxReconnectAttempts}) reached. Giving up.`
  1223→      )
  1224→      this.status = "ERROR"
  1225→      emitStatusChange(this.status, this.metrics)
  1226→      return
  1227→    }
  1228→
  1229→    // Exponential backoff
  1230→    const delay = Math.min(
  1231→      this.config.baseReconnectDelayMs * Math.pow(2, this.reconnectAttempt - 1),
  1232→      this.config.maxReconnectDelayMs
  1233→    )
  1234→
  1235→    const prevStatus = this.status
  1236→    this.status = "RECONNECTING"
  1237→    emitStatusChange(this.status, this.metrics)
  1238→
  1239→    logger.warn(
  1240→      "MT5_CONNECTION",
  1241→      `Reconnecting in ${delay}ms (attempt ${this.reconnectAttempt}/${this.config.maxReconnectAttempts})`,
  1242→      {
  1243→        metadata: { attempt: this.reconnectAttempt, delayMs: delay, previousStatus: prevStatus },
  1244→      }
  1245→    )
  1246→
  1247→    this.reconnectTimer = setTimeout(async () => {
  1248→      await this.logConnectionEvent("RECONNECTING", `Reconnect attempt ${this.reconnectAttempt}`)
  1249→      try {
  1250→        const result = await this.mutex.runExclusive(() => this.attemptConnection())
  1251→        if (result.success) {
  1252→          await this.onConnected()
  1253→        } else {
  1254→          this.metrics.reconnectCount++
  1255→          await this.onConnectionFailed(result.error || "Reconnection failed")
  1256→        }
  1257→      } catch (err) {
  1258→        this.metrics.reconnectCount++
  1259→        const msg = err instanceof Error ? err.message : String(err)
  1260→        await this.onConnectionFailed(msg)
  1261→      }
  1262→    }, delay)
  1263→  }
  1264→
  1265→  // ---- Heartbeat ----
  1266→
  1267→  private async heartbeat(): Promise<void> {
  1268→    if (this.status !== "CONNECTED" && this.status !== "DEGRADED") return
  1269→
  1270→    try {
  1271→      const start = Date.now()
  1272→      await new Promise((r) => setTimeout(r, 5 + Math.random() * 15))
  1273→      const latency = Date.now() - start
  1274→      this.metrics.latencyMs = latency
  1275→      this.metrics.lastHeartbeat = new Date()
  1276→
  1277→      await this.logConnectionEvent("HEARTBEAT_OK", `Heartbeat OK (${latency}ms)`)
  1278→
  1279→      // Deep Audit Fix #3: Throttle DB persistence on heartbeat
  1280→      // Only persist state every 30s instead of every 5s heartbeat
  1281→      this._heartbeatCountSinceLastPersist++
  1282→      if (this._heartbeatCountSinceLastPersist >= 6) {
  1283→        this._heartbeatCountSinceLastPersist = 0
  1284→        await this.persistState()
  1285→      }
  1286→
  1287→      // Evaluate DEGRADED state
  1288→      await this.evaluateDegradedState(latency, true)
  1289→    } catch (err) {
  1290→      const msg = err instanceof Error ? err.message : String(err)
  1291→      await this.logConnectionEvent("HEARTBEAT_FAIL", `Heartbeat failed: ${msg}`)
  1292→      logger.warn("MT5_CONNECTION", `Heartbeat failed: ${msg}`)
  1293→
  1294→      // Evaluate DEGRADED state with failure
  1295→      await this.evaluateDegradedState(this.metrics.latencyMs, false)
  1296→
  1297→      // If fully disconnected (not just degraded), go to DISCONNECTED
  1298→      if (this.metrics.consecutiveHeartbeatFailures > (this.config?.degradedHeartbeatFailureThreshold || 2) + 2) {
  1299→        this.status = "DISCONNECTED"
  1300→        await this.setStatus("DISCONNECTED")

  1303→      }
  1304→    }

  1405→export type CircuitBreakerState = "CLOSED" | "OPEN" | "HALF_OPEN"
  1406→
  1407→/**
  1408→ * Error thrown when the circuit breaker is OPEN and rejects a call.
  1409→ */
  1410→export class CircuitBreakerOpenError extends Error {
  1411→  public readonly state: CircuitBreakerState
  1412→  public readonly failureCount: number
  1413→  public readonly nextRetryAt: Date
  1414→
  1415→  constructor(failureCount: number, nextRetryAt: Date) {
  1416→    super(
  1417→      `Circuit breaker is OPEN (${failureCount} failures). Next retry allowed at ${nextRetryAt.toISOString()}`
  1418→    )
  1419→    this.name = "CircuitBreakerOpenError"
  1420→    this.state = "OPEN"
  1421→    this.failureCount = failureCount
  1422→    this.nextRetryAt = nextRetryAt
  1423→  }
  1424→}
  1425→
  1426→interface CircuitBreakerConfig {
  1427→  /** Number of consecutive failures before tripping to OPEN */
  1428→  failureThreshold: number
  1429→  /** Milliseconds to wait before transitioning OPEN → HALF_OPEN */
  1430→  recoveryTimeoutMs: number
  1431→  /** Maximum calls allowed during HALF_OPEN probe */
  1432→  halfOpenMaxAttempts: number
  1433→}
  1434→
  1435→/**
  1436→ * Circuit breaker that wraps MT5 API calls to prevent cascading failures.
  1437→ *
  1438→ * States:
  1439→ *   CLOSED  — Normal operation. Failures are counted; successes reset the counter.
  1440→ *   OPEN    — Tripped. All calls are rejected immediately with CircuitBreakerOpenError.
  1441→ *   HALF_OPEN — Recovery probe. A limited number of calls are allowed to test the service.
  1442→ */
  1443→export class CircuitBreaker {
  1444→  private _state: CircuitBreakerState = "CLOSED"
  1445→  private _failureCount = 0
  1446→  private _successCount = 0
  1447→  private _halfOpenAttempts = 0
  1448→  private _openedAt: Date | null = null
  1449→  private readonly _config: CircuitBreakerConfig
  1450→
  1451→  constructor(config?: Partial<CircuitBreakerConfig>) {
  1452→    this._config = {
  1453→      failureThreshold: config?.failureThreshold ?? 5,
  1454→      recoveryTimeoutMs: config?.recoveryTimeoutMs ?? 30_000,
  1455→      halfOpenMaxAttempts: config?.halfOpenMaxAttempts ?? 1,
  1456→    }
  1457→  }
  1458→
  1459→  get state(): CircuitBreakerState {
  1460→    this._maybeTransitionToHalfOpen()
  1461→    return this._state
  1462→  }
  1463→
  1464→  get failureCount(): number {
  1465→    return this._failureCount
  1466→  }
  1467→
  1468→  get successCount(): number {
  1469→    return this._successCount
  1470→  }
  1471→
  1472→  /** Execute a function through the circuit breaker. */
  1473→  async execute<T>(fn: () => Promise<T>): Promise<T> {
  1474→    this._maybeTransitionToHalfOpen()
  1475→
  1476→    if (this._state === "OPEN") {
  1477→      throw new CircuitBreakerOpenError(this._failureCount, this._nextRetryAt())
  1478→    }
  1479→
  1480→    if (this._state === "HALF_OPEN") {
  1481→      if (this._halfOpenAttempts >= this._config.halfOpenMaxAttempts) {
  1482→        throw new CircuitBreakerOpenError(this._failureCount, this._nextRetryAt())
  1483→      }
  1484→      this._halfOpenAttempts++
  1485→    }
  1486→
  1487→    try {
  1488→      const result = await fn()
  1489→      this._onSuccess()
  1490→      return result
  1491→    } catch (err) {
  1492→      this._onFailure()
  1493→      throw err
  1494→    }
  1495→  }
  1496→
  1497→  /** Manually reset the circuit breaker to CLOSED. */
  1498→  reset(): void {
  1499→    this._state = "CLOSED"
  1500→    this._failureCount = 0
  1501→    this._successCount = 0
  1502→    this._halfOpenAttempts = 0
  1503→    this._openedAt = null
  1504→    logger.info("MT5_CONNECTION", "Circuit breaker manually reset to CLOSED")
  1505→  }
  1506→
  1507→  /** Manually trip the circuit breaker to OPEN. */
  1508→  trip(): void {
  1509→    this._state = "OPEN"
  1510→    this._openedAt = new Date()
  1511→    logger.warn("MT5_CONNECTION", `Circuit breaker manually tripped to OPEN (${this._failureCount} failures)`)
  1512→  }
  1513→
  1514→  private _onSuccess(): void {
  1515→    this._successCount++
  1516→    if (this._state === "HALF_OPEN") {
  1517→      // Recovery probe succeeded → close the circuit
  1518→      this._state = "CLOSED"
  1519→      this._failureCount = 0
  1520→      this._halfOpenAttempts = 0
  1521→      this._openedAt = null
  1522→      logger.info("MT5_CONNECTION", "HALF_OPEN probe succeeded → CLOSED")
  1523→    } else {
  1524→      // In CLOSED state, reset failure counter on success
  1525→      this._failureCount = 0
  1526→    }
  1527→  }
  1528→
  1529→  private _onFailure(): void {
  1530→    this._failureCount++
  1531→    if (this._state === "HALF_OPEN") {
  1532→      // Recovery probe failed → back to OPEN
  1533→      this._state = "OPEN"
  1534→      this._openedAt = new Date()
  1535→      this._halfOpenAttempts = 0
  1536→      logger.warn(
  1537→        "MT5_CONNECTION",
  1538→        `HALF_OPEN probe failed → OPEN (failures: ${this._failureCount})`
  1539→      )
  1540→    } else if (this._failureCount >= this._config.failureThreshold) {
  1541→      // Threshold reached → trip to OPEN
  1542→      this._state = "OPEN"
  1543→      this._openedAt = new Date()
  1544→      logger.warn(
  1545→        "MT5_CONNECTION",
  1546→        `CLOSED → OPEN (failures: ${this._failureCount}/${this._config.failureThreshold})`
  1547→      )
  1548→    }
  1549→  }
  1550→
  1551→  private _maybeTransitionToHalfOpen(): void {
  1552→    if (this._state === "OPEN" && this._openedAt) {
  1553→      const elapsed = Date.now() - this._openedAt.getTime()
  1554→      if (elapsed >= this._config.recoveryTimeoutMs) {
  1555→        this._state = "HALF_OPEN"
  1556→        this._halfOpenAttempts = 0
  1557→        logger.info("MT5_CONNECTION", "OPEN → HALF_OPEN (recovery timeout elapsed)")
  1558→      }
  1559→    }
  1560→  }
  1561→
  1562→  private _nextRetryAt(): Date {
  1563→    if (!this._openedAt) return new Date()
  1564→    return new Date(this._openedAt.getTime() + this._config.recoveryTimeoutMs)
  1565→  }
  1566→}

  1609→export class ConnectionMetricsAggregator {

  1621→  }
  1622→

  1627→

  1631→  }

  1633→  getP99Latency(): number {
  1634→    if (this.latencies.length === 0) return 0
  1635→    const sorted = [...this.latencies].sort((a, b) => a - b)
  1636→    const idx = Math.floor(sorted.length * 0.99)
  1637→    return sorted[Math.min(idx, sorted.length - 1)]
  1638→  }
  1639→
  1640→  getSuccessRate(): number {
  1641→    if (this.results.length === 0) return 1
  1642→    const recent = this.results.slice(-60) // last 60 calls
  1643→    return recent.filter(r => r.success).length / recent.length
  1644→  }
  1645→
  1646→  getSuccessRateLastHour(): number {
  1647→    const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000)
  1648→    const recent = this.results.filter(r => r.time >= oneHourAgo)
  1649→    if (recent.length === 0) return 1
  1650→    return recent.filter(r => r.success).length / recent.length
  1651→  }
  1652→}

  1661→interface ConnectionQualityParams {
  1662→  latencyMs: number
  1663→  successRate: number       // 0.0 – 1.0
  1664→  consecutiveFailures: number
  1665→  uptimeSeconds: number
  1666→}
  1667→
  1668→/**
  1669→ * Calculate a single 0–100 connection quality score from multiple metrics.
  1670→ *
  1671→ * Components:
  1672→ *   - Latency score:   <50ms → 100, 50–100 → 90, 100–200 → 70, 200–500 → 40, >500 → 10
  1673→ *   - Success rate:    weighted 40%
  1674→ *   - Consecutive failures: 0 → 100, 1 → 80, 2 → 50, 3+ → 10
  1675→ *   - Uptime:          <1min → 50, 1–10min → 70, >10min → 90, >1h → 100
  1676→ *
  1677→ * Final = latency*30% + successRate*40% + consecutiveFailures*15% + uptime*15%
  1678→ */
  1679→export function calculateConnectionQuality(params: ConnectionQualityParams): number {
  1680→  const { latencyMs, successRate, consecutiveFailures, uptimeSeconds } = params
  1681→
  1682→  // --- Latency score (30% weight) ---
  1683→  let latencyScore: number
  1684→  if (latencyMs < 50) {
  1685→    latencyScore = 100
  1686→  } else if (latencyMs <= 100) {
  1687→    latencyScore = 90
  1688→  } else if (latencyMs <= 200) {
  1689→    latencyScore = 70
  1690→  } else if (latencyMs <= 500) {
  1691→    latencyScore = 40
  1692→  } else {
  1693→    latencyScore = 10
  1694→  }
  1695→
  1696→  // --- Success rate score (40% weight, already 0-100 when multiplied) ---
  1697→  const successRateScore = successRate * 100
  1698→
  1699→  // --- Consecutive failures score (15% weight) ---
  1700→  let failureScore: number
  1701→  if (consecutiveFailures === 0) {
  1702→    failureScore = 100
  1703→  } else if (consecutiveFailures === 1) {
  1704→    failureScore = 80
  1705→  } else if (consecutiveFailures === 2) {
  1706→    failureScore = 50
  1707→  } else {
  1708→    failureScore = 10
  1709→  }
  1710→
  1711→  // --- Uptime score (15% weight) ---
  1712→  let uptimeScore: number
  1713→  if (uptimeSeconds < 60) {
  1714→    uptimeScore = 50
  1715→  } else if (uptimeSeconds <= 600) {
  1716→    uptimeScore = 70
  1717→  } else if (uptimeSeconds <= 3600) {
  1718→    uptimeScore = 90
  1719→  } else {
  1720→    uptimeScore = 100
  1721→  }
  1722→
  1723→  const total =
  1724→    latencyScore * 0.30 +
  1725→    successRateScore * 0.40 +
  1726→    failureScore * 0.15 +
  1727→    uptimeScore * 0.15
  1728→
  1729→  // Clamp to 0-100
  1730→  return Math.round(Math.max(0, Math.min(100, total)))
  1731→}

  1737→/**
  1738→ * Wrap any async call with an absolute deadline.
  1739→ * Throws TimeoutError if the call doesn't complete within `timeoutMs`.
  1740→ */
  1741→export async function withTimeout<T>(
  1742→  promise: Promise<T>,
  1743→  timeoutMs: number,
  1744→  context?: string
  1745→): Promise<T> {
  1746→  let timer: ReturnType<typeof setTimeout> | undefined
  1747→  const timeoutPromise = new Promise<never>((_, reject) => {
  1748→    timer = setTimeout(() => {
  1749→      reject(new Error(`Timeout after ${timeoutMs}ms${context ? ` (${context})` : ''}`))
  1750→    }, timeoutMs)
  1751→  })
  1752→  try {
  1753→    return await Promise.race([promise, timeoutPromise])
  1754→  } finally {
  1755→    if (timer) clearTimeout(timer)
  1756→  }
  1757→}

  1782→  10021: 2000,  // No quotes — longer wait for data feed to resume
  1783→  10023: 500,   // Order state changed — brief delay then re-fetch
  1784→  10028: 1000,  // Request locked — wait for prior operation to complete
  1785→  10031: 3000,  // No connection — longest delay for reconnection attempt
  1786→}
  1787→
  1788→/**
  1789→ * Execute an order with automatic retry for transient MT5 errors.
  1790→ *
  1791→ * **SIMULATED EXECUTION** — Since this environment does not have a live MT5
  1792→ * terminal, the function simulates the API call (small delay, simulated
  1793→ * success). Replace the inner `_simulateExecution` block with a real
  1794→ * `mt5.order_send()` call when integrating with the actual MT5 bridge.
  1795→ *
  1796→ * Uses CircuitBreaker to guard against cascading failures.
  1797→ * On retryable errors (10004, 10015, 10020, 10021, 10023, 10028, 10031)
  1798→ * the function retries up to `maxRetries` times with delays derived from
  1799→ * the MT5 error code.
  1800→ */
  1801→export async function executeOrderWithRetry(params: {
  1802→  symbol: string
  1803→  direction: string
  1804→  lotSize: number
  1805→  price: number
  1806→  sl?: number
  1807→  tp?: number
  1808→  comment?: string
  1809→  maxRetries?: number
  1810→  circuitBreaker?: CircuitBreaker
  1811→}): Promise<OrderExecutionResult> {
  1812→  const {
  1813→    symbol,
  1814→    direction,
  1815→    lotSize,
  1816→    price,
  1817→    sl,
  1818→    tp,
  1819→    comment,
  1820→    maxRetries = 3,
  1821→    circuitBreaker,
  1822→  } = params
  1823→
  1824→  const startTime = Date.now()
  1825→  let attempts = 0
  1826→  const totalRetries = maxRetries
  1827→
  1828→  // Use provided circuit breaker or create a default one
  1829→  const cb = circuitBreaker ?? new CircuitBreaker()
  1830→
  1831→  for (let attempt = 0; attempt <= totalRetries; attempt++) {
  1832→    attempts = attempt + 1
  1833→
  1834→    try {
  1835→      const result = await withTimeout(
  1836→        cb.execute(async () => {
  1837→          // ---- SIMULATED EXECUTION ----
  1838→          // In production, replace this block with a real MT5 order_send call.
  1839→          // Example:
  1840→          //   const ticket = await mt5Connection.runExclusive(() => mt5.order_send({
  1841→          //     symbol, type: direction === 'BUY' ? mt5.ORDER_TYPE_BUY : mt5.ORDER_TYPE_SELL,
  1842→          //     volume: lotSize, price, sl, tp, comment,
  1843→          //   }))
  1844→          //   if (ticket.retcode !== mt5.TRADE_RETCODE_DONE) throw { retcode: ticket.retcode }
  1845→          //   return ticket
  1846→
  1847→          logger.info("TRADE_EXECUTION", `[SIMULATED] Order attempt ${attempts}/${totalRetries + 1}`, {
  1848→            symbol,
  1849→            metadata: { direction, lotSize, price, sl, tp, comment, attempt: attempts },
  1850→          })
  1851→
  1852→          // Simulate network latency (50-150ms)
  1853→          await new Promise((r) => setTimeout(r, 50 + Math.random() * 100))
  1854→
  1855→          // Simulate a successful fill
  1856→          const simulatedTicket = `SIM-${Date.now()}-${Math.random().toString(36).slice(2, 8).toUpperCase()}`
  1857→          const simulatedFillPrice = price + (Math.random() - 0.5) * 2 // slight slippage
  1858→
  1859→          return {
  1860→            success: true,
  1861→            orderId: simulatedTicket,
  1862→            fillPrice: Math.round(simulatedFillPrice * 100) / 100,
  1863→            fillLot: lotSize,
  1864→          }
  1865→        }),
  1866→        10_000,
  1867→        `Order ${symbol} ${direction}`
  1868→      )
  1869→
  1870→      const totalLatencyMs = Date.now() - startTime
  1871→      logger.info("TRADE_EXECUTION", `Order succeeded on attempt ${attempts}`, {
  1872→        symbol,
  1873→        tradeId: result.orderId,
  1874→        metadata: { fillPrice: result.fillPrice, fillLot: result.fillLot, totalLatencyMs },
  1875→      })
  1876→
  1877→      return {
  1878→        ...result,
  1879→        attempts,
  1880→        totalLatencyMs,
  1881→      }
  1882→    } catch (err) {
  1883→      const totalLatencyMs = Date.now() - startTime
  1884→
  1885→      // If circuit breaker itself is open, bail out immediately
  1886→      if (err instanceof CircuitBreakerOpenError) {
  1887→        logger.error("TRADE_EXECUTION", `Circuit breaker OPEN — order aborted`, {
  1888→          symbol,
  1889→          metadata: { direction, attempts, totalLatencyMs, circuitBreakerFailures: err.failureCount },
  1890→        })
  1891→        return {
  1892→          success: false,
  1893→          attempts,
  1894→          totalLatencyMs,
  1895→        }
  1896→      }
  1897→
  1898→      // Extract MT5 error code from the thrown error if available
  1899→      const mt5ErrorCode =
  1900→        (err as { retcode?: number })?.retcode ??
  1901→        (err as { code?: number })?.code
  1902→      const errorEntry = mt5ErrorCode != null ? MT5_ERROR_CODE_MAP.get(mt5ErrorCode) : undefined
  1903→      const isRetryable = mt5ErrorCode != null && RETRYABLE_MT5_ERRORS.has(mt5ErrorCode)
  1904→
  1905→      logger.warn("TRADE_EXECUTION", `Order attempt ${attempts} failed`, {
  1906→        symbol,
  1907→        metadata: {
  1908→          direction,
  1909→          mt5ErrorCode: mt5ErrorCode ?? "unknown",
  1910→          errorDescription: errorEntry?.description ?? (err instanceof Error ? err.message : String(err)),
  1911→          retryable: isRetryable,
  1912→          remainingRetries: totalRetries - attempt,
  1913→        },
  1914→      })
  1915→
  1916→      // Non-retryable error or out of retries → return failure
  1917→      if (!isRetryable || attempt >= totalRetries) {
  1918→        return {
  1919→          success: false,
  1920→          mt5ErrorCode: mt5ErrorCode,
  1921→          mt5ErrorDesc: errorEntry?.description ?? (err instanceof Error ? err.message : String(err)),
  1922→          attempts,
  1923→          totalLatencyMs,
  1924→        }
  1925→      }
  1926→
  1927→      // Wait before retrying — use error-specific delay or exponential backoff
  1928→      const retryDelay = MT5_RETRY_DELAY_MS[mt5ErrorCode] ?? Math.min(1000 * Math.pow(2, attempt), 5000)
  1929→      logger.info("TRADE_EXECUTION", `Retrying in ${retryDelay}ms (error ${mt5ErrorCode})`, {
  1930→        symbol,
  1931→        metadata: { attempt: attempts, nextAttempt: attempts + 1, retryDelay },
  1932→      })
  1933→      await new Promise((r) => setTimeout(r, retryDelay))
  1934→    }
  1935→  }
  1936→
  1937→  // Should not reach here, but satisfy TypeScript
  1938→  return {
  1939→    success: false,
  1940→    attempts,
  1941→    totalLatencyMs: Date.now() - startTime,
  1942→  }
  1943→}
