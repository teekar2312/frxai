    44→interface SignalResult {
    45→  id: string;
    46→  symbol: Pair;
    47→  side: "BUY" | "SELL";
    48→  entry: number;
    49→  stopLoss: number;
    50→  takeProfit: number;
    51→  confidence: number;
    52→  reason: string;
    53→}
    54→
    55→interface ChatMessage {
    56→  id: string;
    57→  role: "user" | "ai";
    58→  text: string;
    59→}
    60→
    61→type DirectionMeta = {
    62→  tone: "up" | "down" | "default";
    63→  color: string;
    64→  bg: string;
    65→  label: string;
    66→  Icon: typeof TrendingUp;
    67→};
    68→
    69→function directionMeta(d: SignalDirection): DirectionMeta {
    70→  if (d === "BUY")
    71→    return {
    72→      tone: "up",
    73→      color: "text-emerald-400",
    74→      bg: "bg-emerald-500/10 border-emerald-500/30",
    75→      label: "BUY",
    76→      Icon: TrendingUp,
    77→    };
    78→  if (d === "SELL")
    79→    return {
    80→      tone: "down",
    81→      color: "text-rose-400",
    82→      bg: "bg-rose-500/10 border-rose-500/30",
    83→      label: "SELL",
    84→      Icon: TrendingDown,
    85→    };
    86→  return {
    87→    tone: "default",
    88→    color: "text-muted-foreground",
    89→    bg: "bg-muted/40 border-border",
    90→    label: "NEUTRAL",
    91→    Icon: Minus,
    92→  };
    93→}

    95→export function AiSection() {
    96→  const tradingCfg = useStore((s) => s.tradingCfg);
    97→  const [symbol, setSymbol] = useState<Pair>(tradingCfg.pairs[0] ?? "EURUSD");
    98→  const [providerLabel, setProviderLabel] = useState("Z.ai");
    99→  const [loading, setLoading] = useState(false);
   100→  const [result, setResult] = useState<AiAnalysisResult | null>(null);
   101→  const [lastTime, setLastTime] = useState<Date | null>(null);
   102→  const [executing, setExecuting] = useState(false);
   103→  const [signal, setSignal] = useState<SignalResult | null>(null);
   104→  const [signalReason, setSignalReason] = useState<string | null>(null);
   105→
   106→  // chat state
   107→  const [messages, setMessages] = useState<ChatMessage[]>([
   108→    {
   109→      id: "intro",
   110→      role: "ai",
   111→      text: "Halo! Saya asisten AI trading FINEX Indonesia. Tanyakan apapun tentang pasar forex, strategi scalping, atau analisa pair.",
   112→    },
   113→  ]);
   114→  const [chatInput, setChatInput] = useState("");
   115→  const [chatSending, setChatSending] = useState(false);
   116→  const chatEndRef = useRef<HTMLDivElement>(null);
   117→
   118→  // fetch active provider on mount
   119→  useEffect(() => {
   120→    fetch("/api/config/keys")
   121→      .then((r) => r.json())
   122→      .then((d) => {
   123→        const ap = d?.keys?.activeProvider as string | undefined;
   124→        const found = AI_PROVIDERS.find((p) => p.key === ap);
   125→        setProviderLabel(found ? found.label.split(" (")[0] : "Z.ai");
   126→      })
   127→      .catch(() => setProviderLabel("Z.ai"));
   128→  }, []);
   129→
   130→  useEffect(() => {
   131→    chatEndRef.current?.scrollIntoView({ behavior: "smooth" });
   132→  }, [messages]);
   133→
   134→  async function runAnalysis() {
   135→    setLoading(true);
   136→    setResult(null);
   137→    setSignal(null);
   138→    setSignalReason(null);
   139→    try {
   140→      const res = await fetch("/api/ai/analyze", {
   141→        method: "POST",
   142→        headers: { "Content-Type": "application/json" },
   143→        body: JSON.stringify({ symbol }),
   144→      });
   145→      if (!res.ok) throw new Error("Analisa gagal");
   146→      const data = await res.json();
   147→      setResult(data.analysis as AiAnalysisResult);
   148→      setLastTime(new Date());
   149→      toast.success("Analisa AI selesai", {
   150→        description: `${symbol}: ${data.analysis.signal} @ ${data.analysis.confidence}% confidence`,
   151→      });
   152→    } catch (e) {
   153→      toast.error("Gagal menganalisa", {
   154→        description: e instanceof Error ? e.message : "Unknown error",
   155→      });
   156→    } finally {
   157→      setLoading(false);
   158→    }
   159→  }
   160→
   161→  async function executeSignal() {
   162→    if (!result) return;
   163→    setExecuting(true);
   164→    setSignal(null);
   165→    setSignalReason(null);
   166→    try {
   167→      const res = await fetch("/api/ai/signal", {
   168→        method: "POST",
   169→        headers: { "Content-Type": "application/json" },
   170→        body: JSON.stringify({ symbol }),
   171→      });
   172→      if (!res.ok) throw new Error("Eksekusi sinyal gagal");
   173→      const data = await res.json();
   174→      if (data.signal) {
   175→        setSignal(data.signal as SignalResult);
   176→        toast.success("Sinyal dihasilkan", {
   177→          description: `${data.signal.side} ${symbol} @ ${data.signal.entry}`,
   178→        });
   179→      } else {
   180→        setSignalReason((data.reason as string) ?? "Sinyal dilewati");
   181→        toast.info("Sinyal dilewati", {
   182→          description: (data.reason as string) ?? "Confidence terlalu rendah",
   183→        });
   184→      }
   185→    } catch (e) {
   186→      toast.error("Gagal eksekusi sinyal", {
   187→        description: e instanceof Error ? e.message : "Unknown error",
   188→      });
   189→    } finally {
   190→      setExecuting(false);
   191→    }
   192→  }
   193→
   194→  async function sendChat() {
   195→    const text = chatInput.trim();
   196→    if (!text || chatSending) return;
   197→    const userMsg: ChatMessage = {
   198→      id: Math.random().toString(36).slice(2),
   199→      role: "user",
   200→      text,
   201→    };
   202→    setMessages((m) => [...m, userMsg]);
   203→    setChatInput("");
   204→    setChatSending(true);
   205→    try {
   206→      const res = await fetch("/api/ai/chat", {
   207→        method: "POST",
   208→        headers: { "Content-Type": "application/json" },
   209→        body: JSON.stringify({ message: text }),
   210→      });
   211→      if (!res.ok) throw new Error("Chat gagal");
   212→      const data = await res.json();
   213→      setMessages((m) => [
   214→        ...m,
   215→        {
   216→          id: Math.random().toString(36).slice(2),
   217→          role: "ai",
   218→          text: (data.reply as string) ?? "Tidak ada balasan.",
   219→        },
   220→      ]);
   221→    } catch {
   222→      setMessages((m) => [
   223→        ...m,
   224→        {
   225→          id: Math.random().toString(36).slice(2),
   226→          role: "ai",
   227→          text: "Maaf, terjadi kesalahan koneksi. Silakan coba lagi.",
   228→        },
   229→      ]);
   230→      toast.error("Chat AI gagal");
   231→    } finally {
   232→      setChatSending(false);
   233→    }
   234→  }
   235→
   236→  const sig = result ? directionMeta(result.signal) : null;
   237→
   238→  return (
   239→    <div className="space-y-5">
   240→      <SectionHeader
   241→        title="Pusat Analisa AI Multi-Faktor"
   242→        description="Analisa kebijakan bank sentral, data ekonomi, geopolitik, komoditas, sentimen & breaking news."
   243→        icon={<Brain className="size-5" />}
   244→        actions={
   245→          <>
   246→            <Pill tone="accent">
   247→              <Sparkles className="size-3" /> {providerLabel}
   248→            </Pill>
   249→            <Select value={symbol} onValueChange={(v) => setSymbol(v as Pair)}>
   250→              <SelectTrigger size="sm" className="w-[120px]">
   251→                <SelectValue />
   252→              </SelectTrigger>
   253→              <SelectContent>
   254→                {PAIRS.map((p) => (
   255→                  <SelectItem key={p.symbol} value={p.symbol}>
   256→                    {p.symbol}
   257→                  </SelectItem>
   258→                ))}
   259→              </SelectContent>
   260→            </Select>
   261→            <Button size="sm" onClick={runAnalysis} disabled={loading}>
   262→              {loading ? (
   263→                <Loader2 className="size-4 animate-spin" />
   264→              ) : (
   265→                <Zap className="size-4" />
   266→              )}
   267→              Analisa Sekarang
   268→            </Button>
   269→          </>
   270→        }
   271→      />
   272→
   273→      {/* Faktor strip */}
   274→      <div className="flex flex-wrap items-center gap-2">
   275→        <span className="text-xs font-medium text-muted-foreground">
   276→          Faktor dianalisa:
   277→        </span>
   278→        {FACTOR_LIST.map((f, i) => (
   279→          <Pill key={i} className="text-[10px]">
   280→            {f}
   281→          </Pill>
   282→        ))}
   283→      </div>
   284→
   285→      {/* Main grid */}
   286→      <div className="grid grid-cols-1 gap-5 lg:grid-cols-3">
   287→        {/* LEFT — Hasil Analisa */}
   288→        <div className="lg:col-span-2">
   289→          <Panel
   290→            title="Hasil Analisa"
   291→            description={
   292→              lastTime
   293→                ? `Terakhir dianalisa: ${lastTime.toLocaleTimeString("id-ID")}`
   294→                : "Belum ada analisa"
   295→            }
   296→            actions={
   297→              result && sig ? (
   298→                <Pill tone={sig.tone}>
   299→                  <sig.Icon className="size-3" /> {sig.label}
   300→                </Pill>
   301→              ) : undefined
   302→            }
   303→          >
   304→            {loading ? (
   305→              <LoadingState symbol={symbol} />
   306→            ) : !result ? (
   307→              <EmptyState
   308→                icon={<Brain className="size-10" />}
   309→                title="Belum ada hasil analisa"
   310→                description="Pilih pair lalu klik 'Analisa Sekarang'. AI memproses 7 faktor pasar (~10-20 detik)."
   311→              />
   312→            ) : (
   313→              <ResultBody
   314→                result={result}
   315→                sig={sig!}
   316→                executing={executing}
   317→                signal={signal}
   318→                signalReason={signalReason}
   319→                onExecute={executeSignal}
   320→              />
   321→            )}
   322→          </Panel>
   323→        </div>
   324→
   325→        {/* RIGHT — Chat & Riwayat */}
   326→        <div>
   327→          <Tabs defaultValue="chat">
   328→            <TabsList className="w-full">
   329→              <TabsTrigger value="chat" className="flex-1">
   330→                <MessageSquare className="size-3.5" /> Chat AI
   331→              </TabsTrigger>
   332→              <TabsTrigger value="history" className="flex-1">
   333→                <History className="size-3.5" /> Riwayat
   334→              </TabsTrigger>
   335→            </TabsList>
   336→            <TabsContent value="chat" className="mt-2">
   337→              <div className="flex h-[420px] flex-col overflow-hidden rounded-xl border border-border bg-card shadow-sm">
   338→                <div className="flex-1 space-y-3 overflow-y-auto scroll-thin p-4">
   339→                  {messages.map((m) => (
   340→                    <ChatBubble key={m.id} msg={m} />
   341→                  ))}
   342→                  {chatSending && (
   343→                    <div className="flex items-center gap-2 text-xs text-muted-foreground">
   344→                      <Bot className="size-4 text-violet-400" />
   345→                      <Loader2 className="size-3 animate-spin" /> AI sedang mengetik...
   346→                    </div>
   347→                  )}
   348→                  <div ref={chatEndRef} />
   349→                </div>
   350→                <div className="border-t border-border p-3">
   351→                  <div className="flex items-center gap-2">
   352→                    <Input
   353→                      value={chatInput}
   354→                      onChange={(e) => setChatInput(e.target.value)}
   355→                      onKeyDown={(e) => {
   356→                        if (e.key === "Enter") sendChat();
   357→                      }}
   358→                      placeholder="Tanya AI tentang pasar..."
   359→                      disabled={chatSending}
   360→                    />
   361→                    <Button
   362→                      size="icon"
   363→                      onClick={sendChat}
   364→                      disabled={chatSending || !chatInput.trim()}
   365→                    >
   366→                      {chatSending ? (
   367→                        <Loader2 className="size-4 animate-spin" />
   368→                      ) : (
   369→                        <Send className="size-4" />
   370→                      )}
   371→                    </Button>
   372→                  </div>
   373→                </div>
   374→              </div>
   375→            </TabsContent>
   376→            <TabsContent value="history" className="mt-2">
   377→              <Panel title="Riwayat Analisa" description="Tersimpan untuk self-learning AI">
   378→                {lastTime ? (
   379→                  <div className="space-y-3">
   380→                    <div className="rounded-lg border border-border bg-muted/30 p-3">
   381→                      <div className="flex items-center justify-between">
   382→                        <span className="text-sm font-medium">{symbol}</span>
   383→                        <span className="text-xs text-muted-foreground">
   384→                          {lastTime.toLocaleString("id-ID")}
   385→                        </span>
   386→                      </div>
   387→                      <div className="mt-2 flex items-center gap-2">
   388→                        {sig && (
   389→                          <Pill tone={sig.tone}>
   390→                            <sig.Icon className="size-3" /> {sig.label}
   391→                          </Pill>
   392→                        )}
   393→                        <span className="text-xs text-muted-foreground">
   394→                          Confidence{" "}
   395→                          <span className="tnum">{result?.confidence}%</span>
   396→                        </span>
   397→                      </div>
   398→                    </div>
   399→                    <p className="text-xs text-muted-foreground">
   400→                      Setiap analisa disimpan di database untuk memory
   401→                      self-learning AI. Model memperbaiki keputusan dari setiap
   402→                      hasil yang tersimpan.
   403→                    </p>
   404→                  </div>
   405→                ) : (
   406→                  <EmptyState
   407→                    icon={<History className="size-8" />}
   408→                    title="Belum ada riwayat"
   409→                    description="Jalankan analisa pertama Anda."
   410→                  />
   411→                )}
   412→              </Panel>
   413→            </TabsContent>
   414→          </Tabs>
   415→        </div>
   416→      </div>
   417→
   418→      {/* Status Model AI */}
   419→      <Panel
   420→        title="Status Model AI"
   421→        description="Multi-provider LLM dengan self-learning memory"
   422→        actions={
   423→          <Pill tone="accent">
   424→            <Cpu className="size-3" /> Self-Learning Aktif
   425→          </Pill>
   426→        }
   427→      >
   428→        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-5">
   429→          {AI_PROVIDERS.map((p) => {
   430→            const active = p.label
   431→              .toLowerCase()
   432→              .startsWith(providerLabel.toLowerCase());
   433→            return (
   434→              <div
   435→                key={p.key}
   436→                className={cn(
   437→                  "rounded-lg border p-3 transition-colors",
   438→                  active
   439→                    ? "border-violet-500/40 bg-violet-500/5"
   440→                    : "border-border bg-muted/20",
   441→                )}
   442→              >
   443→                <div className="flex items-center justify-between">
   444→                  <span className="text-sm font-medium">
   445→                    {p.label.split(" (")[0]}
   446→                  </span>
   447→                  {active && (
   448→                    <span className="h-2 w-2 rounded-full bg-violet-400 live-dot" />
   449→                  )}
   450→                </div>
   451→                <p className="mt-1 text-xs text-muted-foreground">{p.note}</p>
   452→                {active && (
   453→                  <p className="mt-2 text-[10px] font-medium text-violet-400">
   454→                    AKTIF
   455→                  </p>
   456→                )}
   457→              </div>
   458→            );
   459→          })}
   460→        </div>
   461→        <div className="mt-4 flex items-start gap-2 rounded-lg border border-violet-500/20 bg-violet-500/5 p-3">
   462→          <Brain className="mt-0.5 size-4 shrink-0 text-violet-400" />
   463→          <p className="text-xs text-muted-foreground">
   464→            Model belajar mandiri dari setiap analisa (self-learning memory
   465→            tersimpan di database). Semua hasil analisa multi-faktor dipersist
   466→            untuk meningkatkan akurasi keputusan di masa depan.
   467→          </p>
   468→        </div>
   469→      </Panel>
   470→    </div>
   471→  );
   472→}

   479→        <div className="relative flex h-16 w-16 items-center justify-center rounded-full border border-violet-500/30 bg-violet-500/10">
   480→          <Brain className="size-8 text-violet-400" />
   481→        </div>
   482→      </div>
   483→      <div className="text-center">
   484→        <p className="text-sm font-medium">
   485→          AI sedang menganalisa multi-faktor...
   486→        </p>

   488→          {symbol} — proses ~10-20 detik
   489→        </p>
   490→      </div>
   491→      <div className="grid w-full max-w-md grid-cols-1 gap-2">

   506→function ResultBody({
   507→  result,
   508→  sig,
   509→  executing,
   510→  signal,
   511→  signalReason,
   512→  onExecute,
   513→}: {
   514→  result: AiAnalysisResult;
   515→  sig: DirectionMeta;
   516→  executing: boolean;
   517→  signal: SignalResult | null;
   518→  signalReason: string | null;
   519→  onExecute: () => void;
   520→}) {
   521→  return (
   522→    <div className="space-y-5">
   523→      {/* Signal banner */}
   524→      <div className={cn("rounded-xl border p-4", sig.bg)}>
   525→        <div className="flex flex-wrap items-center justify-between gap-3">
   526→          <div className="flex items-center gap-3">
   527→            <sig.Icon className={cn("size-8", sig.color)} />
   528→            <div>
   529→              <div className="text-[10px] uppercase tracking-wider text-muted-foreground">
   530→                Sinyal AI
   531→              </div>
   532→              <div className={cn("text-2xl font-bold", sig.color)}>
   533→                {sig.label}
   534→              </div>
   535→            </div>
   536→          </div>
   537→          <div className="text-right">
   538→            <div className="text-[10px] uppercase tracking-wider text-muted-foreground">
   539→              Confidence
   540→            </div>
   541→            <div className={cn("text-3xl font-bold tnum", sig.color)}>
   542→              {result.confidence}%
   543→            </div>
   544→          </div>
   545→        </div>
   546→        <div className="mt-3">
   547→          <Progress
   548→            value={result.confidence}
   549→            className={cn(
   550→              "h-2",
   551→              sig.tone === "up"
   552→                ? "[&_[data-slot=progress-indicator]]:bg-emerald-500"
   553→                : sig.tone === "down"
   554→                  ? "[&_[data-slot=progress-indicator]]:bg-rose-500"
   555→                  : "[&_[data-slot=progress-indicator]]:bg-violet-500",
   556→            )}
   557→          />
   558→        </div>
   559→      </div>
   560→
   561→      {/* Summary */}
   562→      <div>
   563→        <div className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground">
   564→          Ringkasan
   565→        </div>
   566→        <p className="mt-1.5 text-sm leading-relaxed text-foreground/90">
   567→          {result.summary}
   568→        </p>
   569→      </div>
   570→
   571→      {/* Entry / SL / TP */}
   572→      <div className="grid grid-cols-3 gap-3">
   573→        <MiniStat
   574→          label="Entry"
   575→          value={result.suggestedEntry}
   576→          icon={<Crosshair className="size-3.5" />}
   577→          tone="default"
   578→        />
   579→        <MiniStat
   580→          label="Stop Loss"
   581→          value={result.suggestedStopLoss}
   582→          icon={<Shield className="size-3.5" />}
   583→          tone="down"
   584→        />
   585→        <MiniStat
   586→          label="Take Profit"
   587→          value={result.suggestedTakeProfit}
   588→          icon={<Target className="size-3.5" />}
   589→          tone="up"
   590→        />
   591→      </div>
   592→
   593→      {/* Execute */}
   594→      <div className="flex flex-wrap items-center gap-3">
   595→        <Button onClick={onExecute} disabled={executing}>
   596→          {executing ? (
   597→            <Loader2 className="size-4 animate-spin" />
   598→          ) : (
   599→            <Zap className="size-4" />
   600→          )}
   601→          Eksekusi Sinyal
   602→        </Button>
   603→        {signal && (

   683→function FactorRow({ factor }: { factor: FactorScore }) {
   684→  const meta = directionMeta(factor.direction);
   685→  const score = Math.max(-100, Math.min(100, factor.score));
   686→  const widthPct = Math.abs(score) / 2; // 0..50 (half-width bar)
   687→  return (
   688→    <div className="rounded-lg border border-border bg-muted/20 p-3">
   689→      <div className="flex items-center justify-between gap-2">
   690→        <span className="text-sm font-medium">{factor.factor}</span>
   691→        <Pill tone={meta.tone === "up" ? "up" : meta.tone === "down" ? "down" : "default"}>
   692→          <meta.Icon className="size-3" /> {meta.label}
   693→        </Pill>
   694→      </div>
   695→      <div className="mt-2 flex items-center gap-2">
   696→        <span className="w-8 text-right text-[10px] tnum text-rose-400/70">
   697→          -100
   698→        </span>
   699→        <div className="relative h-2.5 flex-1 rounded-full bg-muted">
   700→          <div className="absolute left-1/2 top-0 h-full w-px bg-border" />
   701→          {score >= 0 ? (
   702→            <div
   703→              className="absolute left-1/2 top-0 h-full rounded-r-full bg-emerald-500"
   704→              style={{ width: `${widthPct}%` }}
   705→            />
   706→          ) : (
   707→            <div
   708→              className="absolute top-0 h-full rounded-l-full bg-rose-500"
   709→              style={{ right: "50%", width: `${widthPct}%` }}
   710→            />
   711→          )}
   712→        </div>
   713→        <span className="w-8 text-[10px] tnum text-emerald-400/70">+100</span>
   714→        <span
   715→          className={cn(
   716→            "w-12 text-right text-xs font-semibold tnum",
   717→            meta.color,
   718→          )}
   719→        >
   720→          {score > 0 ? "+" : ""}
   721→          {score}
   722→        </span>
   723→      </div>
   724→      {factor.detail && (
   725→        <p className="mt-2 text-xs leading-relaxed text-muted-foreground">
   726→          {factor.detail}
   727→        </p>
   728→      )}
   729→    </div>
   730→  );
   731→}
