    61→function toggleInArray<T>(arr: T[], item: T): T[] {
    62→  return arr.includes(item) ? arr.filter((x) => x !== item) : [...arr, item];
    63→}

    65→function formatMoney(n: number): string {
    66→  return n.toLocaleString("en-US", {
    67→    minimumFractionDigits: 2,
    68→    maximumFractionDigits: 2,
    69→  });
    70→}

    72→export function SettingsSection() {
    73→  const tradingCfg = useStore((s) => s.tradingCfg);
    74→  const setTradingCfg = useStore((s) => s.setTradingCfg);
    75→  const riskCfg = useStore((s) => s.riskCfg);
    76→  const setRiskCfg = useStore((s) => s.setRiskCfg);
    77→  const apiKeys = useStore((s) => s.apiKeys);
    78→  const setApiKeys = useStore((s) => s.setApiKeys);
    79→  const account = useStore((s) => s.account);
    80→  const setAccount = useStore((s) => s.setAccount);
    81→  const setSection = useStore((s) => s.setSection);
    82→
    83→  const [keyDraft, setKeyDraft] = useState<ApiKeys>(apiKeys);
    84→  const [savingKeys, setSavingKeys] = useState(false);
    85→
    86→  // MT5 credentials state
    87→  interface Mt5Creds {
    88→    mt5Account: string;
    89→    mt5Password: string;
    90→    mt5Server: string;
    91→    mt5AccountType: "demo" | "real";
    92→    mt5Terminal: string;
    93→    hasPassword: boolean;
    94→  }
    95→  const [mt5Creds, setMt5Creds] = useState<Mt5Creds>({
    96→    mt5Account: "",
    97→    mt5Password: "",
    98→    mt5Server: "FINEX-Live01",
    99→    mt5AccountType: "demo",
   100→    mt5Terminal: "MetaTrader5",
   101→    hasPassword: false,
   102→  });
   103→  const [mt5Connecting, setMt5Connecting] = useState(false);
   104→  const [showMt5Password, setShowMt5Password] = useState(false);
   105→
   106→  // MT5 terminal app state
   107→  interface Mt5Terminal {
   108→    path: string;
   109→    running: boolean;
   110→    pid: number | null;
   111→    autoStart: boolean;
   112→  }
   113→  const [mt5Terminal, setMt5Terminal] = useState<Mt5Terminal>({
   114→    path: "",
   115→    running: false,
   116→    pid: null,
   117→    autoStart: true,
   118→  });
   119→  const [terminalBusy, setTerminalBusy] = useState(false);
   120→
   121→  // Fetch all configs on mount
   122→  useEffect(() => {
   123→    let mounted = true;
   124→    (async () => {
   125→      try {
   126→        const [tr, rk, ks, mt5] = await Promise.all([
   127→          fetch("/api/config/trading").then((r) => r.json()),
   128→          fetch("/api/config/risk").then((r) => r.json()),
   129→          fetch("/api/config/keys").then((r) => r.json()),
   130→          fetch("/api/mt5/credentials").then((r) => r.json()),
   131→        ]);
   132→        if (!mounted) return;
   133→        if (tr?.config) setTradingCfg(tr.config);
   134→        if (rk?.config) setRiskCfg(rk.config);
   135→        if (ks?.keys) {
   136→          setApiKeys(ks.keys);
   137→          setKeyDraft(ks.keys);
   138→        }
   139→        if (mt5?.credentials) {
   140→          setMt5Creds({
   141→            mt5Account: mt5.credentials.mt5Account ?? "",
   142→            mt5Password: mt5.credentials.mt5Password ?? "",
   143→            mt5Server: mt5.credentials.mt5Server ?? "FINEX-Live01",
   144→            mt5AccountType: mt5.credentials.mt5AccountType ?? "demo",
   145→            mt5Terminal: mt5.credentials.mt5Terminal ?? "MetaTrader5",
   146→            hasPassword: mt5.credentials.hasPassword ?? false,
   147→          });
   148→        }
   149→        if (mt5?.terminal) {
   150→          setMt5Terminal({
   151→            path: mt5.terminal.path ?? "",
   152→            running: mt5.terminal.running ?? false,
   153→            pid: mt5.terminal.pid ?? null,
   154→            autoStart: mt5.terminal.autoStart ?? true,
   155→          });
   156→        }
   157→      } catch {
   158→        /* ignore */
   159→      }
   160→    })();
   161→    return () => {
   162→      mounted = false;
   163→    };
   164→  }, [setTradingCfg, setRiskCfg, setApiKeys]);
   165→
   166→  // --- Trading config helpers ---
   167→  const updateTrading = async (patch: Partial<TradingConfig>) => {
   168→    setTradingCfg(patch);
   169→    try {
   170→      await fetch("/api/config/trading", {
   171→        method: "PUT",
   172→        headers: { "Content-Type": "application/json" },
   173→        body: JSON.stringify(patch),
   174→      });
   175→      toast.success("Konfigurasi trading disimpan");
   176→    } catch {
   177→      toast.error("Gagal menyimpan konfigurasi");
   178→    }
   179→  };
   180→
   181→  const handlePairsChange = (vals: string[]) => {
   182→    updateTrading({ pairs: vals as Pair[] });
   183→  };
   184→  const handleTimeframesChange = (vals: string[]) => {
   185→    updateTrading({ timeframes: vals as Timeframe[] });
   186→  };
   187→  const handleSessionsToggle = (s: Session) => {
   188→    updateTrading({ sessions: toggleInArray(tradingCfg.sessions, s) });
   189→  };
   190→
   191→  // --- API keys helpers ---
   192→  const handleKeyChange = (key: keyof ApiKeys, value: string) => {
   193→    setKeyDraft((d) => ({ ...d, [key]: value }));
   194→  };
   195→
   196→  const handleActiveProvider = async (provider: ApiKeys["activeProvider"]) => {
   197→    setApiKeys({ activeProvider: provider });
   198→    setKeyDraft((d) => ({ ...d, activeProvider: provider }));
   199→    try {
   200→      await fetch("/api/config/keys", {
   201→        method: "PUT",
   202→        headers: { "Content-Type": "application/json" },
   203→        body: JSON.stringify({ activeProvider: provider }),
   204→      });
   205→      toast.success(`Provider aktif: ${provider}`, {
   206→        description: "Perubahan diterapkan pada MT5 bridge.",
   207→      });
   208→    } catch {
   209→      toast.error("Gagal mengubah provider");
   210→    }
   211→  };
   212→
   213→  const handleSaveKeys = async () => {
   214→    setSavingKeys(true);
   215→    try {
   216→      // Send all keyDraft values — backend ignores masked (•) values
   217→      await fetch("/api/config/keys", {
   218→        method: "PUT",
   219→        headers: { "Content-Type": "application/json" },
   220→        body: JSON.stringify(keyDraft),
   221→      });
   222→      toast.success("API keys disimpan", {
   223→        description: "Perubahan diterapkan pada MT5 bridge.",
   224→      });
   225→      // Re-fetch to re-mask
   226→      const res = await fetch("/api/config/keys");
   227→      const data = await res.json();
   228→      if (data?.keys) {
   229→        setApiKeys(data.keys);
   230→        setKeyDraft(data.keys);
   231→      }
   232→    } catch {
   233→      toast.error("Gagal menyimpan keys");
   234→    } finally {
   235→      setSavingKeys(false);
   236→    }
   237→  };
   238→
   239→  // --- MT5 connect (with credentials) ---
   240→  const handleMt5Connect = async () => {
   241→    // Validate
   242→    if (!/^\d{4,12}$/.test(mt5Creds.mt5Account)) {
   243→      toast.error("Nomor akun MT5 tidak valid", {
   244→        description: "Harus 4-12 digit angka.",
   245→      });
   246→      return;
   247→    }
   248→    const pwd = mt5Creds.mt5Password.includes("•") ? "" : mt5Creds.mt5Password;
   249→    if (!pwd && !mt5Creds.hasPassword) {
   250→      toast.error("Password MT5 wajib diisi");
   251→      return;
   252→    }
   253→    if (!mt5Creds.mt5Server.trim()) {
   254→      toast.error("Server MT5 wajib diisi");
   255→      return;
   256→    }
   257→    setMt5Connecting(true);
   258→    try {
   259→      // 1) Save credentials first
   260→      await fetch("/api/mt5/credentials", {
   261→        method: "PUT",
   262→        headers: { "Content-Type": "application/json" },
   263→        body: JSON.stringify({
   264→          mt5Account: mt5Creds.mt5Account,
   265→          mt5Password: pwd || undefined, // don't send masked
   266→          mt5Server: mt5Creds.mt5Server,
   267→          mt5AccountType: mt5Creds.mt5AccountType,
   268→          mt5Terminal: mt5Creds.mt5Terminal,
   269→        }),
   270→      });
   271→      // 2) Connect (bridge reads saved creds)
   272→      const res = await fetch("/api/mt5/connect", {
   273→        method: "POST",
   274→        headers: { "Content-Type": "application/json" },
   275→        body: JSON.stringify({
   276→          mt5Account: mt5Creds.mt5Account,
   277→          mt5Password: pwd || undefined,
   278→          mt5Server: mt5Creds.mt5Server,
   279→          mt5AccountType: mt5Creds.mt5AccountType,
   280→          mt5Terminal: mt5Creds.mt5Terminal,
   281→        }),
   282→      });
   283→      const data = await res.json();
   284→      if (!res.ok) {
   285→        throw new Error(data?.error ?? "Gagal menyambungkan MT5");
   286→      }
   287→      if (data?.account) {
   288→        setAccount(data.account);
   289→        toast.success("MT5 Terhubung", {
   290→          description: `Akun ${mt5Creds.mt5Account} @ ${mt5Creds.mt5Server} (${mt5Creds.mt5AccountType})`,
   291→        });
   292→        // refresh terminal status (auto-start may have launched the app)
   293→        refreshTerminalStatus();
   294→      }
   295→    } catch (e) {
   296→      toast.error("Gagal menyambungkan MT5", {
   297→        description: e instanceof Error ? e.message : "Unknown error",
   298→      });
   299→    } finally {
   300→      setMt5Connecting(false);
   301→    }
   302→  };
   303→
   304→  // --- MT5 disconnect ---
   305→  const handleMt5Disconnect = async () => {
   306→    try {
   307→      const res = await fetch("/api/mt5/disconnect", { method: "POST" });
   308→      const data = await res.json();
   309→      if (data?.account) {
   310→        setAccount(data.account);
   311→        toast.info("MT5 Diputus", {
   312→          description: "Kredensial tetap tersimpan untuk reconnect cepat.",
   313→        });
   314→      }
   315→    } catch {
   316→      toast.error("Gagal memutus MT5");
   317→    }
   318→  };
   319→
   320→  // --- MT5 terminal: save path + auto-start ---
   321→  const handleTerminalConfigSave = async () => {
   322→    try {
   323→      await fetch("/api/mt5/credentials", {
   324→        method: "PUT",
   325→        headers: { "Content-Type": "application/json" },
   326→        body: JSON.stringify({
   327→          mt5TerminalPath: mt5Terminal.path,
   328→          mt5AutoStartTerminal: mt5Terminal.autoStart,
   329→        }),
   330→      });
   331→      toast.success("Konfigurasi terminal disimpan");
   332→    } catch {
   333→      toast.error("Gagal menyimpan konfigurasi terminal");
   334→    }
   335→  };
   336→
   337→  // --- MT5 terminal: launch the app ---
   338→  const handleStartTerminal = async () => {
   339→    if (!mt5Terminal.path.trim()) {
   340→      toast.error("Path terminal MT5 belum diisi", {
   341→        description: "Isi path ke terminal64.exe terlebih dahulu, lalu Simpan.",
   342→      });
   343→      return;
   344→    }
   345→    // save path first
   346→    setTerminalBusy(true);
   347→    try {
   348→      await fetch("/api/mt5/credentials", {
   349→        method: "PUT",
   350→        headers: { "Content-Type": "application/json" },
   351→        body: JSON.stringify({
   352→          mt5TerminalPath: mt5Terminal.path,
   353→          mt5AutoStartTerminal: mt5Terminal.autoStart,
   354→        }),
   355→      });
   356→      const res = await fetch("/api/mt5/start-terminal", { method: "POST" });
   357→      const data = await res.json();
   358→      if (!res.ok) {
   359→        throw new Error(data?.error ?? "Gagal menjalankan terminal");
   360→      }
   361→      setMt5Terminal((t) => ({ ...t, running: true, pid: data.pid ?? null }));
   362→      toast.success("Terminal MT5 dijalankan", {
   363→        description: data.alreadyRunning
   364→          ? "Terminal sudah berjalan sebelumnya"
   365→          : `PID ${data.pid} · ${mt5Terminal.path}`,
   366→      });
   367→    } catch (e) {
   368→      toast.error("Gagal menjalankan terminal MT5", {
   369→        description: e instanceof Error ? e.message : "Unknown error",
   370→      });
   371→    } finally {
   372→      setTerminalBusy(false);
   373→    }
   374→  };
   375→
   376→  // --- MT5 terminal: stop the app ---
   377→  const handleStopTerminal = async () => {
   378→    setTerminalBusy(true);
   379→    try {
   380→      const res = await fetch("/api/mt5/stop-terminal", { method: "POST" });
   381→      const data = await res.json();
   382→      if (!res.ok) throw new Error("Gagal menghentikan terminal");
   383→      setMt5Terminal((t) => ({ ...t, running: false, pid: null }));
   384→      // also reflect disconnected account state
   385→      const st = await fetch("/api/config/account").then((r) => r.json());
   386→      if (st?.account) setAccount(st.account);
   387→      toast.info("Terminal MT5 dihentikan", {
   388→        description: "Koneksi bridge juga diputus.",
   389→      });
   390→    } catch (e) {
   391→      toast.error("Gagal menghentikan terminal");
   392→    } finally {
   393→      setTerminalBusy(false);
   394→    }
   395→  };
   396→
   397→  // --- MT5 terminal: refresh status ---
   398→  const refreshTerminalStatus = async () => {
   399→    try {
   400→      const res = await fetch("/api/mt5/terminal-status");
   401→      const data = await res.json();
   402→      setMt5Terminal((t) => ({
   403→        ...t,
   404→        running: data.running ?? false,
   405→        pid: data.pid ?? null,
   406→        path: data.path ?? t.path,
   407→        autoStart: data.autoStart ?? t.autoStart,
   408→      }));
   409→    } catch {
   410→      /* ignore */
   411→    }
   412→  };
   413→
   414→  return (
   415→    <div className="space-y-5">
   416→      <SectionHeader
   417→        title="Pengaturan Sistem"
   418→        description="Konfigurasi broker, AI provider, pair/sesi/timeframe, dan API keys."
   419→        icon={<SettingsIcon className="h-4 w-4" />}
   420→      />
   421→
   422→      <Tabs defaultValue="trading" className="w-full">
   423→        <TabsList className="h-auto w-full justify-start overflow-x-auto p-1">
   424→          <TabsTrigger value="trading">Trading</TabsTrigger>
   425→          <TabsTrigger value="ai">AI Provider</TabsTrigger>
   426→          <TabsTrigger value="broker">Broker / MT5</TabsTrigger>
   427→          <TabsTrigger value="risk">Risiko</TabsTrigger>
   428→        </TabsList>
   429→
   430→        {/* ============ Trading Tab ============ */}
   431→        <TabsContent value="trading" className="space-y-5">
   432→          <Panel
   433→            title="Pair Aktif"
   434→            description="Pilih pasangan mata uang yang dipantau dan diperdagangkan"
   435→          >
   436→            <ToggleGroup
   437→              type="multiple"
   438→              value={tradingCfg.pairs as string[]}
   439→              onValueChange={handlePairsChange}
   440→              variant="outline"
   441→              className="flex w-full flex-wrap"
   442→            >
   443→              {PAIRS.map((p) => (
   444→                <ToggleGroupItem
   445→                  key={p.symbol}
   446→                  value={p.symbol}
   447→                  className="m-0.5 flex-1 px-3"
   448→                  aria-label={p.symbol}
   449→                >
   450→                  {p.symbol}
   451→                </ToggleGroupItem>
   452→              ))}
   453→            </ToggleGroup>
   454→            <p className="mt-2 text-xs text-muted-foreground">
   455→              <span className="tnum text-foreground">{tradingCfg.pairs.length}</span>{" "}
   456→              pair dipilih · <span className="tnum">{PAIRS.length}</span> total tersedia
   457→            </p>
   458→          </Panel>
   459→
   460→          <Panel
   461→            title="Timeframe"
   462→            description="Pilih timeframe untuk analisis & scalping"
   463→          >
   464→            <ToggleGroup
   465→              type="multiple"
   466→              value={tradingCfg.timeframes as string[]}
   467→              onValueChange={handleTimeframesChange}
   468→              variant="outline"
   469→              className="flex w-full flex-wrap"
   470→            >
   471→              {TIMEFRAMES.map((t) => (
   472→                <ToggleGroupItem
   473→                  key={t.value}
   474→                  value={t.value}
   475→                  className="m-0.5 flex-1 px-3"
   476→                  aria-label={t.value}
   477→                >
   478→                  {t.value}
   479→                </ToggleGroupItem>
   480→              ))}
   481→            </ToggleGroup>
   482→            <p className="mt-2 text-xs text-muted-foreground">
   483→              <span className="tnum text-foreground">
   484→                {tradingCfg.timeframes.length}
   485→              </span>{" "}
   486→              timeframe dipilih
   487→            </p>
   488→          </Panel>
   489→
   490→          <Panel
   491→            title="Sesi Trading"
   492→            description="Sesi pasar yang aktif untuk eksekusi (waktu UTC)"
   493→          >
   494→            <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3">
   495→              {SESSIONS.map((s) => {
   496→                const active = tradingCfg.sessions.includes(s.value);
   497→                return (
   498→                  <button
   499→                    key={s.value}
   500→                    type="button"
   501→                    onClick={() => handleSessionsToggle(s.value)}
   502→                    aria-pressed={active}
   503→                    className={`flex items-center gap-2 rounded-lg border px-3 py-2 text-left text-sm transition-colors ${
   504→                      active
   505→                        ? "border-primary bg-primary/10 text-foreground"
   506→                        : "border-border bg-card hover:bg-accent"
   507→                    }`}
   508→                  >
   509→                    <Clock className="h-3.5 w-3.5 shrink-0 text-muted-foreground" />
   510→                    <div className="min-w-0 flex-1">
   511→                      <div className="truncate font-medium">{s.label}</div>
   512→                      <div className="text-[10px] text-muted-foreground tnum">
   513→                        {String(s.utcStart).padStart(2, "0")}:00–
   514→                        {String(s.utcEnd).padStart(2, "0")}:00 UTC
   515→                      </div>
   516→                    </div>
   517→                    {active && (
   518→                      <CheckCircle2 className="h-3.5 w-3.5 shrink-0 text-primary" />
   519→                    )}
   520→                  </button>
   521→                );
   522→              })}
   523→            </div>
   524→          </Panel>
   525→
   526→          <Panel
   527→            title="Aturan & Mode Otomatis"
   528→            description="Pengaturan eksekusi & auto-mode"
   529→          >
   530→            <div className="grid gap-3 md:grid-cols-2">
   531→              <ToggleRow
   532→                icon={<Calendar className="h-4 w-4 text-amber-400" />}
   533→                title="Hindari Akhir Pekan"
   534→                desc="Skip trading Sabtu/Minggu"
   535→                checked={tradingCfg.avoidWeekends}
   536→                onChange={(v) => updateTrading({ avoidWeekends: v })}
   537→              />
   538→              <ToggleRow
   539→                icon={<TrendingUp className="h-4 w-4 text-emerald-400" />}
   540→                title="Trailing Stop Auto"
   541→                desc="Otomatis geser SL mengikuti harga"
   542→                checked={tradingCfg.trailingAuto}
   543→                onChange={(v) => updateTrading({ trailingAuto: v })}
   544→              />
   545→              <ToggleRow
   546→                icon={<Activity className="h-4 w-4 text-violet-400" />}
   547→                title="Indikator Auto"
   548→                desc="Optimasi parameter indikator otomatis"
   549→                checked={tradingCfg.indicatorAuto}
   550→                onChange={(v) => updateTrading({ indicatorAuto: v })}
   551→              />
   552→              <ToggleRow
   553→                icon={<ShieldAlert className="h-4 w-4 text-rose-400" />}
   554→                title="Risk Auto"
   555→                desc="Penyesuaian parameter risiko otomatis"
   556→                checked={tradingCfg.riskAuto}
   557→                onChange={(v) => updateTrading({ riskAuto: v })}
   558→              />
   559→            </div>
   560→          </Panel>
   561→        </TabsContent>
   562→
   563→        {/* ============ AI Provider Tab ============ */}
   564→        <TabsContent value="ai" className="space-y-5">
   565→          <Panel
   566→            title="Provider Aktif"
   567→            description="Pilih AI provider untuk analisis pasar"
   568→          >
